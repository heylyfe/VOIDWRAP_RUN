//! Movement tuning. All durations are in physics frames (1 / PHYSICS_HZ seconds).

use protocol::PHYSICS_HZ;

/// Fixed physics timestep in seconds.
pub const DT: f32 = 1.0 / PHYSICS_HZ as f32;

/// Player collision box half-width (x and z). `ControllerState.pos` is the feet point.
pub const PLAYER_HALF_WIDTH: f32 = 0.4;
/// Player collision box height.
pub const PLAYER_HEIGHT: f32 = 1.8;

/// Ground run speed, units/s. Applied instantly (no acceleration mush).
pub const RUN_SPEED: f32 = 9.0;
/// Gravity, units/s².
pub const GRAVITY: f32 = 32.0;
/// Initial jump velocity, units/s (apex ≈ v²/2g ≈ 2.2 units).
pub const JUMP_VELOCITY: f32 = 12.0;
/// Terminal fall speed, units/s.
pub const MAX_FALL_SPEED: f32 = 40.0;

/// Frames of coyote time after walking off a ledge (~120 ms at 60 Hz).
pub const COYOTE_FRAMES: u8 = 7;
/// Frames a jump press is remembered while waiting for a landing (~100 ms).
pub const JUMP_BUFFER_FRAMES: u8 = 6;
/// Upward speed is clamped to this when jump is released early (variable height).
pub const JUMP_CUT_VELOCITY: f32 = 4.0;

/// Dash speed, units/s.
pub const DASH_SPEED: f32 = 28.0;
/// Frames a dash lasts (28 * 9 / 60 = 4.2 units of travel).
pub const DASH_FRAMES: u8 = 9;
/// Frames between dashes.
pub const DASH_COOLDOWN_FRAMES: u16 = 45;

/// Feet-to-surface distance still treated as "resting on" a solid top.
pub const GROUND_EPSILON: f32 = 0.001;

//! Kinematic character controller.
//!
//! M1 controller: instant run speed (ground and air), gravity with terminal
//! velocity, coyote time, jump buffering, variable jump height (jump cut) and a
//! dash class ability. Collision is axis-separated AABB against [`World`].
//!
//! [`step`] is the contract shared by the authoritative server and the WASM
//! client prediction. Both must produce bit-identical results, so the step path
//! only uses `+ - * /`, `sqrt`, `abs`, `min`, `max` and comparisons: no
//! transcendental functions, no randomness, no time, no allocation.

use crate::tuning::*;
use crate::world::{World, overlaps};
use protocol::{Aabb, BTN_ABILITY, BTN_JUMP, ControllerState, InputFrame, Vec3};

/// A fresh controller standing at `pos`.
pub fn spawn_state(pos: Vec3) -> ControllerState {
    ControllerState {
        pos,
        vel: Vec3::ZERO,
        grounded: false,
        coyote_frames: 0,
        jump_buffer_frames: 0,
        prev_buttons: 0,
        air_jumps_used: 0,
        ability_cooldown_frames: 0,
        ability_active_frames: 0,
        ability_dir: Vec3::ZERO,
    }
}

/// The player's collision box for a given feet position.
pub fn player_box(pos: &Vec3) -> Aabb {
    Aabb {
        min: Vec3::new(pos.x - PLAYER_HALF_WIDTH, pos.y, pos.z - PLAYER_HALF_WIDTH),
        max: Vec3::new(
            pos.x + PLAYER_HALF_WIDTH,
            pos.y + PLAYER_HEIGHT,
            pos.z + PLAYER_HALF_WIDTH,
        ),
    }
}

/// Normalised wish direction from the quantised input (length ≤ 1).
pub fn wish_dir(input: &InputFrame) -> (f32, f32) {
    // Clamp -128 to -127 so the range is symmetric.
    let x = f32::from(input.move_x.max(-127)) / 127.0;
    let z = f32::from(input.move_z.max(-127)) / 127.0;
    let len_sq = x * x + z * z;
    if len_sq > 1.0 {
        let len = len_sq.sqrt();
        (x / len, z / len)
    } else {
        (x, z)
    }
}

/// Advance one fixed physics frame of [`DT`] seconds.
///
/// Step order (identical on server and client):
///   1. timers & edge detection
///   2. dash start
///   3. jump (buffer + coyote)
///   4. horizontal velocity
///   5. vertical velocity (gravity / jump cut / dash)
///   6. collide Y, then X, then Z
///   7. grounded & coyote refresh, ability timer tick
///   8. `prev_buttons = input.buttons`
pub fn step(state: &mut ControllerState, input: &InputFrame, world: &World) {
    let was_grounded = state.grounded;

    // 1. Timers and button edges.
    state.ability_cooldown_frames = state.ability_cooldown_frames.saturating_sub(1);
    let jump_pressed = input.buttons & BTN_JUMP != 0 && state.prev_buttons & BTN_JUMP == 0;
    let ability_pressed = input.buttons & BTN_ABILITY != 0 && state.prev_buttons & BTN_ABILITY == 0;
    if jump_pressed {
        state.jump_buffer_frames = JUMP_BUFFER_FRAMES;
    }
    let (wx, wz) = wish_dir(input);

    // 2. Dash start.
    if ability_pressed && state.ability_cooldown_frames == 0 && state.ability_active_frames == 0 {
        state.ability_dir = dash_dir(wx, wz, &state.vel);
        state.ability_active_frames = DASH_FRAMES;
        state.ability_cooldown_frames = DASH_COOLDOWN_FRAMES;
    }

    // 3. Jump: consume the buffer on the first step a jump is allowed.
    if state.jump_buffer_frames > 0 && (state.grounded || state.coyote_frames > 0) {
        state.vel.y = JUMP_VELOCITY;
        state.grounded = false;
        state.coyote_frames = 0;
        state.jump_buffer_frames = 0;
        state.ability_active_frames = 0; // jumping cancels a dash
    } else if state.jump_buffer_frames > 0 && !jump_pressed {
        // The arming step itself does not tick, so a press N steps before
        // touchdown still fires on the first step a jump is allowed.
        state.jump_buffer_frames -= 1;
    }

    let dashing = state.ability_active_frames > 0;

    // 4. Horizontal velocity (instant, ground and air).
    if dashing {
        state.vel.x = state.ability_dir.x * DASH_SPEED;
        state.vel.z = state.ability_dir.z * DASH_SPEED;
    } else {
        state.vel.x = wx * RUN_SPEED;
        state.vel.z = wz * RUN_SPEED;
    }

    // 5. Vertical velocity.
    if dashing {
        state.vel.y = 0.0;
    } else {
        state.vel.y = (state.vel.y - GRAVITY * DT).max(-MAX_FALL_SPEED);
        if input.buttons & BTN_JUMP == 0 && state.vel.y > 0.0 {
            state.vel.y = state.vel.y.min(JUMP_CUT_VELOCITY);
        }
    }

    // 6. Collide, one axis at a time.
    let landed = move_axis(state, world, Axis::Y, state.vel.y * DT);
    move_axis(state, world, Axis::X, state.vel.x * DT);
    move_axis(state, world, Axis::Z, state.vel.z * DT);

    // 7. Grounded & coyote refresh.
    state.grounded = landed || resting_on_solid(state, world);
    if state.grounded {
        state.coyote_frames = COYOTE_FRAMES;
    } else if !was_grounded {
        // Only steps that began airborne consume coyote time, so the step that
        // walks off a ledge still leaves the whole window available.
        state.coyote_frames = state.coyote_frames.saturating_sub(1);
    }
    state.ability_active_frames = state.ability_active_frames.saturating_sub(1);

    // 8. Remember buttons for edge detection.
    state.prev_buttons = input.buttons;
}

/// Unit horizontal dash direction: wish direction, else current horizontal
/// velocity, else straight back (-Z).
fn dash_dir(wx: f32, wz: f32, vel: &Vec3) -> Vec3 {
    let wish_len_sq = wx * wx + wz * wz;
    if wish_len_sq > 0.0 {
        let len = wish_len_sq.sqrt();
        return Vec3::new(wx / len, 0.0, wz / len);
    }
    let vel_len_sq = vel.x * vel.x + vel.z * vel.z;
    if vel_len_sq > 0.0 {
        let len = vel_len_sq.sqrt();
        return Vec3::new(vel.x / len, 0.0, vel.z / len);
    }
    Vec3::new(0.0, 0.0, -1.0)
}

/// True when the feet rest on a solid top (within [`GROUND_EPSILON`]) while
/// overlapping it in x/z and not moving upwards. Needed because a dash zeroes
/// `vel.y`, which makes the downward Y move (and therefore `landed`) zero.
fn resting_on_solid(state: &ControllerState, world: &World) -> bool {
    if state.vel.y > 0.0 {
        return false;
    }
    let b = player_box(&state.pos);
    for solid in &world.solids {
        let overlaps_xz =
            b.min.x < solid.max.x && b.max.x > solid.min.x && b.min.z < solid.max.z && b.max.z > solid.min.z;
        if overlaps_xz && (state.pos.y - solid.max.y).abs() <= GROUND_EPSILON {
            return true;
        }
    }
    false
}

#[derive(Clone, Copy)]
enum Axis {
    X,
    Y,
    Z,
}

fn get(v: &Vec3, a: Axis) -> f32 {
    match a {
        Axis::X => v.x,
        Axis::Y => v.y,
        Axis::Z => v.z,
    }
}

fn set(v: &mut Vec3, a: Axis, value: f32) {
    match a {
        Axis::X => v.x = value,
        Axis::Y => v.y = value,
        Axis::Z => v.z = value,
    }
}

/// Moves along one axis, stopping flush against the first solid hit.
/// Returns true if the move was a downward move that landed on something.
fn move_axis(state: &mut ControllerState, world: &World, axis: Axis, delta: f32) -> bool {
    if delta == 0.0 {
        return false;
    }
    let mut pos = state.pos.clone();
    let target = get(&pos, axis) + delta;
    set(&mut pos, axis, target);
    let mut hit = false;
    for solid in &world.solids {
        let b = player_box(&pos);
        if !overlaps(&b, solid) {
            continue;
        }
        hit = true;
        // Push back flush against the face we entered through.
        let p = get(&pos, axis);
        let resolved = match axis {
            Axis::Y if delta < 0.0 => solid.max.y,
            Axis::Y => solid.min.y - PLAYER_HEIGHT,
            _ if delta < 0.0 => get(&solid.max, axis) + PLAYER_HALF_WIDTH,
            _ => get(&solid.min, axis) - PLAYER_HALF_WIDTH,
        };
        // Only ever push backwards along the motion, never forwards.
        let clamped = if delta < 0.0 { resolved.max(p) } else { resolved.min(p) };
        set(&mut pos, axis, clamped);
    }
    if hit {
        set(&mut state.vel, axis, 0.0);
    }
    state.pos = pos;
    hit && matches!(axis, Axis::Y) && delta < 0.0
}

//! Pure, deterministic M1 game rules: gift pickups, the beacon, and chaser enemies.
//!
//! Everything here is a plain function over protocol data so the authoritative server
//! and the client-side predictor can run the exact same code. Only `+ - * /` and
//! `sqrt` are used, so results are bit-identical on every target.

use protocol::{Aabb, Gift, TICK_HZ, Vec3};

use crate::controller::player_box;
use crate::world::overlaps;

/// A gift is collected when the player centre is within this distance of it.
pub const GIFT_RADIUS: f32 = 1.1;

/// Horizontal radius in which the beacon counts as reached.
pub const BEACON_RADIUS: f32 = 1.5;

/// Vertical tolerance for standing on the beacon's platform.
pub const BEACON_HEIGHT: f32 = 1.0;

/// Enemy kind tag for the flying chaser.
pub const ENEMY_KIND_CHASER: u8 = 1;

/// Chaser flight speed in units per second (the player runs at 9.0).
pub const CHASER_SPEED: f32 = 5.5;

/// A chaser only pursues targets within this 3D distance.
pub const CHASER_AGGRO_RADIUS: f32 = 22.0;

/// Half extent of the chaser's cubic hit box, centred on its position.
pub const CHASER_HALF_SIZE: f32 = 0.5;

/// Ticks a freshly spawned chaser stays dormant (3 s at 20 Hz).
pub const CHASER_GRACE_TICKS: u32 = 60;

/// Height of the player's centre above their feet.
const PLAYER_CENTRE_HEIGHT: f32 = 0.9;

/// A single enemy. `home` is where it drifts back to when nothing is in range.
#[derive(Debug, Clone, PartialEq)]
pub struct Enemy {
    pub id: u32,
    pub kind: u8,
    pub pos: Vec3,
    pub home: Vec3,
    pub active_in: u32,

/// Euclidean distance between two points.
fn distance(a: &Vec3, b: &Vec3) -> f32 {
    let dx = a.x - b.x;
    let dy = a.y - b.y;
    let dz = a.z - b.z;
    (dx * dx + dy * dy + dz * dz).sqrt()
}

/// The player's centre point, derived from their feet position.
fn player_centre(player_feet: &Vec3) -> Vec3 {
    Vec3::new(player_feet.x, player_feet.y + PLAYER_CENTRE_HEIGHT, player_feet.z)
}

/// Move `pos` toward `target` by at most `max_step`, never overshooting.
fn move_toward(pos: &mut Vec3, target: &Vec3, max_step: f32) {
    let dx = target.x - pos.x;
    let dy = target.y - pos.y;
    let dz = target.z - pos.z;
    let d = (dx * dx + dy * dy + dz * dz).sqrt();
    if d <= max_step {
        *pos = target.clone();
        return;
    }
    let scale = max_step / d;
    pos.x += dx * scale;
    pos.y += dy * scale;
    pos.z += dz * scale;
}

/// True when the player's centre is within [`GIFT_RADIUS`] of the gift centre.
pub fn gift_touched(player_feet: &Vec3, gift: &Gift) -> bool {
    distance(&player_centre(player_feet), &gift.pos) <= GIFT_RADIUS
}

/// True when the player stands on the beacon: close horizontally and roughly level.
pub fn beacon_reached(player_feet: &Vec3, beacon: &Vec3) -> bool {
    let dx = player_feet.x - beacon.x;
    let dz = player_feet.z - beacon.z;
    let horizontal = (dx * dx + dz * dz).sqrt();
    horizontal <= BEACON_RADIUS && (player_feet.y - beacon.y).abs() <= BEACON_HEIGHT
}

/// Create an enemy sitting at `pos`, with `pos` as its home and a full grace period.
pub fn spawn_enemy(id: u32, kind: u8, pos: Vec3) -> Enemy {
    Enemy {
        id,
        kind,
        home: pos.clone(),
        pos,
        active_in: CHASER_GRACE_TICKS,
    }
}

/// Advance one enemy by one server tick (`dt = 1 / TICK_HZ`).
///
/// `targets` are the centres of living players. While `active_in > 0` the enemy only
/// counts down. Afterwards it flies straight at the nearest target inside
/// [`CHASER_AGGRO_RADIUS`] (ties go to the lowest index), or drifts home at half speed.
/// Unknown kinds do nothing at all.
pub fn step_enemy(enemy: &mut Enemy, targets: &[Vec3]) {
    if enemy.kind != ENEMY_KIND_CHASER {
        return;
    }
    if enemy.active_in > 0 {
        enemy.active_in -= 1;
        return;
    }

    let step = CHASER_SPEED / TICK_HZ as f32;

    let mut nearest: Option<(usize, f32)> = None;
    for (index, target) in targets.iter().enumerate() {
        let d = distance(&enemy.pos, target);
        if d > CHASER_AGGRO_RADIUS {
            continue;
        }
        match nearest {
            Some((_, best)) if best <= d => {}
            _ => nearest = Some((index, d)),
        }
    }

    match nearest {
        Some((index, _)) => {
            let target = targets[index].clone();
            move_toward(&mut enemy.pos, &target, step);
        }
        None => {
            let home = enemy.home.clone();
            move_toward(&mut enemy.pos, &home, step * 0.5);
        }
    }
}

/// True when the enemy's cube overlaps the player's collision box.
pub fn enemy_touches(enemy: &Enemy, player_feet: &Vec3) -> bool {
    let half = CHASER_HALF_SIZE;
    let cube = Aabb {
        min: Vec3::new(enemy.pos.x - half, enemy.pos.y - half, enemy.pos.z - half),
        max: Vec3::new(enemy.pos.x + half, enemy.pos.y + half, enemy.pos.z + half),
    };
    overlaps(&cube, &player_box(player_feet))
}

//! WASM bridge for client-side prediction. Runs the exact same [`sim::controller::step`]
//! as the server, so prediction and authority never drift.
//!
//! Data crosses the boundary only as protocol-encoded bytes (bincode), written by the
//! generated TypeScript codec, through one shared I/O buffer:
//!
//! 1. JS calls `vw_io_buf(len)` → pointer to a `len`-byte buffer in WASM memory.
//! 2. JS writes encoded bytes there, then calls `vw_load_level(len)` / `vw_set_state(len)`.
//! 3. `vw_step(...)` advances the predicted controller one physics frame.
//! 4. `vw_get_state()` encodes the state into the I/O buffer and returns its length;
//!    JS reads it from the pointer returned by `vw_io_ptr()`.
//!
//! `#[unsafe(no_mangle)]` is the only unsafe-flagged construct; there are no `unsafe` blocks.
#![allow(unsafe_code)]

use protocol::{ControllerState, InputFrame, LevelData, Vec3};
use sim::controller;
use sim::world::World;
use std::cell::RefCell;

/// Upper bound on bytes JS may ask us to hold (level data is the largest message).
const MAX_IO_BYTES: u32 = 1 << 20;

struct Bridge {
    io: Vec<u8>,
    world: World,
    state: ControllerState,
}

thread_local! {
    static BRIDGE: RefCell<Bridge> = RefCell::new(Bridge {
        io: Vec::new(),
        world: World::default(),
        state: controller::spawn_state(Vec3::ZERO),
    });
}

/// Resize the I/O buffer to `len` bytes and return its address (null if too large).
#[unsafe(no_mangle)]
pub extern "C" fn vw_io_buf(len: u32) -> *mut u8 {
    if len > MAX_IO_BYTES {
        return std::ptr::null_mut();
    }
    BRIDGE.with_borrow_mut(|b| {
        b.io.clear();
        b.io.resize(len as usize, 0);
        b.io.as_mut_ptr()
    })
}

/// Address of the I/O buffer (valid until the next `vw_io_buf` / `vw_get_state`).
#[unsafe(no_mangle)]
pub extern "C" fn vw_io_ptr() -> *const u8 {
    BRIDGE.with_borrow(|b| b.io.as_ptr())
}

fn io_bytes(b: &Bridge, len: u32) -> Option<&[u8]> {
    b.io.get(..len as usize)
}

/// Load collision geometry from an encoded `LevelData`. Returns 0 on success, -1 on error.
#[unsafe(no_mangle)]
pub extern "C" fn vw_load_level(len: u32) -> i32 {
    BRIDGE.with_borrow_mut(
        |b| match io_bytes(b, len).and_then(|x| protocol::decode::<LevelData>(x).ok()) {
            Some(level) => {
                b.world = World::from_level(&level);
                0
            }
            None => -1,
        },
    )
}

/// Replace the predicted state with an encoded `ControllerState`. Returns 0 / -1.
#[unsafe(no_mangle)]
pub extern "C" fn vw_set_state(len: u32) -> i32 {
    BRIDGE.with_borrow_mut(
        |b| match io_bytes(b, len).and_then(|x| protocol::decode::<ControllerState>(x).ok()) {
            Some(state) => {
                b.state = state;
                0
            }
            None => -1,
        },
    )
}

/// Advance the predicted state one physics frame. Out-of-range inputs are clamped
/// exactly as the server would see them after i8/u8 decoding.
#[unsafe(no_mangle)]
pub extern "C" fn vw_step(move_x: i32, move_z: i32, buttons: i32) {
    let input = InputFrame {
        seq: 0,
        move_x: move_x.clamp(-128, 127) as i8,
        move_z: move_z.clamp(-128, 127) as i8,
        buttons: buttons.clamp(0, 255) as u8,
    };
    BRIDGE.with_borrow_mut(|b| {
        let Bridge { world, state, .. } = b;
        controller::step(state, &input, world);
    });
}

/// Encode the predicted state into the I/O buffer; returns its byte length.
#[unsafe(no_mangle)]
pub extern "C" fn vw_get_state() -> u32 {
    BRIDGE.with_borrow_mut(|b| {
        b.io = protocol::encode(&b.state).unwrap_or_default();
        b.io.len() as u32
    })
}

//! Kinematic character controller.
//!
//! M1 controller: instant run speed (ground and air), gravity with terminal
//! velocity, coyote time, jump buffering, variable jump height (jump cut) and a
//! dash class ability. Collision is axis-separated AABB against [`World`].
//!
//! [`step_with`] applies a [`Tuning`] on top of that (upgrades and curses): run,
//! jump, gravity and dash-speed multipliers, a dash-cooldown multiplier and extra
//! air jumps. [`step`] is the same step with the neutral default tuning.
//!
//! [`step`] is the contract shared by the authoritative server and the WASM
//! client prediction. Both must produce bit-identical results, so the step path
//! only uses `+ - * /`, `sqrt`, `abs`, `min`, `max`, `clamp` and comparisons: no
//! transcendental functions, no randomness, no time, no allocation.

use crate::world::{World, overlaps};

    ControllerState {
        pos,
        vel: Vec3::ZERO,
        grounded: false,
        coyote_frames: 0,
        jump_buffer_frames: 0,
        prev_buttons: 0,
        air_jumps_used: 0,
        ability_cooldown_frames: 0,
        ability_active_frames: 0,

        max: Vec3::new(

            pos.z + PLAYER_HALF_WIDTH,
        ),
    }

/// Normalised wish direction from the quantised input (length ≤ 1).
pub fn wish_dir(input: &InputFrame) -> (f32, f32) {
    // Clamp -128 to -127 so the range is symmetric.
    let x = f32::from(input.move_x.max(-127)) / 127.0;
    let z = f32::from(input.move_z.max(-127)) / 127.0;
    let len_sq = x * x + z * z;
    if len_sq > 1.0 {
        let len = len_sq.sqrt();
        (x / len, z / len)
    } else {
        (x, z)
    }
}

/// Advance one fixed physics frame of [`DT`] seconds under `tuning`.
///
/// Step order (identical on server and client):
///   1. timers & edge detection
///   2. dash start
///   3. jump (buffer + coyote, then air jumps)
///   4. horizontal velocity
///   5. vertical velocity (gravity / jump cut / dash)
///   6. collide Y, then X, then Z
///   7. grounded & coyote refresh, air jump reset, ability timer tick
///   8. `prev_buttons = input.buttons`
///
/// Every multiplier is clamped to [`TUNING_MUL_MIN`]..=[`TUNING_MUL_MAX`] (NaN
/// becomes the neutral 1.0) and `air_jumps` to `0..=TUNING_AIR_JUMPS_MAX`, so a
/// malformed tuning can never produce a non-finite or absurd step.
pub fn step_with(state: &mut ControllerState, input: &InputFrame, world: &World, tuning: &Tuning) {
    let was_grounded = state.grounded;

    // Defensive, deterministic clamps. With the default tuning every product
    // below is exactly the untuned constant (`x * 1.0 == x`), which is what keeps
    // `step` bit-identical to the pre-tuning controller.
    let run_speed = RUN_SPEED * clamp_mul(tuning.run_speed_mul);
    let jump_velocity = JUMP_VELOCITY * clamp_mul(tuning.jump_mul);
    let gravity = GRAVITY * clamp_mul(tuning.gravity_mul);
    let dash_speed = DASH_SPEED * clamp_mul(tuning.dash_speed_mul);
    let dash_cooldown = tuned_dash_cooldown(clamp_mul(tuning.dash_cooldown_mul));
    let air_jumps = tuning.air_jumps.min(TUNING_AIR_JUMPS_MAX);

    // 1. Timers and button edges.
    state.ability_cooldown_frames = state.ability_cooldown_frames.saturating_sub(1);
    let jump_pressed = input.buttons & BTN_JUMP != 0 && state.prev_buttons & BTN_JUMP == 0;
    let ability_pressed = input.buttons & BTN_ABILITY != 0 && state.prev_buttons & BTN_ABILITY == 0;
    if jump_pressed {
        state.jump_buffer_frames = JUMP_BUFFER_FRAMES;
    }
    let (wx, wz) = wish_dir(input);

    // 2. Dash start.
    if ability_pressed && state.ability_cooldown_frames == 0 && state.ability_active_frames == 0 {
        state.ability_dir = dash_dir(wx, wz, &state.vel);
        state.ability_active_frames = DASH_FRAMES;
        state.ability_cooldown_frames = dash_cooldown;
    }

    // 3. Jump: consume the buffer on the first step a jump is allowed.
    if state.jump_buffer_frames > 0 && (state.grounded || state.coyote_frames > 0) {
        state.vel.y = jump_velocity;
        state.grounded = false;
        state.coyote_frames = 0;
        state.jump_buffer_frames = 0;
        state.ability_active_frames = 0; // jumping cancels a dash
    } else if state.jump_buffer_frames > 0 && state.air_jumps_used < air_jumps {
        // Air jump: only reachable when the ground/coyote rules said no. Coyote
        // jumps take the branch above and therefore never spend an air jump.
        state.vel.y = jump_velocity;
        state.grounded = false;
        state.coyote_frames = 0;
        state.jump_buffer_frames = 0;
        state.ability_active_frames = 0; // an air jump cancels a dash too
        state.air_jumps_used += 1;
    } else if state.jump_buffer_frames > 0 && !jump_pressed {
        // The arming step itself does not tick, so a press N steps before
        // touchdown still fires on the first step a jump is allowed.
        state.jump_buffer_frames -= 1;
    }

    let dashing = state.ability_active_frames > 0;

    // 4. Horizontal velocity (instant, ground and air).
    if dashing {
        state.vel.x = state.ability_dir.x * dash_speed;
        state.vel.z = state.ability_dir.z * dash_speed;
    } else {
        state.vel.x = wx * run_speed;
        state.vel.z = wz * run_speed;
    }

    // 5. Vertical velocity. Terminal fall speed is deliberately not tuned.
    if dashing {
        state.vel.y = 0.0;
    } else {
        state.vel.y = (state.vel.y - gravity * DT).max(-MAX_FALL_SPEED);
        if input.buttons & BTN_JUMP == 0 && state.vel.y > 0.0 {
            state.vel.y = state.vel.y.min(JUMP_CUT_VELOCITY);
        }
    }

    // 6. Collide, one axis at a time.
    let landed = move_axis(state, world, Axis::Y, state.vel.y * DT);
    move_axis(state, world, Axis::X, state.vel.x * DT);
    move_axis(state, world, Axis::Z, state.vel.z * DT);

    // 7. Grounded & coyote refresh.
    state.grounded = landed || resting_on_solid(state, world);
    if state.grounded {
        state.coyote_frames = COYOTE_FRAMES;
        state.air_jumps_used = 0;
    } else if !was_grounded {
        // Only steps that began airborne consume coyote time, so the step that
        // walks off a ledge still leaves the whole window available.
        state.coyote_frames = state.coyote_frames.saturating_sub(1);
    }
    state.ability_active_frames = state.ability_active_frames.saturating_sub(1);

    // 8. Remember buttons for edge detection.
    state.prev_buttons = input.buttons;
}

/// Clamp a tuning multiplier into [`TUNING_MUL_MIN`]..=[`TUNING_MUL_MAX`], mapping
/// NaN to the neutral 1.0.
///
/// Written with comparisons plus `clamp` only: `min`/`max` return the *other*
/// operand when one side is NaN, so they cannot detect NaN, while comparisons are
/// false for NaN — which is exactly what the final `else` branch keys on. No
/// transcendental function is involved, so the result is identical on every target.
fn clamp_mul(value: f32) -> f32 {
    if value >= TUNING_MUL_MIN {
        value.clamp(TUNING_MUL_MIN, TUNING_MUL_MAX)
    } else if value <= TUNING_MUL_MAX {
        TUNING_MUL_MIN
    } else {
        1.0
    }
}

/// Dash cooldown in frames for a multiplier already passed through [`clamp_mul`].
///
/// `(x + 0.5) as u16` is a deterministic round-half-up: `as` truncates toward zero
/// and saturates, and `x` is non-negative, so `f32::round` (whose behaviour is not
/// guaranteed to be identical on every target) is not needed. The result is then
/// clamped to [`DASH_COOLDOWN_MIN_FRAMES`]..=[`DASH_COOLDOWN_MAX_FRAMES`].
fn tuned_dash_cooldown(mul: f32) -> u16 {
    let scaled = DASH_COOLDOWN_FRAMES as f32 * mul;
    let rounded = (scaled + 0.5) as u16;
    rounded.clamp(DASH_COOLDOWN_MIN_FRAMES, DASH_COOLDOWN_MAX_FRAMES)
}

/// Unit horizontal dash direction: wish direction, else current horizontal
/// velocity, else straight back (-Z).
fn dash_dir(wx: f32, wz: f32, vel: &Vec3) -> Vec3 {
    let wish_len_sq = wx * wx + wz * wz;
    if wish_len_sq > 0.0 {
        let len = wish_len_sq.sqrt();
        return Vec3::new(wx / len, 0.0, wz / len);
    }
    let vel_len_sq = vel.x * vel.x + vel.z * vel.z;
    if vel_len_sq > 0.0 {
        let len = vel_len_sq.sqrt();
        return Vec3::new(vel.x / len, 0.0, vel.z / len);
    }
    Vec3::new(0.0, 0.0, -1.0)
}

/// True when the feet rest on a solid top (within [`GROUND_EPSILON`]) while
/// overlapping it in x/z and not moving upwards. Needed because a dash zeroes
/// `vel.y`, which makes the downward Y move (and therefore `landed`) zero.
fn resting_on_solid(state: &ControllerState, world: &World) -> bool {
    if state.vel.y > 0.0 {
        return false;
    }
    let b = player_box(&state.pos);
    for solid in &world.solids {
        let overlaps_xz =
            b.min.x < solid.max.x && b.max.x > solid.min.x && b.min.z < solid.max.z && b.max.z > solid.min.z;
        if overlaps_xz && (state.pos.y - solid.max.y).abs() <= GROUND_EPSILON {
            return true;
        }
    }
    false
}

#[derive(Clone, Copy)]
enum Axis {
    X,
    Y,
    Z,
}

fn get(v: &Vec3, a: Axis) -> f32 {
    match a {
        Axis::X => v.x,
        Axis::Y => v.y,
        Axis::Z => v.z,
    }
}

fn set(v: &mut Vec3, a: Axis, value: f32) {
    match a {
        Axis::X => v.x = value,
        Axis::Y => v.y = value,
        Axis::Z => v.z = value,
    }
}

/// Moves along one axis, stopping flush against the first solid hit.
/// Returns true if the move was a downward move that landed on something.
fn move_axis(state: &mut ControllerState, world: &World, axis: Axis, delta: f32) -> bool {
    if delta == 0.0 {
        return false;
    }
    let mut pos = state.pos.clone();
    let target = get(&pos, axis) + delta;
    set(&mut pos, axis, target);
    let mut hit = false;
    for solid in &world.solids {
        let b = player_box(&pos);
        if !overlaps(&b, solid) {
            continue;
        }
        hit = true;
        // Push back flush against the face we entered through.
        let p = get(&pos, axis);
        let resolved = match axis {
            Axis::Y if delta < 0.0 => solid.max.y,
            Axis::Y => solid.min.y - PLAYER_HEIGHT,
            _ if delta < 0.0 => get(&solid.max, axis) + PLAYER_HALF_WIDTH,
            _ => get(&solid.min, axis) - PLAYER_HALF_WIDTH,
        };
        // Only ever push backwards along the motion, never forwards.
        let clamped = if delta < 0.0 { resolved.max(p) } else { resolved.min(p) };
        set(&mut pos, axis, clamped);
    }
    if hit {
        set(&mut state.vel, axis, 0.0);
    }
    state.pos = pos;
    hit && matches!(axis, Axis::Y) && delta < 0.0
}

        assert_eq!(s.air_jumps_used, 1);
        assert!(s.vel.y > 0.0);

        // A second press while still airborne must not fire.
        step_with(&mut s, &idle(seq), &w, &tuning);
        seq += 1;
        step_with(&mut s, &input(seq, 0, 0, BTN_JUMP), &w, &tuning);
        seq += 1;
        assert_eq!(s.air_jumps_used, 1, "the air jump budget is spent");

        // Land: the budget comes back.
        let mut guard = 0;
        while !s.grounded && guard < 2000 {
            step_with(&mut s, &idle(seq), &w, &tuning);
            seq += 1;
            guard += 1;
        }
        assert!(s.grounded, "controller never landed");
        assert_eq!(s.air_jumps_used, 0, "landing must reset the air jump budget");

        // A ground jump does not consume an air jump; the next airborne press does.
        step_with(&mut s, &input(seq, 0, 0, BTN_JUMP), &w, &tuning);
        seq += 1;
        assert!(s.vel.y > 0.0, "ground jump after landing");
        assert_eq!(s.air_jumps_used, 0, "a ground jump must not consume an air jump");
        step_with(&mut s, &idle(seq), &w, &tuning);
        seq += 1;
        step_with(&mut s, &input(seq, 0, 0, BTN_JUMP), &w, &tuning);
        assert_eq!(s.air_jumps_used, 1, "the restored budget is usable again");
    }

    fn jump_mul_raises_apex() {

        let apex = |tuning: &Tuning| -> f32 {
            let mut s = spawn_state(Vec3::ZERO);

            for i in 0..120u32 {

                }
            }

        let base = apex(&Tuning::default());

            jump_mul: 1.5,
            ..Tuning::default()
        });
        assert!(boosted > base, "jump_mul 1.5 apex {boosted} must exceed base apex {base}");
    }

    #[test]
    fn gravity_mul_lengthens_airtime() {
        let w = flat_world();
        let airtime = |tuning: &Tuning| -> u32 {
            let mut s = spawn_state(Vec3::ZERO);

            while !s.grounded && n < 2000 {

            gravity_mul: 0.8,
            ..Tuning::default()
        });
        assert!(
            floaty > base,
            "gravity_mul 0.8 airtime {floaty} must exceed base airtime {base}"
        );
    }
