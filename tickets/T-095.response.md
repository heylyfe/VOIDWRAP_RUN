    // ---- T-095: dead-end patterns may terminate a plan -------------------

    /// Seeds swept by the dead-end tests.
    const DEAD_END_SWEEP: u64 = 64;

    /// Level / player count / tier used by the dead-end sweep. The level is high
    /// enough that every dead-end pattern in the test catalog is past its
    /// `min_level`.
    const SWEEP_LEVEL: u32 = 18;
    const SWEEP_PLAYERS: u8 = 4;

    /// Build the plan for `seed` against the catalog that carries dead-ends and
    /// return the engine alongside the selected chunk indices.
    fn dead_end_plan(seed: u64) -> (PatternEngine, Vec<usize>) {
        let engine = PatternEngine::new(dead_end_catalog());
        let plan = engine.compose(seed, SWEEP_LEVEL, SWEEP_PLAYERS, Tier::Standard);
        (engine, plan.chunks)
    }

    /// True when chunk `idx` is a dead-end pattern.
    fn is_dead_end(engine: &PatternEngine, idx: usize) -> bool {
        matches!(engine.metas[idx].role, ChunkRole::DeadEnd)
    }

    #[test]
    fn a_dead_end_can_terminate_a_plan() {
        let mut terminated = 0usize;
        for seed in 0..DEAD_END_SWEEP {
            let (engine, plan) = dead_end_plan(seed);
            assert!(!plan.is_empty(), "seed {seed} produced an empty plan");
            for (pos, &idx) in plan.iter().enumerate() {
                if is_dead_end(&engine, idx) {
                    assert_eq!(
                        pos,
                        plan.len() - 1,
                        "seed {seed}: dead-end at {pos} of {}",
                        plan.len()
                    );
                    terminated += 1;
                }
            }
        }
        assert!(
            terminated > 0,
            "no plan in the {DEAD_END_SWEEP}-seed sweep ended on a dead-end"
        );
    }

    #[test]
    fn a_dead_end_never_appears_mid_plan() {
        for seed in 0..DEAD_END_SWEEP {

            for (pos, &idx) in plan.iter().enumerate() {
                if is_dead_end(&engine, idx) {
                    assert_eq!(

        }
