import {
  BoxGeometry,
  CanvasTexture,
  RepeatWrapping,
  SRGBColorSpace,
} from "three";
import type { Theme } from "./theme";

/** Pixel size of the generated checker canvas (2x2 checks). */
const CANVAS_SIZE = 256;
/** One check = one world unit, so half the canvas on a 256px tile. */
const CHECK_SIZE = CANVAS_SIZE / 2;
/** Per-pixel brightness noise amplitude (+/-6%). */
const NOISE_AMPLITUDE = 0.06;
/** 2px darker groove along every check border. */
const GROOVE_WIDTH = 2;
const GROOVE_ALPHA = 0.25;
/** Faint marble veins (~10% opacity). */
const VEIN_ALPHA = 0.1;
const VEIN_MIN = 3;
const VEIN_MAX = 6;

const TOP_SEED_SUFFIX = ":top";
const SIDE_SEED_SUFFIX = ":side";

export interface ThemeTexturePair {
  top: CanvasTexture | null;
  side: CanvasTexture | null;
}

/**
 * A BoxGeometry whose per-face UVs are rescaled so ONE uv unit equals ONE
 * world unit (one check per world unit with the 2-check-per-tile texture),
 * and centred on the box so checks line up with the world grid for boxes
 * whose edges sit on integer coordinates.
 */
export function worldScaledBoxGeometry(
  width: number,
  height: number,
  depth: number,
): BoxGeometry {
  const geometry = new BoxGeometry(width, height, depth);
  const uv = geometry.attributes.uv;
  if (!uv) {
    throw new Error("BoxGeometry is missing its uv attribute");
  }
  for (let face = 0; face < 6; face++) {
    const [extentU, extentV] = faceExtents(face, width, height, depth);
    for (let corner = 0; corner < 4; corner++) {
      const index = face * 4 + corner;
      const u = uv.getX(index);
      const v = uv.getY(index);
      // Old uv is in [0, 1] across the face extent; remap to world units
      // centred on the box origin: [-extent / 2, +extent / 2].
      uv.setXY(index, u * extentU - extentU / 2, v * extentV - extentV / 2);
    }
  }
  uv.needsUpdate = true;
  return geometry;
}

/** u/v world extents per BoxGeometry face (order +x, -x, +y, -y, +z, -z). */
function faceExtents(
  face: number,
  width: number,
  height: number,
  depth: number,
): [number, number] {
  switch (face) {
    case 0: // +x: u spans depth, v spans height
    case 1: // -x: u spans depth, v spans height
      return [depth, height];
    case 2: // +y: u spans width, v spans depth
    case 3: // -y: u spans width, v spans depth
      return [width, depth];
    default: // +z / -z: u spans width, v spans height
      return [width, height];
  }
}

/**
 * Grungy 2x2 checker canvas texture (each check = one world unit; pair with
 * `texture.repeat = (0.5, 0.5)` and world-scaled UVs). Deterministic for a
 * given (light, dark, seed). Returns null when there is no DOM.
 */
export function makeCheckerTexture(
  light: string,
  dark: string,
  seed: number,
): CanvasTexture | null {
  if (typeof document === "undefined") {
    return null;
  }
  const canvas = document.createElement("canvas");
  canvas.width = CANVAS_SIZE;
  canvas.height = CANVAS_SIZE;
  const ctx = canvas.getContext("2d");
  if (!ctx) {
    return null;
  }
  const rand = mulberry32(seed);

  drawChecks(ctx, light, dark);
  drawGrooves(ctx);
  drawVeins(ctx, rand);
  applyNoise(ctx, rand);

  const texture = new CanvasTexture(canvas);
  texture.wrapS = RepeatWrapping;
  texture.wrapT = RepeatWrapping;
  texture.colorSpace = SRGBColorSpace;
  texture.anisotropy = 8;
  texture.generateMipmaps = true;
  texture.repeat.set(0.5, 0.5);
  texture.needsUpdate = true;
  return texture;
}

const textureCache = new Map<string, ThemeTexturePair>();

/** Top/side checker textures for a theme, memoised per theme.id. */
export function themeTextures(theme: Theme): ThemeTexturePair {
  const cached = textureCache.get(theme.id);
  if (cached) {
    return cached;
  }
  const pair: ThemeTexturePair = {
    top: makeCheckerTexture(
      theme.tile.light,
      theme.tile.dark,
      stringSeed(theme.id + TOP_SEED_SUFFIX),
    ),
    side: makeCheckerTexture(
      theme.tile.side,
      theme.tile.dark,
      stringSeed(theme.id + SIDE_SEED_SUFFIX),
    ),
  };
  textureCache.set(theme.id, pair);
  return pair;
}

function drawChecks(
  ctx: CanvasRenderingContext2D,
  light: string,
  dark: string,
): void {
  ctx.fillStyle = light;
  ctx.fillRect(0, 0, CANVAS_SIZE, CANVAS_SIZE);
  ctx.fillStyle = dark;
  ctx.fillRect(CHECK_SIZE, 0, CHECK_SIZE, CHECK_SIZE);
  ctx.fillRect(0, CHECK_SIZE, CHECK_SIZE, CHECK_SIZE);
}

/** 2px darker groove along every check border, wrap-safe. */
function drawGrooves(ctx: CanvasRenderingContext2D): void {
  ctx.fillStyle = `rgba(0, 0, 0, ${GROOVE_ALPHA})`;
  ctx.fillRect(0, 0, CANVAS_SIZE, GROOVE_WIDTH);
  ctx.fillRect(0, CHECK_SIZE - GROOVE_WIDTH / 2, CANVAS_SIZE, GROOVE_WIDTH);
  ctx.fillRect(0, 0, GROOVE_WIDTH, CANVAS_SIZE);
  ctx.fillRect(CHECK_SIZE - GROOVE_WIDTH / 2, 0, GROOVE_WIDTH, CANVAS_SIZE);
}

/** 3-6 faint thin bezier "marble" veins, wrapped so they tile seamlessly. */
function drawVeins(ctx: CanvasRenderingContext2D, rand: () => number): void {
  ctx.strokeStyle = `rgba(0, 0, 0, ${VEIN_ALPHA})`;
  const count = VEIN_MIN + Math.floor(rand() * (VEIN_MAX - VEIN_MIN + 1));
  for (let i = 0; i < count; i++) {
    ctx.lineWidth = 1 + rand() * 1.5;
    const horizontal = rand() < 0.5;
    const p0 = rand() * CANVAS_SIZE;
    const p1 = rand() * CANVAS_SIZE;
    const c1 = rand() * CANVAS_SIZE;
    const c2 = rand() * CANVAS_SIZE;
    const draw = (offset: number): void => {
      ctx.beginPath();
      if (horizontal) {
        ctx.moveTo(offset, p0);
        ctx.bezierCurveTo(
          offset + CANVAS_SIZE / 3,
          c1,
          offset + (2 * CANVAS_SIZE) / 3,
          c2,
          offset + CANVAS_SIZE,
          p1,
        );
      } else {
        ctx.moveTo(p0, offset);
        ctx.bezierCurveTo(
          c1,
          offset + CANVAS_SIZE / 3,
          c2,
          offset + (2 * CANVAS_SIZE) / 3,
          p1,
          offset + CANVAS_SIZE,
        );
      }
      ctx.stroke();
    };
    draw(-CANVAS_SIZE);
    draw(0);
    draw(CANVAS_SIZE);
  }
}

/** Deterministic per-pixel brightness noise of +/-6% over the whole tile. */
function applyNoise(ctx: CanvasRenderingContext2D, rand: () => number): void {
  const image = ctx.getImageData(0, 0, CANVAS_SIZE, CANVAS_SIZE);
  const data = image.data;
  for (let i = 0; i < data.length; i += 4) {
    const r = data[i];
    const g = data[i + 1];
    const b = data[i + 2];
    if (r === undefined || g === undefined || b === undefined) {
      continue; // Unreachable for a full-canvas ImageData.
    }
    const factor = 1 + (rand() * 2 - 1) * NOISE_AMPLITUDE;
    data[i] = scaleChannel(r, factor);
    data[i + 1] = scaleChannel(g, factor);
    data[i + 2] = scaleChannel(b, factor);
  }
  ctx.putImageData(image, 0, 0);
}

function scaleChannel(value: number, factor: number): number {
  return Math.max(0, Math.min(255, Math.round(value * factor)));
}

/** Small deterministic PRNG (mulberry32). */
function mulberry32(seed: number): () => number {
  let state = seed >>> 0;
  return () => {
    state = (state + 0x6d2b79f5) >>> 0;
    let t = state;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/** FNV-1a style stable string hash -> uint32, for per-theme texture seeds. */
function stringSeed(text: string): number {
  let hash = 2166136261;
  for (let i = 0; i < text.length; i++) {
    hash ^= text.charCodeAt(i);
    hash = Math.imul(hash, 16777619);
  }
  return hash >>> 0;
}

"use client";

import { useEffect, useMemo } from "react";
import type { JSX } from "react";
import { BoxGeometry, MeshStandardMaterial } from "three";
import type { Material } from "three";
import type { Platform } from "@/protocol/generated";
import { themeTextures, worldScaledBoxGeometry } from "./checker";
import type { Theme } from "./theme";

interface EdgeBarData {
  size: [number, number, number];
  position: [number, number, number];
}

interface PlatformBoxData {
  id: number;
  center: [number, number, number];
  size: [number, number, number];
  edgeBars: EdgeBarData[];
}

interface ThemeMaterials {
  /** One material per BoxGeometry face, in face order: +x, -x, +y, -y, +z, -z. */
  byFace: [Material, Material, Material, Material, Material, Material];
  /** Glowing perimeter trim material. */
  edgeTrim: Material;
  /** Every material created for the current theme, for disposal. */
  all: readonly Material[];
}

const EDGE_BAR_THICKNESS = 0.08;
const EDGE_BAR_HEIGHT = 0.08;
const EDGE_BAR_OVERHANG = 0.04;
const EDGE_BAR_EMISSIVE = 1.4;
const SIDE_TINT = "#9a9a9a";
const TOP_ROUGHNESS = 0.85;
const SIDE_ROUGHNESS = 0.9;
const BOTTOM_ROUGHNESS = 1;

// Four thin bars along the top perimeter. The bars along X straddle the
// top edge (overhang EDGE_BAR_OVERHANG on each side); the bars along Z
// span exactly the gap between them so nothing overlaps at the corners.
function toEdgeBars(size: readonly [number, number, number]): EdgeBarData[] {
  const [width, height, depth] = size;
  const halfWidth = width / 2;
  const halfDepth = depth / 2;
  const barY = height / 2 + EDGE_BAR_HEIGHT / 2;
  const longLength = width + EDGE_BAR_OVERHANG * 2;
  const shortLength = depth - EDGE_BAR_OVERHANG * 2;
  return [
    {
      size: [longLength, EDGE_BAR_HEIGHT, EDGE_BAR_THICKNESS],
      position: [0, barY, halfDepth],
    },
    {
      size: [longLength, EDGE_BAR_HEIGHT, EDGE_BAR_THICKNESS],
      position: [0, barY, -halfDepth],
    },
    {
      size: [EDGE_BAR_THICKNESS, EDGE_BAR_HEIGHT, shortLength],
      position: [halfWidth, barY, 0],
    },
    {
      size: [EDGE_BAR_THICKNESS, EDGE_BAR_HEIGHT, shortLength],
      position: [-halfWidth, barY, 0],
    },
  ];
}

function toBoxData(platform: Platform): PlatformBoxData {
  const { min, max } = platform.bounds;
  const size: [number, number, number] = [
    max.x - min.x,
    max.y - min.y,
    max.z - min.z,
  ];
  const center: [number, number, number] = [
    (min.x + max.x) / 2,
    (min.y + max.y) / 2,
    (min.z + max.z) / 2,
  ];
  return {
    id: platform.id,
    center,
    size,
    edgeBars: toEdgeBars(size),
  };
}

/** Materials shared by every platform of a theme (textures are module-cached). */
function createThemeMaterials(theme: Theme): ThemeMaterials {
  const textures = themeTextures(theme);
  const top = new MeshStandardMaterial(
    textures.top
      ? { map: textures.top, roughness: TOP_ROUGHNESS }
      : { color: theme.tile.light, roughness: TOP_ROUGHNESS },
  );
  const side = new MeshStandardMaterial(
    textures.side
      ? { map: textures.side, color: SIDE_TINT, roughness: SIDE_ROUGHNESS }
      : { color: theme.tile.side, roughness: SIDE_ROUGHNESS },
  );
  const bottom = new MeshStandardMaterial({
    color: theme.tile.dark,
    roughness: BOTTOM_ROUGHNESS,
  });
  const edgeTrim = new MeshStandardMaterial({
    color: theme.tile.edge,
    emissive: theme.tile.edge,
    emissiveIntensity: EDGE_BAR_EMISSIVE,
  });
  return {
    // BoxGeometry face order: +x, -x, +y, -y, +z, -z.
    byFace: [side, side, top, bottom, side, side],
    edgeTrim,
    all: [top, side, bottom, edgeTrim],
  };
}

function disposeMaterials(materials: ThemeMaterials): void {
  for (const material of materials.all) {
    material.dispose();
  }
}

function PlatformMesh(props: {
  box: PlatformBoxData;
  materials: ThemeMaterials;
}): JSX.Element {
  const { box, materials } = props;
  // World-scaled UVs are memoised per platform box; disposed on change/unmount.
  const geometry = useMemo(
    () => worldScaledBoxGeometry(box.size[0], box.size[1], box.size[2]),
    [box.size],
  );
  useEffect(() => () => geometry.dispose(), [geometry]);
  return (
    <group position={box.center}>
      <mesh geometry={geometry} material={materials.byFace} />
      {box.edgeBars.map((bar, index) => (
        <EdgeTrimBar key={index} bar={bar} material={materials.edgeTrim} />
      ))}
    </group>
  );
}

function EdgeTrimBar(props: { bar: EdgeBarData; material: Material }): JSX.Element {
  const [x, y, z] = props.bar.size;
  // World-scaled UVs do not matter for the flat trim bars, so a plain box does.
  const geometry = useMemo(() => new BoxGeometry(x, y, z), [x, y, z]);
  useEffect(() => () => geometry.dispose(), [geometry]);
  return (
    <mesh geometry={geometry} position={props.bar.position} material={props.material} />
  );
}

export function Platforms(props: {
  platforms: readonly Platform[];
  theme: Theme;
}): JSX.Element {
  const boxes = useMemo(() => props.platforms.map(toBoxData), [props.platforms]);
  // Materials are shared between all platforms and memoised per theme.
  const materials = useMemo(() => createThemeMaterials(props.theme), [props.theme]);
  useEffect(() => () => disposeMaterials(materials), [materials]);

  return (
    <group>
      {boxes.map((box) => (
        <PlatformMesh key={box.id} box={box} materials={materials} />
      ))}
    </group>
  );
}

import { BoxGeometry } from "three";
import { describe, expect, it } from "vitest";
import { makeCheckerTexture, themeTextures, worldScaledBoxGeometry } from "./checker";
import { themeForLevel } from "./theme";

interface UvRange {
  minU: number;
  maxU: number;
  spanU: number;
  spanV: number;
}

function faceUvRange(geometry: BoxGeometry, face: number): UvRange {
  const uv = geometry.attributes.uv;
  if (!uv) {
    throw new Error("expected BoxGeometry to have a uv attribute");
  }
  let minU = Number.POSITIVE_INFINITY;
  let maxU = Number.NEGATIVE_INFINITY;
  let minV = Number.POSITIVE_INFINITY;
  let maxV = Number.NEGATIVE_INFINITY;
  for (let i = face * 4; i < face * 4 + 4; i++) {
    const u = uv.getX(i);
    const v = uv.getY(i);
    minU = Math.min(minU, u);
    maxU = Math.max(maxU, u);
    minV = Math.min(minV, v);
    maxV = Math.max(maxV, v);
  }
  return { minU, maxU, spanU: maxU - minU, spanV: maxV - minV };
}

describe("worldScaledBoxGeometry", () => {
  it("maps one uv unit to one world unit on every face", () => {
    const geometry = worldScaledBoxGeometry(4, 1, 2);

    // +y face (index 2): u spans width (4), v spans depth (2).
    const top = faceUvRange(geometry, 2);
    expect(top.spanU).toBeCloseTo(4);
    expect(top.spanV).toBeCloseTo(2);

    // +x face (index 0): u spans depth (2), v spans height (1).
    const side = faceUvRange(geometry, 0);
    expect(side.spanU).toBeCloseTo(2);
    expect(side.spanV).toBeCloseTo(1);

    geometry.dispose();
  });

  it("centres each face's uvs on the box so checks align to the world grid", () => {
    const geometry = worldScaledBoxGeometry(4, 1, 2);

    // +y face: u should range from -2 to +2 (half its width extent on each side).
    const top = faceUvRange(geometry, 2);
    expect(top.minU).toBeCloseTo(-2);
    expect(top.maxU).toBeCloseTo(2);

    // +x face: u should range from -1 to +1 (half its depth extent on each side).
    const side = faceUvRange(geometry, 0);
    expect(side.minU).toBeCloseTo(-1);
    expect(side.maxU).toBeCloseTo(1);

    geometry.dispose();
  });
});

describe("makeCheckerTexture", () => {
  it("returns null in the node test environment (no DOM)", () => {
    expect(makeCheckerTexture("#ffffff", "#000000", 1)).toBeNull();
  });
});

describe("themeTextures", () => {
  it("returns the same memoised object for the same theme id", () => {
    const theme = themeForLevel(1);
    const first = themeTextures(theme);
    const second = themeTextures(theme);
    expect(second).toBe(first);
  });

  it("returns null textures without a DOM", () => {
    const pair = themeTextures(themeForLevel(1));
    expect(pair.top).toBeNull();
    expect(pair.side).toBeNull();
  });
});

=== NOTES ===
- `worldScaledBoxGeometry` remaps each face's four UVs from [0,1] to (u - 0.5) * extent, so uv coordinates equal local world coordinates; with `texture.repeat = (0.5, 0.5)` and the 2x2-check tile, exactly one check is one world unit, and boxes with integer edges align to the world grid. Face extents per the documented order (+x/-x: depth×height, +y/-y: width×depth, +z/-z: width×height).
- `makeCheckerTexture` is fully deterministic: all randomness (tile noise, vein count/shape/width) comes from one mulberry32 stream whose call order is fixed; PRNG call counts are branch-independent. Grunge = per-pixel ±6% noise via ImageData + 3-6 thin ~10%-opacity black bezier veins (drawn at -SIZE/0/+SIZE offsets so they wrap seamlessly) + 2px darker grooves on every check border. Returns null when `typeof document === "undefined"`.
- `themeTextures` uses a module-level Map keyed by `theme.id`; seeds derived from a stable FNV-1a hash of `themeId + ":top"` / `":side"`.
- Platforms: materials created once per `theme` via `useMemo` and shared across all platforms (one material set total, not per platform); the mesh uses a 6-entry array in face order [side, side, top, bottom, side, side]. Textures are module-cached and shared, so material `dispose()` never disposes them (and r3f owns no texture lifetime here). Per-platform box geometry and per-edge-bar trim geometry are `useMemo`-cached with cleanup effects that `dispose()` them on change/unmount. No `useFrame`, no refs during render, no setState in effects — fully static scene graph, so no per-frame work or allocations.
- Textured side uses `color: "#9a9a9a"` (multiplies the map, dimming it); texture-less fallback (tests/SSR) uses plain `tile.light` / `tile.side`. Edge trim bars unchanged geometrically, now `theme.tile.edge` with emissiveIntensity 1.4 (3.0 emissive tint omitted per spec; bloom-ready).
- Removed the old `PALETTE`/`platformColor` imports and per-platform tinting; `palette.ts` untouched.
- Tests cover: +y face spans 4×2, +x face spans 2×1 uv-world units; uv centring; null texture in node; `themeTextures` identity per theme id. All files avoid `any` and non-null assertions (indexed reads of `ImageData.data` are explicitly guarded instead).
