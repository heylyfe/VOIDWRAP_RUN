#!/usr/bin/env node
/**
 * tools/playthrough/run.mjs — end-to-end playthrough sweep.
 *
 * Connects N headless clients to a running server, drives them through a run
 * and prints one `PASS/SKIP/FAIL - name (detail)` line per check.
 *
 * Navigation: the bot used to random-walk, so it never collected the gifts a
 * level requires and the run stalled at level 1. It now keeps the `Level`
 * payload (spawn / beacon / kill_y / platforms / gifts) and the player position
 * from `Snapshot.you.pos`, and steers to the nearest uncollected gift. The
 * random walk is kept as the fallback for servers that never send a `Level`.
 */

function parseArgs(argv) {
  const opts = { ...DEFAULTS };
  for (let i = 0; i < argv.length; i += 1) {
    const arg = argv[i];
    if (arg === '--url') {
      i += 1;
      opts.url = argv[i];
    } else if (arg === '--clients') {
      i += 1;
      opts.clients = Number(argv[i]);
    } else if (arg === '--seconds') {
      i += 1;
      opts.seconds = Number(argv[i]);
    } else if (arg === '--help' || arg === '-h') {
      console.log(usage());
      process.exit(0);
    } else {
      throw new Error(`unknown argument: ${arg}\n${usage()}`);
    }

    throw new Error('--clients must be a positive integer');
  }

// Wire codec (msgpack)

const TEXT_DECODER = new TextDecoder();
const TEXT_ENCODER = new TextEncoder();

function decodeMsgpack(bytes) {
  const view = new DataView(bytes.buffer, bytes.byteOffset, bytes.byteLength);
  let off = 0;

  function need(n) {
    if (off + n > bytes.length) throw new Error('truncated msgpack frame');
  }

  function readStr(len) {
    need(len);
    const s = TEXT_DECODER.decode(bytes.subarray(off, off + len));
    off += len;
    return s;
  }

  function readBin(len) {
    need(len);
    const b = bytes.subarray(off, off + len);
    off += len;
    return b;
  }

  function readArray(len) {
    const out = new Array(len);
    for (let i = 0; i < len; i += 1) out[i] = read();
    return out;
  }

  function readMap(len) {
    const out = {};
    for (let i = 0; i < len; i += 1) {
      const key = read();
      out[typeof key === 'string' ? key : String(key)] = read();
    }
    return out;
  }

  function read() {
    need(1);
    const b = view.getUint8(off);
    off += 1;

    if (b <= 0x7f) return b;
    if (b >= 0xe0) return b - 0x100;
    if ((b & 0xf0) === 0x80) return readMap(b & 0x0f);
    if ((b & 0xf0) === 0x90) return readArray(b & 0x0f);
    if ((b & 0xe0) === 0xa0) return readStr(b & 0x1f);

    switch (b) {
      case 0xc0: return null;
      case 0xc2: return false;
      case 0xc3: return true;
      case 0xc4: { need(1); const n = view.getUint8(off); off += 1; return readBin(n); }
      case 0xc5: { need(2); const n = view.getUint16(off); off += 2; return readBin(n); }
      case 0xc6: { need(4); const n = view.getUint32(off); off += 4; return readBin(n); }
      case 0xca: { need(4); const v = view.getFloat32(off); off += 4; return v; }
      case 0xcb: { need(8); const v = view.getFloat64(off); off += 8; return v; }
      case 0xcc: { need(1); const v = view.getUint8(off); off += 1; return v; }
      case 0xcd: { need(2); const v = view.getUint16(off); off += 2; return v; }
      case 0xce: { need(4); const v = view.getUint32(off); off += 4; return v; }
      case 0xcf: { need(8); const v = Number(view.getBigUint64(off)); off += 8; return v; }
      case 0xd0: { need(1); const v = view.getInt8(off); off += 1; return v; }
      case 0xd1: { need(2); const v = view.getInt16(off); off += 2; return v; }
      case 0xd2: { need(4); const v = view.getInt32(off); off += 4; return v; }
      case 0xd3: { need(8); const v = Number(view.getBigInt64(off)); off += 8; return v; }
      case 0xd9: { need(1); const n = view.getUint8(off); off += 1; return readStr(n); }
      case 0xda: { need(2); const n = view.getUint16(off); off += 2; return readStr(n); }
      case 0xdb: { need(4); const n = view.getUint32(off); off += 4; return readStr(n); }
      case 0xdc: { need(2); const n = view.getUint16(off); off += 2; return readArray(n); }
      case 0xdd: { need(4); const n = view.getUint32(off); off += 4; return readArray(n); }
      case 0xde: { need(2); const n = view.getUint16(off); off += 2; return readMap(n); }
      case 0xdf: { need(4); const n = view.getUint32(off); off += 4; return readMap(n); }
      default: throw new Error(`unsupported msgpack byte 0x${b.toString(16)}`);
    }
  }

  return read();
}

function encodeMsgpack(value) {
  const out = [];

  function push(...bytes) {
    for (const b of bytes) out.push(b & 0xff);
  }

  function write(v) {
    if (v === null || v === undefined) { push(0xc0); return; }
    if (v === true) { push(0xc3); return; }
    if (v === false) { push(0xc2); return; }

    if (typeof v === 'number') {
      if (Number.isInteger(v)) {
        if (v >= 0 && v <= 0x7f) { push(v); return; }
        if (v < 0 && v >= -32) { push(v); return; }
        if (v >= 0 && v <= 0xff) { push(0xcc, v); return; }
        if (v >= 0 && v <= 0xffff) { push(0xcd, v >>> 8, v); return; }
        if (v >= 0 && v <= 0xffffffff) { push(0xce, v >>> 24, v >>> 16, v >>> 8, v); return; }
        if (v < 0 && v >= -128) { push(0xd0, v); return; }
        if (v < 0 && v >= -32768) { push(0xd1, v >> 8, v); return; }
        push(0xd2, v >>> 24, v >>> 16, v >>> 8, v);
        return;
      }
      const buf = new ArrayBuffer(8);
      new DataView(buf).setFloat64(0, v);
      push(0xcb, ...new Uint8Array(buf));
      return;
    }

    if (typeof v === 'string') {
      const bytes = TEXT_ENCODER.encode(v);
      if (bytes.length <= 31) push(0xa0 | bytes.length);
      else if (bytes.length <= 0xff) push(0xd9, bytes.length);
      else push(0xda, bytes.length >>> 8, bytes.length);
      for (const b of bytes) push(b);
      return;
    }

    if (Array.isArray(v)) {
      if (v.length <= 15) push(0x90 | v.length);
      else push(0xdc, v.length >>> 8, v.length);
      for (const item of v) write(item);
      return;
    }

    if (typeof v === 'object') {
      const keys = Object.keys(v);
      if (keys.length <= 15) push(0x80 | keys.length);
      else push(0xde, keys.length >>> 8, keys.length);
      for (const k of keys) { write(k); write(v[k]); }
      return;
    }

    throw new Error(`cannot encode value of type ${typeof v}`);
  }

  write(value);
  return Uint8Array.from(out);
}

/**
 * Normalise a decoded frame to `[variantName, body]`.
 *
 * Handles the three serde enum taggings so the sweep does not care which one
 * the protocol crate uses: externally tagged `{ Level: { data } }`, internally
 * tagged `{ t: 'Level', data }`, and adjacently tagged `['Level', { data }]`.
 */
function variant(msg) {
  if (Array.isArray(msg) && msg.length === 2 && typeof msg[0] === 'string') {
    return [msg[0], msg[1]];
  }
  if (msg && typeof msg === 'object') {
    if (typeof msg.t === 'string') return [msg.t, msg];
    if (typeof msg.type === 'string') return [msg.type, msg];
    const keys = Object.keys(msg);
    if (keys.length === 1) return [keys[0], msg[keys[0]]];
  }
  return [null, msg];
}

/** A gift is considered collected once the player has been this close to it. */
const GIFT_PICKUP_RADIUS = 1.5;
/** Jump when the target sits more than this far above the player. */
const JUMP_HEIGHT_DELTA = 0.5;
/** Floor probes ahead of the player, in world units. */
const GAP_PROBE_NEAR = 1.0;
const GAP_PROBE_FAR = 3.0;
/** Minimum horizontal gap width that is worth a jump. */
const GAP_MIN_WIDTH = 2.0;
/** How long the fallback random walk keeps one heading, in ms. */
const WALK_HEADING_MS = 400;

function clampInt(value, lo, hi) {
  if (value < lo) return lo;
  if (value > hi) return hi;
  return value;
}

function dist2(a, b) {
  const dx = a.x - b.x;
  const dy = a.y - b.y;
  const dz = a.z - b.z;
  return dx * dx + dy * dy + dz * dz;
}

/** True when some platform provides a floor at (x, z) near the player's feet. */
function hasFloorAt(level, x, y, z) {
  const platforms = level.platforms;
  if (!Array.isArray(platforms)) return false;
  for (const platform of platforms) {
    const bounds = platform && platform.bounds;
    if (!bounds || !bounds.min || !bounds.max) continue;
    if (x < bounds.min.x || x > bounds.max.x) continue;
    if (z < bounds.min.z || z > bounds.max.z) continue;
    // The platform top has to be at, or just below, the player's feet.
    if (bounds.max.y <= y + 0.5 && bounds.max.y >= y - 1.5) return true;
  }
  return false;
}

/**
 * True when the floor is missing across the whole probe span ahead, i.e. the
 * gap in front of the player is at least GAP_MIN_WIDTH wide.
 */
function gapAhead(level, pos, dir) {
  if (!level || !Array.isArray(level.platforms) || level.platforms.length === 0) return false;
  if (GAP_PROBE_FAR - GAP_PROBE_NEAR < GAP_MIN_WIDTH) return false;
  const probes = [GAP_PROBE_NEAR, (GAP_PROBE_NEAR + GAP_PROBE_FAR) / 2, GAP_PROBE_FAR];
  for (const ahead of probes) {
    if (hasFloorAt(level, pos.x + dir.x * ahead, pos.y, pos.z + dir.z * ahead)) return false;
  }
  return true;
}

/** Nearest gift that has not been collected yet, or null. */
function nearestGift(level, pos, taken) {
  const gifts = level.gifts;
  if (!Array.isArray(gifts)) return null;
  let best = null;
  let bestD2 = Infinity;
  for (const gift of gifts) {
    if (!gift || !gift.pos) continue;
    if (taken.has(gift.id)) continue;
    const d2 = dist2(gift.pos, pos);
    if (d2 < bestD2) {
      bestD2 = d2;
      best = gift;
    }
  }
  return best;
}

/** Mark every gift the player is standing on as collected. */
function markCollected(level, pos, taken) {
  const gifts = level.gifts;
  if (!Array.isArray(gifts)) return;
  const radius2 = GIFT_PICKUP_RADIUS * GIFT_PICKUP_RADIUS;
  for (const gift of gifts) {
    if (!gift || !gift.pos) continue;
    if (taken.has(gift.id)) continue;
    if (dist2(gift.pos, pos) <= radius2) taken.add(gift.id);
  }
}

/**
 * Convert a world direction to the target into an `Inputs` payload.
 * Returns null when there is nothing left to collect.
 */
function steerToNearestGift(level, pos, taken, grounded) {
  const target = nearestGift(level, pos, taken);
  if (!target) return null;

  let dx = target.pos.x - pos.x;
  let dz = target.pos.z - pos.z;
  const len = Math.hypot(dx, dz);

  let moveX = 0;
  let moveZ = 0;
  if (len > 1e-6) {
    dx /= len;
    dz /= len;
    moveX = clampInt(Math.round(dx * 127), -127, 127);
    moveZ = clampInt(Math.round(dz * 127), -127, 127);
  }

  let buttons = 0;
  const targetAbove = target.pos.y - pos.y > JUMP_HEIGHT_DELTA;
  if (targetAbove && grounded) buttons |= 1;
  if (len > 1e-6 && gapAhead(level, pos, { x: dx, z: dz })) buttons |= 1;

  return { move_x: moveX, move_z: moveZ, buttons };
}

const stats = {
  connected: 0,
  welcome: 0,
  runstate: 0,
  draft: 0,
  voteOpen: 0,
  shop: 0,
  error: 0,
  undecodable: 0,
  classRejected: 0,
  maxLevel: 0,
  phases: new Set(),
};

const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

// Client

class Client {
  constructor(index, url) {
    this.index = index;
    this.url = url;
    this.ws = null;
    this.seq = 0;
    this.level = null;
    this.pos = null;
    this.grounded = true;
    this.taken = new Set();
    this.walkDir = { x: 1, z: 0 };
    this.walkUntil = 0;
    this.classSelected = false;
    this.classRejected = false;
  }

  connect() {
    return new Promise((resolve, reject) => {
      const ws = new WebSocket(this.url);
      ws.binaryType = 'arraybuffer';
      this.ws = ws;

      ws.addEventListener('open', () => {
        stats.connected += 1;
        resolve();
      });
      ws.addEventListener('error', () => {
        reject(new Error(`client ${this.index}: socket error`));
      });
      ws.addEventListener('message', (event) => this.onMessage(event.data));
      ws.addEventListener('close', () => {
        this.ws = null;
      });
    });
  }

  close() {
    if (!this.ws) return;
    try {
      this.ws.close();
    } catch {
      // already closing
    }
    this.ws = null;
  }

  send(value) {
    if (!this.ws || this.ws.readyState !== 1) return;
    this.ws.send(encodeMsgpack(value));
  }

  onMessage(data) {
    let msg;
    try {
      if (typeof data === 'string') {
        msg = JSON.parse(data);
      } else {
        const bytes = data instanceof ArrayBuffer
          ? new Uint8Array(data)
          : new Uint8Array(data.buffer, data.byteOffset, data.byteLength);
        msg = decodeMsgpack(bytes);
      }
    } catch {
      stats.undecodable += 1;
      return;
    }
    this.react(msg);
  }

  react(msg) {
    const [name, body] = variant(msg);
    switch (name) {
      case 'Welcome':
        stats.welcome += 1;
        this.selectClass();
        break;
      case 'RunState':
        stats.runstate += 1;
        this.observeRunState(body);
        break;
      case 'Level':
        this.level = body && body.data ? body.data : body;
        this.taken.clear();
        break;
      case 'Snapshot':
        this.onSnapshot(body);
        break;
      case 'Draft':
        stats.draft += 1;
        this.autoDraft(body);
        break;
      case 'VoteOpen':
        stats.voteOpen += 1;
        this.autoVote(body);
        break;
      case 'Shop':
        stats.shop += 1;
        this.autoShop(body);
        break;
      case 'Error':
        stats.error += 1;
        if (this.classSelected && !this.classRejected) {
          this.classRejected = true;
          stats.classRejected += 1;
        }
        break;
      default:
        break;
    }
  }

  observeRunState(body) {
    if (!body || typeof body !== 'object') return;
    if (typeof body.level === 'number' && body.level > stats.maxLevel) {
      stats.maxLevel = body.level;
    }
    if (typeof body.phase === 'number') {
      stats.phases.add(body.phase);
    }
  }

  onSnapshot(body) {
    const you = body && body.you;
    if (!you || !you.pos) return;

    this.pos = { x: you.pos.x, y: you.pos.y, z: you.pos.z };
    this.grounded = typeof you.grounded === 'boolean' ? you.grounded : true;

    let inputs = null;
    if (this.level) {
      markCollected(this.level, this.pos, this.taken);
      inputs = steerToNearestGift(this.level, this.pos, this.taken, this.grounded);
    }
    if (!inputs) inputs = this.randomWalk(Date.now());

    this.send({ Inputs: { seq: this.seq, move_x: inputs.move_x, move_z: inputs.move_z, buttons: inputs.buttons } });
    this.seq += 1;
  }

  /** Fallback used before a `Level` arrives, or once every gift is collected. */
  randomWalk(now) {
    if (now >= this.walkUntil) {
      const angle = Math.random() * Math.PI * 2;
      this.walkDir = { x: Math.cos(angle), z: Math.sin(angle) };
      this.walkUntil = now + WALK_HEADING_MS;
    }
    return {
      move_x: clampInt(Math.round(this.walkDir.x * 127), -127, 127),
      move_z: clampInt(Math.round(this.walkDir.z * 127), -127, 127),
      buttons: Math.random() < 0.05 ? 1 : 0,
    };
  }

  selectClass() {
    this.classSelected = true;
    this.send({ SelectClass: { class: this.index % 4 } });
  }

  autoDraft(body) {
    const options = body && Array.isArray(body.options) ? body.options : [];
    const first = options.length > 0 ? options[0] : null;
    const index = first && typeof first === 'object' && typeof first.id === 'number' ? first.id : 0;
    this.send({ DraftPick: { index } });
  }

  autoVote(body) {
    const options = body && Array.isArray(body.options) ? body.options : [];
    const first = options.length > 0 ? options[0] : null;
    const target = first && typeof first === 'object' && typeof first.id === 'number' ? first.id : 0;
    this.send({ Vote: { target } });
  }

  autoShop() {
    this.send({ ShopClose: {} });
  }
}

function printReport(opts) {
  const total = 11;
  let passed = 0;
  let skipped = 0;
  const lines = [];

  function emit(name, status, detail) {
    if (status === 'PASS') passed += 1;
    if (status === 'SKIP') skipped += 1;
    lines.push(`${status} - ${name} (${detail})`);
  }

  emit(
    'connect',
    stats.connected === opts.clients ? 'PASS' : 'FAIL',
    `${stats.connected}/${opts.clients} clients connected`,
  );

  emit('welcome', stats.welcome > 0 ? 'PASS' : 'FAIL', `Welcome frames: ${stats.welcome}`);
  emit('runstate', stats.runstate > 0 ? 'PASS' : 'FAIL', `RunState frames: ${stats.runstate}`);

  const levelOk = stats.maxLevel >= 2;
  emit('level-progression', levelOk ? 'PASS' : 'FAIL', `highest level reached: ${stats.maxLevel}`);

  emit('draft', stats.draft > 0 ? 'PASS' : 'SKIP', `Draft frames: ${stats.draft}`);
  emit('vote', stats.voteOpen > 0 ? 'PASS' : 'SKIP', `VoteOpen frames: ${stats.voteOpen}`);
  emit('shop', stats.shop > 0 ? 'PASS' : 'SKIP', `Shop frames: ${stats.shop}`);

  const phases = [...stats.phases].sort((a, b) => a - b);
  emit(
    'phase-coverage',
    phases.length >= 2 ? 'PASS' : 'FAIL',
    `distinct phases: ${phases.length} [${phases.join(',')}]`,
  );

  emit('decode', stats.undecodable === 0 ? 'PASS' : 'FAIL', `${stats.undecodable} undecodable frames`);
  emit(
    'class-select',
    stats.classRejected === 0 ? 'PASS' : 'FAIL',
    `${stats.classRejected}/${opts.clients} clients had their class rejected`,
  );
  emit('error-frames', stats.error === 0 ? 'PASS' : 'FAIL', `${stats.error} unexpected Error frames`);

  for (const line of lines) console.log(line);

  const reason = levelOk ? '' : ' (level never completed)';
  console.log(`playthrough: ${passed}/${total} checks passed, ${skipped} skipped${reason}`);

  return passed === total && skipped === 0;
}

async function main() {
  const opts = parseArgs(process.argv.slice(2));
  const clients = [];
  for (let i = 0; i < opts.clients; i += 1) clients.push(new Client(i, opts.url));

  const started = Date.now();
  const results = await Promise.allSettled(clients.map((client) => client.connect()));
  for (const result of results) {
    if (result.status === 'rejected') console.error(`playthrough: ${result.reason.message}`);
  }

  await sleep(opts.seconds * 1000);

  for (const client of clients) client.close();

  const elapsed = ((Date.now() - started) / 1000).toFixed(1);
  const ok = printReport(opts);
  console.error(`playthrough: sweep finished in ${elapsed}s`);

  process.exitCode = ok ? 0 : 1;
}

main().catch((error) => {
  console.error(`playthrough: ${error.message}`);
  process.exitCode = 1;
