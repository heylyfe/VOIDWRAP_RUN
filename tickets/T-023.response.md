  position: [number, number, number];
  scale: number;
  spin: [number, number, number];
  orbitSpeed: number;
  bobPhase: number;

const MIN_SCALE = 0.15;
const MAX_SCALE = 0.9;
const MIN_SPIN = -1.2;
const MAX_SPIN = 1.2;
const MIN_ORBIT = -0.03;
const MAX_ORBIT = 0.03;
const BOB_AMPLITUDE = 0.6;
const BOB_SPEED = 0.8;
const CLUSTER_FRACTION = 0.3;
const CLUSTER_SPREAD = 2.0;

/** Seed of the single shared rock silhouette; every Debris instance reuses it. */
const ROCK_JITTER_SEED = 0x5eed;
const ROCK_MIN_FACTOR = 0.7;
const ROCK_MAX_FACTOR = 1.15;
/** Positions are quantised before hashing so duplicated vertices agree. */
const ROCK_QUANTISATION = 1000;

/** Per-instance axis stretch so no two rocks share the same silhouette. */
const MIN_AXIS_SCALE = 0.75;
const MAX_AXIS_SCALE = 1.25;

const RIM_STRENGTH = 0.6;

const RIM_FRAGMENT_PARS = `
uniform vec3 uRimColor;
uniform float uRimStrength;
`;

const RIM_FRAGMENT_APPLY = `
  {
    vec3 rimViewDirection = normalize( vViewPosition );
    float rimFresnel = pow( 1.0 - clamp( dot( normal, rimViewDirection ), 0.0, 1.0 ), 3.0 );
    outgoingLight += uRimColor * rimFresnel * uRimStrength;
  }
`;

function clamp(value: number, min: number, max: number): number {
  return Math.min(Math.max(value, min), max);
}

/** Deterministic float in [0, 1) derived from a string key. */
function unitHash(key: string): number {
  return hashString(key) / 4294967296;
}

/**
 * Deterministic debris layout: a thick shell of cubes around `center`, with
 * roughly 30% of them grouped into small swarms.
 */
export function debrisLayout(
  seed: number,
  count: number,
  center: { x: number; y: number; z: number },
  radius: number,
): DebrisItem[] {
  const rand = mulberry32(seed);
  const total = Math.max(0, Math.floor(count));
  const items: DebrisItem[] = [];
  if (total === 0) return items;

  const innerRadius = radius * 0.55;
  const outerRadius = radius * 1.15;
  const minY = center.y - 12;
  const maxY = center.y + 18;

  const makeItem = (x: number, y: number, z: number): DebrisItem => {
    const scaleBias = rand();
    return {
      position: [x, y, z],
      scale: MIN_SCALE + scaleBias * scaleBias * (MAX_SCALE - MIN_SCALE),
      spin: [
        MIN_SPIN + rand() * (MAX_SPIN - MIN_SPIN),
        MIN_SPIN + rand() * (MAX_SPIN - MIN_SPIN),
        MIN_SPIN + rand() * (MAX_SPIN - MIN_SPIN),
      ],
      orbitSpeed: MIN_ORBIT + rand() * (MAX_ORBIT - MIN_ORBIT),
      bobPhase: rand() * Math.PI * 2,
    };
  };

  const shellPoint = (): [number, number, number] => {
    const angle = rand() * Math.PI * 2;
    const distance = innerRadius + rand() * (outerRadius - innerRadius);
    const y = minY + rand() * (maxY - minY);
    return [center.x + Math.cos(angle) * distance, y, center.z + Math.sin(angle) * distance];
  };

  const clusterPoint = (anchor: [number, number, number]): [number, number, number] => {
    const offsetX = anchor[0] - center.x;
    const offsetZ = anchor[2] - center.z;
    const baseAngle = Math.atan2(offsetZ, offsetX);
    const baseDistance = Math.hypot(offsetX, offsetZ);
    const tangential = (rand() - 0.5) * CLUSTER_SPREAD * 2;
    const angle = baseAngle + tangential / Math.max(baseDistance, 1);
    const distance = clamp(
      baseDistance + (rand() - 0.5) * CLUSTER_SPREAD * 2,
      innerRadius,
      outerRadius,
    );
    const y = clamp(anchor[1] + (rand() - 0.5) * CLUSTER_SPREAD * 2, minY, maxY);
    return [center.x + Math.cos(angle) * distance, y, center.z + Math.sin(angle) * distance];
  };

  const clusterTarget = Math.round(total * CLUSTER_FRACTION);
  while (items.length < clusterTarget) {
    const anchor = shellPoint();
    const clusterSize = 3 + Math.floor(rand() * 4);
    const remaining = Math.min(clusterSize, clusterTarget - items.length, total - items.length);
    for (let i = 0; i < remaining; i += 1) {
      const point = i === 0 ? anchor : clusterPoint(anchor);
      items.push(makeItem(point[0], point[1], point[2]));
    }
  }

  while (items.length < total) {
    const point = shellPoint();
    items.push(makeItem(point[0], point[1], point[2]));
  }

  return items;
}

/**
 * One shared jagged rock: an icosahedron whose vertices are pushed in and out
 * by a per-position hash. `IcosahedronGeometry` is non-indexed, so the same
 * corner appears once per adjacent face; hashing the quantised position (not
 * the vertex index) gives every copy the same factor, which keeps the faces
 * welded and the silhouette crack-free.
 */
function buildRockGeometry(): BufferGeometry {
  const geometry = new IcosahedronGeometry(1, 0);
  const attribute = geometry.getAttribute("position");
  if (!(attribute instanceof BufferAttribute)) {
    throw new Error("IcosahedronGeometry is missing a position attribute");
  }

  const factors = new Map<string, number>();
  for (let i = 0; i < attribute.count; i += 1) {
    const x = attribute.getX(i);
    const y = attribute.getY(i);
    const z = attribute.getZ(i);
    const key = [
      Math.round(x * ROCK_QUANTISATION),
      Math.round(y * ROCK_QUANTISATION),
      Math.round(z * ROCK_QUANTISATION),
    ].join(":");

    let factor = factors.get(key);
    if (factor === undefined) {
      factor =
        ROCK_MIN_FACTOR +
        unitHash(`rock:${ROCK_JITTER_SEED}:${key}`) * (ROCK_MAX_FACTOR - ROCK_MIN_FACTOR);
      factors.set(key, factor);
    }

    attribute.setXYZ(i, x * factor, y * factor, z * factor);
  }

  attribute.needsUpdate = true;
  geometry.computeVertexNormals();
  return geometry;
}

"use client";

import { useEffect, useLayoutEffect, useMemo, useRef } from "react";
import type { JSX } from "react";
import { Color, MeshStandardMaterial, Object3D, OctahedronGeometry } from "three";
import type { InstancedMesh } from "three";
import type { Platform } from "@/protocol/generated";
import type { Theme } from "./theme";
import { mulberry32 } from "./rng";

export interface CrystalSpot {
  position: [number, number, number];
  scale: number;
  tilt: [number, number];
  yaw: number;
}

const MIN_CLUSTER_SIZE = 3;
const MAX_CLUSTER_SIZE = 7;
const MIN_CRYSTAL_SCALE = 0.25;
const MAX_CRYSTAL_SCALE = 1.1;
const MAX_TILT = 0.5;

/** Hanging clusters: anchored below the underside, staggered further down. */
const HANGING_DROP = 0.18;
const HANGING_STAGGER = 0.24;
const HANGING_SPREAD = 0.3;
const HANGING_INSET = 0.45;

/** Standing clusters: pushed just past a corner, never back over the top. */
const STANDING_OFFSET = 0.4;
const STANDING_SPREAD = 0.18;
const STANDING_VERTICAL_SPREAD = 0.2;
const STANDING_LIFT = 0.12;
const STANDING_HEADROOM = 0.12;

const PLACEMENT_ATTEMPTS = 8;
const DEFAULT_PER_PLATFORM = 2;

const CRYSTAL_WIDTH = 0.35;
const CRYSTAL_HEIGHT = 1.4;
const EMISSIVE_INTENSITY = 1.6;
const CRYSTAL_ROUGHNESS = 0.3;
/** Slack used when deciding whether a spot sits under a platform. */
const HANGING_FOOTPRINT_MARGIN = 0.3;

function insideFootprint(x: number, z: number, platform: Platform): boolean {
  const { min, max } = platform.bounds;
  return x >= min.x && x <= max.x && z >= min.z && z <= max.z;
}

/**
 * A position is blocked when it sits inside a platform footprint at or above
 * that platform's underside - that is, on the walkable top or inside the block.
 */
function isBlocked(
  position: readonly [number, number, number],
  platforms: readonly Platform[],
): boolean {
  for (const platform of platforms) {
    if (position[1] < platform.bounds.min.y) continue;
    if (insideFootprint(position[0], position[2], platform)) return true;
  }
  return false;
}

function hangingAnchor(rand: () => number, platform: Platform): [number, number, number] {
  const { min, max } = platform.bounds;
  const width = Math.max(max.x - min.x, 0);
  const depth = Math.max(max.z - min.z, 0);
  const insetX = Math.min(HANGING_INSET, width * 0.25);
  const insetZ = Math.min(HANGING_INSET, depth * 0.25);
  return [
    min.x + insetX + rand() * Math.max(width - insetX * 2, 0),
    min.y - HANGING_DROP,
    min.z + insetZ + rand() * Math.max(depth - insetZ * 2, 0),
  ];
}

function standingAnchor(rand: () => number, platform: Platform): [number, number, number] {
  const { min, max } = platform.bounds;
  const x = rand() < 0.5 ? min.x - STANDING_OFFSET : max.x + STANDING_OFFSET;
  const z = rand() < 0.5 ? min.z - STANDING_OFFSET : max.z + STANDING_OFFSET;
  const low = min.y + STANDING_LIFT;
  const high = max.y - STANDING_HEADROOM;
  const y = high > low ? low + rand() * (high - low) : low;
  return [x, y, z];
}

function pickAnchor(
  rand: () => number,
  platform: Platform,
  hanging: boolean,
  platforms: readonly Platform[],
): [number, number, number] | null {
  for (let attempt = 0; attempt < PLACEMENT_ATTEMPTS; attempt += 1) {
    const candidate = hanging ? hangingAnchor(rand, platform) : standingAnchor(rand, platform);
    if (!isBlocked(candidate, platforms)) return candidate;
  }
  return null;
}

function placeSpot(
  rand: () => number,
  anchor: readonly [number, number, number],
  spread: number,
  verticalSpread: number,
  downward: boolean,
  platforms: readonly Platform[],
): [number, number, number] {
  for (let attempt = 0; attempt < PLACEMENT_ATTEMPTS; attempt += 1) {
    const jitter = rand() * verticalSpread;
    const candidate: [number, number, number] = [
      anchor[0] + (rand() - 0.5) * spread * 2,
      downward ? anchor[1] - jitter : anchor[1] + jitter - verticalSpread * 0.5,
      anchor[2] + (rand() - 0.5) * spread * 2,
    ];
    if (!isBlocked(candidate, platforms)) return candidate;
  }
  // The anchor itself is always valid, so this keeps the cluster size intact.
  return [anchor[0], anchor[1], anchor[2]];
}

/**
 * Deterministic crystal clusters around `platforms`. Every cluster either hangs
 * under a platform (strictly below `bounds.min.y`) or stands just outside a
 * corner of it, between its bottom and its top. Nothing is ever placed on or
 * above a walkable top.
 */
export function crystalLayout(
  seed: number,
  platforms: readonly Platform[],
  perPlatform: number,
): CrystalSpot[] {
  const spots: CrystalSpot[] = [];
  const maxClusters = Math.max(0, Math.floor(perPlatform));
  if (maxClusters === 0 || platforms.length === 0) return spots;

  const rand = mulberry32(seed);

  for (const platform of platforms) {
    const clusterCount = Math.floor(rand() * (maxClusters + 1));
    for (let cluster = 0; cluster < clusterCount; cluster += 1) {
      const hanging = rand() < 0.5;
      const size =
        MIN_CLUSTER_SIZE + Math.floor(rand() * (MAX_CLUSTER_SIZE - MIN_CLUSTER_SIZE + 1));
      const anchor = pickAnchor(rand, platform, hanging, platforms);
      if (!anchor) continue;

      for (let i = 0; i < size; i += 1) {
        const position =
          i === 0
            ? anchor
            : placeSpot(
                rand,
                anchor,
                hanging ? HANGING_SPREAD : STANDING_SPREAD,
                hanging ? HANGING_STAGGER : STANDING_VERTICAL_SPREAD,
                hanging,
                platforms,
              );

        spots.push({
          position,
          scale: MIN_CRYSTAL_SCALE + rand() * (MAX_CRYSTAL_SCALE - MIN_CRYSTAL_SCALE),
          tilt: [(rand() - 0.5) * 2 * MAX_TILT, (rand() - 0.5) * 2 * MAX_TILT],
          yaw: rand() * Math.PI * 2,
        });
      }
    }
  }

  return spots;
}

/**
 * A spot hangs when it sits below a platform's underside and under (or just
 * past) its footprint. Hanging crystals are flipped so their tip points down.
 */
function isHangingSpot(spot: CrystalSpot, platforms: readonly Platform[]): boolean {
  for (const platform of platforms) {
    const { min, max } = platform.bounds;
    if (spot.position[1] >= min.y) continue;
    if (spot.position[0] < min.x - HANGING_FOOTPRINT_MARGIN) continue;
    if (spot.position[0] > max.x + HANGING_FOOTPRINT_MARGIN) continue;
    if (spot.position[2] < min.z - HANGING_FOOTPRINT_MARGIN) continue;
    if (spot.position[2] > max.z + HANGING_FOOTPRINT_MARGIN) continue;
    return true;
  }
  return false;
}

    [geometry],
  );

  useEffect(
    () => () => {
      material.dispose();

    [material],
  );

  return (
    <instancedMesh
      ref={meshRef}
      args={[undefined, undefined, layout.length]}
      geometry={geometry}
      material={material}
      frustumCulled={false}
    />
  );

varying vec3 vWorldPosition;

void main() {
  vec4 worldPosition = modelMatrix * vec4(position, 1.0);
  vWorldPosition = worldPosition.xyz;
  gl_Position = projectionMatrix * viewMatrix * worldPosition;
}
`;

const SKY_FRAGMENT_SHADER = `
uniform vec3 uZenith;
uniform vec3 uHorizon;
uniform vec3 uCloud;
uniform vec3 uCloudDark;
uniform vec3 uCenter;
uniform float uTime;

varying vec3 vWorldPosition;

// Hash-based value noise (no textures, fully procedural).
float hash13(vec3 p) {
  p = fract(p * 0.1031);
  p += dot(p, p.zyx + 31.32);
  return fract((p.x + p.y) * p.z);
}

float valueNoise(vec3 p) {
  vec3 i = floor(p);
  vec3 f = fract(p);
  vec3 u = f * f * (3.0 - 2.0 * f);

  float n000 = hash13(i + vec3(0.0, 0.0, 0.0));
  float n100 = hash13(i + vec3(1.0, 0.0, 0.0));
  float n010 = hash13(i + vec3(0.0, 1.0, 0.0));
  float n110 = hash13(i + vec3(1.0, 1.0, 0.0));
  float n001 = hash13(i + vec3(0.0, 0.0, 1.0));
  float n101 = hash13(i + vec3(1.0, 0.0, 1.0));
  float n011 = hash13(i + vec3(0.0, 1.0, 1.0));
  float n111 = hash13(i + vec3(1.0, 1.0, 1.0));

  return mix(
    mix(mix(n000, n100, u.x), mix(n010, n110, u.x), u.y),
    mix(mix(n001, n101, u.x), mix(n011, n111, u.x), u.y),
    u.z
  );
}

float fbm(vec3 p) {
  float sum = 0.0;
  float amplitude = 0.5;
  for (int i = 0; i < 5; i++) {
    sum += amplitude * valueNoise(p);
    p = p * 2.02 + vec3(11.3, 7.7, 5.1);
    amplitude *= 0.5;
  }
  return sum;
}

void main() {
  vec3 dir = normalize(vWorldPosition - uCenter);

  // Vertical gradient: horizon colour low, zenith colour high. The direction is
  // continuous over the whole sphere, so the dome has no hard seam.
  float height = smoothstep(-0.15, 0.85, dir.y);
  vec3 base = mix(uHorizon, uZenith, height);

  // Keep the zenith clearly brighter than the horizon band so the dome still
  // reads as sky under the heavy cloud cover.
  base = mix(base, uZenith, smoothstep(0.45, 1.0, dir.y) * 0.35);

  // Hard-edged billows: a narrow smoothstep band turns the fbm into crisp cloud
  // silhouettes instead of a milky wash.
  vec3 cloudDomain = dir * 3.0 + vec3(uTime * 0.01, 0.0, uTime * 0.006);
  float cloudNoise = fbm(cloudDomain);
  float cloudMask = smoothstep(0.42, 0.62, cloudNoise);
  vec3 colour = mix(base, uCloud, cloudMask * 0.85);

  // Broad, low-frequency masses: big dark shapes that dominate the lower sky.
  vec3 massDomain = dir * 0.85 + vec3(uTime * 0.003, 0.0, uTime * 0.002);
  float massNoise = fbm(massDomain);
  float massMask = smoothstep(0.38, 0.72, massNoise);
  float lowerSky = 1.0 - smoothstep(-0.05, 0.75, dir.y);
  colour = mix(colour, uCloudDark, massMask * lowerSky * 0.75);

  // Storm wall: noise stretched along x so the bands read as long horizontal
  // streaks rather than blobs.
  vec3 streakDomain = vec3(dir.x * 0.6, dir.y * 6.0, dir.z * 0.6) + vec3(uTime * 0.02, 0.0, 0.0);
  float streakNoise = fbm(streakDomain);
  float streakMask = smoothstep(0.45, 0.75, streakNoise);
  float streakBand = 1.0 - smoothstep(0.0, 0.55, abs(dir.y - 0.12));
  colour = mix(colour, uCloudDark, streakMask * streakBand * 0.35);

  gl_FragColor = vec4(colour, 1.0);

  #include <colorspace_fragment>
}
`;

  uHorizon: { value: Color };
  uCloud: { value: Color };
  uCloudDark: { value: Color };
  uCenter: { value: Vector3 };
  uTime: { value: number };
};

const DOME_RADIUS = 450;

export function Atmosphere(props: { theme: Theme }): JSX.Element {
  const { theme } = props;

  const domeRef = useRef<Mesh>(null);

  const uniforms = useMemo<SkyUniforms>(
    () => ({
      uZenith: { value: new Color() },
      uHorizon: { value: new Color() },
      uCloud: { value: new Color() },
      uCloudDark: { value: new Color() },
      uCenter: { value: new Vector3() },
      uTime: { value: 0 },
    }),
    [],
  );

  const geometry = useMemo(() => new SphereGeometry(DOME_RADIUS, 48, 24), []);

  const material = useMemo(
    () =>
      new ShaderMaterial({
        uniforms,
        vertexShader: SKY_VERTEX_SHADER,
        fragmentShader: SKY_FRAGMENT_SHADER,
        side: BackSide,
        depthWrite: false,
        fog: false,
      }),
    [uniforms],
  );

  // Theme changes only mutate uniform values through the live material: the
  // material (and its uniform record) is never re-created. A layout effect is
  // used so the colours are in place before the first rendered frame.
  useLayoutEffect(() => {
    const dome = domeRef.current;
    if (!dome) return;
    const mat = dome.material;
    if (!(mat instanceof ShaderMaterial)) return;

    const zenith = mat.uniforms.uZenith;
    if (zenith && zenith.value instanceof Color) zenith.value.set(theme.sky.zenith);

    const horizon = mat.uniforms.uHorizon;
    if (horizon && horizon.value instanceof Color) horizon.value.set(theme.sky.horizon);

    const cloud = mat.uniforms.uCloud;
    if (cloud && cloud.value instanceof Color) cloud.value.set(theme.sky.cloud);

    const cloudDark = mat.uniforms.uCloudDark;
    if (cloudDark && cloudDark.value instanceof Color) cloudDark.value.set(theme.sky.cloudDark);
  }, [theme]);

  useEffect(
    () => () => {
      geometry.dispose();
    },
    [geometry],
  );

  useEffect(
    () => () => {
      material.dispose();
    },
    [material],
  );

  useFrame((state) => {
    const dome = domeRef.current;
    if (!dome) return;

    const { x, y, z } = state.camera.position;
    dome.position.set(x, y, z);

    const mat = dome.material;
    if (!(mat instanceof ShaderMaterial)) return;

    const center = mat.uniforms.uCenter;
    if (center && center.value instanceof Vector3) center.value.set(x, y, z);

    const time = mat.uniforms.uTime;
    if (time && typeof time.value === "number") time.value = state.clock.elapsedTime;
  });

  return (
    <>
      <color attach="background" args={[theme.sky.horizon]} />
      <fog attach="fog" args={[theme.fog.color, theme.fog.near, theme.fog.far]} />
      <mesh
        ref={domeRef}
        geometry={geometry}
        material={material}
        frustumCulled={false}
        renderOrder={-1}
      />
    </>
  );
}

import { describe, expect, it } from "vitest";
import type { Platform } from "@/protocol/generated";
import { crystalLayout } from "./Crystals";
import type { CrystalSpot } from "./Crystals";

const PLATFORMS: readonly Platform[] = [
  { id: 1, kind: 0, bounds: { min: { x: -4, y: 0, z: -4 }, max: { x: 4, y: 1, z: 4 } } },
  { id: 2, kind: 0, bounds: { min: { x: 6, y: 2, z: -3 }, max: { x: 12, y: 3.5, z: 3 } } },
  { id: 3, kind: 1, bounds: { min: { x: -2, y: 4, z: 6 }, max: { x: 2, y: 5, z: 10 } } },
];

const SEEDS: readonly number[] = [1, 2, 3, 5, 8, 13, 21, 34];

function insideFootprint(spot: CrystalSpot, platform: Platform): boolean {
  const { min, max } = platform.bounds;
  return (
    spot.position[0] >= min.x &&
    spot.position[0] <= max.x &&
    spot.position[2] >= min.z &&
    spot.position[2] <= max.z
  );
}

describe("crystalLayout", () => {
  it("is deterministic for a given seed", () => {
    const first = crystalLayout(7, PLATFORMS, 2);
    const second = crystalLayout(7, PLATFORMS, 2);
    expect(first).toEqual(second);
  });

  it("changes when the seed changes", () => {
    const signatures = SEEDS.map((seed) => JSON.stringify(crystalLayout(seed, PLATFORMS, 2)));
    expect(new Set(signatures).size).toBeGreaterThan(1);
  });

  it("produces clusters for at least one seed", () => {
    const total = SEEDS.reduce((sum, seed) => sum + crystalLayout(seed, PLATFORMS, 2).length, 0);
    expect(total).toBeGreaterThan(0);
  });

  it("never places a crystal on or above a walkable top", () => {
    for (const seed of SEEDS) {
      const spots = crystalLayout(seed, PLATFORMS, 2);
      for (const spot of spots) {
        for (const platform of PLATFORMS) {
          if (!insideFootprint(spot, platform)) continue;
          // Inside a footprint the only legal place is strictly underneath it.
          expect(spot.position[1]).toBeLessThan(platform.bounds.min.y);
        }
      }
    }
  });

  it("hangs some clusters below a platform underside", () => {
    let hanging = 0;
    for (const seed of SEEDS) {
      const spots = crystalLayout(seed, PLATFORMS, 2);
      hanging += spots.filter((spot) =>
        PLATFORMS.some(
          (platform) => insideFootprint(spot, platform) && spot.position[1] < platform.bounds.min.y,
        ),
      ).length;
    }
    expect(hanging).toBeGreaterThan(0);
  });

  it("keeps scale and tilt inside their ranges", () => {
    for (const seed of SEEDS) {
      for (const spot of crystalLayout(seed, PLATFORMS, 2)) {
        expect(spot.scale).toBeGreaterThanOrEqual(0.25);
        expect(spot.scale).toBeLessThanOrEqual(1.1);
        expect(Math.abs(spot.tilt[0])).toBeLessThanOrEqual(0.5);
        expect(Math.abs(spot.tilt[1])).toBeLessThanOrEqual(0.5);
        expect(Number.isFinite(spot.yaw)).toBe(true);
      }
    }
  });

  it("respects perPlatform and empty inputs", () => {
    expect(crystalLayout(3, PLATFORMS, 0)).toEqual([]);
    expect(crystalLayout(3, [], 2)).toEqual([]);

    for (const seed of SEEDS) {
      const spots = crystalLayout(seed, PLATFORMS, 1);
      expect(spots.length).toBeLessThanOrEqual(PLATFORMS.length * 1 * 7);
    }
  });
});
