// Minimal RFC 6455 WebSocket client. Deliberately hand-rolled: the global
// WebSocket only exists on newer Node versions, and the kit must be dependency
// free and version independent.
// ---------------------------------------------------------------------------
const WS_GUID = "258EAFA5-E914-47DA-95CA-C5AB0DC85B11";

class WsClient extends EventEmitter {
  constructor(url) {
    super();
    this.url = url instanceof URL ? url : new URL(url);
    this.messages = []; // every text frame, in arrival order: { t, msg }
    this.open = false;
    this.clientId = null;
    this.token = null;
    this.welcome = null;
    this.socket = null;
    this.buffer = Buffer.alloc(0);
  }

  async connect(timeoutMs = CONNECT_TIMEOUT_MS) {
    const secure = this.url.protocol === "wss:";
    const port = Number(this.url.port) || (secure ? 443 : 80);
    const hostname = this.url.hostname;
    const path = (this.url.pathname || "/") + this.url.search;
    const key = crypto.randomBytes(16).toString("base64");
    const expectedAccept = crypto.createHash("sha1").update(key + WS_GUID).digest("base64");
    await new Promise((resolve, reject) => {
      let settled = false;
      let upgraded = false;
      let head = Buffer.alloc(0);
      const socket = secure
        ? tls.connect({ port, host: hostname, servername: hostname })
        : net.connect({ port, host: hostname });
      this.socket = socket;
      const timer = setTimeout(() => {
        socket.destroy();
        settle(new Error(`connect/handshake timed out after ${timeoutMs} ms (${this.url})`));
      }, timeoutMs);
      const settle = (error) => {
        if (settled) return;
        settled = true;
        clearTimeout(timer);
        if (error) {
          socket.destroy();
          reject(error);
        } else {
          resolve();
        }
      };
      socket.on("error", (error) => settle(error instanceof Error ? error : new Error(String(error))));
      socket.on("close", () => {
        if (!this.open) settle(new Error(`connection closed during handshake (${this.url})`));
        else this.handleClose();
      });
      socket.on("data", (chunk) => {
        if (this.open) {
          this.buffer = Buffer.concat([this.buffer, chunk]);
          this.pump();
          return;
        }
        if (upgraded) return;
        head = Buffer.concat([head, chunk]);
        const idx = head.indexOf("\r\n\r\n");
        if (idx === -1) return;
        upgraded = true;
        const header = head.subarray(0, idx).toString("latin1").toLowerCase();
        if (!/^http\/1\.1 101 /.test(header) || !header.includes(`sec-websocket-accept: ${expectedAccept}`)) {
          settle(new Error(`bad websocket handshake from ${this.url}`));
          return;
        }
        this.open = true;
        this.buffer = head.subarray(idx + 4);
        this.pump();
        settle();
      });
    });
  }

  pump() {
    for (;;) {
      const buf = this.buffer;
      if (buf.length < 2) return;
      const opcode = buf[0] & 0x0f;
      let len = buf[1] & 0x7f;
      let pos = 2;
      if (len === 126) {
        if (buf.length < 4) return;
        len = buf.readUInt16BE(2);
        pos = 4;
      } else if (len === 127) {
        if (buf.length < 10) return;
        const big = buf.readBigUInt64BE(2);
        if (big > 0x7fffffffn) {
          this.handleClose();
          return;
        }
        len = Number(big);
        pos = 10;
      }
      if (buf.length - pos < len) return;
      const payload = buf.subarray(pos, pos + len);
      this.buffer = buf.subarray(pos + len);
      if (opcode === 0x1) {
        let msg;
        try {
          msg = JSON.parse(payload.toString("utf8"));
        } catch {
          msg = { type: "__raw__", text: payload.toString("utf8") };
        }
        const rec = { t: Date.now(), msg };
        this.messages.push(rec);
        this.emit("message", rec);
      } else if (opcode === 0x9) {
        this.sendFrame(0xA, Buffer.from(payload)); // pong
      } else if (opcode === 0x8) {
        this.handleClose();
        return;
      }
    }
  }

  sendFrame(opcode, payload) {
    if (!this.socket || this.socket.destroyed) return;
    const mask = crypto.randomBytes(4);
    const masked = Buffer.allocUnsafe(payload.length);
    for (let i = 0; i < payload.length; i++) masked[i] = payload[i] ^ mask[i & 3];
    let header;
    if (payload.length < 126) {
      header = Buffer.from([0x80 | opcode, 0x80 | payload.length]);
    } else if (payload.length < 65536) {
      header = Buffer.alloc(4);
      header[0] = 0x80 | opcode;
      header[1] = 0x80 | 126;
      header.writeUInt16BE(payload.length, 2);
    } else {
      header = Buffer.alloc(10);
      header[0] = 0x80 | opcode;
      header[1] = 0x80 | 127;
      header.writeBigUInt64BE(BigInt(payload.length), 2);
    }
    this.socket.write(Buffer.concat([header, mask, masked]));
  }

  send(message) {
    this.sendFrame(0x1, Buffer.from(JSON.stringify(message), "utf8"));
  }

  close() {
    try {
      this.sendFrame(0x8, Buffer.alloc(0));
    } catch {
      // socket already gone
    }
    this.open = false;
    const socket = this.socket;
    if (socket && !socket.destroyed) setTimeout(() => socket.destroy(), 50).unref?.();
  }

  handleClose() {
    if (!this.open) return;
    this.open = false;
    this.emit("close");
  }
}

// Waiting helpers. Every message ever received stays in ws.messages, so waits
// also see messages that arrived before the wait started.
// ---------------------------------------------------------------------------
function messageMatches(msg, type, match) {
  if (msg?.type !== type) return false;
  for (const [k, v] of Object.entries(match ?? {})) {
    if (msg[k] !== v) return false;
  }
  return true;
}

const isMsg = (type, match = {}) => (rec) => messageMatches(rec.msg, type, match);

function waitFor(clients, predicate, timeoutMs, what) {
  const list = Array.isArray(clients) ? clients : [clients];
  return new Promise((resolve, reject) => {
    let done = false;
    let timer = null;
    const onMessage = () => {
      scan();
    };
    const onClose = () => finish(null, new Error(`${what}: connection closed before the expected message arrived`));
    const cleanup = () => {
      clearTimeout(timer);
      for (const ws of list) {
        ws.off("message", onMessage);
        ws.off("close", onClose);
      }
    };
    const finish = (value, error) => {
      if (done) return;
      done = true;
      cleanup();
      if (error) reject(error);
      else resolve(value);
    };
    const scan = () => {
      for (const ws of list) {
        for (const rec of ws.messages) {
          if (predicate(rec)) {
            finish(rec);
            return true;
          }
        }
      }
      return false;
    };
    timer = setTimeout(() => finish(null, new Error(`${what}: timed out after ${timeoutMs} ms`)), timeoutMs);
    for (const ws of list) {
      ws.on("message", onMessage);
      ws.on("close", onClose);
    }
    scan();
  });
}

function normalizeWsUrl(raw) {
  let s = String(raw).trim().replace(/\/+$/, "");
  if (/^https:\/\//.test(s)) s = "wss://" + s.slice("https://".length);
  else if (/^http:\/\//.test(s)) s = "ws://" + s.slice("http://".length);
  let url;
  try {
    url = new URL(s);
  } catch {
    throw new Error(`invalid WebSocket URL: ${raw}`);
  }
  if (url.protocol !== "ws:" && url.protocol !== "wss:") {
    throw new Error(`unsupported URL ${raw}: want ws:// or wss://`);
  }
  return url.toString();
}

async function connectClient(base, name, token = undefined) {
  const ws = new WsClient(base);
  await ws.connect();
  ws.send({ type: "hello", name, ...(token ? { token } : {}) });
  const rec = await waitFor(ws, (r) => r.msg.type === "welcome" || r.msg.type === "error", CONNECT_TIMEOUT_MS, `hello ${name}`);
  if (rec.msg.type === "error") fail(`hello ${name} rejected: ${rec.msg.reason ?? "(no reason given)"}`);
  ws.clientId = rec.msg.clientId;
  ws.token = rec.msg.token;
  ws.welcome = rec.msg;
  return ws;
}

async function openRoom(base, size = 3) {
  const clients = [];
  try {
    for (let i = 0; i < size; i++) clients.push(await connectClient(base, i === 0 ? "kit-host" : `kit-${i}`));
    if (clients[0].welcome.host !== true) {
      fail(`the first connected client is not the host (welcome.host = ${JSON.stringify(clients[0].welcome.host)})`);
    }
    for (let i = 1; i < size; i++) {
      if (clients[i].welcome.host !== false) fail(`late joiner kit-${i} unexpectedly reported host = true`);
    }
    return clients;
  } catch (error) {
    for (const c of clients) c.close();
    throw error;
  }
}

function closeAll(clients) {
  for (const c of clients) if (c) c.close();
}

async function startRun(host, { seed, permanents = null, klass = "striker" } = {}) {
  host.send({ type: "start_run", seed, class: klass, ...(permanents ? { permanents } : {}) });
  const rec = await waitFor(host, (r) => r.msg.type === "run_started" || r.msg.type === "error", CONNECT_TIMEOUT_MS, "start_run");
  if (rec.msg.type === "error") fail(`start_run rejected: ${rec.msg.reason ?? "(no reason given)"}`);
  return rec.msg.runId;
}

async function runLevel(host, runId, level) {
  host.send({ type: "start_level", level });
  const spawnRec = await waitFor(host, isMsg("spawn_set", { runId, level }), SPAWN_OR_OPEN_TIMEOUT_MS, `spawn_set for level ${level}`);
  const openRec = await waitFor(host, isMsg("vote_open", { runId, level }), SPAWN_OR_OPEN_TIMEOUT_MS, `vote_open for level ${level}`);
  if (spawnRec.t > openRec.t) {
    fail(`level ${level}: vote_open arrived before spawn_set — the session must open during the collapse window, after the level starts`);
  }
  return { spawn: spawnRec.msg, open: openRec.msg };
}

async function awaitClose(host, runId, sessionId) {
  const rec = await waitFor(host, isMsg("vote_closed", { runId, sessionId }), CLOSE_TIMEOUT_MS, `vote_closed for session ${sessionId}`);
  return rec.msg;
}

async function voteAll(entries, runId, sessionId) {
  for (const [client, optionId] of entries) {
    client.send({ type: "vote", optionId });
    await waitFor(client, isMsg("vote_update", { runId, sessionId, voter: client.clientId }), CONNECT_TIMEOUT_MS, "ack of a vote");
  }
}

const countIn = (list, id) => list.filter((x) => x === id).length;

function multisetEqual(a, b) {
  if (a.length !== b.length) return false;
  const s = [...a].sort();
  const t = [...b].sort();
  return s.every((v, i) => v === t[i]);
}

function fullCapSeeding() {
  const out = [];
  for (const entry of ENEMIES) for (let i = 0; i < entry.cap; i++) out.push(entry.id);
  return out;
}

async function checkSessionAtEveryLevel(base) {
  const [host, two, three] = await openRoom(base);
  try {
    const runId = await startRun(host, { seed: 1101 });
    for (const level of [1, 2, 3, 4]) {
      const { open } = await runLevel(host, runId, level);
      await awaitClose(host, runId, open.sessionId);
    }
  } finally {
    closeAll([host, two, three]);
  }
}

async function checkOptionsShape(base) {
  const [host, two, three] = await openRoom(base);
  try {
    const runId = await startRun(host, { seed: 1201 });
    for (const level of [1, 2]) {
      const { open } = await runLevel(host, runId, level);
      const options = open.options;
      if (!Array.isArray(options) || options.length !== 3) {
        fail(`level ${level}: offered ${JSON.stringify(options?.length)} options, want exactly 3`);
      }
      const ids = new Set();
      for (const option of options) {
        if (typeof option?.id !== "string" || option.id === "") fail(`level ${level}: option without an id: ${JSON.stringify(option)}`);
        if (!isKnownEnemy(option.id)) fail(`level ${level}: option id ${JSON.stringify(option.id)} is not an enemy from design/ENEMY_REGISTRY.md`);
        if (typeof option?.name !== "string" || option.name === "") fail(`level ${level}: option ${option.id} has no name`);
        if (option?.kind !== "permanent" && option?.kind !== "temporary") {
          fail(`level ${level}: option ${option.id} kind ${JSON.stringify(option?.kind)} is neither "permanent" nor "temporary"`);
        }
        if (ids.has(option.id)) fail(`level ${level}: duplicate option ${option.id}`);
        ids.add(option.id);
      }
      await awaitClose(host, runId, open.sessionId);
    }
  } finally {
    closeAll([host, two, three]);
  }
}

async function checkMajorityWins(base) {
  const [host, two, three] = await openRoom(base);
  try {
    const runId = await startRun(host, { seed: 1301 });
    const { open } = await runLevel(host, runId, 1);
    const want = open.options[0].id;
    const other = open.options[1].id;
    await voteAll([[host, want], [two, want], [three, other]], runId, open.sessionId);
    const closed = await awaitClose(host, runId, open.sessionId);
    if (closed.winnerId !== want) {
      fail(`three clients voted 2-1 for ${JSON.stringify(want)}, but the winner was ${JSON.stringify(closed.winnerId)} — the majority option must win`);
    }
  } finally {
    closeAll([host, two, three]);
  }
}

async function checkTieRules(base) {
  // (a) 1-1 tie between two options: the host's vote breaks the tie.
  {
    const [host, two, three] = await openRoom(base);
    try {
      const runId = await startRun(host, { seed: 1401 });
      const { open } = await runLevel(host, runId, 1);
      const hostPick = open.options[0].id;
      const otherPick = open.options[1].id;
      await voteAll([[host, hostPick], [two, otherPick]], runId, open.sessionId);
      const closed = await awaitClose(host, runId, open.sessionId);
      if (closed.winnerId !== hostPick) {
        fail(`1-1 tie with the host voting ${JSON.stringify(hostPick)} resolved to ${JSON.stringify(closed.winnerId)} — a tie must go to the host's vote`);
      }
    } finally {
      closeAll([host, two, three]);
    }
  }
  // (b) tie with no host vote: the same seed and level must pick the same winner on a re-run.
  {
    const winners = [];
    for (let attempt = 0; attempt < 2; attempt++) {
      const [host, two, three] = await openRoom(base);
      try {
        const runId = await startRun(host, { seed: 1402 });
        const { open } = await runLevel(host, runId, 1);
        await voteAll([[two, open.options[0].id], [three, open.options[1].id]], runId, open.sessionId);
        const closed = await awaitClose(host, runId, open.sessionId);
        winners.push(closed.winnerId);
      } finally {
        closeAll([host, two, three]);
      }
    }
    if (winners[0] !== winners[1]) {
      fail(`no-host-vote ties resolved differently on identical re-runs (seed 1402, level 1): ${JSON.stringify(winners)} — the server pick must be deterministic`);
    }
  }
}

async function checkPermanence(base) {
  const [host, two, three] = await openRoom(base);
  try {
    const runId = await startRun(host, { seed: 1501 });
    const first = await runLevel(host, runId, 1);
    const winner = first.open.options[0].id;
    await voteAll([[host, winner], [two, winner], [three, winner]], runId, first.open.sessionId);
    const closed = await awaitClose(host, runId, first.open.sessionId);
    if (closed.winnerId !== winner) fail(`unanimous vote for ${JSON.stringify(winner)} resolved to ${JSON.stringify(closed.winnerId)}`);
    const second = await runLevel(host, runId, 2);
    if (!second.spawn.enemies.includes(winner)) {
      fail(`winner ${winner} is missing from level 2's spawn set — the vote winner must become permanent`);
    }
    if (countIn(second.spawn.permanents, winner) !== 1) {
      fail(`level 2 spawn set holds ${JSON.stringify(second.spawn.permanents)} — want exactly one permanent ${winner}`);
    }
    await awaitClose(host, runId, second.open.sessionId);
    const third = await runLevel(host, runId, 3);
    if (countIn(third.spawn.permanents, winner) < 1) {
      fail(`winner ${winner} vanished by level 3 — permanence must hold for later levels`);
    }
  } finally {
    closeAll([host, two, three]);
  }
}

async function checkCapAwarePool(base) {
  const [host, two, three] = await openRoom(base);
  try {
    const seeded = ["pallbearer", "pallbearer"]; // selection_cap 2 -> already reached
    const runId = await startRun(host, { seed: 1601, permanents: seeded });
    const { open } = await runLevel(host, runId, 1);
    for (const option of open.options) {
      if (!isKnownEnemy(option.id)) fail(`offered option ${JSON.stringify(option.id)} is not in the frozen registry`);
      const held = countIn(seeded, option.id);
      if (held >= capOf(option.id)) {
        fail(`${option.id} offered although ${held} permanents already reach its selection_cap (${capOf(option.id)})`);
      }
      if (option.kind !== "permanent") {
        fail(`option ${option.id} offered as ${JSON.stringify(option.kind)} — a normal-pool option must be permanent`);
      }
    }
    await awaitClose(host, runId, open.sessionId);
  } finally {
    closeAll([host, two, three]);
  }
}

async function checkMinLevelGate(base) {
  const [host, two, three] = await openRoom(base);
  try {
    const runId = await startRun(host, { seed: 1701 });
    const { open } = await runLevel(host, runId, 1);
    for (const option of open.options) {
      const min = minLevelOf(option.id);
      if (min === undefined) fail(`offered option ${JSON.stringify(option.id)} is not in the frozen registry`);
      if (min > 1) fail(`level 1 offered ${option.id} whose min_level is ${min} — nothing above the current level may be offered`);
    }
    await awaitClose(host, runId, open.sessionId);
  } finally {
    closeAll([host, two, three]);
  }
}

async function checkOverflowTemporary(base) {
  const [host, two, three] = await openRoom(base);
  try {
    const seeded = fullCapSeeding(); // every entry at its selection_cap -> empty pool
    const runId = await startRun(host, { seed: 1801, permanents: seeded });
    const first = await runLevel(host, runId, 1);
    if (!multisetEqual(first.spawn.permanents, seeded)) {
      fail(`level 1 spawn set permanents ${JSON.stringify(first.spawn.permanents)} do not match the ${seeded.length} seeded permanents`);
    }
    if (first.spawn.temporaries.length !== 0) {
      fail(`level 1 already reports temporaries ${JSON.stringify(first.spawn.temporaries)}`);
    }
    if (!Array.isArray(first.open.options) || first.open.options.length < 1) {
      fail("overflow session offered no options — the run must not skip the enemy slot");
    }
    for (const option of first.open.options) {
      if (!isKnownEnemy(option.id)) fail(`overflow option ${JSON.stringify(option.id)} is not a known enemy`);
      if (option.kind !== "temporary") {
        fail(`with every entry capped, option ${option.id} is offered as ${JSON.stringify(option.kind)} — overflow options must be temporary`);
      }
    }
    const winner = first.open.options[0].id;
    await voteAll([[host, winner], [two, winner], [three, winner]], runId, first.open.sessionId);
    const closed = await awaitClose(host, runId, first.open.sessionId);
    if (closed.temporary !== true) {
      fail(`vote_closed reports temporary = ${JSON.stringify(closed.temporary)} — an overflow winner must be temporary`);
    }
    const second = await runLevel(host, runId, 2);
    if (countIn(second.spawn.permanents, winner) !== countIn(seeded, winner)) {
      fail(`overflow winner ${winner} added a permanent copy (level 2 permanents ${JSON.stringify(second.spawn.permanents)}) — overflow must never add a permanent`);
    }
    if (!second.spawn.temporaries.includes(winner)) {
      fail(`overflow winner ${winner} missing from level 2's temporaries (${JSON.stringify(second.spawn.temporaries)})`);
    }
    await awaitClose(host, runId, second.open.sessionId);
    const third = await runLevel(host, runId, 3);
    if (third.spawn.temporaries.includes(winner)) {
      fail(`overflow winner ${winner} still present at level 3 — temporary copies are removed when their level ends`);
    }
    if (!multisetEqual(third.spawn.permanents, seeded)) {
      fail(`level 3 permanents drifted from the seeded set: ${JSON.stringify(third.spawn.permanents)}`);
    }
  } finally {
    closeAll([host, two, three]);
  }
}

async function checkRejoinSafety(base) {
  const [host, two, three] = await openRoom(base);
  const extras = [];
  try {
    const runId = await startRun(host, { seed: 1901 });
    const { open } = await runLevel(host, runId, 1);
    const preserved = open.options[0].id; // the rejoiner's pre-disconnect pick
    const majorityPick = open.options[1].id;
    two.send({ type: "vote", optionId: preserved });
    await waitFor(two, isMsg("vote_update", { runId, sessionId: open.sessionId, voter: two.clientId }), CONNECT_TIMEOUT_MS, "ack of the first vote");
    two.close();
    const rejoiner = await connectClient(base, "kit-1", two.token);
    extras.push(rejoiner);
    if (rejoiner.welcome.rejoin !== true) {
      fail(`server treated the reconnecting client as new (welcome.rejoin = ${JSON.stringify(rejoiner.welcome.rejoin)})`);
    }
    if (rejoiner.welcome.clientId !== two.clientId) {
      fail(`rejoined client kept identity ${JSON.stringify(two.clientId)} but was assigned ${JSON.stringify(rejoiner.welcome.clientId)}`);
    }
    if (rejoiner.welcome.host !== false) fail("rejoined client unexpectedly became the host");
    // host votes the other option; the third client votes the rejoiner's option. If the
    // pre-disconnect vote was kept: 2-1 for it. If it was forgotten: 1-1 tie that the host
    // tie-breaks toward the other option. The winner tells the two cases apart.
    await voteAll([[host, majorityPick], [three, preserved]], runId, open.sessionId);
    const closed = await awaitClose(host, runId, open.sessionId);
    if (closed.winnerId !== preserved) {
      fail(`a client that disconnects and rejoins keeps its vote: expected ${JSON.stringify(preserved)} (its vote + client 2 = 2-1), got ${JSON.stringify(closed.winnerId)}`);
    }
  } finally {
    closeAll([host, two, three, ...extras]);
  }
}

// It implements design/ENEMY_REGISTRY.md §9 (pool rules) against the same wire contract the kit
// speaks (see tools/voting-conformance/README.md) — plus one deliberately broken behaviour at a
// time, selected with VOTE_STUB_VIOLATION:
//
//   ""                    faithful
//   majority_inverted     the trailing option wins instead of the majority option
//   tie_ignored           a tied vote ignores the host's vote
//   cap_ignored           capped entries stay in the pool
//   permanence_dropped    winners are never added as permanents (or as pending temporaries)
//   rejoin_forgotten      a reconnecting client is treated as brand new
//   late_vote_accepted    votes cast after the countdown are accepted
//
// Usage: node stub/server.mjs [port]   (port 0 or omitted -> ephemeral)

function insertClient(r, conn) {

// Frames (server side: client->server frames are masked, server->client are not)

function decodeFrames(buffer) {
  const frames = [];
  let off = 0;
  for (;;) {
    if (buffer.length - off < 2) break;
    const opcode = buffer[off] & 0x0f;
    const masked = (buffer[off + 1] & 0x80) !== 0;
    let len = buffer[off + 1] & 0x7f;
    let pos = off + 2;
    if (len === 126) {
      if (buffer.length - pos < 2) break;
      len = buffer.readUInt16BE(pos);
      pos += 2;
    } else if (len === 127) {
      if (buffer.length - pos < 8) break;
      const big = buffer.readBigUInt64BE(pos);
      pos += 8;
      if (big > 0x4000000n) return { frames, rest: null, oversized: true };
      len = Number(big);
    }
    let mask = null;
    if (masked) {
      if (buffer.length - pos < 4) break;
      mask = buffer.subarray(pos, pos + 4);
      pos += 4;
    }
    if (buffer.length - pos < len) break;
    let payload = buffer.subarray(pos, pos + len);
    if (mask) {
      payload = Buffer.from(payload);
      for (let i = 0; i < payload.length; i++) payload[i] ^= mask[i & 3];
    }
    frames.push({ opcode, payload });
    off = pos + len;
  }
  return { frames, rest: buffer.subarray(off), oversized: false };
}

function writeFrame(conn, opcode, payload) {
  if (!conn.socket || conn.socket.destroyed) return;
  let header;
  if (payload.length < 126) {
    header = Buffer.from([0x80 | opcode, payload.length]);
  } else if (payload.length < 65536) {
    header = Buffer.alloc(4);
    header[0] = 0x80 | opcode;
    header[1] = 126;
    header.writeUInt16BE(payload.length, 2);
  } else {
    header = Buffer.alloc(10);
    header[0] = 0x80 | opcode;
    header[1] = 127;
    header.writeBigUInt64BE(BigInt(payload.length), 2);
  }
  try {
    conn.socket.write(Buffer.concat([header, payload]));
  } catch {
    // client went away; the close handler cleans up
  }
}

    .sort((a, b) => (b.min_level - a.min_level) || (ENEMIES.indexOf(a) - ENEMIES.indexOf(b)));
  const picks = holders.slice(0, POOL_OFFER).map((e) => ({ id: e.id, name: e.name, kind: "temporary" }));

function closeSession(r, run, session) {
  session.closeTimer = null;
  const entries = session.options.map((o) => ({ id: o.id, count: session.tally.get(o.id) ?? 0 }));
  const max = Math.max(...entries.map((e) => e.count));
  let leaders = entries.filter((e) => e.count === max);
  if (VIOLATION === "majority_inverted") {
    const min = Math.min(...entries.map((e) => e.count));
    leaders = entries.filter((e) => e.count === min);
  }
  const host = r.clients[0] ?? null;
  const hostOption = host ? session.votes.get(host.clientId) : undefined;
  let winner;
  if (VIOLATION !== "tie_ignored" && hostOption !== undefined && leaders.some((e) => e.id === hostOption)) {
    winner = hostOption; // a tie goes to the host's vote
  } else {
    winner = leaders.length > 0 ? leaders[leaders.length - 1].id : session.options[0]?.id ?? null;
  }
  session.open = false;
  const winnerOption = winner === null ? null : session.options.find((o) => o.id === winner) ?? null;
  const temporary = winnerOption ? winnerOption.kind === "temporary" : false;
  if (winnerOption) {
    if (temporary) {
      run.pendingTemp = { id: winner, level: session.level + 1 }; // next level only
    } else if (VIOLATION !== "permanence_dropped") {
      // §4: a grant that would exceed the cap grants nothing for that enemy.

    }

    type: "vote_closed",
    runId: run.id,
    sessionId: session.id,

    for (const [id, n] of run.permanents) {
      if (n > ENEMY_BY_ID.get(id).cap) {
        return send(conn, { type: "error", reason: `permanents seeding exceeds selection_cap for ${id} (${n} > ${ENEMY_BY_ID.get(id).cap})` });
      }
    }
  }
  r.run = run;
  broadcast(r, { type: "run_started", runId: run.id, seed: run.seed, class: run.class });
}

function handleStartLevel(r, conn, msg) {
  const run = r.run;
  if (!run) return send(conn, { type: "error", reason: "start_level before start_run" });
  const level = Math.floor(Number(msg.level));
  if (!Number.isFinite(level) || level < 1) {
    return send(conn, { type: "error", reason: `bad level ${JSON.stringify(msg.level)}` });
  }
  if (run.session?.closeTimer) clearTimeout(run.session.closeTimer);
  if (run.session) run.session.open = false;
  if (run.sessionTimer) clearTimeout(run.sessionTimer);
  run.level = level;
  if (run.pendingTemp && run.pendingTemp.level < level) run.pendingTemp = null; // its level ended
  const temporaries = run.pendingTemp && run.pendingTemp.level === level ? [run.pendingTemp.id] : [];
  const permanents = expandPermanents(run);
  broadcast(r, {
    type: "spawn_set",
    runId: run.id,
    level,
    permanents,
    temporaries,
    enemies: [...permanents, ...temporaries],
  });
  run.session = null;
  run.sessionTimer = setTimeout(() => openSession(r, run, level), COLLAPSE_MS);
}

function handleVote(r, conn, msg) {
  const run = r.run;
  const session = run ? run.session : null;
  const optionId = typeof msg.optionId === "string" ? msg.optionId : null;
  if (!session || !session.open) {
    if (VIOLATION === "late_vote_accepted" && session && !session.open
      && optionId && session.options.some((o) => o.id === optionId) && !session.votes.has(conn.clientId)) {
      session.votes.set(conn.clientId, optionId);
      session.tally.set(optionId, (session.tally.get(optionId) ?? 0) + 1);
      return broadcast(r, {
        type: "vote_update",
        runId: run.id,
        sessionId: session.id,
        tally: tallyOut(session),
        voter: conn.clientId,
        late: true,
      });
    }
    return send(conn, {
      type: "vote_rejected",
      reason: session ? "the vote session is closed" : "no vote session is open",
      runId: run ? run.id : null,
    });
  }
  if (!optionId || !session.options.some((o) => o.id === optionId)) {
    return send(conn, { type: "vote_rejected", reason: `unknown option ${JSON.stringify(msg.optionId)}`, runId: run.id });
  }
  if (session.votes.has(conn.clientId)) {
    return send(conn, { type: "vote_rejected", reason: "already voted in this session", runId: run.id });
  }
  castVote(r, conn, session, optionId);
}

function detach(roomKey, conn) {
  const r = roomOf(roomKey);
  const idx = r.clients.indexOf(conn);
  if (idx === -1) return;
  r.clients.splice(idx, 1);
  if (conn.token) {
    r.detached.set(conn.token, {
      clientId: conn.clientId,
      name: conn.name,
      joinSeq: conn.joinSeq,
      pendingVote: conn.lastVoted,
    });
  }
  if (r.clients.length === 0 && !r.run) rooms.delete(roomKey);
}

const server = http.createServer((req, res) => {
  res.writeHead(404, { "content-type": "text/plain" });
  res.end("voting-conformance stub speaks WebSocket only\n");
});

server.on("upgrade", (req, socket) => {
  const key = req.headers["sec-websocket-key"];
  if (!key) {
    socket.destroy();
    return;
  }
  const accept = crypto.createHash("sha1").update(key + WS_GUID).digest("base64");
  socket.write(
    "HTTP/1.1 101 Switching Protocols\r\n"
    + "Upgrade: websocket\r\n"
    + "Connection: Upgrade\r\n"
    + `Sec-WebSocket-Accept: ${accept}\r\n`
    + "\r\n",
  );
  socket.setNoDelay(true);
  const url = new URL(req.url ?? "/", "http://localhost");
  const roomKey = url.searchParams.get("room") ?? "";
  const conn = {
    socket,
    clientId: 0,
    name: "",
    token: null,
    joinSeq: 0,
    rejoin: false,
    lastVoted: null,
  };
  let buffer = Buffer.alloc(0);
  socket.on("data", (chunk) => {
    buffer = Buffer.concat([buffer, chunk]);
    const { frames, rest, oversized } = decodeFrames(buffer);
    if (oversized) {
      socket.destroy();
      return;
    }
    buffer = rest;
    for (const frame of frames) {
      if (frame.opcode === 0x1) {
        let msg = null;
        try {
          msg = JSON.parse(frame.payload.toString("utf8"));
        } catch {
          continue;
        }
        handleMessage(roomKey, conn, msg);
      } else if (frame.opcode === 0x8) {
        detach(roomKey, conn);
        socket.end();
      } else if (frame.opcode === 0x9) {
        writeFrame(conn, 0xA, Buffer.from(frame.payload)); // pong
      }
    }
  });
  const gone = () => detach(roomKey, conn);
  socket.on("close", gone);
  socket.on("error", gone);
});

const portArg = process.argv[2];
const port = portArg === undefined || portArg === "0" ? 0 : Number(portArg);
server.on("error", (error) => {
  console.error(String(error));
  process.exit(1);
});
server.listen(Number.isFinite(port) && port >= 0 ? port : 0, "127.0.0.1", () => {
  console.log(`listening on ws://127.0.0.1:${server.address().port}`);
});

# Voting conformance kit

Black-box conformance kit for the end-of-level vote session described in T-081 and
`design/ENEMY_REGISTRY.md` §9 ("Pool rules (the voting system)"). It never reads the voting core's
source; it drives real WebSocket clients and prints `PASS - <check>` / `FAIL - <check>` per check.
Exit code 0 only on a full pass.

```sh
VOTING_URL=ws://127.0.0.1:9001 node tools/voting-conformance/run.mjs
node tools/voting-conformance/run.mjs --url ws://127.0.0.1:9001
node tools/voting-conformance/run.mjs --selftest   # kit-vs-stub, ~1 min
```

Without `--url` / `VOTING_URL` the kit prints `skipped` and exits 0, so `just verify` stays green
when no server is running (matches the matchmaker kit's convention).

## Files

- `run.mjs` — the check suite, selftest driver and a hand-rolled RFC 6455 WebSocket client
  (works on any Node ≥ 18; the global `WebSocket` needs Node ≥ 22, and the kit must stay
  dependency free).
- `registry.mjs` — the frozen enemy registry (§10) transcribed from `design/ENEMY_REGISTRY.md`;
  shared by the kit's assertions and the stub's pool builder. Spec data, not server source.
- `stub/server.mjs` — a faithful implementation of §9 on this wire contract, plus single
  deliberate violations for `--selftest` (`VOTE_STUB_VIOLATION`).

## Wire contract (what the server under test must implement)

JSON text frames. The kit connects three clients (all HTTP Upgrade handshakes to the given URL,
optionally with a shared `?room=<id>` query parameter). The earliest-joined connected client of a
room is the host.

Client → server:

| message | meaning |
|---|---|
| `{"type":"hello","name":str,"token"?:str}` | join; with a previous `welcome.token` reclaims the old session |
| `{"type":"start_run","seed":num,"class"?:str,"permanents"?:[enemyId,...]}` | begin a fresh run (resets permanents/sessions); duplicate ids in `permanents` seed permanent copies and must not exceed `selection_cap` — needed by the cap (`cap-aware-pool`) and overflow (`overflow-temporary`) checks |
| `{"type":"start_level","level":num}` | begin a level (usually the host); the server broadcasts the spawn set, then opens the vote session during the collapse window |
| `{"type":"vote","optionId":str}` | cast a vote in the open session; one vote per client per session |

Server → client:

| message | meaning |
|---|---|
| `{"type":"welcome","clientId","token","host":bool,"rejoin":bool,"runId"?}` | reply to `hello`; `rejoin:true` means the session (including an already-cast vote) was restored |
| `{"type":"run_started","runId","seed","class"}` | run created; all subsequent messages carry this `runId` |
| `{"type":"spawn_set","runId","level","permanents":[ids],"temporaries":[ids],"enemies":[ids]}` | level start; `permanents` lists permanent copies (with repeats), `temporaries` the overflow copies living this level only |
| `{"type":"vote_open","runId","sessionId","level","options":[{id,name,kind}],"countdown_ms"?}` | session opened during the collapse window; `kind` is `"permanent"` or `"temporary"` |
| `{"type":"vote_update","runId","sessionId","tally":{optionId:count},"voter"}` | a vote registered (including a restored vote after rejoin) |
| `{"type":"vote_closed","runId","sessionId","level","winnerId","winnerName"?,"temporary":bool}` | countdown elapsed or resolved; `temporary:true` for an overflow winner |
| `{"type":"vote_rejected","reason","runId"?}` (or `{"type":"error","reason"}`) | late/invalid vote rejected |

Sequencing per level: `start_level` → `spawn_set(level)` → `vote_open(level)` during the collapse →
votes → `vote_closed` (countdown, even with zero votes) → next `start_level`. The kit always
awaits `vote_closed` before starting the next level. The server, not the kit, simulates the level
and its collapse window.

## The ten checks

| check | asserts |
|---|---|
| `session-at-every-level` | a vote session opens at level 1 and at every later level, after the level's spawn set (i.e. in the collapse window) |
| `options-shape` | every session offers exactly 3 options, each with a nonempty `id` (a registry enemy), a nonempty `name` and a `kind` of `"permanent"`/`"temporary"`, ids unique |
| `majority-wins` | 2-1 vote ⇒ the 2-option wins |
| `tie-rules` | 1-1 tie ⇒ host's vote wins; 1-1 tie with no host vote ⇒ same winner on an identical re-run (same seed + level) |
| `permanence` | unanimous winner appears in the next level's spawn set (exactly once) and is still there two levels later |
| `cap-aware-pool` | with 2 permanents of `pallbearer` (its cap), it is never offered; offered options stay `permanent` |
| `min-level-gate` | a level-1 session never offers a registry enemy with `min_level` > 1 |
| `overflow-temporary` | with every entry capped, all offered options are `temporary`; the winner is flagged temporary, appears among level 2's temporaries, adds no permanent, and is gone at level 3 |
| `rejoin-safety` | a disconnect + reconnect with the old token keeps the client's identity (`rejoin:true`) and its pre-disconnect vote (2-1 outcome proves it was counted) |
| `countdown-late-vote` | with no votes the countdown still closes the session with a winner, and a vote sent afterwards is rejected (`vote_rejected`/`error`) |

## Selftest

`--selftest` spawns `stub/server.mjs` on an ephemeral port, runs the full suite against the
faithful stub (must pass everything), then, for each violation, runs the violation's paired check
and requires it to FAIL:

| `VOTE_STUB_VIOLATION` | paired check |
|---|---|
| `majority_inverted` | `majority-wins` |
| `tie_ignored` | `tie-rules` |
| `cap_ignored` | `cap-aware-pool` |
| `permanence_dropped` | `permanence` |
| `rejoin_forgotten` | `rejoin-safety` |
| `late_vote_accepted` | `countdown-late-vote` |

Exit 0 only when the faithful stub fully passes and every violation trips its paired check.
(Each violation run is scoped to its paired check; other checks are irrelevant there because only
one behaviour is broken at a time.)

## Assumptions and limits

- The kit defines the wire contract above; a server speaking a different protocol fails loudly
  — that failure is a reportable implementation/kit mismatch, per T-081.
- `start_run.permanents` seeding must be honoured for `cap-aware-pool` and `overflow-temporary`;
  organically filling 24 permanents through the vote would make those checks prohibitively slow.
- `vote_rejected` must be explicit for a late vote; silently ignoring it is a FAIL (check
  `countdown-late-vote`).
- Determinism is compared across two identical runs; a pick that depends only on the offered set
  (not the seed) also passes — the ticket's observable requirement is re-run stability.
