//! Client for the VOIDWRAP RUN matchmaker (frozen contract: `POST /matches`,
//! `POST /matches/{id}`, `DELETE /matches/{id}`, `GET /matches?version=..`).
//!
//! The matchmaker is optional infrastructure. Nothing in this module may make the
//! game fail: every request has a short timeout, unreachable or unhelpful answers are
//! logged at debug and reported as plain outcomes, and callers retry on the next
//! heartbeat. Delisting on shutdown is best effort and never blocks the game.

use serde::{Deserialize, Serialize};
use std::time::Duration;

/// Default matchmaker base URL when `MATCHMAKER_URL` is not set.
pub const DEFAULT_MATCHMAKER_URL: &str = "http://localhost:8080";

/// Short timeout for matchmaker requests, connect and total alike. The game must never wait on the matchmaker.
const REQUEST_TIMEOUT: Duration = Duration::from_secs(2);

/// Body of `POST /matches`: field-for-field the frozen contract.
#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
pub struct RegisterRequest {
    /// Display name of the listing.
    pub name: String,
    /// Workspace game version.
    pub version: String,
    /// Port friends connect to.
    pub port: u16,
    /// Current players.
    pub players: u32,
    /// Maximum players.
    pub max_players: u32,
    /// Current level.
    pub level: u32,
    /// Current phase (short lowercase word).
    pub phase: String,
    /// True when the match accepts new players.
    pub joinable: bool,
    /// Address friends may use, or null when unknown.
    pub address: Option<String>,
}

/// Body of the `POST /matches/{id}` heartbeat: field-for-field the frozen contract.
#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
pub struct HeartbeatRequest {
    /// Secret returned by registration.
    pub secret: String,
    /// Current players.
    pub players: u32,
    /// Current level.
    pub level: u32,
    /// Current phase.
    pub phase: String,
    /// True when the match accepts new players.
    pub joinable: bool,
}

/// Body of `DELETE /matches/{id}`: field-for-field the frozen contract.
#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
struct DelistRequest {
    /// Secret returned by registration.
    secret: String,
}

/// Response of `POST /matches`. A missing `address` decodes as `None`.
#[derive(Clone, Debug, Deserialize)]
pub struct RegisterResponse {
    /// Opaque match id.
    pub id: String,
    /// Secret used for the heartbeat and the delist.
    pub secret: String,
    /// Address the matchmaker recorded, if it reports one.
    #[serde(default)]
    pub address: Option<String>,
}

/// An active matchmaker listing: the id and secret returned by registration.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MatchmakerSession {
    /// Opaque match id.
    pub id: String,
    /// Secret used for the heartbeat and the delist.
    pub secret: String,
    /// Address the matchmaker recorded (its echo wins on mismatch).
    pub address: Option<String>,
}

/// Outcome of a heartbeat, so the caller can re-register when the entry is gone.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum HeartbeatOutcome {
    /// The matchmaker accepted the heartbeat (200).
    Ok,
    /// The entry expired (404) or the secret was rejected (403); register again.
    Expired,
    /// Unreachable, timed out or answered something unexpected; try again next beat.
    Unreachable,
}

/// Best-effort snapshot of the running game, sent with every heartbeat.
///
/// Nothing updates the snapshot yet; it stays at [`GameSnapshot::default`] until the
/// game task reports its state (this change touches only the hosting layer). Kept small
/// and cloneable so the heartbeat never blocks on the game.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct GameSnapshot {
    /// Current number of connected players.
    pub players: u32,
    /// Level of the current run.
    pub level: u32,
    /// Phase of the game ("unknown" until the game task reports).
    pub phase: String,
}

impl Default for GameSnapshot {
    /// Zero players, level 1, phase "unknown".
    fn default() -> Self {
        Self {
            players: 0,
            level: 1,
            phase: "unknown".to_string(),
        }
    }
}

/// Configuration for the matchmaker connection, read from the environment.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MatchmakerConfig {
    /// Base URL of the matchmaker, trailing slashes trimmed (`MATCHMAKER_URL`).
    pub url: String,
    /// Public origin advertised by forced-open hosts (`PUBLIC_ORIGIN`), if set.
    pub public_origin: Option<String>,
}

impl MatchmakerConfig {
    /// Read `MATCHMAKER_URL` (defaulting to `http://localhost:8080`) and `PUBLIC_ORIGIN`.
    ///
    /// Empty values are treated as unset.
    pub fn from_env() -> Self {
        let url = std::env::var("MATCHMAKER_URL")
            .ok()
            .map(|value| value.trim().trim_end_matches('/').to_string())
            .filter(|value| !value.is_empty())
            .unwrap_or_else(|| DEFAULT_MATCHMAKER_URL.to_string());
        let public_origin = std::env::var("PUBLIC_ORIGIN")
            .ok()
            .map(|value| value.trim().trim_end_matches('/').to_string())
            .filter(|value| !value.is_empty());
        Self { url, public_origin }
    }
}

impl Default for MatchmakerConfig {
    /// The default configuration: localhost matchmaker, no public origin.
    fn default() -> Self {
        Self {
            url: DEFAULT_MATCHMAKER_URL.to_string(),
            public_origin: None,
        }
    }
}

/// Matchmaker HTTP client. Cheap to clone; every method logs failures at debug and never panics.
#[derive(Clone)]
pub struct MatchmakerClient {
    http: reqwest::Client,
    base_url: String,
}

impl MatchmakerClient {
    /// Create a client for `base_url` (trailing slashes trimmed).
    pub fn new(base_url: &str) -> Self {
        let http = reqwest::Client::builder()
            .timeout(REQUEST_TIMEOUT)
            .connect_timeout(REQUEST_TIMEOUT)
            .build()
            .unwrap_or_else(|_| reqwest::Client::new());
        let base_url = base_url.trim().trim_end_matches('/').to_string();
        Self { http, base_url }
    }

    /// `POST /matches`: register this host. `None` (logged at debug) when anything fails; retry next beat.
    pub async fn register(&self, body: &RegisterRequest) -> Option<RegisterResponse> {
        let url = format!("{}/matches", self.base_url);
        let response = match self.http.post(&url).json(body).send().await {
            Ok(response) => response,
            Err(e) => {
                tracing::debug!("matchmaker unreachable: {e}");
                return None;
            }
        };
        if !response.status().is_success() {
            tracing::debug!("matchmaker refused the registration: HTTP {}", response.status());
            return None;
        }
        match response.json::<RegisterResponse>().await {
            Ok(parsed) => Some(parsed),
            Err(e) => {
                tracing::debug!("matchmaker answered with an unreadable registration: {e}");
                None
            }
        }
    }

    /// `POST /matches/{id}`: heartbeat. 404/403 are reported as [`HeartbeatOutcome::Expired`].
    pub async fn heartbeat(&self, id: &str, body: &HeartbeatRequest) -> HeartbeatOutcome {
        let url = format!("{}/matches/{id}", self.base_url);
        let response = match self.http.post(&url).json(body).send().await {
            Ok(response) => response,
            Err(e) => {
                tracing::debug!("matchmaker unreachable: {e}");
                return HeartbeatOutcome::Unreachable;
            }
        };
        match response.status().as_u16() {
            200 => HeartbeatOutcome::Ok,
            404 | 403 => HeartbeatOutcome::Expired,
            status => {
                tracing::debug!("matchmaker answered the heartbeat with HTTP {status}");
                HeartbeatOutcome::Unreachable
            }
        }
    }

    /// `DELETE /matches/{id}`: delist (best effort). True when the entry is gone.
    ///
    /// A missing entry (404) counts as delisted: the goal is that no listing remains.
    pub async fn delist(&self, id: &str, secret: &str) -> bool {
        let url = format!("{}/matches/{id}", self.base_url);
        let body = DelistRequest {
            secret: secret.to_string(),
        };
        let response = match self.http.delete(&url).json(&body).send().await {
            Ok(response) => response,
            Err(e) => {
                tracing::debug!("matchmaker unreachable: {e}");
                return false;
            }
        };
        match response.status().as_u16() {
            200 | 404 => true,
            status => {
                tracing::debug!("matchmaker refused the delist: HTTP {status}");
                false
            }
        }
    }
}

//! In-game hosting for friends: an extra listener on `0.0.0.0:<HOST_PORT>` serving the
//! same router and the same game over the LAN, plus a best-effort UPnP port mapping
//! ("VOIDWRAP RUN", TCP HOST_PORT, 2 hours) so friends on the internet can join too.
//!
//! Also lists the host on the matchmaker once hosting settles (after the UPnP
//! attempt), heartbeats every 10 seconds, and delists on stop/shutdown. The
//! matchmaker is optional infrastructure: being unable to reach it never affects the game.

use crate::matchmaker_client::{
    GameSnapshot, HeartbeatOutcome, HeartbeatRequest, MatchmakerClient, MatchmakerConfig, MatchmakerSession,
    RegisterRequest,
};
use serde::{Deserialize, Serialize};
use std::net::{IpAddr, Ipv4Addr, SocketAddr};
use std::sync::Arc;
use std::time::Duration;
use tokio::sync::Mutex;

/// Default port for the hosting listener; override with the `VOIDWRAP_HOST_PORT` environment variable.
pub const DEFAULT_HOST_PORT: u16 = 7879;
/// How long the router is asked to keep the UPnP port mapping.
const UPNP_LEASE_SECS: u64 = 2 * 60 * 60;
/// How often an open UPnP mapping is renewed (well inside the 2 hour lease).
///
/// Lease-expiry choice: the mapping is re-asserted every 90 minutes (it never
/// silently lapses). If a re-assert fails, the UPnP state becomes `Failed` and the
/// next heartbeat carries `joinable: false` until the router is asked again.
const UPNP_RENEW_INTERVAL: Duration = Duration::from_secs(90 * 60);
/// How long to wait for the UPnP gateway to answer the gateway search.
const UPNP_SEARCH_TIMEOUT: Duration = Duration::from_secs(3);
/// How long to wait for the router to remove the UPnP mapping when hosting stops or the program exits.
const UPNP_REMOVE_TIMEOUT: Duration = Duration::from_secs(5);
/// How long to wait for the hosting listener to close before aborting it.
const HOST_STOP_TIMEOUT: Duration = Duration::from_secs(2);
/// Description attached to the UPnP port mapping (shown in the router's port-forwarding table).
const UPNP_DESCRIPTION: &str = "VOIDWRAP RUN";
/// Message used when the server itself listens on all interfaces (`VOIDWRAP_HOST=0.0.0.0`).
const FORCED_OPEN_MESSAGE: &str = "Hosting is on because VOIDWRAP_HOST=0.0.0.0 was set when the server started. Friends can join with the addresses below.";
/// How often the matchmaker is heartbeated while hosting.
const HEARTBEAT_INTERVAL: Duration = Duration::from_secs(10);
/// Display name reported to the matchmaker.
const MATCHMAKER_NAME: &str = "VOIDWRAP RUN";
/// Maximum players reported to the matchmaker (the game has no hard cap today).
const MAX_PLAYERS: u32 = 8;
/// Workspace game version reported to the matchmaker.
const GAME_VERSION: &str = env!("CARGO_PKG_VERSION");

/// UPnP state shown to the player: "off" | "trying" | "open" | "failed".
#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
#[serde(rename_all = "lowercase")]
pub enum UpnpStatus {
    /// Hosting is off, or UPnP has not been attempted.
    Off,
    /// Asking the router to open the port.
    Trying,
    /// Router opened the port; friends anywhere can join.
    Open,
    /// Router refused or was unreachable; LAN friends can still join.
    Failed,
}

impl UpnpStatus {
    /// Plain-words sentence for the player (for non-technical players).
    pub fn message(self) -> &'static str {
        match self {
            UpnpStatus::Off => "Hosting is off. Turn it on so friends can join.",
            UpnpStatus::Trying => "Asking your router to open the door for friends...",
            UpnpStatus::Open => "Friends anywhere can join with the 'Friends anywhere' address.",
            UpnpStatus::Failed => {
                "Your router did not allow automatic setup. Friends on your Wi-Fi can still join; for internet play see HOW-TO-PLAY.md."
            }
        }
    }
}

/// Parse the hosting port from `VOIDWRAP_HOST_PORT`.
/// Missing, empty, zero or unparseable values fall back to [`DEFAULT_HOST_PORT`].
pub fn parse_host_port(raw: Option<&str>) -> u16 {
    raw.and_then(|v| v.trim().parse::<u16>().ok())
        .filter(|&p| p != 0)
        .unwrap_or(DEFAULT_HOST_PORT)
}

/// True when the peer is on this machine (127.0.0.0/8 or `[::1]`).
/// Used to keep `/api/host` local-only, so friends can never toggle hosting.
pub fn is_loopback_peer(peer: &SocketAddr) -> bool {
    peer.ip().is_loopback()
}

/// True for RFC 1918 private IPv4 addresses (10/8, 172.16/12, 192.168/16).
pub fn is_private_ipv4(ip: Ipv4Addr) -> bool {
    let o = ip.octets();
    o[0] == 10 || (o[0] == 172 && (16..=31).contains(&o[1])) || (o[0] == 192 && o[1] == 168)
}

/// Filter to private addresses, deduplicate and sort them (stable output, testable).
pub fn private_ipv4_strings(addrs: &[Ipv4Addr]) -> Vec<String> {
    let mut list: Vec<String> = addrs
        .iter()
        .filter(|ip| is_private_ipv4(**ip))
        .map(|ip| ip.to_string())
        .collect();
    list.sort();
    list.dedup();
    list
}

/// Local IPv4 address this machine would use to reach `target`
/// (a UDP `connect` selects the route; no packet is ever sent).
fn outbound_ipv4(target: Ipv4Addr) -> Option<Ipv4Addr> {
    let socket = std::net::UdpSocket::bind((Ipv4Addr::UNSPECIFIED, 0)).ok()?;
    socket.connect((target, 80)).ok()?;
    match socket.local_addr().ok()?.ip() {
        IpAddr::V4(v4) => Some(v4),
        IpAddr::V6(_) => None,
    }
}

/// This machine's private IPv4 addresses, best effort. Sends no traffic.
pub fn lan_ipv4_addresses() -> Vec<String> {
    // Two well-known public resolvers usually route via the primary interface; the
    // connect only picks a route, it does not send a packet.
    let probes = [Ipv4Addr::new(8, 8, 8, 8), Ipv4Addr::new(1, 1, 1, 1)];
    let found: Vec<Ipv4Addr> = probes.into_iter().filter_map(outbound_ipv4).collect();
    private_ipv4_strings(&found)
}

/// Compute the `joinable` flag reported to the matchmaker.
///
/// A forced-open host (the Render deployment) is always joinable. A normal host is
/// joinable only while its UPnP mapping is open; waiting on the router or a lapsed
/// mapping reports `joinable: false`.
fn matchmaker_joinable(forced_open: bool, upnp: UpnpStatus) -> bool {
    forced_open || upnp == UpnpStatus::Open
}

/// Cross-check the address we send with the matchmaker's echo.
///
/// On a mismatch the matchmaker's address wins (it may know how the world reaches us,
/// for example when the router reports no public address) and a warning is logged.
/// The sent address is kept only when the matchmaker reports none.
fn reconcile_echo(sent: Option<&str>, echo: Option<&str>) -> Option<String> {
    match (sent, echo) {
        (Some(sent), Some(echo)) => {
            if sent != echo {
                tracing::warn!("matchmaker reports address {echo}, we sent {sent}; trusting the matchmaker");
            }
            Some(echo.to_string())
        }
        (None, Some(echo)) => Some(echo.to_string()),
        (Some(_), None) => {
            tracing::debug!("matchmaker did not report the address it recorded; keeping {sent:?}");
            sent.map(str::to_string)
        }
        (None, None) => None,
    }
}

}

struct HostInner {
    /// Shutdown signal for the hosting listener, present while it runs.
    shutdown: Option<tokio::sync::oneshot::Sender<()>>,
    /// Handle of the hosting listener task, present while it runs.
    listener: Option<tokio::task::JoinHandle<()>>,
    /// UPnP gateway connection, kept so the mapping can be removed on stop/exit.
    gateway: Option<igd_next::aio::Gateway<igd_next::aio::tokio::Tokio>>,
    /// Background task that renews the UPnP lease while the mapping is open.
    renewal: Option<tokio::task::JoinHandle<()>>,
    /// Current UPnP state.
    upnp: UpnpStatus,
    /// Public IP as reported by the router, if any.
    public: Option<IpAddr>,
    /// Matchmaker listing session, present while registered; taken on stop/delist.
    matchmaker: Option<MatchmakerSession>,
    /// Background registration/heartbeat task, spawned once hosting settles; aborted on stop.
    matchmaker_task: Option<tokio::task::JoinHandle<()>>,
    /// Best-effort game snapshot reported to the matchmaker.
    game: GameSnapshot,
}

impl HostManager {
    /// Create a manager. `forced_open` must be true when the server itself already
    /// listens on all interfaces (`VOIDWRAP_HOST=0.0.0.0` or `PORT` set).
    ///
    /// The matchmaker is configured from `MATCHMAKER_URL` (default
    /// `http://localhost:8080`) and `PUBLIC_ORIGIN`.
    pub fn new(main_port: u16, host_port: u16, forced_open: bool) -> Self {
        Self::with_matchmaker(main_port, host_port, forced_open, MatchmakerConfig::from_env())
    }

    /// Create a manager with explicit matchmaker settings (used by tests).
    pub fn with_matchmaker(main_port: u16, host_port: u16, forced_open: bool, matchmaker: MatchmakerConfig) -> Self {
        let matchmaker_client = MatchmakerClient::new(&matchmaker.url);
        Self {
            main_port,
            host_port,
            forced_open,
            inner: Mutex::new(HostInner {
                shutdown: None,
                listener: None,
                gateway: None,
                renewal: None,
                upnp: UpnpStatus::Off,
                public: None,
                matchmaker: None,
                matchmaker_task: None,
                game: GameSnapshot::default(),
            }),
            matchmaker,
            matchmaker_client,
        }
    }

    /// Port reported to friends.
    fn reported_port(&self) -> u16 {
        if self.forced_open {
            self.main_port
        } else {
            self.host_port
        }
    }

    /// Address to register: forced-open hosts advertise `PUBLIC_ORIGIN`; everyone else
    /// prefers the router-reported public address (`inner.public`).
    fn registration_address(&self, inner: &HostInner) -> Option<String> {
        if self.forced_open {
            return self.matchmaker.public_origin.clone();
        }
        inner.public.map(|ip| ip.to_string())
    }

    /// Build the JSON status from a locked inner state (synchronous).
    fn snapshot(&self, inner: &HostInner) -> HostStatus {
        let hosting = self.forced_open || inner.listener.is_some();
        let upnp = if hosting { inner.upnp } else { UpnpStatus::Off };
        let message = if self.forced_open {
            FORCED_OPEN_MESSAGE.to_string()
        } else {
            upnp.message().to_string()
        };
        HostStatus {
            hosting,
            port: self.reported_port(),
            lan: lan_ipv4_addresses(),
            public: if hosting {
                inner.public.map(|ip| ip.to_string())
            } else {
                None
            },
            upnp,
            message,
        }
    }

    /// Build the JSON status on failure to open the hosting port.
    fn bind_failure(&self, e: &std::io::Error) -> HostStatus {
        HostStatus {
            hosting: false,
            port: self.host_port,
            lan: lan_ipv4_addresses(),
            public: None,
            upnp: UpnpStatus::Off,
            message: format!("Could not open port {} so friends can join ({e}).", self.host_port),
        }
    }

    /// Current hosting status.
    pub async fn status(&self) -> HostStatus {
        let inner = self.inner.lock().await;
        self.snapshot(&inner)
    }

    /// Register with the matchmaker when hosting was forced open at startup (the Render
    /// deployment): a forced-open server never calls [`HostManager::start_hosting`], so
    /// this is the hook that lists it (`joinable: true`, address from `PUBLIC_ORIGIN`).
    /// A no-op otherwise; never fails the server.
    pub async fn register_if_forced_open(self: &Arc<Self>) {
        if self.forced_open {
            self.matchmaker_settle().await;
        }
    }

    /// Start the hosting listener (no-op when already hosting or when forced open).
    /// `app` must serve the same router as the main listener, with the same game.
    pub async fn start_hosting(self: &Arc<Self>, app: axum::Router) -> HostStatus {
        if self.forced_open {
            return self.status().await;
        }
        {
            let inner = self.inner.lock().await;
            if inner.listener.is_some() {
                return self.snapshot(&inner);
            }
        }
        let addr = SocketAddr::new(IpAddr::V4(Ipv4Addr::UNSPECIFIED), self.host_port);
        let listener = match tokio::net::TcpListener::bind(addr).await {
            Ok(listener) => listener,
            Err(e) => {
                tracing::error!("cannot start hosting on {addr}: {e}");
                return self.bind_failure(&e);
            }
        };
        let (shutdown_tx, shutdown_rx) = tokio::sync::oneshot::channel::<()>();
        let task = tokio::spawn(async move {
            let server = axum::serve(listener, app.into_make_service_with_connect_info::<SocketAddr>())
                .with_graceful_shutdown(async move {
                    let _ = shutdown_rx.await;
                });
            if let Err(e) = server.await {
                tracing::error!("hosting listener error: {e}");
            }
        });
        {
            let mut inner = self.inner.lock().await;
            inner.shutdown = Some(shutdown_tx);
            inner.listener = Some(task);
            inner.upnp = UpnpStatus::Trying;
            inner.public = None;
        }
        tracing::info!(
            "hosting listener on 0.0.0.0:{} (LAN); asking the router for a UPnP mapping",
            self.host_port
        );
        let manager = Arc::clone(self);
        tokio::spawn(async move {
            manager.upnp_try().await;
        });
        self.status().await
    }

    /// Stop hosting: close the listener, remove the UPnP mapping (best effort) and delist
    /// from the matchmaker (best effort; the request is time-bounded and shutdown is never blocked).
    pub async fn stop_hosting(&self) -> HostStatus {
        if self.forced_open {
            // Keep today's behaviour: the main listener stays open; only the report changes.
            return self.status().await;
        }
        let (shutdown, task, gateway, matchmaker_session) = {
            let mut inner = self.inner.lock().await;
            if let Some(renewal) = inner.renewal.take() {
                renewal.abort();
            }
            if let Some(matchmaker_task) = inner.matchmaker_task.take() {
                matchmaker_task.abort();
            }
            inner.upnp = UpnpStatus::Off;
            inner.public = None;
            let session = inner.matchmaker.take();
            (inner.shutdown.take(), inner.listener.take(), inner.gateway.take(), session)
        };
        if let Some(tx) = shutdown {
            let _ = tx.send(());
        }
        if let Some(mut handle) = task
            && tokio::time::timeout(HOST_STOP_TIMEOUT, &mut handle).await.is_err()
        {
            // A friend still has an open connection; close anyway.
            handle.abort();
            let _ = handle.await;
        }
        if let Some(gateway) = gateway {
            let _ = tokio::time::timeout(UPNP_REMOVE_TIMEOUT, upnp_close(gateway, self.host_port)).await;
        }
        if let Some(session) = matchmaker_session {
            if !self.matchmaker_client.delist(&session.id, &session.secret).await {
                tracing::debug!("matchmaker: delist did not confirm; the entry expires on its own");
            }
        }
        self.status().await
    }

    /// Ask the router to forward our hosting port to this machine's LAN address.
    /// Runs in the background; on success friends anywhere can join with `public`.
    async fn upnp_try(self: Arc<Self>) {
        let port = self.host_port;
        let result = upnp_open(port).await;
        let mut inner = self.inner.lock().await;
        if inner.listener.is_none() {
            // Hosting was stopped while we were asking the router; undo the mapping.
            drop(inner);
            if let Ok((_, gateway)) = result {
                let _ = tokio::time::timeout(UPNP_REMOVE_TIMEOUT, upnp_close(gateway, port)).await;
            }
            return;
        }
        match result {
            Ok((external, gateway)) => {
                inner.upnp = UpnpStatus::Open;
                inner.public = Some(external);
                inner.gateway = Some(gateway);
                if let Some(old) = inner.renewal.take() {
                    old.abort();
                }
                let manager = Arc::clone(&self);
                inner.renewal = Some(tokio::spawn(async move {
                    manager.upnp_renew_loop().await;
                }));
            }
            Err(e) => {
                tracing::warn!("UPnP setup failed: {e}");
                inner.upnp = UpnpStatus::Failed;
                inner.public = None;
            }
        }
        // Hosting has settled (the router answered, or not): register with the matchmaker.
        drop(inner);
        self.matchmaker_settle().await;
    }

    /// The router only keeps the mapping for the lease (2 hours): re-open it every
    /// [`UPNP_RENEW_INTERVAL`] while it is open. A failed renewal marks UPnP as failed
    /// (the status message then tells the player) and flips the next heartbeat's
    /// `joinable` to false; LAN friends keep working.
    async fn upnp_renew_loop(self: Arc<Self>) {
        loop {
            tokio::time::sleep(UPNP_RENEW_INTERVAL).await;
            if self.inner.lock().await.listener.is_none() {
                return;
            }
            let result = upnp_open(self.host_port).await;
            let mut inner = self.inner.lock().await;
            if inner.listener.is_none() {
                return;
            }
            match result {
                Ok((external, gateway)) => {
                    tracing::info!("UPnP mapping renewed");
                    inner.upnp = UpnpStatus::Open;
                    inner.public = Some(external);
                    inner.gateway = Some(gateway);
                }
                Err(e) => {
                    tracing::warn!("UPnP renewal failed: {e}");
                    inner.upnp = UpnpStatus::Failed;
                    inner.public = None;
                    inner.renewal = None;
                    return;
                }
            }
        }
    }

    /// Spawn the registration/heartbeat task once, now that hosting is active (called when
    /// hosting settles, i.e. after the UPnP attempt for normal hosts, and at startup for
    /// forced-open hosts). A no-op when the task already runs or hosting is off.
    async fn matchmaker_settle(self: &Arc<Self>) {
        {
            let inner = self.inner.lock().await;
            if !self.matchmaker_active(&inner) || inner.matchmaker_task.is_some() {
                return;
            }
        }
        let mut inner = self.inner.lock().await;
        if !self.matchmaker_active(&inner) || inner.matchmaker_task.is_some() {
            return;
        }
        let manager = Arc::clone(self);
        inner.matchmaker_task = Some(tokio::spawn(async move {
            manager.matchmaker_loop().await;
        }));
    }

    /// True while this host should be listed on the matchmaker.
    fn matchmaker_active(&self, inner: &HostInner) -> bool {
        self.forced_open || inner.listener.is_some()
    }

    /// Registration/heartbeat loop. Registers first when there is no session, then heartbeats
    /// every [`HEARTBEAT_INTERVAL`]. An unreachable matchmaker is logged at debug and retried
    /// on the next tick; a 404/403 response drops the session so the next tick re-registers.
    /// Returns (and is never re-spawned by itself) once hosting is no longer active.
    async fn matchmaker_loop(self: Arc<Self>) {
        let mut interval = tokio::time::interval(HEARTBEAT_INTERVAL);
        loop {
            interval.tick().await;
            let (players, level, phase, joinable, port, address) = {
                let inner = self.inner.lock().await;
                if !self.matchmaker_active(&inner) {
                    return;
                }
                (
                    inner.game.players,
                    inner.game.level,
                    inner.game.phase.clone(),
                    matchmaker_joinable(self.forced_open, inner.upnp),
                    self.reported_port(),
                    self.registration_address(&inner),
                )
            };
            let session = self.inner.lock().await.matchmaker.clone();
            match session {
                None => {
                    let body = RegisterRequest {
                        name: MATCHMAKER_NAME.to_string(),
                        version: GAME_VERSION.to_string(),
                        port,
                        players,
                        max_players: MAX_PLAYERS,
                        level,
                        phase,
                        joinable,
                        address: address.clone(),
                    };
                    match self.matchmaker_client.register(&body).await {
                        Some(response) => {
                            let address = reconcile_echo(address.as_deref(), response.address.as_deref());
                            let mut inner = self.inner.lock().await;
                            inner.matchmaker = Some(MatchmakerSession {
                                id: response.id,
                                secret: response.secret,
                                address,
                            });
                            tracing::info!("registered with the matchmaker (joinable: {joinable})");
                        }
                        None => {
                            // Debug details were logged inside the client; the next beat registers again.
                        }
                    }
                }
                Some(session) => {
                    let body = HeartbeatRequest {
                        secret: session.secret.clone(),
                        players,
                        level,
                        phase,
                        joinable,
                    };
                    match self.matchmaker_client.heartbeat(&session.id, &body).await {
                        HeartbeatOutcome::Ok => {}
                        HeartbeatOutcome::Expired => {
                            tracing::debug!("matchmaker: entry expired or rejected; registering again on the next beat");
                            self.inner.lock().await.matchmaker = None;
                        }
                        HeartbeatOutcome::Unreachable => {}
                    }
                }
            }
        }
    }
}

/// One UPnP setup: find the gateway (3 s timeout), map TCP `port` for 2 hours
/// ("VOIDWRAP RUN") and read our public IP. The router maps the port to the device
/// that asked for the mapping, which is this machine's LAN address (the address used
/// to reach the gateway). Isolated in one function so the igd-next calls are easy to audit.
async fn upnp_open(port: u16) -> Result<(IpAddr, igd_next::aio::Gateway<igd_next::aio::tokio::Tokio>), String> {
    let mut options = igd_next::SearchOptions::default();
    options.timeout = Some(UPNP_SEARCH_TIMEOUT);
    let gateway = igd_next::aio::tokio::search_gateway(options)
        .await
        .map_err(|e| format!("gateway search failed: {e}"))?;
    let external = gateway
        .get_external_ip()
        .await
        .map_err(|e| format!("could not read the external IP: {e}"))?;
    // The router forwards to this computer's LAN address (the one used to reach the internet).
    let lan: std::net::Ipv4Addr = lan_ipv4_addresses()
        .first()
        .and_then(|a| a.parse().ok())
        .ok_or_else(|| "no local network address found".to_string())?;
    let lease = u32::try_from(UPNP_LEASE_SECS).unwrap_or(u32::MAX);
    gateway
        .add_port(
            igd_next::PortMappingProtocol::TCP,
            port,
            std::net::SocketAddr::from((lan, port)),
            lease,
            UPNP_DESCRIPTION,
        )
        .await
        .map_err(|e| format!("router refused the port mapping: {e}"))?;
    Ok((external, gateway))
}

/// Remove the UPnP port mapping (best effort).
async fn upnp_close(gateway: igd_next::aio::Gateway<igd_next::aio::tokio::Tokio>, port: u16) {
    if let Err(e) = gateway.remove_port(igd_next::PortMappingProtocol::TCP, port).await {
        tracing::warn!("could not remove the UPnP mapping for port {port}: {e}");
    }
}

//! VOIDWRAP RUN authoritative game server.
//!
//! Architecture: one game task owns all simulation state and runs the fixed 20 Hz
//! tick loop. Each WebSocket connection is a pair of small tasks that only exchange
//! bounded channel messages with the game task — no shared mutable state, no locks.
//!
//! Desktop builds (`--features desktop`) open a native window instead of a browser,
//! unless `--browser`, `--headless` or `PORT` are used; see `desktop.rs`.

use tokio::sync::mpsc;

/// Inbound queue from all connections to the game task.
const GAME_INBOX_CAPACITY: usize = 1024;

/// Everything `main` needs before choosing between the windowed and the normal path.
struct Startup {
    /// Address the normal listener binds.
    addr: SocketAddr,
    /// Built web client directory, if found.
    web_dir: Option<PathBuf>,
    /// First level of a new run (dev/QA aid).
    start_level: u32,
    /// True when the server itself listens on all interfaces (`VOIDWRAP_HOST=0.0.0.0` or `PORT` set).
    forced_open: bool,
    /// Port the hosting listener uses (`VOIDWRAP_HOST_PORT`, default 7879).
    host_port: u16,
}

/// Read the environment once and compute the listen address, web directory and ports.
fn startup() -> Startup {
    // Where to listen. Hosting platforms set PORT and need every interface; locally the default is this
    // computer only. VOIDWRAP_HOST=0.0.0.0 opens the game to other computers (LAN multiplayer).
    let hosted_port: Option<u16> = std::env::var("PORT").ok().and_then(|p| p.parse().ok());
    let port: u16 = hosted_port
        .or_else(|| std::env::var("VOIDWRAP_PORT").ok().and_then(|p| p.parse().ok()))
        .unwrap_or(7878);
    let default_host = if hosted_port.is_some() { "0.0.0.0" } else { "127.0.0.1" };
    let host: std::net::IpAddr = std::env::var("VOIDWRAP_HOST")
        .ok()
        .and_then(|h| h.parse().ok())
        .unwrap_or_else(|| default_host.parse().unwrap_or(std::net::IpAddr::from([127, 0, 0, 1])));
    let addr = SocketAddr::from((host, port));
    let forced_open = host.is_unspecified();
    // Dev/QA aid: start every run at a later level (e.g. to review each colour theme).
    let start_level: u32 = std::env::var("VOIDWRAP_START_LEVEL")
        .ok()
        .and_then(|v| v.parse().ok())
        .unwrap_or(1)
        .clamp(1, 10_000);
    let host_port = host::parse_host_port(std::env::var("VOIDWRAP_HOST_PORT").ok().as_deref());
    Startup {
        addr,
        web_dir: net::find_web_dir(),
        start_level,
        forced_open,
        host_port,
    }
}

fn main() {
    init_logging();
    let startup = startup();

    // Desktop window mode: server on a background thread, tao/wry window on the main thread.
    #[cfg(feature = "desktop")]
    {
        let args: Vec<String> = std::env::args().skip(1).collect();
        let port_env_set = std::env::var_os("PORT").is_some();
        if desktop::should_open_window(&args, port_env_set) {
            run_desktop(startup);
            return;
        }
    }

    let runtime = match tokio::runtime::Builder::new_multi_thread().enable_all().build() {
        Ok(runtime) => runtime,
        Err(e) => {
            tracing::error!("could not create the tokio runtime: {e}");
            std::process::exit(1);
        }
    };
    if let Err(e) = runtime.block_on(run_server(startup, None)) {
        tracing::error!("fatal: {e}");
        std::process::exit(1);
    }
}

/// Desktop path: server on a background thread; window on the main thread; closing the window
/// shuts the server down (best effort) and then exits.
#[cfg(feature = "desktop")]
fn run_desktop(startup: Startup) {
    let port = startup.addr.port();
    let (shutdown_tx, shutdown_rx) = tokio::sync::oneshot::channel::<()>();
    // The window waits on this after a close so cleanup (UPnP removal) finishes: tao exits the
    // process as soon as its event loop ends, so code after open_window may never run.
    let (done_tx, done_rx) = std::sync::mpsc::channel::<()>();
    let server_thread = std::thread::Builder::new()
        .name("voidwrap-server".to_string())
        .spawn(move || {
            let runtime = match tokio::runtime::Builder::new_multi_thread().enable_all().build() {
                Ok(runtime) => runtime,
                Err(e) => {
                    tracing::error!("could not create the tokio runtime: {e}");
                    return;
                }
            };
            if let Err(e) = runtime.block_on(run_server(startup, Some(shutdown_rx))) {
                tracing::error!("server stopped with an error: {e}");
            }
            let _ = done_tx.send(());
        });
    let server_thread = match server_thread {
        Ok(thread) => Some(thread),
        Err(e) => {
            tracing::error!("could not start the server thread: {e}");
            None
        }
    };
    desktop::wait_for_server(port, std::time::Duration::from_secs(10));
    desktop::open_window(port, shutdown_tx, done_rx);
    // Best effort: let the server finish graceful shutdown (removes any UPnP mapping).
    if let Some(thread) = server_thread {
        let _ = thread.join();
    }
}

/// Run the server until the shutdown signal (Ctrl-C, or the desktop window closing).
/// Returns an error message when startup fails.
async fn run_server(startup: Startup, window_closed: Option<tokio::sync::oneshot::Receiver<()>>) -> Result<(), String> {
    let Startup {
        addr,
        web_dir,
        start_level,
        forced_open,
        host_port,
    } = startup;

    if start_level != 1 {
        tracing::warn!("VOIDWRAP_START_LEVEL={start_level}: runs start at level {start_level} (dev/QA only)");
    }
    let (game_tx, game_rx) = mpsc::channel(GAME_INBOX_CAPACITY);
    // Seed draft offers and level layouts from the clock: every run is different.
    let seed = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .map(|d| d.subsec_nanos() ^ (d.as_secs() as u32))
        .unwrap_or(0x5eed);
    let state = game::Game::starting_at(start_level, seed).map_err(|e| format!("cannot start: {e}"))?;
    let game = tokio::spawn(game::run(game_rx, state));

    let listener = tokio::net::TcpListener::bind(addr)
        .await
        .map_err(|e| format!("cannot bind {addr}: {e}"))?;
    tracing::info!("VOIDWRAP RUN server listening on ws://{addr}/ws");
    match &web_dir {
        Some(dir) => {
            let shown = if addr.ip().is_unspecified() || addr.ip().is_loopback() {
                "localhost".to_string()
            } else {
                addr.ip().to_string()
            };
            tracing::info!("serving the game from {}", dir.display());
            tracing::info!(">>> open http://{shown}:{} in your browser to play <<<", addr.port());
            if std::env::var_os("VOIDWRAP_OPEN_BROWSER").is_some() {
                open_browser(&format!("http://localhost:{}", addr.port()));
            }
        }
        None => tracing::info!("no built web client found (dev mode): run the web dev server separately"),
    }

    let host_manager = Arc::new(host::HostManager::new(addr.port(), host_port, forced_open));
    {
        let status = host_manager.status().await;
        if let Ok(json) = serde_json::to_string(&status) {
            tracing::info!("host status at startup: {json}");
        }
        if status.hosting {
            tracing::info!(
                "hosting is ON: friends can reach this computer on port {} (started with VOIDWRAP_HOST=0.0.0.0)",
                status.port
            );
        } else {
            tracing::info!("hosting is off: only this computer can play. Turn it on in the game.");
        }
    }
    // Forced-open hosts (the Render deployment) are listed on the matchmaker from the start;
    // normal hosts register when the hosting toggle settles. Never fails the server.
    host_manager.register_if_forced_open().await;

    let app = net::router(game_tx, web_dir, Arc::clone(&host_manager));
    let serve = axum::serve(listener, app.into_make_service_with_connect_info::<SocketAddr>())
        .with_graceful_shutdown(shutdown_signal(window_closed));
    if let Err(e) = serve.await {
        tracing::error!("server error: {e}");
    }
    // Best effort: close the hosting listener and remove any UPnP mapping before exit.
    let status = host_manager.stop_hosting().await;
    if let Ok(json) = serde_json::to_string(&status) {
        tracing::info!("host status at shutdown: {json}");
    }
    game.abort();
    Ok(())
}

/// Wait for Ctrl-C, or for the desktop window to close.
async fn shutdown_signal(window_closed: Option<tokio::sync::oneshot::Receiver<()>>) {
    match window_closed {
        Some(rx) => {
            tokio::select! {
                res = tokio::signal::ctrl_c() => {
                    if let Err(e) = res {
                        tracing::warn!("ctrl-c handler failed: {e}");
                    } else {
                        tracing::info!("shutting down (Ctrl-C)");
                    }
                }
                _ = rx => tracing::info!("the window was closed; shutting down"),
            }
        }
        None => {
            let _ = tokio::signal::ctrl_c().await;
            tracing::info!("shutting down");
        }
    }
}

/// Windows desktop builds have no console; log to `voidwrap.log` next to the executable so
/// problems can be reported. The file is truncated on each start.
#[cfg(all(feature = "desktop", target_os = "windows"))]
fn init_logging() {
    let filter = EnvFilter::try_from_default_env().unwrap_or_else(|_| EnvFilter::new("info"));
    let path = std::env::current_exe()
        .ok()
        .and_then(|exe| exe.parent().map(|dir| dir.join("voidwrap.log")))
        .unwrap_or_else(|| PathBuf::from("voidwrap.log"));
    match std::fs::File::create(&path) {
        Ok(file) => {
            let shared = std::sync::Arc::new(std::sync::Mutex::new(file));
            tracing_subscriber::fmt()
                .with_env_filter(filter)
                .with_ansi(false)
                .with_writer(move || SharedFileWriter(std::sync::Arc::clone(&shared)))
                .init();
            tracing::info!("logging to {}", path.display());
        }
        Err(e) => {
            tracing_subscriber::fmt()
                .with_env_filter(filter)
                .with_writer(std::io::stderr)
                .init();
            tracing::warn!("could not open {} for logging ({e}); logging to stderr", path.display());
        }
    }
}

/// `std::io::Write` adapter around a shared file handle, for tracing's `with_writer`.
#[cfg(all(feature = "desktop", target_os = "windows"))]
struct SharedFileWriter(std::sync::Arc<std::sync::Mutex<std::fs::File>>);

#[cfg(all(feature = "desktop", target_os = "windows"))]
impl std::io::Write for SharedFileWriter {
    fn write(&mut self, buf: &[u8]) -> std::io::Result<usize> {
        match self.0.lock() {
            Ok(mut file) => std::io::Write::write(&mut *file, buf),
            Err(_) => Err(std::io::Error::other("voidwrap.log lock poisoned")),
        }
    }
    fn flush(&mut self) -> std::io::Result<()> {
        match self.0.lock() {
            Ok(mut file) => std::io::Write::flush(&mut *file),
            Err(_) => Err(std::io::Error::other("voidwrap.log lock poisoned")),
        }
    }
}

/// Best-effort: open the game in the default browser (used by the PLAY launchers). Failure is harmless.

        std::process::Command::new("cmd").args(["/C", "start", "", url]).spawn()
    } else if cfg!(target_os = "macos") {
        std::process::Command::new("open").arg(url).spawn()
    } else {
        std::process::Command::new("xdg-open").arg(url).spawn()
    };
    if let Err(e) = result {
        tracing::warn!("could not open a browser ({e}); open {url} yourself");
    }
}
