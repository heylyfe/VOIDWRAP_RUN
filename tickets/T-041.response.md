//! Pure, deterministic game rules: gift pickups, the beacon, and enemy behaviours.
//!
//! Everything here is a plain function over protocol data so the authoritative server
//! and the client-side predictor can run the exact same code. Only `+ - * /` and
//! `sqrt` are used, so results are bit-identical on every target.

use protocol::{ENEMY_DISGUISED, ENEMY_DORMANT, ENEMY_HUNTING, Gift, TICK_HZ, Vec3};

use crate::controller::player_box;

/// A gift is collected when the player centre is within this distance of it.
pub const GIFT_RADIUS: f32 = 1.1;

/// Horizontal radius in which the beacon counts as reached.
pub const BEACON_RADIUS: f32 = 1.5;

/// Vertical tolerance for standing on the beacon's platform.
pub const BEACON_HEIGHT: f32 = 1.0;

/// Enemy kind tag for the flying chaser.
pub const ENEMY_KIND_CHASER: u8 = 1;

/// Chaser flight speed in units per second (the player runs at 9.0).
pub const CHASER_SPEED: f32 = 5.5;

/// A chaser only pursues targets within this 3D distance.
pub const CHASER_AGGRO_RADIUS: f32 = 22.0;

/// Body radius of the default chaser (the hit sphere is this times [`ENEMY_HIT_FORGIVENESS`]).
pub const CHASER_HALF_SIZE: f32 = 0.5;

/// Ticks a freshly spawned chaser stays dormant (3 s at 20 Hz).
pub const CHASER_GRACE_TICKS: u32 = 60;

/// Behaviour tag: flying chaser (the M1 default).
pub const BEHAVIOR_CHASER: u8 = 0;

/// Behaviour tag: stationary turret that fires projectiles at players in range.
pub const BEHAVIOR_TURRET: u8 = 1;

/// Behaviour tag: gift mimic that reveals itself when a player gets close.
pub const BEHAVIOR_MIMIC: u8 = 2;

/// Behaviour tag: chaser that corrupts the nearest valid platform to crumble.
pub const BEHAVIOR_CORRUPTER: u8 = 3;

/// Default turret firing cadence in ticks (1 s at 20 Hz).
pub const TURRET_FIRE_INTERVAL_TICKS: u32 = 20;

/// Default turret projectile speed in units per second.
pub const TURRET_PROJECTILE_SPEED: f32 = 12.0;

/// Default distance at which a mimic reveals itself.
pub const MIMIC_REVEAL_RADIUS: f32 = 3.0;

/// Default corrupter corruption cadence in ticks (2 s at 20 Hz).
pub const CORRUPTER_INTERVAL_TICKS: u32 = 40;

/// Height of the player's centre above their feet.
const PLAYER_CENTRE_HEIGHT: f32 = 0.9;

/// Per-enemy tuning: flight speed, aggro radius, body radius and grace ticks.
///
/// [`EnemyParams::default`] is exactly the M1 chaser, so an enemy spawned without
/// explicit params behaves as it always did.
#[derive(Debug, Clone, PartialEq)]
pub struct EnemyParams {
    /// Flight speed in units per second.
    pub speed: f32,
    /// 3D distance within which the enemy pursues a target.
    pub aggro: f32,
    /// Body radius: the client draws the enemy this big; see [`enemy_hit_radius`].
    pub size: f32,
    /// Ticks the enemy stays dormant after spawning (does not apply to mimics).
    pub grace_ticks: u32,
    /// Which behaviour drives this enemy (`BEHAVIOR_*`).
    pub behavior: u8,
    /// Ticks between shots for `BEHAVIOR_TURRET`.
    pub fire_interval_ticks: u32,
    /// Projectile speed in units per second for `BEHAVIOR_TURRET`.
    pub projectile_speed: f32,
    /// Distance within which a `BEHAVIOR_MIMIC` reveals itself.
    pub reveal_radius: f32,
    /// Ticks between platform corruption for `BEHAVIOR_CORRUPTER`.
    pub corrupt_interval_ticks: u32,
}

impl Default for EnemyParams {
    fn default() -> Self {
        Self {
            speed: CHASER_SPEED,
            aggro: CHASER_AGGRO_RADIUS,
            size: CHASER_HALF_SIZE,
            grace_ticks: CHASER_GRACE_TICKS,
            behavior: BEHAVIOR_CHASER,
            fire_interval_ticks: TURRET_FIRE_INTERVAL_TICKS,
            projectile_speed: TURRET_PROJECTILE_SPEED,
            reveal_radius: MIMIC_REVEAL_RADIUS,
            corrupt_interval_ticks: CORRUPTER_INTERVAL_TICKS,
        }
    }
}

/// A single enemy. `home` is where it drifts back to when nothing is in range.
#[derive(Debug, Clone, PartialEq)]
pub struct Enemy {
    pub id: u32,
    pub kind: u8,
    pub pos: Vec3,
    pub home: Vec3,
    pub active_in: u32,
    pub params: EnemyParams,
    /// Ticks until a turret may fire / a corrupter may corrupt again (counting down).
    pub cooldown_ticks: u32,
    /// Mimics only: set once a target has got close enough to reveal the enemy for good.
    pub revealed: bool,
}

/// Euclidean distance between two points.
pub(crate) fn distance(a: &Vec3, b: &Vec3) -> f32 {
    let dx = a.x - b.x;
    let dy = a.y - b.y;
    let dz = a.z - b.z;
    (dx * dx + dy * dy + dz * dz).sqrt()
}

/// The player's centre point, derived from their feet position.
pub(crate) fn player_centre(player_feet: &Vec3) -> Vec3 {
    Vec3::new(player_feet.x, player_feet.y + PLAYER_CENTRE_HEIGHT, player_feet.z)
}

/// Move `pos` toward `target` by at most `max_step`, never overshooting.
pub(crate) fn move_toward(pos: &mut Vec3, target: &Vec3, max_step: f32) {
    let dx = target.x - pos.x;
    let dy = target.y - pos.y;
    let dz = target.z - pos.z;
    let d = (dx * dx + dy * dy + dz * dz).sqrt();
    if d <= max_step {
        *pos = target.clone();
        return;
    }
    let scale = max_step / d;
    pos.x += dx * scale;
    pos.y += dy * scale;
    pos.z += dz * scale;
}

/// Index of the nearest target within `aggro`, or `None`. Ties go to the lowest index.
pub(crate) fn nearest_target_within(pos: &Vec3, targets: &[Vec3], aggro: f32) -> Option<usize> {
    let mut nearest: Option<(usize, f32)> = None;
    for (index, target) in targets.iter().enumerate() {
        let d = distance(pos, target);
        if d > aggro {
            continue;
        }
        match nearest {
            Some((_, best)) if best <= d => {}
            _ => nearest = Some((index, d)),
        }
    }
    nearest.map(|(index, _)| index)
}

/// True when any target is within `radius` of `pos`.
pub(crate) fn target_within(pos: &Vec3, targets: &[Vec3], radius: f32) -> bool {
    targets.iter().any(|target| distance(pos, target) <= radius)
}

/// Chaser-family movement shared by chaser/mimic/corrupter: while grace lasts only
/// count down; afterwards fly at the nearest target in aggro, else drift home at
/// half speed.
pub(crate) fn step_hunter(enemy: &mut Enemy, targets: &[Vec3]) {
    if enemy.active_in > 0 {
        enemy.active_in -= 1;
        return;
    }
    let step = enemy.params.speed / TICK_HZ as f32;
    let aggro = enemy.params.aggro;
    match nearest_target_within(&enemy.pos, targets, aggro) {
        Some(index) => {
            let target = targets[index].clone();
            move_toward(&mut enemy.pos, &target, step);
        }
        None => {
            let home = enemy.home.clone();
            move_toward(&mut enemy.pos, &home, step * 0.5);
        }
    }
}

/// Behaviour stepping shared by [`step_enemy`] and `crate::hazards`: turret grace
/// countdown, mimic reveal and chaser-family movement. Firing turrets and
/// corrupting corrupters need level context and are handled in `crate::hazards`.
/// Unknown behaviours do nothing.
pub(crate) fn step_behavior(enemy: &mut Enemy, targets: &[Vec3]) {
    match enemy.params.behavior {
        BEHAVIOR_TURRET => {
            if enemy.active_in > 0 {
                enemy.active_in -= 1;
            }
        }
        BEHAVIOR_MIMIC => {
            if !enemy.revealed {
                if target_within(&enemy.pos, targets, enemy.params.reveal_radius) {
                    enemy.revealed = true;
                } else {
                    return;
                }
            }
            step_hunter(enemy, targets);
        }
        BEHAVIOR_CORRUPTER | BEHAVIOR_CHASER => step_hunter(enemy, targets),
        _ => {}
    }
}

/// True when the player's centre is within [`GIFT_RADIUS`] of the gift centre.
pub fn gift_touched(player_feet: &Vec3, gift: &Gift) -> bool {
    gift_touched_within(player_feet, gift, GIFT_RADIUS)
}

/// True when the player's centre is within `radius` of the gift centre.
pub fn gift_touched_within(player_feet: &Vec3, gift: &Gift, radius: f32) -> bool {
    distance(&player_centre(player_feet), &gift.pos) <= radius
}

/// True when the player stands on the beacon: close horizontally and roughly level.
pub fn beacon_reached(player_feet: &Vec3, beacon: &Vec3) -> bool {
    beacon_reached_within(player_feet, beacon, BEACON_RADIUS)
}

/// True when the player is within `radius` horizontally of the beacon and roughly
/// level with it. The vertical tolerance stays [`BEACON_HEIGHT`].
pub fn beacon_reached_within(player_feet: &Vec3, beacon: &Vec3, radius: f32) -> bool {
    let dx = player_feet.x - beacon.x;
    let dz = player_feet.z - beacon.z;
    let horizontal = (dx * dx + dz * dz).sqrt();
    horizontal <= radius && (player_feet.y - beacon.y).abs() <= BEACON_HEIGHT
}

/// Create an enemy sitting at `pos`, with `pos` as its home, default params and a
/// full grace period (mimics never get one).
pub fn spawn_enemy(id: u32, kind: u8, pos: Vec3) -> Enemy {
    spawn_enemy_with(id, kind, pos, EnemyParams::default())
}

/// Create an enemy with explicit [`EnemyParams`]; `active_in` starts at
/// `params.grace_ticks`, except for mimics which never wait.
pub fn spawn_enemy_with(id: u32, kind: u8, pos: Vec3, params: EnemyParams) -> Enemy {
    let active_in = if params.behavior == BEHAVIOR_MIMIC {
        0
    } else {
        params.grace_ticks
    };
    Enemy {
        id,
        kind,
        home: pos.clone(),
        pos,
        active_in,
        params,
        cooldown_ticks: 0,
        revealed: false,
    }
}

/// The protocol state to report for `enemy`: `ENEMY_DISGUISED` for an unrevealed
/// mimic, `ENEMY_DORMANT` while grace lasts (and for unknown behaviours), else
/// `ENEMY_HUNTING`.
pub fn enemy_state(enemy: &Enemy) -> u8 {
    match enemy.params.behavior {
        BEHAVIOR_MIMIC => {
            if enemy.revealed {
                ENEMY_HUNTING
            } else {
                ENEMY_DISGUISED
            }
        }
        BEHAVIOR_CHASER | BEHAVIOR_TURRET | BEHAVIOR_CORRUPTER => {
            if enemy.active_in > 0 {
                ENEMY_DORMANT
            } else {
                ENEMY_HUNTING
            }
        }
        _ => ENEMY_DORMANT,
    }
}

/// Advance one enemy by one server tick (`dt = 1 / TICK_HZ`).
///
/// `targets` are the centres of living players. While `active_in > 0` the enemy only
/// counts down. Afterwards it flies straight at the nearest target inside
/// `enemy.params.aggro` (ties go to the lowest index), or drifts home at half speed.
/// Unknown kinds (and unknown behaviours) do nothing at all.
///
/// Turret firing and corrupter platform marking need level context; the server must
/// call `crate::hazards::Hazards::step`, which dispatches behaviour for every enemy
/// (regardless of `kind`) and replaces this function for all of them.
pub fn step_enemy(enemy: &mut Enemy, targets: &[Vec3]) {
    if enemy.kind != ENEMY_KIND_CHASER {
        return;
    }
    step_behavior(enemy, targets);
}

/// Fraction of the drawn body radius that actually kills. Slightly under 1 so a hit
/// always looks like a hit: grazing the glowing edge of the model is survivable.
pub const ENEMY_HIT_FORGIVENESS: f32 = 0.85;

/// Radius of the lethal sphere around the enemy centre.
pub fn enemy_hit_radius(enemy: &Enemy) -> f32 {
    enemy.params.size * ENEMY_HIT_FORGIVENESS
}

/// True when the enemy's hit sphere (see [`enemy_hit_radius`]) overlaps the player's
/// collision box. The sphere matches the drawn body, so there are no invisible corners.
pub fn enemy_touches(enemy: &Enemy, player_feet: &Vec3) -> bool {
    let b = player_box(player_feet);
    let c = &enemy.pos;
    let dx = c.x - c.x.clamp(b.min.x, b.max.x);
    let dy = c.y - c.y.clamp(b.min.y, b.max.y);
    let dz = c.z - c.z.clamp(b.min.z, b.max.z);
    let r = enemy_hit_radius(enemy);
    dx * dx + dy * dy + dz * dz <= r * r
}

//! Deterministic enemy hazards for M2: turret projectiles and crumbling tiles.
//!
//! Owns all projectile and tile state and advances it once per server tick, alongside
//! the enemy behaviours from [`crate::rules`]. `Hazards::step` steps enemy movement for
//! every enemy (chaser, turret, mimic and corrupter) and replaces the server's direct
//! call to [`crate::rules::step_enemy`]. Only `+ - * /`, `sqrt` and `clamp` are used on
//! floats, so results are bit-identical on every target.

use protocol::{
    DEATH_ENEMY, DEATH_PROJECTILE, ENEMY_HUNTING, LevelData, PLATFORM_CRUMBLE, Platform,
    ProjectileView, TILE_GONE, TILE_WARN, TICK_HZ, TileState, Vec3,
};

use crate::controller::player_box;
use crate::rules::{
    BEACON_HEIGHT, BEHAVIOR_CORRUPTER, BEHAVIOR_TURRET, Enemy, enemy_state, enemy_touches,
    nearest_target_within, player_centre, step_behavior,
};
use crate::world::World;

/// Ticks a projectile survives after being fired before it is removed (4 s at 20 Hz).
pub const PROJECTILE_LIFETIME_TICKS: u32 = 80;

/// Projectiles kill when they come within this distance of the closest point on the player box.
pub const PROJECTILE_RADIUS: f32 = 0.35;

/// Ticks a platform stays `TILE_WARN` before going `TILE_GONE` (0.6 s at 20 Hz).
pub const TILE_WARN_TICKS: u32 = 12;

/// Ticks a platform stays `TILE_GONE` before returning to normal (3 s at 20 Hz).
pub const TILE_GONE_TICKS: u32 = 60;

/// Most projectiles alive at once; extra shots are dropped on the floor.
pub const MAX_PROJECTILES: usize = 64;

/// Feet must be within this height of the platform top to count as standing on it.
const STAND_HEIGHT_TOLERANCE: f32 = 0.05;

/// A flying projectile. `ticks_alive` counts from 0 on the firing tick.
#[derive(Debug, Clone, PartialEq)]
struct Projectile {
    id: u32,
    pos: Vec3,
    vel: Vec3,
    ticks_alive: u32,
}

/// A platform that is currently `TILE_WARN` or `TILE_GONE`, with ticks left in that state.
#[derive(Debug, Clone, PartialEq)]
struct TileEntry {
    id: u16,
    state: u8,
    ticks_left: u32,
}

/// Owns projectiles and tile states. Call [`Hazards::step`] once per server tick.
///
/// `new`/`reset` produce an empty state (fresh level); `tiles` only reports
/// non-normal platforms, sorted by id; `projectiles` reports live projectiles in id
/// order. Projectile ids increase monotonically from 0 and wrap around `u32`.
#[derive(Debug, Default)]
pub struct Hazards {
    projectiles: Vec<Projectile>,
    tiles: Vec<TileEntry>,
    next_projectile_id: u32,
}

impl Hazards {
    /// Create an empty `Hazards`.
    pub fn new() -> Self {
        Self::default()
    }

    /// Drop all projectiles and tile state. Call when the level changes.
    pub fn reset(&mut self) {
        self.projectiles.clear();
        self.tiles.clear();
        self.next_projectile_id = 0;
    }

    /// Advance tiles, enemies, projectiles and crumble detection by one server tick.
    ///
    /// `players` are the feet positions of the living players; everything the step
    /// function needs (target centres, aim points) is derived from them. Enemy
    /// behaviour and movement for all enemies is stepped here; this replaces the
    /// server's call to [`crate::rules::step_enemy`]. `world` is consulted for the
    /// current set of solid platforms (projectiles removed inside them); `level`
    /// supplies the platforms (ids, kinds, bounds, beacon, kill_y).
    pub fn step(&mut self, enemies: &mut [Enemy], world: &World, level: &LevelData, players: &[Vec3]) {
        // 1. Age existing tile state so newly marked platforms always get their full duration.
        self.age_tiles();

        // 2. Enemies: behaviour movement/timers, turret firing, corrupter marking.
        let centres: Vec<Vec3> = players.iter().map(player_centre).collect();
        for enemy in enemies.iter_mut() {
            step_behavior(enemy, &centres);
            match enemy.params.behavior {
                BEHAVIOR_TURRET => step_turret(self, enemy, &centres),
                BEHAVIOR_CORRUPTER => step_corrupter(self, enemy, level),
                _ => {}
            }
        }

        // 3. Projectiles: straight-line flight, then removal for age, void or walls.
        let dt = 1.0 / TICK_HZ as f32;
        for p in self.projectiles.iter_mut() {
            p.pos.x += p.vel.x * dt;
            p.pos.y += p.vel.y * dt;
            p.pos.z += p.vel.z * dt;
            p.ticks_alive += 1;
        }
        let kill_y = level.kill_y;
        self.projectiles.retain(|p| {
            p.pos.y >= kill_y
                && p.ticks_alive < PROJECTILE_LIFETIME_TICKS
                && !point_in_solid(&p.pos, world)
        });

        // 4. Players standing on a crumbling platform warn it.
        for platform in &level.platforms {
            if platform.kind != PLATFORM_CRUMBLE {
                continue;
            }
            if self.tiles.iter().any(|t| t.id == platform.id) {
                continue;
            }
            let top = platform.bounds.max.y;
            let standing = players.iter().any(|feet| {
                feet.x >= platform.bounds.min.x
                    && feet.x <= platform.bounds.max.x
                    && feet.z >= platform.bounds.min.z
                    && feet.z <= platform.bounds.max.z
                    && (feet.y - top).abs() <= STAND_HEIGHT_TOLERANCE
            });
            if standing {
                self.tiles.push(TileEntry {
                    id: platform.id,
                    state: TILE_WARN,
                    ticks_left: TILE_WARN_TICKS,
                });
            }
        }
    }

    /// Deaths caused by this hazard set for a player standing at `feet`.
    ///
    /// `Some(DEATH_ENEMY)` for an active, non-disguised enemy touch (checked first),
    /// otherwise `Some(DEATH_PROJECTILE)` for a projectile within [`PROJECTILE_RADIUS`]
    /// of the player box, otherwise `None`.
    pub fn hit(&self, enemies: &[Enemy], feet: &Vec3) -> Option<u8> {
        for enemy in enemies {
            if enemy_state(enemy) == ENEMY_HUNTING && enemy_touches(enemy, feet) {
                return Some(DEATH_ENEMY);
            }
        }
        let b = player_box(feet);
        for p in &self.projectiles {
            let cx = p.pos.x.clamp(b.min.x, b.max.x);
            let cy = p.pos.y.clamp(b.min.y, b.max.y);
            let cz = p.pos.z.clamp(b.min.z, b.max.z);
            let dx = p.pos.x - cx;
            let dy = p.pos.y - cy;
            let dz = p.pos.z - cz;
            if dx * dx + dy * dy + dz * dz <= PROJECTILE_RADIUS * PROJECTILE_RADIUS {
                return Some(DEATH_PROJECTILE);
            }
        }
        None
    }

    /// Views of all live projectiles, in spawn (id) order.
    pub fn projectiles(&self) -> Vec<ProjectileView> {
        self.projectiles
            .iter()
            .map(|p| ProjectileView { id: p.id, pos: p.pos.clone() })
            .collect()
    }

    /// Views of every non-normal platform (`TILE_WARN` / `TILE_GONE`), sorted by id.
    pub fn tiles(&self) -> Vec<TileState> {
        let mut out: Vec<TileState> = self
            .tiles
            .iter()
            .map(|t| TileState { id: t.id, state: t.state })
            .collect();
        out.sort_by_key(|t| t.id);
        out
    }

    /// Age existing tile entries one tick; WARN -> GONE after [`TILE_WARN_TICKS`],
    /// then removed entirely (back to normal) after [`TILE_GONE_TICKS`].
    fn age_tiles(&mut self) {
        let mut i = 0;
        while i < self.tiles.len() {
            self.tiles[i].ticks_left -= 1;
            if self.tiles[i].ticks_left == 0 {
                if self.tiles[i].state == TILE_WARN {
                    self.tiles[i].state = TILE_GONE;
                    self.tiles[i].ticks_left = TILE_GONE_TICKS;
                    i += 1;
                } else {
                    self.tiles.remove(i);
                }
            } else {
                i += 1;
            }
        }
    }

    /// Push a new projectile if at most [`MAX_PROJECTILES`] are alive; ids wrap around.
    fn spawn_projectile(&mut self, pos: Vec3, vel: Vec3) {
        if self.projectiles.len() >= MAX_PROJECTILES {
            return;
        }
        let id = self.next_projectile_id;
        self.next_projectile_id = self.next_projectile_id.wrapping_add(1);
        self.projectiles.push(Projectile { id, pos, vel, ticks_alive: 0 });
    }

    /// Mark `id` WARN unless it is already non-normal.
    fn mark_warn(&mut self, id: u16) {
        if self.tiles.iter().any(|t| t.id == id) {
            return;
        }
        self.tiles.push(TileEntry { id, state: TILE_WARN, ticks_left: TILE_WARN_TICKS });
    }
}

/// True when `p` is inside any of the world's current solid platforms (inclusive).
fn point_in_solid(p: &Vec3, world: &World) -> bool {
    world.solids.iter().any(|b| {
        p.x >= b.min.x
            && p.x <= b.max.x
            && p.y >= b.min.y
            && p.y <= b.max.y
            && p.z >= b.min.z
            && p.z <= b.max.z
    })
}

/// True when the platform's top face contains the beacon: horizontally inside the
/// top rectangle and within [`BEACON_HEIGHT`] of its height.
fn platform_contains_beacon(platform: &Platform, beacon: &Vec3) -> bool {
    let top = platform.bounds.max.y;
    beacon.x >= platform.bounds.min.x
        && beacon.x <= platform.bounds.max.x
        && beacon.z >= platform.bounds.min.z
        && beacon.z <= platform.bounds.max.z
        && (beacon.y - top).abs() <= BEACON_HEIGHT
}

/// Id of the nearest platform whose top-face centre is closest to `pos`, excluding
/// platform id 0 (the spawn), the platform holding the beacon and any platform that
/// is not currently normal. Ties go to the platform appearing first in `level.platforms`.
fn nearest_corruptible(level: &LevelData, tiles: &[TileEntry], pos: &Vec3) -> Option<u16> {
    let mut best: Option<(u16, f32)> = None;
    for platform in &level.platforms {
        if platform.id == 0 {
            continue;
        }
        if tiles.iter().any(|t| t.id == platform.id) {
            continue;
        }
        if platform_contains_beacon(platform, &level.beacon) {
            continue;
        }
        let cx = (platform.bounds.min.x + platform.bounds.max.x) * 0.5;
        let cy = platform.bounds.max.y;
        let cz = (platform.bounds.min.z + platform.bounds.max.z) * 0.5;
        let dx = cx - pos.x;
        let dy = cy - pos.y;
        let dz = cz - pos.z;
        let d = (dx * dx + dy * dy + dz * dz).sqrt();
        let replace = match best {
            None => true,
            Some((_, best_d)) => d < best_d,
        };
        if replace {
            best = Some((platform.id, d));
        }
    }
    best.map(|(id, _)| id)
}

/// Turret behaviour: after grace, fire at the nearest target's centre (feet + 0.9 up)
/// every `fire_interval_ticks`. Target centres are passed in via `targets`.
fn step_turret(hazards: &mut Hazards, enemy: &mut Enemy, targets: &[Vec3]) {
    if enemy.active_in != 0 {
        return; // still in grace period
    }
    if enemy.cooldown_ticks > 0 {
        enemy.cooldown_ticks -= 1;
    }
    if enemy.cooldown_ticks != 0 {
        return; // not time yet
    }
    let Some(index) = nearest_target_within(&enemy.pos, targets, enemy.params.aggro) else {
        return; // nobody in range; cadence counter stays at zero
    };
    enemy.cooldown_ticks = enemy.params.fire_interval_ticks;
    let target = targets[index].clone();
    let dx = target.x - enemy.pos.x;
    let dy = target.y - enemy.pos.y;
    let dz = target.z - enemy.pos.z;
    let d = (dx * dx + dy * dy + dz * dz).sqrt();
    if d <= 0.0 {
        return; // target exactly on the turret: shot fizzles, cadence consumed
    }
    let scale = enemy.params.projectile_speed / d;
    hazards.spawn_projectile(enemy.pos.clone(), Vec3::new(dx * scale, dy * scale, dz * scale));
}

/// Corrupter behaviour: after grace, mark the nearest valid platform `TILE_WARN`
/// every `corrupt_interval_ticks`.
fn step_corrupter(hazards: &mut Hazards, enemy: &mut Enemy, level: &LevelData) {
    if enemy.active_in != 0 {
        return; // still in grace period
    }
    if enemy.cooldown_ticks > 0 {
        enemy.cooldown_ticks -= 1;
    }
    if enemy.cooldown_ticks != 0 {
        return; // not time yet
    }
    let Some(id) = nearest_corruptible(level, &hazards.tiles, &enemy.pos) else {
        return; // no valid platform; try again next tick
    };
    hazards.mark_warn(id);
    enemy.cooldown_ticks = enemy.params.corrupt_interval_ticks;
}

        // Target far away: stays disguised, sits still and is harmless on touch.
        let far = [feet(9.9, 0.0, 0.0)];
        hazards.step(&mut enemies, &world, &level, &far);
        assert_eq!(enemy_state(&enemies[0]), ENEMY_DISGUISED);
        assert_eq!(enemies[0].pos, feet(0.0, 0.0, 0.0));
        assert_eq!(hazards.hit(&enemies, &feet(0.0, 0.0, 0.0)), None);

        let near = [feet(1.0, 0.0, 0.0)];

        assert_eq!(enemy_state(&enemies[0]), ENEMY_HUNTING);

        // Target retreats: never disguises again.
        hazards.step(&mut enemies, &world, &level, &far);
        assert_eq!(enemy_state(&enemies[0]), ENEMY_HUNTING);

        // Revealed mimic on top of the player is lethal.
        enemies[0].pos = feet(0.0, 0.9, 0.0);
        assert_eq!(hazards.hit(&enemies, &feet(0.0, 0.0, 0.0)), Some(DEATH_ENEMY));

    #[test]
    fn tile_warn_gone_normal_cycle_timing() {
        let platforms = vec![platform(2, PLATFORM_CRUMBLE, [-5.0, 10.0, -5.0], [5.0, 11.0, 5.0])];
        let level = test_level(platforms, feet(50.0, 0.0, 50.0), -100.0);
        let world = World::from_level(&level);
        let mut enemies: Vec<Enemy> = Vec::new();
        let mut hazards = Hazards::new();

        let standing = [feet(0.0, 11.0, 0.0)];
        hazards.step(&mut enemies, &world, &level, &standing); // becomes WARN
        assert_eq!(hazards.tiles(), vec![TileState { id: 2, state: TILE_WARN }]);

        // 11 aging ticks keep it WARN (it was created with 12).
        for _ in 0..11 {
            hazards.step(&mut enemies, &world, &level, &[]);
        }
        assert_eq!(hazards.tiles(), vec![TileState { id: 2, state: TILE_WARN }]);

        // The 12th aging tick flips WARN -> GONE.
        hazards.step(&mut enemies, &world, &level, &[]);
        assert_eq!(hazards.tiles(), vec![TileState { id: 2, state: TILE_GONE }]);

        // 59 aging ticks keep it GONE; the 60th removes it entirely.
        for _ in 0..59 {
            hazards.step(&mut enemies, &world, &level, &[]);
        }
        assert_eq!(hazards.tiles(), vec![TileState { id: 2, state: TILE_GONE }]);

        hazards.step(&mut enemies, &world, &level, &[]);
        assert!(hazards.tiles().is_empty(), "back to normal");

        // With the player standing again, a returned-to-normal platform crumbles anew.
        hazards.step(&mut enemies, &world, &level, &standing);
        assert_eq!(hazards.tiles(), vec![TileState { id: 2, state: TILE_WARN }]);
    }
