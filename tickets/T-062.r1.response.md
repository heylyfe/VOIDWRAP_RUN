// Pure lookup + curve helpers for the reworked enemy roster (ticket T-057a).
// No three.js / React runtime imports — unit-testable in isolation.
import type { MutableRefObject } from "react";
import {
  ENEMY_ATTACK,
  ENEMY_TELEGRAPH,
} from "../protocol/generated";

/** Stable behaviour id (matches design/content.json enemy ids). */
export type EnemyBehaviorId =
  | "pallbearer"
  | "snapshrike"
  | "moteling"
  | "rift_warden"
  | "gale_wight"
  | "lantern_leech"
  | "turret"
  | "mimic"
  | "corrupter";

export interface EnemyLook {
  /** Behaviour id; also the roster order used for the wire `kind` index. */
  readonly behavior: EnemyBehaviorId;
  /** Primary accent: emissive trim + telegraph colour (design doc §2). */
  readonly accent: string;
  /** Secondary emissive: hot core / rim / inner glow. */
  readonly glow: string;
}

/** Order matches the wire `kind` index: pallbearer 0 … corrupter 8. */
export const ENEMY_LOOKS: readonly EnemyLook[] = [
  { behavior: "pallbearer", accent: "#ff3355", glow: "#ffdf9e" },
  { behavior: "snapshrike", accent: "#ff4436", glow: "#ffb36b" },
  { behavior: "moteling", accent: "#ffc3d9", glow: "#ff5fa8" },
  { behavior: "rift_warden", accent: "#b0313f", glow: "#ff5a3c" },
  { behavior: "gale_wight", accent: "#d94fff", glow: "#b47aff" },
  { behavior: "lantern_leech", accent: "#ff7ab8", glow: "#ffb3d1" },
  { behavior: "turret", accent: "#ff8a3c", glow: "#ffd07a" },
  { behavior: "mimic", accent: "#ff3df0", glow: "#b46bff" },
  { behavior: "corrupter", accent: "#39ffb0", glow: "#7dffb9" },
];

export const ENEMY_KIND_COUNT = ENEMY_LOOKS.length;

const FALLBACK_LOOK: EnemyLook = { behavior: "pallbearer", accent: "#ff3355", glow: "#ffdf9e" };

export function isEnemyKind(kind: number): boolean {
  return Number.isInteger(kind) && kind >= 0 && kind < ENEMY_LOOKS.length;
}

/** Unknown/missing kinds fall back to the pallbearer look (never crash a render). */
export function enemyLookFor(kind: number | undefined): EnemyLook {
  if (kind === undefined) return FALLBACK_LOOK;
  const look = ENEMY_LOOKS[kind];
  return look ?? FALLBACK_LOOK;
}

export function isTelegraphState(state: number | undefined): boolean {
  return state === ENEMY_TELEGRAPH || state === ENEMY_ATTACK;
}

export function clamp01(v: number): number {
  if (!Number.isFinite(v)) return 0;
  return v < 0 ? 0 : v > 1 ? 1 : v;
}

/** Wire `prog` (0–255) → 0..1. */
export function prog01(prog: number | undefined): number {
  if (prog === undefined || !Number.isFinite(prog)) return 0;
  return clamp01(prog / 255);
}

export function lerp(a: number, b: number, t: number): number {
  const c = clamp01(t);
  return a + (b - a) * c;
}

/* ---------- telegraph curves ---------- */

/** Alpha ramp shared by every telegraph: dim at prog 0, solid at 255. */
export function telegraphAlpha(t01: number): number {
  return 0.3 + 0.7 * clamp01(t01);
}

/** Moteling (Stillness Shrieker) INHALE: warning ring starts at orbit radius and collapses onto the victim. */
export const MOTELING_RING_START = 5.0;
export const MOTELING_RING_END = 0.45;
export function motelingRingRadius(t01: number): number {
  return lerp(MOTELING_RING_START, MOTELING_RING_END, t01);
}

/** Rings that grow outward as an attack resolves (shriek snap / warden shove / fret spread). */
export function expandingRingRadius(t01: number, maxRadius: number): number {
  return Math.max(0.25, clamp01(t01) * maxRadius);
}

/** Fixed threat radii from the design doc (§2). */

/** MIMIC CACHED tell: a 0.5 s magenta shimmer every 3.0 s. Purely client-clock driven. */

export const MIMIC_SHIMMER_WINDOW = 0.5;
function shimmerPhase(timeS: number): number {
  return ((timeS % MIMIC_SHIMMER_PERIOD) + MIMIC_SHIMMER_PERIOD) % MIMIC_SHIMMER_PERIOD;
}
export function mimicShimmerVisible(timeS: number): boolean {
  if (!Number.isFinite(timeS)) return false;
  return shimmerPhase(timeS) < MIMIC_SHIMMER_WINDOW;
}
/** 0 → peak → 0 across the shimmer window; 0 outside it. */
export function mimicShimmerOpacity(timeS: number): number {
  if (!mimicShimmerVisible(timeS)) return 0;
  return Math.sin((shimmerPhase(timeS) / MIMIC_SHIMMER_WINDOW) * Math.PI) * 0.9;
}

=== FN: server/crates/sim/src/levelgen.rs :: set_gift_spacing_mul ===
/// Set the process-wide gift-spacing multiplier (T-062, Clustered Gifts curse).
///
/// Multiplies `GIFT_MIN_SPACING` for gifts placed by subsequent generation calls.
/// `0.6` places gifts down to 1.2 u apart instead of 2.0 u, so gifts gather in denser
/// clusters instead of spreading evenly. Values are clamped to [0.25, 1.0] (the curse can
/// only cluster, never spread); non-finite or non-positive input resets the multiplier to 1.0.
///
/// Determinism contract: this is a single stored scalar, never combined with time, HashMap
/// iteration or extra randomness — the sim stays deterministic. Callers that set a
/// non-default multiplier must reset it with `set_gift_spacing_mul(1.0)` after generating
/// (the server's `Run::start_level` does) so other callers in the same process are unaffected.
/// Same seed + same multiplier -> identical output.
pub fn set_gift_spacing_mul(mul: f32) {
    let bits = if mul.is_finite() && mul > 0.0 {
        mul.clamp(GIFT_SPACING_MUL_MIN, 1.0).to_bits()
    } else {
        1.0_f32.to_bits()
    };
    GIFT_SPACING_MUL_BITS.store(bits, std::sync::atomic::Ordering::Relaxed);
}

=== FN: server/crates/sim/src/levelgen.rs :: accept_candidate ===
/// Accepts a gift candidate iff it stands on a NORMAL platform top (edge inset >= 0.5 u),
/// its body box does not overlap any wall, jump pad or ramp (any non-NORMAL platform volume),
/// it is >= the effective minimum spacing from other gifts horizontally and >= 2.5 u from
/// the beacon. Rejected candidates are not placed; the callers re-sample (the chunk-slot
/// passes skip the slot; the top-up pass samples a fresh random site), so a stuck slot never
/// yields a stuck gift.
#[allow(clippy::too_many_arguments)]
fn accept_candidate(
    platforms: &[Platform],
    grid: &mut PointGrid,
    beacon: &Vec3,
    pos: Vec3,
    elevated: bool,
    gifts: &mut Vec<Gift>,
    elevated_flags: &mut Vec<bool>,
) -> bool {
    let top = pos.y - 0.3;
    let Some(i) = platform_top_under(platforms, pos.x, pos.z, top, 0.5) else {
        return false;
    };
    if platforms[i].kind != PLATFORM_NORMAL {
        return false;
    }
    // T-062: spawn safety — no gift may spawn stuck inside/overlapping a wall, jump pad or ramp.
    if !gift_body_clear(
        platforms
            .iter()
            .filter(|p| p.kind != PLATFORM_NORMAL)
            .map(|p| &p.bounds),
        pos,
    ) {
        return false;
    }
    if grid.any_within(pos.x, pos.z, gift_min_spacing() - EPS) {
        return false;
    }
    if h_dist(pos.x, pos.z, beacon.x, beacon.z) < GIFT_BEACON_CLEAR - EPS {
        return false;
    }
    let y = platforms[i].bounds.max.y + 0.3;
    grid.insert(Vec3::new(pos.x, y, pos.z));
    gifts.push(Gift {
        id: gifts.len() as u16,
        pos: Vec3::new(pos.x, y, pos.z),
        kind: GIFT_NORMAL,
    });
    elevated_flags.push(elevated);
    true
}

=== FN: server/crates/sim/src/levelgen.rs :: fill_gifts ===
/// A3: fill up to `min(gift_target(level, players), eligible_area / 9)` gifts from chunk
/// slots (>= 20% elevated preferred first) then top up on random eligible platform tops.
/// Minimum gift spacing honours the Clustered Gifts curse knob (`set_gift_spacing_mul`);
/// the grid cell width stays `GIFT_MIN_SPACING`, which remains >= the reduced spacing, so the
/// `PointGrid` neighbour search stays correct under clustering.
fn fill_gifts(
    platforms: &[Platform],
    slots: &[(Vec3, bool)],
    beacon: &Vec3,
    level: u32,
    players: u32,
    rng: &mut Rng,
) -> GiftFill {
    let mut gifts = Vec::new();
    let mut elevated_flags: Vec<bool> = Vec::new();
    let mut grid = PointGrid::new();
    let eligible_area: f32 = platforms
        .iter()
        .filter(|p| p.kind == PLATFORM_NORMAL)
        .map(|p| aabb_area(&p.bounds))
        .sum();
    let cap = ((eligible_area + 1e-3) / GIFT_AREA_PER).floor().max(0.0) as u32;
    let target = gift_target(level, players).min(cap) as usize;
    let quota = ((target as f32) * GIFT_ELEVATED_QUOTA).ceil() as usize;
    let mut accepted_slot = vec![false; slots.len()];
    let mut order: Vec<usize> = (0..slots.len()).collect();
    rng.shuffle(&mut order);
    // Pass A: elevated slots first so the quota is met when slots exist.
    for &si in &order {
        let elevated_taken = elevated_flags.iter().filter(|&&e| e).count();
        if elevated_taken >= quota || gifts.len() >= target {
            break;
        }
        if !slots[si].1 || accepted_slot[si] {
            continue;
        }
        let (pos, elev) = slots[si];
        if accept_candidate(platforms, &mut grid, beacon, pos, elev, &mut gifts, &mut elevated_flags) {
            accepted_slot[si] = true;
        }
    }
    // Pass B: all remaining slots.
    for &si in &order {
        if gifts.len() >= target {
            break;
        }
        if accepted_slot[si] {
            continue;
        }
        let (pos, elev) = slots[si];
        if accept_candidate(platforms, &mut grid, beacon, pos, elev, &mut gifts, &mut elevated_flags) {
            accepted_slot[si] = true;
        }
    }
    // Pass C: top up on random eligible platform tops (area-weighted).
    let normal_idx: Vec<usize> = platforms
        .iter()
        .enumerate()
        .filter(|(_, p)| p.kind == PLATFORM_NORMAL)
        .map(|(i, _)| i)
        .collect();
    let mut cum: Vec<f32> = Vec::with_capacity(normal_idx.len());
    let mut acc = 0.0_f32;
    for &i in &normal_idx {
        acc += aabb_area(&platforms[i].bounds);
        cum.push(acc);
    }
    let mut attempts = 0usize;
    while gifts.len() < target && attempts < TOPUP_ATTEMPTS {
        attempts += 1;
        if normal_idx.is_empty() {
            break;
        }
        let idx = weighted_platform_index(rng, &normal_idx, &cum);
        let p = &platforms[idx];
        let min_x = p.bounds.min.x + 0.5;
        let max_x = p.bounds.max.x - 0.5;
        let min_z = p.bounds.min.z + 0.5;
        let max_z = p.bounds.max.z - 0.5;
        if min_x > max_x - EPS || min_z > max_z - EPS {
            continue;
        }
        let pos = Vec3::new(
            rng.range_f32(min_x, max_x),
            p.bounds.max.y + 0.3,
            rng.range_f32(min_z, max_z),
        );
        accept_candidate(
            platforms,
            &mut grid,
            beacon,
            pos,
            false,
            &mut gifts,
            &mut elevated_flags,
        );
    }
    GiftFill { gifts }
}

                roster.push(self.drafted_enemies[k % drafted].clone());

            self.level_num,
            &self.drafted_enemies,
            self.most_voted_enemy.as_deref(),
        ));

            roster.clear();

        for (index, id) in roster.iter().enumerate() {

                params,
            ));

        self.phase = PHASE_PLAYING;
