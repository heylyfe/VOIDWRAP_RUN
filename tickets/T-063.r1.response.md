/** Side length of a gift (design: 0.34 u). Golden gifts are identical in size. */
const GIFT_SIZE = 0.34;
/** Bob amplitude while hovering. */
const BOB_AMP = 0.08;
const BOB_HZ = 1.6;
const SPIN_RAD_PER_S = 0.9;
/** Golden sparkle octahedron hovers this far above the gift centre. */
const SPARKLE_LIFT = 0.34;
const SPARKLE_RADIUS = 0.085;

const GOLD_BODY = "#ffd75e";
const GOLD_RIBBON = "#8f6a10";
const GOLD_BOW = "#fff6c8";
const GOLD_EMISSIVE = "#ffc63f";
const GOLD_EMISSIVE_INTENSITY = 1.5;
/**
 * Normal gifts now glow like golden ones (DO 1): pushed above 1.0 so the
 * toneMapped:false emissive clears typical bloom thresholds instead of the
 * previous 0.6, which never glowed. Golden gifts stay brighter and keep their
 * sparkle so they remain distinct.
 */
const NORMAL_EMISSIVE_INTENSITY = 1.35;
const SPARKLE_COLOR = "#fff3b0";

interface GiftAssets {
  box: BoxGeometry;
  sparkle: OctahedronGeometry;
  normalMaterial: MeshStandardMaterial;
  goldenMaterial: MeshStandardMaterial;
  sparkleMaterial: MeshBasicMaterial;
  dispose(): void;
}

/** Deterministic little gift texture: body colour with a ribbon cross and a bow dot. */
function createGiftTexture(body: string, ribbon: string, bow: string): CanvasTexture | null {
  if (typeof document === "undefined") return null;
  const size = 128;
  const canvas = document.createElement("canvas");
  canvas.width = size;
  canvas.height = size;
  const ctx = canvas.getContext("2d");
  if (!ctx) return null;
  ctx.fillStyle = body;
  ctx.fillRect(0, 0, size, size);
  ctx.fillStyle = "rgba(0, 0, 0, 0.18)";
  ctx.fillRect(0, size * 0.72, size, size * 0.28);
  const band = Math.floor(size * 0.2);
  ctx.fillStyle = ribbon;
  ctx.fillRect((size - band) / 2, 0, band, size);
  ctx.fillRect(0, (size - band) / 2, size, band);
  ctx.fillStyle = bow;
  ctx.beginPath();
  ctx.arc(size / 2, size / 2, band * 0.45, 0, Math.PI * 2);
  ctx.fill();
  const tex = new CanvasTexture(canvas);
  tex.wrapS = RepeatWrapping;
  tex.wrapT = RepeatWrapping;
  tex.colorSpace = SRGBColorSpace;
  tex.anisotropy = 4;
  return tex;
}

function createGiftAssets(): GiftAssets {
  const normalTex = createGiftTexture(COMMON.gift.body, COMMON.gift.ribbon, COMMON.gift.glow);
  const goldenTex = createGiftTexture(GOLD_BODY, GOLD_RIBBON, GOLD_BOW);
  const box = new BoxGeometry(GIFT_SIZE, GIFT_SIZE, GIFT_SIZE);
  const sparkle = new OctahedronGeometry(SPARKLE_RADIUS);
  const normalMaterial = new MeshStandardMaterial({
    map: normalTex ?? undefined,
    color: normalTex ? "#ffffff" : COMMON.gift.body,
    emissive: COMMON.gift.glow,
    emissiveIntensity: NORMAL_EMISSIVE_INTENSITY,
    roughness: 0.55,
    metalness: 0.1,
    toneMapped: false,
  });
  const goldenMaterial = new MeshStandardMaterial({
    map: goldenTex ?? undefined,
    color: goldenTex ? "#ffffff" : GOLD_BODY,
    emissive: GOLD_EMISSIVE,
    emissiveIntensity: GOLD_EMISSIVE_INTENSITY,
    roughness: 0.32,
    metalness: 0.35,
    toneMapped: false,
  });
  const sparkleMaterial = new MeshBasicMaterial({
    color: SPARKLE_COLOR,
    transparent: true,
    opacity: 0.9,
    blending: AdditiveBlending,
    depthWrite: false,
    toneMapped: false,
    fog: false,
  });
  return {
    box,
    sparkle,
    normalMaterial,
    goldenMaterial,
    sparkleMaterial,
    dispose() {
      box.dispose();
      sparkle.dispose();
      normalMaterial.dispose();
      goldenMaterial.dispose();
      sparkleMaterial.dispose();
      normalTex?.dispose();
      goldenTex?.dispose();
    },
  };
}

interface Scratch {
  mat: Matrix4;
  pos: Vector3;
  quat: Quaternion;
  scale: Vector3;
  axis: Vector3;
}

/** The platform supporting a gift: highest platform top below the gift whose XZ contains it. */
function platformUnder(pos: Vec3, platforms: readonly Platform[]): number | null {
  let best: Platform | null = null;
  for (const p of platforms) {
    if (pos.x < p.bounds.min.x || pos.x > p.bounds.max.x) continue;
    if (pos.z < p.bounds.min.z || pos.z > p.bounds.max.z) continue;
    const top = p.bounds.max.y;
    if (top > pos.y + 0.01) continue;
    if (best === null || top > best.bounds.max.y) best = p;
  }
  return best === null ? null : best.id;
}

/**
 * All gifts on the level, drawn as three instanced meshes (normal, golden, sparkle).
 * Phase 1: untaken normal gifts. Phase 2 (levelPhase != LEVEL_COLLECT): one golden gift at
 * every collected normal gift position, unless its id is in store.goldenTaken or its platform
 * is TILE_GONE (golden gifts stay collectable during TILE_WARN, like on the server).
 */
export function Gifts({
  level,
  tiles,
}: {
  level: LevelData;
  tiles: readonly TileState[];
}): JSX.Element {
  const assets = useMemo(() => createGiftAssets(), []);
  useEffect(
    () => () => {
      assets.dispose();
    },
    [assets],
  );

  const scratchRef = useRef<Scratch | null>(null);

  const giftsTaken = useGameStore((s) => s.giftsTaken);
  const goldenTaken = useGameStore((s) => s.goldenTaken);
  const levelPhase = useGameStore((s) => s.levelPhase);
  const phase2 = levelPhase !== LEVEL_COLLECT;

  const normalGifts = useMemo(
    () => level.gifts.filter((g) => g.kind !== GIFT_GOLDEN),
    [level],
  );
  // Resolved once per level; only TILE_GONE re-filters the visible golden set.
  const giftPlatform = useMemo(
    () => normalGifts.map((g) => platformUnder(g.pos, level.platforms)),
    [normalGifts, level.platforms],
  );
  const taken = useMemo(() => new Set<number>(giftsTaken), [giftsTaken]);
  const goldTaken = useMemo(() => new Set<number>(goldenTaken), [goldenTaken]);
  const gone = useMemo(() => {
    const s = new Set<number>();
    for (const t of tiles) {
      if (t.state === TILE_GONE) s.add(t.id);
    }
    return s;
  }, [tiles]);

  const visibleNormal = useMemo(() => {
    const out: number[] = [];
    if (phase2) return out;
    for (let i = 0; i < normalGifts.length; i += 1) {
      const g = normalGifts[i];
      if (!g || taken.has(g.id)) continue;
      out.push(i);
    }
    return out;
  }, [phase2, normalGifts, taken]);

  const visibleGolden = useMemo(() => {
    const out: number[] = [];
    if (!phase2) return out;
    for (let i = 0; i < normalGifts.length; i += 1) {
      const g = normalGifts[i];
      if (!g) continue;
      if (!taken.has(g.id)) continue;
      if (goldTaken.has(g.id)) continue;
      const p = giftPlatform[i];
      if (p !== undefined && p !== null && gone.has(p)) continue;
      out.push(i);
    }
    return out;
  }, [phase2, normalGifts, taken, goldTaken, giftPlatform, gone]);

  const normalRef = useRef<InstancedMesh>(null);
  const goldenRef = useRef<InstancedMesh>(null);
  const sparkleRef = useRef<InstancedMesh>(null);
  useEffect(() => {
    for (const m of [normalRef.current, goldenRef.current, sparkleRef.current]) {
      if (m) {
        m.instanceMatrix.setUsage(DynamicDrawUsage);
        m.count = 0;
        m.instanceMatrix.needsUpdate = true;
      }
    }
  }, [assets, normalGifts.length]);

  useFrame((state) => {
    let s = scratchRef.current;
    if (s === null) {
      s = {
        mat: new Matrix4(),
        pos: new Vector3(),
        quat: new Quaternion(),
        scale: new Vector3(1, 1, 1),
        axis: new Vector3(0, 1, 0),
      };
      scratchRef.current = s;
    }
    const t = state.clock.elapsedTime;

    const normalMesh = normalRef.current;
    if (normalMesh) {
      let n = 0;
      for (const gi of visibleNormal) {
        const g = normalGifts[gi];
        if (!g) continue;
        const phase = (gi * 1.61803398875) % (Math.PI * 2);
        s.pos.set(g.pos.x, g.pos.y + Math.sin(t * BOB_HZ + phase) * BOB_AMP, g.pos.z);
        s.quat.setFromAxisAngle(s.axis, t * SPIN_RAD_PER_S + phase);
        s.scale.set(1, 1, 1);
        s.mat.compose(s.pos, s.quat, s.scale);
        normalMesh.setMatrixAt(n, s.mat);
        n += 1;
      }
      normalMesh.count = n;
      normalMesh.instanceMatrix.needsUpdate = true;
    }

    const goldenMesh = goldenRef.current;
    const sparkleMesh = sparkleRef.current;
    if (goldenMesh) {
      let n = 0;
      for (const gi of visibleGolden) {
        const g = normalGifts[gi];
        if (!g) continue;
        const phase = (gi * 2.39996322973) % (Math.PI * 2);
        s.pos.set(g.pos.x, g.pos.y + Math.sin(t * BOB_HZ + phase) * BOB_AMP, g.pos.z);
        s.quat.setFromAxisAngle(s.axis, t * SPIN_RAD_PER_S * 1.5 + phase);
        s.scale.set(1, 1, 1);
        s.mat.compose(s.pos, s.quat, s.scale);
        goldenMesh.setMatrixAt(n, s.mat);
        if (sparkleMesh) {
          const pulse = 0.75 + 0.45 * Math.sin(t * 3.1 + phase * 2);
          s.pos.set(
            g.pos.x,
            g.pos.y + SPARKLE_LIFT + Math.sin(t * 2.2 + phase) * 0.06,
            g.pos.z,
          );
          s.quat.setFromAxisAngle(s.axis, t * 2.4 + phase);
          s.scale.setScalar(Math.max(0.2, pulse));
          s.mat.compose(s.pos, s.quat, s.scale);
          sparkleMesh.setMatrixAt(n, s.mat);
        }
        n += 1;
      }
      goldenMesh.count = n;
      goldenMesh.instanceMatrix.needsUpdate = true;
      if (sparkleMesh) {
        sparkleMesh.count = n;
        sparkleMesh.instanceMatrix.needsUpdate = true;
      }
    }
  });

  const capacity = Math.max(1, normalGifts.length);
  return (
    <group>
      <instancedMesh
        ref={normalRef}
        args={[assets.box, assets.normalMaterial, capacity]}
        frustumCulled={false}
        dispose={null}
      />
      <instancedMesh
        ref={goldenRef}
        args={[assets.box, assets.goldenMaterial, capacity]}
        frustumCulled={false}
        dispose={null}
      />
      <instancedMesh
        ref={sparkleRef}
        args={[assets.sparkle, assets.sparkleMaterial, capacity]}
        frustumCulled={false}
        dispose={null}
      />
    </group>
  );
}

/**
 * One stand-alone gift with exactly the look and size of the instanced gifts (used by the False Gift
 * enemy while it is disguised, so it is indistinguishable from a real gift).
 */
export function GiftBox(props: { position: Vec3; taken: boolean }): JSX.Element {
  const assets = useMemo(() => createGiftAssets(), []);
  useEffect(() => () => assets.dispose(), [assets]);
  return (
    <mesh
      position={[props.position.x, props.position.y, props.position.z]}
      visible={!props.taken}
      geometry={assets.box}
      material={assets.normalMaterial}
    />
  );
}

// Pure helpers for the gift-pickup burst effect. No DOM, no React, no three.js.
import { hashString, mulberry32 } from "./rng";

/** Any gift-like record; only the id and world position are needed. */
export interface GiftLike {
  id: number;
  pos: { x: number; y: number; z: number };
}

/** Ids present in `next` but not in `prev`, in `next` order, deduplicated. */
export function newlyTaken(prev: readonly number[], next: readonly number[]): number[] {
  const before = new Set(prev);
  const added = new Set<number>();
  const out: number[] = [];
  for (const id of next) {
    if (before.has(id) || added.has(id)) continue;
    added.add(id);
    out.push(id);
  }
  return out;
}

/** World position of the gift with `id`; null when the level has no such gift. */
export function findGiftPosition(
  gifts: readonly GiftLike[],
  id: number,
): { x: number; y: number; z: number } | null {
  for (const gift of gifts) {
    if (gift.id === id) {
      return { x: gift.pos.x, y: gift.pos.y, z: gift.pos.z };
    }
  }
  return null;
}

const TAU = Math.PI * 2;

export interface BurstParticle {
  /** Outward direction (y-biased so the burst arcs upward). Unit length in XZ+Y. */
  dirX: number;
  dirY: number;
  dirZ: number;
  /** Initial speed in units per second. */
  speed: number;
  /** Relative size multiplier. */
  size: number;
}

/**
 * Deterministic spherical burst with an upward bias. Same seed always yields
 * the same particles, so replays (and vitest) agree.
 */
export function burstParticles(seed: string, count: number): BurstParticle[] {
  const total = Math.max(0, Math.floor(count));
  const rand = mulberry32(hashString(`gift-pickup:${seed}`));
  const particles: BurstParticle[] = [];
  for (let i = 0; i < total; i += 1) {
    const theta = rand() * TAU;
    const lift = 0.25 + rand() * 0.75;
    const ring = Math.sqrt(Math.max(0, 1 - lift * lift));
    particles.push({
      dirX: Math.cos(theta) * ring,
      dirY: lift,
      dirZ: Math.sin(theta) * ring,
      speed: 1.6 + rand() * 1.4,
      size: 0.6 + rand() * 0.8,
    });
  }
  return particles;
}

"use client";

// Sparkle burst when a gift is collected. Purely visual. The store is the
// source of truth: `giftsTaken` / `goldenTaken` only grow, so bursts are keyed
// off newly-appeared ids and positioned at the matching level gift.
import { useEffect, useMemo, useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import {
  AdditiveBlending,
  Color,
  DynamicDrawUsage,
  InstancedMesh,
  Matrix4,
  MeshBasicMaterial,
  OctahedronGeometry,
  Quaternion,
  Vector3,
} from "three";
import type { LevelData } from "@/protocol/generated";
import { useGameStore } from "@/game/store";
import { COMMON } from "./theme";
import { burstParticles, findGiftPosition, newlyTaken } from "./giftPickup";
import type { BurstParticle } from "./giftPickup";

const PARTICLES_PER_BURST = 12;
const MAX_BURSTS = 8;
const LIFETIME_S = 0.7;
const GRAVITY = 4.0;
const PARTICLE_RADIUS = 0.06;

const GOLD_BURST = "#ffd75e";

interface LiveBurst {
  golden: boolean;
  origin: { x: number; y: number; z: number };
  /** Set on the first frame so the burst lives on the scene clock. */
  start: number | null;
  particles: readonly BurstParticle[];
}

export function GiftPickup({ level }: { level: LevelData }): JSX.Element {
  const geometry = useMemo(() => new OctahedronGeometry(PARTICLE_RADIUS), []);
  const material = useMemo(
    () =>
      new MeshBasicMaterial({
        color: "#ffffff",
        transparent: true,
        opacity: 0.95,
        blending: AdditiveBlending,
        depthWrite: false,
        toneMapped: false,
        fog: false,
      }),
    [],
  );
  useEffect(() => () => geometry.dispose(), [geometry]);
  useEffect(() => () => material.dispose(), [material]);

  const capacity = MAX_BURSTS * PARTICLES_PER_BURST;
  const meshRef = useRef<InstancedMesh>(null);
  useEffect(() => {
    const mesh = meshRef.current;
    if (mesh) {
      mesh.instanceMatrix.setUsage(DynamicDrawUsage);
      mesh.count = 0;
      mesh.instanceMatrix.needsUpdate = true;
    }
  }, [geometry, material]);

  const burstsRef = useRef<LiveBurst[]>([]);
  const scratchRef = useRef<{
    mat: Matrix4;
    pos: Vector3;
    quat: Quaternion;
    scale: Vector3;
    up: Vector3;
  } | null>(null);
  const colorRef = useRef(new Color());

  const giftsTaken = useGameStore((s) => s.giftsTaken);
  const goldenTaken = useGameStore((s) => s.goldenTaken);
  const prevRef = useRef<{ normal: readonly number[]; golden: readonly number[] } | null>(null);

  useEffect(() => {
    const prev = prevRef.current;
    if (prev === null) {
      // First observation on this level: remember what is already taken
      // instead of bursting every gift when joining an in-progress run.
      prevRef.current = { normal: giftsTaken, golden: goldenTaken };
      return;
    }
    const spawn = (id: number, golden: boolean) => {
      const position = findGiftPosition(level.gifts, id);
      if (position === null) return;
      const bursts = burstsRef.current;
      if (bursts.length >= MAX_BURSTS) bursts.shift();
      bursts.push({
        golden,
        origin: position,
        start: null,
        particles: burstParticles(`${golden ? "golden" : "normal"}:${id}`, PARTICLES_PER_BURST),
      });
    };
    for (const id of newlyTaken(prev.normal, giftsTaken)) spawn(id, false);
    for (const id of newlyTaken(prev.golden, goldenTaken)) spawn(id, true);
    prevRef.current = { normal: giftsTaken, golden: goldenTaken };
  }, [giftsTaken, goldenTaken, level]);

  useFrame((state) => {
    const mesh = meshRef.current;
    if (!mesh) return;
    let scratch = scratchRef.current;
    if (scratch === null) {
      scratch = {
        mat: new Matrix4(),
        pos: new Vector3(),
        quat: new Quaternion(),
        scale: new Vector3(),
        up: new Vector3(0, 1, 0),
      };
      scratchRef.current = scratch;
    }
    const t = state.clock.elapsedTime;
    const live: LiveBurst[] = [];
    let n = 0;
    for (const burst of burstsRef.current) {
      if (burst.start === null) burst.start = t;
      const age = t - burst.start;
      if (age >= LIFETIME_S) continue;
      live.push(burst);
      const lifeT = age / LIFETIME_S;
      const fade = Math.max(0.05, 1 - lifeT);
      colorRef.current.set(burst.golden ? GOLD_BURST : COMMON.gift.glow);
      for (const p of burst.particles) {
        if (n >= capacity) break;
        const travelled = p.speed * age * (1 - 0.45 * lifeT);
        scratch.pos.set(
          burst.origin.x + p.dirX * travelled,
          burst.origin.y + p.dirY * travelled - 0.5 * GRAVITY * age * age,
          burst.origin.z + p.dirZ * travelled,
        );
        scratch.quat.setFromAxisAngle(scratch.up, age * 5 + p.size);
        scratch.scale.setScalar(p.size * fade);
        scratch.mat.compose(scratch.pos, scratch.quat, scratch.scale);
        mesh.setMatrixAt(n, scratch.mat);
        mesh.setColorAt(n, colorRef.current);
        n += 1;
      }
    }
    burstsRef.current = live;
    mesh.count = n;
    mesh.instanceMatrix.needsUpdate = true;
    if (mesh.instanceColor) mesh.instanceColor.needsUpdate = true;
  });

  return (
    <instancedMesh
      ref={meshRef}
      args={[geometry, material, capacity]}
      frustumCulled={false}
      dispose={null}
    />
  );
}

import { describe, expect, it } from "vitest";
import { burstParticles, findGiftPosition, newlyTaken } from "./giftPickup";

describe("newlyTaken", () => {
  it("returns ids present in next but not in prev, in next order", () => {
    expect(newlyTaken([1, 2], [1, 2, 3, 4])).toEqual([3, 4]);
  });

  it("deduplicates within next and never reports removed ids", () => {
    expect(newlyTaken([5], [5, 5, 6])).toEqual([6]);
    expect(newlyTaken([1, 2], [2])).toEqual([]);
  });

  it("treats an empty prev as everything being new", () => {
    expect(newlyTaken([], [7, 8])).toEqual([7, 8]);
  });
});

describe("findGiftPosition", () => {
  const gifts = [
    { id: 3, pos: { x: 1, y: 2, z: 3 } },
    { id: 9, pos: { x: 4, y: 5, z: 6 } },
  ];

  it("finds a gift by id", () => {
    expect(findGiftPosition(gifts, 9)).toEqual({ x: 4, y: 5, z: 6 });
  });

  it("returns null for an unknown id", () => {
    expect(findGiftPosition(gifts, 42)).toBeNull();
  });
});

describe("burstParticles", () => {
  it("is deterministic for the same seed", () => {
    expect(burstParticles("same-seed", 12)).toEqual(burstParticles("same-seed", 12));
  });

  it("produces the requested count with finite, upward-biased directions and positive speed and size", () => {

// Classify drafted upgrade/curse ids against the content manifest so a swapped
// or mislabelled wire array can never show a curse as an upgrade (or vice
// versa). Pure and unit-tested; no DOM, no React.
import { CURSES, UPGRADES } from '../content';

export type ModGroup = 'upgrade' | 'curse';

export interface CategorizedMod {
  id: string;
  group: ModGroup;
}

const UPGRADE_IDS = new Set(UPGRADES.map((card) => card.id));
const CURSE_IDS = new Set(CURSES.map((card) => card.id));

/**
 * Manifest-only classification. Returns null when the id is unknown, or present
 * in both manifests (ambiguous) — callers then fall back to the wire group.
 */
export function classifyModId(id: string): ModGroup | null {
  const inUpgrades = UPGRADE_IDS.has(id);
  const inCurses = CURSE_IDS.has(id);
  if (inUpgrades && !inCurses) {
    return 'upgrade';
  }
  if (inCurses && !inUpgrades) {
    return 'curse';
  }
  return null;
}

/**
 * Merge the wire `upgrades` and `curses` id lists into one manifest-classified
 * list. Duplicates collapse (first wire occurrence wins). Ids the manifest
 * cannot classify keep the group the wire reported, so nothing is dropped.
 */
export function categorizeMods(
  upgrades: readonly string[],
  curses: readonly string[],
): CategorizedMod[] {
  const seen = new Set<string>();
  const out: CategorizedMod[] = [];
  const wire: readonly (readonly [readonly string[], ModGroup])[] = [
    [upgrades, 'upgrade'],
    [curses, 'curse'],
  ];
  for (const [list, fallback] of wire) {
    for (const id of list) {

function levelBounds(level: LevelData): { center: { x: number; y: number; z: number }; radius: number } {
  let minX = Infinity, maxX = -Infinity, minZ = Infinity, maxZ = -Infinity, minY = Infinity, maxY = -Infinity;
  for (const p of level.platforms) {
    minX = Math.min(minX, p.bounds.min.x);
    maxX = Math.max(maxX, p.bounds.max.x);
    minY = Math.min(minY, p.bounds.min.y);
    maxY = Math.max(maxY, p.bounds.max.y);
    minZ = Math.min(minZ, p.bounds.min.z);
    maxZ = Math.max(maxZ, p.bounds.max.z);
  }
  if (!Number.isFinite(minX)) return { center: ORIGIN, radius: 30 };
  const center = { x: (minX + maxX) / 2, y: (minY + maxY) / 2, z: (minZ + maxZ) / 2 };

/** Counts rendered frames and publishes fps to the store twice a second. */
function FpsProbe() {
  const acc = useRef({ frames: 0, time: 0 });
  useFrame((_, dt) => {
    const a = acc.current;
    a.frames += 1;
    a.time += dt;
    if (a.time >= 0.5) {
      useGameStore.getState().set({ fps: Math.round(a.frames / a.time) });
      a.frames = 0;
      a.time = 0;
    }
  });
  return null;
}

function causeOf(cause: number): DeathCause {
  if (cause === DEATH_PROJECTILE) return "projectile";
  return cause === DEATH_ENEMY ? "enemy" : "void";
}

/** The ability hint for the bottom-of-screen controls line, by class. */
function abilityHint(cls: number): string {
  if (cls === CLASS_GRAPPLER) return "Shift grapple";
  if (cls === CLASS_BULWARK) return "Shift Aegis Bloom";
  if (cls === CLASS_GLIDER) return "Space twice double jump · hold Shift glide";
  return "Shift dash";
}

  const [session, setSession] = useState<GameSession | null>(null);
  const [level, setLevel] = useState<LevelData | null>(null);
  const [loadError, setLoadError] = useState<string | null>(null);
  const container = useRef<HTMLDivElement>(null);
  const status = useGameStore((s) => s.status);
  const pointerLocked = useGameStore((s) => s.pointerLocked);
  const runOver = useGameStore((s) => s.runOver);
  const giftsTaken = useGameStore((s) => s.giftsTaken);
  const runInfo = useGameStore((s) => s.runInfo);
  const draft = useGameStore((s) => s.draft);
  const myMode = useGameStore((s) => s.myMode);
  const activeIds = useGameStore((s) => s.activeIds);
  const shop = useGameStore((s) => s.shop);
  const notices = useGameStore((s) => s.notices);
  const myClass = useGameStore((s) => s.myClass);
  const levelPhase = useGameStore((s) => s.levelPhase);
  const goldenTaken = useGameStore((s) => s.goldenTaken);
  const collapseMs = useGameStore((s) => s.collapseMs);
  const [following, setFollowing] = useState<number | null>(null);
  const drafting = !!draft && runInfo?.phase === PHASE_ENEMY_DRAFT;
  // The lobby hub is shown in the colours of the level about to be played.
  const runLevel = runInfo?.level ?? level?.level ?? 1;
  const normalIds = useMemo(() => (level ? level.gifts.filter((g) => g.kind !== GIFT_GOLDEN).map((g) => g.id) : []), [level]);
  const normalTotal = normalIds.length;
  const normalLeft = normalIds.filter((id) => !giftsTaken.includes(id)).length;
  const playing = runInfo ? runInfo.phase === PHASE_PLAYING : true;
  const phase2 = levelPhase !== LEVEL_COLLECT;

  useEffect(() => {
    let disposed = false;
    let s: GameSession | null = null;
    WasmStepper.load()
      .then((stepper) => {
        if (disposed) return;
        s = new GameSession(stepper, defaultServerUrl(), loadPlayerName());
        s.onLevel = setLevel;
        setSession(s);
      })
      .catch((e: unknown) => setLoadError(String(e)));
    return () => {
      disposed = true;
      s?.dispose();
    };
  }, []);

  // Free the mouse for the run summary and the lobby draft.
  useEffect(() => {
    if ((runOver || drafting) && document.pointerLockElement) document.exitPointerLock();
  }, [runOver, drafting]);

  useEffect(() => {
    if (!session) return;
    const key = (down: boolean) => (e: KeyboardEvent) => {
      const st = useGameStore.getState();
      // Ctrl toggles the mouse lock (the desktop window has no browser "Esc to unlock" bar).
      if (down && !e.repeat && (e.code === "ControlLeft" || e.code === "ControlRight")) {
        e.preventDefault();
        const inDraft = !!st.draft && st.runInfo?.phase === PHASE_ENEMY_DRAFT;
        if (document.pointerLockElement) document.exitPointerLock();
        else if (!st.runOver && !inDraft) void container.current?.requestPointerLock();
        return;
      }
      const spectatingNow = st.runInfo?.phase === PHASE_PLAYING && st.myMode !== PLAYER_ACTIVE;
      if (down && spectatingNow && !e.repeat && (e.code === "KeyF" || e.code === "KeyG")) {
        session.followIndex = e.code === "KeyF" ? nextFollow(session.followIndex, st.activeIds.length) : null;
        setFollowing(session.followIndex);
        e.preventDefault();
        return;
      }
      const b = bindingFor(e.code);
      if (!b) return;
      session.keys[b] = down;
      e.preventDefault();
    };
    const onDown = key(true);
    const onUp = key(false);
    const onMove = (e: MouseEvent) => {
      if (document.pointerLockElement !== container.current) return;
      session.yaw -= e.movementX * MOUSE_SENSITIVITY;
      session.pitch = Math.min(1.3, Math.max(-0.2, session.pitch + e.movementY * MOUSE_SENSITIVITY));
    };
    const onMouse = (down: boolean) => (e: MouseEvent) => {
      if (e.button === 2) session.rightMouse = down;
    };
    const onMouseDown = onMouse(true);
    const onMouseUp = onMouse(false);
    const onLock = () => useGameStore.getState().set({ pointerLocked: document.pointerLockElement === container.current });
    const onBlur = () => {
      for (const k of Object.keys(session.keys) as (keyof typeof session.keys)[]) session.keys[k] = false;
      session.rightMouse = false;
    };
    window.addEventListener("keydown", onDown);
    window.addEventListener("keyup", onUp);
    window.addEventListener("mousemove", onMove);
    window.addEventListener("mousedown", onMouseDown);
    window.addEventListener("mouseup", onMouseUp);
    window.addEventListener("blur", onBlur);
    document.addEventListener("pointerlockchange", onLock);
    return () => {
      window.removeEventListener("keydown", onDown);
      window.removeEventListener("keyup", onUp);
      window.removeEventListener("mousemove", onMove);
      window.removeEventListener("mousedown", onMouseDown);
      window.removeEventListener("mouseup", onMouseUp);
      window.removeEventListener("blur", onBlur);
      document.removeEventListener("pointerlockchange", onLock);
    };
  }, [session]);

  const restart = useCallback(() => session?.restart(), [session]);
  const submitDraft = useCallback((picks: number[]) => session?.submitDraft(picks), [session]);
  const buy = useCallback((itemId: string) => session?.buy(itemId), [session]);
  const chooseClass = useCallback((cls: number) => session?.chooseClass(cls), [session]);

  return (
    <div
      ref={container}
      className="relative h-screen w-screen overflow-hidden bg-[#07060d]"
      onClick={() => {
        const st = useGameStore.getState();
        const inDraft = !!st.draft && st.runInfo?.phase === PHASE_ENEMY_DRAFT;
        if (!st.runOver && !inDraft) void container.current?.requestPointerLock();
      }}
      onContextMenu={(e) => e.preventDefault()}
    >
      {session && level ? (
        <Canvas camera={{ fov: 70, near: 0.1, far: 1000 }} dpr={[1, 2]}>
          <World session={session} level={level} themeLevel={runLevel} />
        </Canvas>
      ) : (
        <div className="flex h-full items-center justify-center font-mono text-white/70">
          {loadError ?? "entering the void…"}
        </div>
      )}
      {level && playing && normalTotal > 0 && (
        <>
          {phase2 ? (
            <CollapseHud golden={goldenTaken.length} goldenTotal={normalTotal} collapseMs={collapseMs} />
          ) : (
            <GameHud level={runLevel} giftsRemaining={normalLeft} giftsTotal={normalTotal} />
          )}
          <LevelBanner level={runLevel} />
        </>
      )}
      {playing && myMode !== PLAYER_ACTIVE && !runOver && (
        <SpectatorHud
          reason={myMode === PLAYER_FINISHED ? "beacon" : "dead"}
          playersLeft={activeIds.length}
          following={following === null ? null : `Player ${activeIds[following % Math.max(activeIds.length, 1)] ?? "?"}`}
        />
      )}
      {runInfo && playing && (
        <RunStatus
          currency={runInfo.currency}
          lives={runInfo.lives}
          enemies={runInfo.enemies}
          curses={runInfo.curses}
          upgrades={runInfo.upgrades}
        />
      )}
      {drafting && draft && runInfo && (
        <DraftScreen
          cards={draft.cards}
          enemyPicks={draft.enemyPicks}
          rewardPicks={draft.rewardPicks}
          nextLevel={runInfo.level}
          currency={runInfo.currency}
          lives={runInfo.lives}
          waiting={draft.submitted}
          onSubmit={submitDraft}
          classId={myClass}
          onChooseClass={chooseClass}
          shop={shop}
          onBuy={buy}
        />
      )}
      {!pointerLocked && status === "connected" && !runOver && !drafting && (
        <div className="pointer-events-none absolute bottom-28 left-1/2 -translate-x-1/2 rounded bg-black/60 px-4 py-2 font-mono text-sm text-white">
          click or Ctrl to aim · WASD move · Space jump · {abilityHint(myClass)} · E use altar
        </div>
      )}
      {status === "connected" && !runOver && !drafting && (
        <div
          data-testid="mouse-lock-hint"
          className="pointer-events-none absolute bottom-2 left-1/2 -translate-x-1/2 font-mono text-[11px] tracking-[0.25em] text-white/55"
        >
          CTRL TO LOCK/UNLOCK MOUSE
        </div>
      )}
      {runOver && (
        <RunSummary
          levelReached={runOver.levelReached}
          giftsCollected={runOver.giftsCollected}
          cause={causeOf(runOver.cause)}
          onRestart={restart}
        />
      )}
      <Notices notices={notices} />
      <DebugPanel />
    </div>
  );
}
