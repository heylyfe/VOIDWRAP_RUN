// Pure lookup + curve helpers for the reworked enemy roster (ticket T-057a).
// No three.js / React runtime imports — unit-testable in isolation.
import type { MutableRefObject } from "react";
import {
  ENEMY_ATTACK,
  ENEMY_DORMANT,
  ENEMY_TELEGRAPH,
} from "../protocol/generated";

/** Stable behaviour id (matches design/content.json enemy ids). */
export type EnemyBehaviorId =
  | "pallbearer"
  | "snapshrike"
  | "moteling"
  | "rift_warden"
  | "gale_wight"
  | "lantern_leech"
  | "turret"
  | "mimic"
  | "corrupter";

export interface EnemyLook {
  /** Behaviour id; also the roster order used for the wire `kind` index. */
  readonly behavior: EnemyBehaviorId;
  /** Primary accent: emissive trim + telegraph colour (design doc §2). */
  readonly accent: string;
  /** Secondary emissive: hot core / rim / inner glow. */
  readonly glow: string;
}

/** Order matches the wire `kind` index: pallbearer 0 … corrupter 8. */
export const ENEMY_LOOKS: readonly EnemyLook[] = [
  { behavior: "pallbearer", accent: "#ff3355", glow: "#ffdf9e" },
  { behavior: "snapshrike", accent: "#ff4436", glow: "#ffb36b" },
  { behavior: "moteling", accent: "#ffc3d9", glow: "#ff5fa8" },
  { behavior: "rift_warden", accent: "#b0313f", glow: "#ff5a3c" },
  { behavior: "gale_wight", accent: "#d94fff", glow: "#b47aff" },
  { behavior: "lantern_leech", accent: "#ff7ab8", glow: "#ffb3d1" },
  { behavior: "turret", accent: "#ff8a3c", glow: "#ffd07a" },
  { behavior: "mimic", accent: "#ff3df0", glow: "#b46bff" },
  { behavior: "corrupter", accent: "#39ffb0", glow: "#7dffb9" },
];

export const ENEMY_KIND_COUNT = ENEMY_LOOKS.length;

const FALLBACK_LOOK: EnemyLook = { behavior: "pallbearer", accent: "#ff3355", glow: "#ffdf9e" };

export function isEnemyKind(kind: number): boolean {
  return Number.isInteger(kind) && kind >= 0 && kind < ENEMY_LOOKS.length;
}

/** Unknown/missing kinds fall back to the pallbearer look (never crash a render). */
export function enemyLookFor(kind: number | undefined): EnemyLook {
  if (kind === undefined) return FALLBACK_LOOK;
  const look = ENEMY_LOOKS[kind];
  return look ?? FALLBACK_LOOK;
}

export function isTelegraphState(state: number | undefined): boolean {
  return state === ENEMY_TELEGRAPH || state === ENEMY_ATTACK;
}

export function clamp01(v: number): number {
  if (!Number.isFinite(v)) return 0;
  return v < 0 ? 0 : v > 1 ? 1 : v;
}

/** Wire `prog` (0–255) → 0..1. */
export function prog01(prog: number | undefined): number {
  if (prog === undefined || !Number.isFinite(prog)) return 0;
  return clamp01(prog / 255);
}

export function lerp(a: number, b: number, t: number): number {
  const c = clamp01(t);
  return a + (b - a) * c;
}

/* ---------- telegraph curves ---------- */

/** Alpha ramp shared by every telegraph: dim at prog 0, solid at 255. */
export function telegraphAlpha(t01: number): number {
  return 0.3 + 0.7 * clamp01(t01);
}

/** Moteling INHALE: warning ring starts at orbit radius and collapses onto the victim. */
export const MOTELING_RING_START = 5.0;
export const MOTELING_RING_END = 0.45;
export function motelingRingRadius(t01: number): number {
  return lerp(MOTELING_RING_START, MOTELING_RING_END, t01);
}

/** Rings that grow outward as an attack resolves (shriek snap / warden shove / fret spread). */
export function expandingRingRadius(t01: number, maxRadius: number): number {
  return Math.max(0.25, clamp01(t01) * maxRadius);
}

/** Fixed threat radii from the design doc (§2). */
export const PALLBEARER_LUNGE_RADIUS = 1.1;
export const GALE_WIGHT_BLAST_RADIUS = 2.1;
export const RIFT_WARDEN_SHOVE_RADIUS = 3.5;
export const SNAPSHRIKE_DASH_HALF_WIDTH = 0.7;
export const TURRET_AIM_WIDTH = 0.1;
export const LEECH_FLARE_RADIUS = 0.9;
export const MIMIC_HISS_RADIUS = 0.95;
export const CORRUPTER_AURA_RADIUS = 2.0;

/** Snapshrike freezes its line for the last 0.35 s of the 0.75 s LOCK window. */
export const SNAPSHRIKE_LOCK_PROG = 0.4 / 0.75; // ≈ 0.533
/** Alpha for the LOCK line: ramps while tracking, snaps solid once the line freezes. */
export function snapshrikeAlpha(t01: number): number {
  const t = clamp01(t01);
  return t >= SNAPSHRIKE_LOCK_PROG ? 1 : telegraphAlpha(t) * 0.85;
}

/** MIMIC CACHED tell: a 0.5 s magenta shimmer every 3.0 s. Purely client-clock driven. */
export const MIMIC_SHIMMER_PERIOD = 3.0;
export const MIMIC_SHIMMER_WINDOW = 0.5;
function shimmerPhase(timeS: number): number {
  return ((timeS % MIMIC_SHIMMER_PERIOD) + MIMIC_SHIMMER_PERIOD) % MIMIC_SHIMMER_PERIOD;
}
export function mimicShimmerVisible(timeS: number): boolean {
  if (!Number.isFinite(timeS)) return false;
  return shimmerPhase(timeS) < MIMIC_SHIMMER_WINDOW;
}
/** 0 → peak → 0 across the shimmer window; 0 outside it. */
export function mimicShimmerOpacity(timeS: number): number {
  if (!mimicShimmerVisible(timeS)) return 0;
  return Math.sin((shimmerPhase(timeS) / MIMIC_SHIMMER_WINDOW) * Math.PI) * 0.9;
}

/** Gifts are 0.34 u; the shared GiftBox still draws a 0.7 u box, so the False Gift
 *  wraps it in a fixed scale wrapper (independent of the body-radius scale). */
export const MIMIC_GIFT_SCALE = 0.34 / 0.7;

/** Yaw that faces a model (+Z forward) from (fx,fz) toward (tx,tz). */
export function yawToward(fx: number, fz: number, tx: number, tz: number): number {
  return Math.atan2(tx - fx, tz - fz);
}

/** Shared per-slot animation payload the parent writes every frame (no re-render).
 *  `tx/ty/tz` is the telegraph anchor copy, `ex/ey/ez` the enemy's world position. */
export interface EnemyAnim {
  state: number;
  prog: number;
  flags: number;
  ex: number;
  ey: number;
  ez: number;
  tx: number;
  ty: number;
  tz: number;
}

/** Stable across frames — parent mutates fields, never replaces the object. */
export type EnemyAnimRef = MutableRefObject<EnemyAnim>;

export interface EnemyModelProps {
  anim: EnemyAnimRef;
}

// Pure routing logic shared by the enemy pool: which roster model a wire `kind` maps
// to, and which readability mode an enemy is in (dormant / hunting / disguised).
// Behaviour ids now come from enemyLook.ts (single source of truth for the 9-kind roster).
import { ENEMY_DISGUISED, ENEMY_DORMANT } from "../protocol/generated";
import { enemyLookFor } from "./enemyLook";
import type { EnemyBehaviorId } from "./enemyLook";

export type EnemyMode = "dormant" | "hunting" | "disguised";
export type EnemyBehavior = EnemyBehaviorId;

/** `kind` indexes the roster order (see enemyLook.ENEMY_LOOKS); unknown kinds fall back to pallbearer. */
export function enemyBehaviorFor(kind: number | undefined): EnemyBehavior {
  return enemyLookFor(kind).behavior;
}

/** Map the wire ENEMY_* state to the visual readability mode. Missing info reads as hunting. */
export function modeForState(state: number | undefined): EnemyMode {
  if (state === ENEMY_DORMANT) return "dormant";
  if (state === ENEMY_DISGUISED) return "disguised";
  return "hunting";
}

"use client";

import { useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import { DoubleSide } from "three";
import type { Group, MeshStandardMaterial, PointLight } from "three";
import { ENEMY_ATTACK, ENEMY_DORMANT, ENEMY_TELEGRAPH } from "@/protocol/generated";
import type { EnemyModelProps } from "../enemyLook";
import { prog01, yawToward } from "../enemyLook";

/**
 * The Pallbearer — black-lacquered coffin-barge, brass lantern prow, hanging shroud.
 * Accent #ff3355 (deep crimson). Unit-radius model; the parent scales by body radius.
 * Idle: low bob / sway. Attack (KNEEL→LUNGE): rears up, lantern whiteouts.
 */
export function Pallbearer({ anim }: EnemyModelProps): JSX.Element {
  const root = useRef<Group>(null);
  const inner = useRef<Group>(null);
  const shroud = useRef<Group>(null);
  const seamMats = useRef<(MeshStandardMaterial | null)[]>([]);
  const lanternMat = useRef<MeshStandardMaterial>(null);
  const lanternLight = useRef<PointLight>(null);

  useFrame((state) => {
    const r = root.current;
    if (!r || (r.parent !== null && !r.parent.visible)) return;
    const a = anim.current;
    const t = state.clock.elapsedTime;
    const dormant = a.state === ENEMY_DORMANT;
    const telegraph = a.state === ENEMY_TELEGRAPH;
    const attack = a.state === ENEMY_ATTACK;
    const p = prog01(a.prog);
    const dim = dormant ? 0.16 : 1;

    if (telegraph || attack) {
      r.rotation.y = yawToward(a.ex, a.ez, a.tx, a.tz);
    }

    if (inner.current) {
      const bobSpeed = dormant ? 0.55 : 1.35;
      inner.current.position.y =
        Math.sin(t * bobSpeed) * (dormant ? 0.05 : 0.09) + (telegraph || attack ? p * 0.32 : 0);
      inner.current.rotation.x = -(telegraph ? p * 0.55 : attack ? 0.55 : 0); // rear-up
      inner.current.rotation.z = Math.sin(t * 0.85) * (dormant ? 0.015 : 0.045);
    }
    if (shroud.current) {
      shroud.current.rotation.x = Math.sin(t * 1.7 + 1.3) * 0.24;
    }
    const ember = (0.9 + Math.sin(t * 2.1) * 0.35) * dim;
    for (const m of seamMats.current) {
      if (m) m.emissiveIntensity = ember;
    }
    if (lanternMat.current) {
      lanternMat.current.emissiveIntensity =
        (dormant ? 0.3 : 1.6) + (telegraph || attack ? p * 3.4 : 0);
    }
    if (lanternLight.current) {
      lanternLight.current.intensity = dormant ? 0.12 : 1.15 + (telegraph || attack ? p * 2.3 : 0);
    }
  });

  return (
    <group ref={root}>
      <group ref={inner}>
        {/* hull — stretched black-lacquer barge */}
        <mesh>
          <boxGeometry args={[0.95, 0.5, 2.15]} />
          <meshStandardMaterial color="#0d0d12" roughness={0.32} metalness={0.55} />
        </mesh>
        {/* coffin lid */}
        <mesh position={[0, 0.36, -0.08]}>
          <boxGeometry args={[0.72, 0.24, 1.7]} />
          <meshStandardMaterial color="#17171f" roughness={0.42} metalness={0.35} />
        </mesh>
        {/* ember-crack seams down both flanks */}
        {[0.485, -0.485].map((x, i) => (
          <mesh key={i} position={[x, 0.02, 0]}>
            <boxGeometry args={[0.02, 0.07, 1.75]} />
            <meshStandardMaterial
              ref={(m) => {
                seamMats.current[i] = m;
              }}
              color="#1a0508"
              emissive="#ff3355"
              emissiveIntensity={0.9}
              toneMapped={false}
            />
          </mesh>
        ))}
        {/* brass lantern prow */}
        <group position={[0, 0.3, 1.18]}>
          <mesh>
            <cylinderGeometry args={[0.14, 0.2, 0.42, 10]} />
            <meshStandardMaterial color="#8a6a2f" roughness={0.35} metalness={0.85} />
          </mesh>
          <mesh position={[0, 0.02, 0]}>
            <sphereGeometry args={[0.11, 10, 10]} />
            <meshStandardMaterial
              ref={lanternMat}
              color="#fff2cf"
              emissive="#ffdf9e"
              emissiveIntensity={1.6}
              toneMapped={false}
            />
          </mesh>
          <mesh position={[0, 0.29, 0]}>
            <coneGeometry args={[0.17, 0.16, 10]} />
            <meshStandardMaterial color="#8a6a2f" roughness={0.4} metalness={0.8} />
          </mesh>
          <pointLight ref={lanternLight} color="#ffd9a0" intensity={1.15} distance={6} decay={2} />
        </group>
        {/* hanging shroud stern */}
        <group ref={shroud} position={[0, 0.05, -1.04]}>
          <mesh position={[0, -0.42, 0]}>
            <planeGeometry args={[0.88, 0.95]} />
            <meshStandardMaterial color="#26262e" roughness={0.9} metalness={0.05} side={DoubleSide} />
          </mesh>
        </group>
      </group>
    </group>
  );
}

"use client";

import { useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import { DoubleSide } from "three";
import type { Group, Mesh, MeshStandardMaterial, PointLight } from "three";
import { ENEMY_ATTACK, ENEMY_DORMANT, ENEMY_TELEGRAPH } from "@/protocol/generated";
import type { EnemyModelProps } from "../enemyLook";
import { SNAPSHRIKE_LOCK_PROG, prog01, yawToward } from "../enemyLook";

/**
 * Snapshrike — two angled chitin shards, scythe forelimbs, spark tail.
 * Accent #ff4436 (scarlet). Unit-radius model; the parent scales by body radius.
 * Idle: shiver + tail whip. LOCK: compresses; the frozen end of the lock flares the
 * eyes. DASH (ATTACK): springs forward, over-stretched.
 */
export function Snapshrike({ anim }: EnemyModelProps): JSX.Element {
  const root = useRef<Group>(null);
  const body = useRef<Group>(null);
  const tail = useRef<Group>(null);
  const sparks = useRef<Group>(null);
  const eyeMats = useRef<(MeshStandardMaterial | null)[]>([]);
  const light = useRef<PointLight>(null);

  useFrame((state) => {
    const r = root.current;
    if (!r || (r.parent !== null && !r.parent.visible)) return;
    const a = anim.current;
    const t = state.clock.elapsedTime;
    const dormant = a.state === ENEMY_DORMANT;
    const telegraph = a.state === ENEMY_TELEGRAPH;
    const attack = a.state === ENEMY_ATTACK;
    const p = prog01(a.prog);
    const locked = p >= SNAPSHRIKE_LOCK_PROG;
    const dim = dormant ? 0.15 : 1;

    if (telegraph || attack) {
      r.rotation.y = yawToward(a.ex, a.ez, a.tx, a.tz);
    }

    if (body.current) {
      body.current.position.y = Math.sin(t * (dormant ? 0.7 : 2.1)) * 0.06;
      if (attack) {
        body.current.scale.set(0.92, 0.9, 1.45); // spring release
      } else if (telegraph) {
        const squash = 1 - p * 0.24;
        body.current.scale.set(squash, 1 + p * 0.32, squash); // compress
      } else {
        body.current.scale.set(1, 1, 1);
      }
    }
    // shiver — amplitude climbs with lock progress
    r.rotation.z = Math.sin(t * 13.7) * 0.022 * (1 + p * 2.2);
    if (tail.current) {
      tail.current.rotation.y = Math.sin(t * (attack ? 7.5 : 3.4)) * (attack ? 0.72 : 0.46);
    }
    if (sparks.current) {
      sparks.current.scale.setScalar(1 + Math.sin(t * 9.3) * 0.14);
    }
    const eye = dormant ? 0.4 : attack ? 4.2 : telegraph ? (locked ? 5 : 1.6 + p * 1.6) : 1.7;
    for (const m of eyeMats.current) {
      if (m) m.emissiveIntensity = eye * dim;
    }
    if (light.current) {
      light.current.intensity = dormant ? 0.1 : 0.85 + (telegraph ? p * 0.8 : 0) + (attack ? 1.1 : 0);
    }
  });

  return (
    <group ref={root}>
      {/* shard body — two angled chitin cones */}
      <group ref={body}>
        <mesh position={[0.15, 0.08, 0.02]} rotation={[Math.PI / 2, 0.42, 0]}>
          <coneGeometry args={[0.3, 1.28, 6]} />
          <meshStandardMaterial color="#231014" roughness={0.55} metalness={0.35} flatShading emissive="#ff4436" emissiveIntensity={0.14} />
        </mesh>

interface TelegraphSlot {
  outer: Group | null;
  ring: Mesh | null;
  disc: Mesh | null;
  line: Mesh | null;
  ringMat: MeshBasicMaterial;
  discMat: MeshBasicMaterial;
  lineMat: MeshBasicMaterial;
}

function createSlot(): TelegraphSlot {
  const base = {
    transparent: true,
    opacity: 0,
    blending: AdditiveBlending,
    depthWrite: false,
    toneMapped: false,
    side: DoubleSide,
  } as const;
  return {
    outer: null,
    ring: null,
    disc: null,
    line: null,
    ringMat: new MeshBasicMaterial({ ...base, color: "#ffffff" }),
    discMat: new MeshBasicMaterial({ ...base, color: "#ffffff" }),
    lineMat: new MeshBasicMaterial({ ...base, color: "#ffffff" }),
  };
}

/**
 * Pooled telegraph drawings, one slot per enemy (max 24). Everything is additive,
 * bright, and tinted with the enemy's accent colour; alpha and radius are driven by
 * the wire `prog` (0..255). Slots exist only while the enemy is TELEGRAPH/ATTACK —
 * otherwise hidden. `target` is the telegraph anchor from the server payload.
 */
export function Telegraphs({ session }: { session: GameSession }): JSX.Element {
  const slots = useRef<TelegraphSlot[]>([]);
  if (slots.current.length === 0) {
    slots.current = Array.from({ length: MAX_TELEGRAPHS }, createSlot);
  }

  const ringGeo = useMemo(() => new RingGeometry(0.86, 1, 48), []);
  const discGeo = useMemo(() => new CircleGeometry(1, 40), []);
  const lineGeo = useMemo(() => new PlaneGeometry(1, 1), []);
  const palette = useMemo(() => ENEMY_LOOKS.map((l) => new Color(l.accent)), []);
  const fallbackColor = useMemo(() => new Color("#ffffff"), []);

  useEffect(
    () => () => {
      for (const slot of slots.current) {
        slot.ringMat.dispose();
        slot.discMat.dispose();
        slot.lineMat.dispose();
      }
      ringGeo.dispose();
      discGeo.dispose();
      lineGeo.dispose();
    },
    [ringGeo, discGeo, lineGeo],
  );

  // Ribbon lines orient with Euler order Y→X→Z (yaw then pitch). Set once.
  useEffect(() => {
    for (const slot of slots.current) {
      if (slot.line) slot.line.rotation.order = "YXZ";
    }
  }, []);

  useFrame((state) => {
    const now = performance.now();
    const t = state.clock.elapsedTime;
    let i = 0;
    for (const [id, o] of session.enemyPositions(now)) {
      if (i >= MAX_TELEGRAPHS) break;
      const slot = slots.current[i];
      if (!slot || !slot.outer) {
        i += 1;
        continue;

      const info = session.enemyInfo.get(id);
      if (!info || !isTelegraphState(info.state)) {
        slot.outer.visible = false;
        i += 1;
        continue;
      }

      const kind = info.kind;
      const look = enemyLookFor(kind);
      const col =
        kind !== undefined && kind >= 0 && kind < palette.length
          ? palette[kind] ?? fallbackColor
          : fallbackColor;

      const attack = info.state === ENEMY_ATTACK;
      const p = prog01(info.prog);
      const anchorX = info.target.x;
      const anchorY = info.target.y;
      const anchorZ = info.target.z;

      let ringVisible = false;
      let discVisible = false;
      let lineVisible = false;
      let ringX = anchorX;
      let ringY = anchorY + 0.06;
      let ringZ = anchorZ;
      let ringRadius = 1;
      let ringAlpha = telegraphAlpha(p);
      let discRadius = 1;
      let discAlpha = 0;
      let lineWidth = 0.2;
      let lineAlpha = telegraphAlpha(p);

      switch (look.behavior) {
        case "pallbearer":
          // KNEEL: straight line barge→victim + ring under the locked point
          lineVisible = true;
          lineWidth = 0.2;
          lineAlpha = attack ? 0.55 * (1 - p) : telegraphAlpha(p);
          ringVisible = true;
          ringRadius = PALLBEARER_LUNGE_RADIUS;
          ringAlpha = telegraphAlpha(p);
          break;
        case "snapshrike":
          // LOCK: full-width dash wedge, snapping brighter once the line freezes
          lineVisible = true;
          if (attack) {
            lineWidth = SNAPSHRIKE_DASH_HALF_WIDTH;
            lineAlpha = 0.6 * (1 - p);
          } else {
            lineWidth = SNAPSHRIKE_DASH_HALF_WIDTH * 2;
            lineAlpha = snapshrikeAlpha(p);
          }
          break;
        case "moteling":
          // INHALE: collapsing ring around the marked player; SHRIEK: ring snap
          ringVisible = true;
          if (attack) {
            ringRadius = expandingRingRadius(p, 4.0);
            ringAlpha = 0.9 * (1 - p);
          } else {
            ringRadius = motelingRingRadius(p);
            ringAlpha = 0.35 + 0.65 * p;
          }
          ringY = anchorY + 0.09;
          discVisible = true;
          discRadius = ringRadius;
          discAlpha = ringAlpha * 0.16;
          break;

          ringVisible = true;
          ringX = o.x;
          ringY = o.y + 0.06;
          ringZ = o.z;
          if (attack) {
            ringRadius = expandingRingRadius(p, RIFT_WARDEN_SHOVE_RADIUS);

          lineVisible = true;
          if (attack) {
            lineWidth = 0.34;
            lineAlpha = Math.max(0, 1 - p * 2.5);
          } else {
            lineWidth = TURRET_AIM_WIDTH;
            lineAlpha = telegraphAlpha(p);
          }
          break;

          ringVisible = true;
          ringX = o.x;
          ringY = o.y + 0.06;
          ringZ = o.z;

          ringAlpha =
            (Math.sin(t * 29 + i * 1.7) > 0 ? 0.95 : 0.28) *
            (attack ? Math.max(0.15, 1 - p) : telegraphAlpha(p));
          if (attack) {
            lineVisible = true;
            lineWidth = 0.5;
            lineAlpha = 0.6 * (1 - p);
          }
          break;

          // fretting spread — veins crawling outward from the marked tile
          ringVisible = true;
          ringRadius = expandingRingRadius(p, CORRUPTER_AURA_RADIUS);
          ringAlpha = attack ? 0.7 * (1 - p) : 0.3 + 0.6 * p;
          discVisible = true;
          discRadius = ringRadius;
          discAlpha = ringAlpha * 0.2;
          break;
      }

      slot.outer.visible = true;

      if (slot.ring) {
        slot.ring.visible = ringVisible;
        if (ringVisible) {
          slot.ring.position.set(ringX, ringY, ringZ);
          slot.ring.scale.set(ringRadius, ringRadius, 1);
          slot.ringMat.opacity = ringAlpha;
          slot.ringMat.color.copy(col);
        }
      }
      if (slot.disc) {
        slot.disc.visible = discVisible;
        if (discVisible) {
          slot.disc.position.set(ringX, ringY - 0.015, ringZ);
          slot.disc.scale.set(discRadius, discRadius, 1);
          slot.discMat.opacity = discAlpha;
          slot.discMat.color.copy(col);
        }
      }
      if (slot.line) {
        slot.line.visible = lineVisible;
        if (lineVisible) {
          const dx = anchorX - o.x;
          const dy = anchorY - o.y;
          const dz = anchorZ - o.z;
          const h = Math.hypot(dx, dz);
          const len = Math.hypot(h, dy);
          if (len < 0.05) {
            slot.line.visible = false;
          } else {
            slot.line.position.set((o.x + anchorX) / 2, (o.y + anchorY) / 2, (o.z + anchorZ) / 2);
            slot.line.rotation.y = Math.atan2(dx, dz);
            slot.line.rotation.x = Math.atan2(h, dy);
            slot.line.scale.set(lineWidth, len, 1);
            slot.lineMat.opacity = lineAlpha;
            slot.lineMat.color.copy(col);
          }
        }
      }
      i += 1;
    }
    for (; i < MAX_TELEGRAPHS; i += 1) {
      const slot = slots.current[i];
      if (slot && slot.outer) slot.outer.visible = false;
    }
  });

  return (
    <group>
      {slots.current.map((slot, i) => (
        <group
          key={i}
          ref={(g) => {
            slot.outer = g;
          }}
          visible={false}
        >
          <mesh
            ref={(m) => {
              slot.ring = m;
            }}
            geometry={ringGeo}
            material={slot.ringMat}
            rotation={[-Math.PI / 2, 0, 0]}
          />
          <mesh
            ref={(m) => {

            material={slot.discMat}

              slot.line = m;
            }}
            geometry={lineGeo}

"use client";

import { useRef } from "react";
import type { JSX } from "react";
import type { ComponentType } from "react";
import { useFrame } from "@react-three/fiber";
import type { Group } from "three";
import { ENEMY_DISGUISED, ENEMY_HUNTING } from "@/protocol/generated";
import type { GameSession } from "@/game/session";
import { Pallbearer } from "./enemies/Pallbearer";
import { Snapshrike } from "./enemies/Snapshrike";
import { Moteling } from "./enemies/Moteling";
import { RiftWarden } from "./enemies/RiftWarden";
import { GaleWight } from "./enemies/GaleWight";
import { LanternLeech } from "./enemies/LanternLeech";
import { BoltSentry } from "./enemies/BoltSentry";
import { FalseGift } from "./enemies/FalseGift";
import { CrumbleWisp } from "./enemies/CrumbleWisp";
import { Telegraphs } from "./Telegraphs";
import { enemyBehaviorFor } from "./enemyBehavior";
import type { EnemyAnim, EnemyModelProps } from "./enemyLook";
import { ENEMY_KIND_COUNT } from "./enemyLook";

export const MAX_ENEMIES = 24;

const MODEL_COMPONENTS: readonly ComponentType<EnemyModelProps>[] = [
  Pallbearer,
  Snapshrike,
  Moteling,
  RiftWarden,
  GaleWight,
  LanternLeech,
  BoltSentry,
  FalseGift,
  CrumbleWisp,
];

interface SlotRefs {
  outer: Group | null;
  models: (Group | null)[];
}

function emptySlot(): SlotRefs {
  return { outer: null, models: Array.from({ length: ENEMY_KIND_COUNT }, () => null) };
}

/**
 * Pooled enemy slots (fixed size, visibility toggled — never re-created).
 * Each slot pre-builds all nine roster models and shows the one matching the
 * server's `kind`. The slot is scaled by the server's hit-sphere radius; a
 * disguised False Gift is left unscaled so it is indistinguishable from a gift.
 * Telegraph drawings live in <Telegraphs /> (own pool, same size).
 * Props are unchanged: { session }.
 */
export function EnemyField({ session }: { session: GameSession }): JSX.Element {
  const slots = useRef<SlotRefs[]>([]);
  if (slots.current.length === 0) {
    slots.current = Array.from({ length: MAX_ENEMIES }, emptySlot);
  }
  const anims = useRef<EnemyAnim[]>([]);
  if (anims.current.length === 0) {
    anims.current = Array.from({ length: MAX_ENEMIES }, () => ({
      state: ENEMY_HUNTING,
      prog: 0,
      flags: 0,
      ex: 0,
      ey: 0,
      ez: 0,
      tx: 0,
      ty: 0,
      tz: 0,
    }));
  }

  useFrame(() => {
    const now = performance.now();
    let i = 0;
    for (const [id, o] of session.enemyPositions(now)) {
      if (i >= MAX_ENEMIES) break;
      const slot = slots.current[i];
      const anim = anims.current[i];
      if (!slot || !anim) {
        i += 1;
        continue;
      }
      const info = session.enemyInfo.get(id);
      const state = info?.state ?? ENEMY_HUNTING;
      const kind = info?.kind ?? 0;
      anim.state = state;
      anim.prog = info?.prog ?? 0;
      anim.flags = info?.flags ?? 0;
      anim.ex = o.x;
      anim.ey = o.y;
      anim.ez = o.z;
      anim.tx = info?.target.x ?? 0;
      anim.ty = info?.target.y ?? 0;
      anim.tz = info?.target.z ?? 0;

      const outer = slot.outer;
      if (outer) {
        outer.visible = true;
        outer.position.set(o.x, o.y, o.z);
        const disguised = enemyBehaviorFor(kind) === "mimic" && state === ENEMY_DISGUISED;
        outer.scale.setScalar(disguised ? 1 : session.enemyRadius.get(id) ?? 0.5);
      }
      for (let k = 0; k < ENEMY_KIND_COUNT; k += 1) {
        const m = slot.models[k];
        if (m) m.visible = k === kind;
      }
      i += 1;
    }
    for (; i < MAX_ENEMIES; i += 1) {
      const slot = slots.current[i];
      if (slot && slot.outer) slot.outer.visible = false;
    }
  });

  return (
    <group>
      {Array.from({ length: MAX_ENEMIES }, (_, i) => {
        const anim = anims.current[i];
        if (!anim) return null;
        return (
          <group
            key={i}
            ref={(g) => {
              const slot = slots.current[i];
              if (slot) slot.outer = g;
            }}
            visible={false}
          >
            {MODEL_COMPONENTS.map((Model, k) => (
              <group
                key={k}
                ref={(g) => {
                  const slot = slots.current[i];
                  if (slot) slot.models[k] = g;
                }}
                visible={false}
              >
                <Model anim={anim} />
              </group>
            ))}
          </group>
        );
      })}
      <Telegraphs session={session} />
    </group>
  );
}

"use client";

import { useEffect, useMemo, useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import { AdditiveBlending, MeshBasicMaterial } from "three";
import type { Group } from "three";
import type { GameSession } from "@/game/session";

export const MAX_PROJECTILES = 64;

/**
 * Pooled falling gale_wight shells (turret bolts share the same pool and visual):
 * a hot glowing shell with a vertical ion streak trailing above it, since shells
 * drop ~12 u onto their marked circle. Positions come from the newest server
 * snapshot; the streak is static-length and always points up.
 */
export function Projectiles({ session }: { session: GameSession }): JSX.Element {
  const slots = useRef<(Group | null)[]>(Array.from({ length: MAX_PROJECTILES }, () => null));
  const coreMat = useMemo(() => new MeshBasicMaterial({ color: "#fff2ff", toneMapped: false }), []);
  const shellMat = useMemo(
    () =>
      new MeshBasicMaterial({
        color: "#d94fff",
        transparent: true,
        opacity: 0.9,
        blending: AdditiveBlending,
        depthWrite: false,
        toneMapped: false,
      }),
    [],
  );
  const haloMat = useMemo(
    () =>
      new MeshBasicMaterial({
        color: "#b47aff",
        transparent: true,
        opacity: 0.42,
        blending: AdditiveBlending,
        depthWrite: false,
        toneMapped: false,
      }),
    [],
  );
  const streakMat = useMemo(
    () =>
      new MeshBasicMaterial({
        color: "#d94fff",
        transparent: true,
        opacity: 0.34,
        blending: AdditiveBlending,
        depthWrite: false,
        toneMapped: false,
      }),
    [],
  );
  const streakCoreMat = useMemo(
    () =>
      new MeshBasicMaterial({
        color: "#eec4ff",
        transparent: true,
        opacity: 0.62,
        blending: AdditiveBlending,
        depthWrite: false,
        toneMapped: false,
      }),
    [],
  );
  useEffect(
    () => () => {
      coreMat.dispose();
      shellMat.dispose();
      haloMat.dispose();
      streakMat.dispose();
      streakCoreMat.dispose();
    },
    [coreMat, shellMat, haloMat, streakMat, streakCoreMat],
  );

  useFrame(() => {
    const now = performance.now();
    let i = 0;
    for (const p of session.projectilePositions(now).values()) {
      if (i >= MAX_PROJECTILES) break;
      const slot = slots.current[i];
      if (slot) {
        slot.visible = true;
        slot.position.set(p.x, p.y, p.z);
      }
      i += 1;
    }
    for (; i < MAX_PROJECTILES; i += 1) {
      const slot = slots.current[i];
      if (slot) slot.visible = false;
    }
  });

  return (
    <group>
      {Array.from({ length: MAX_PROJECTILES }, (_, i) => (
        <group
          key={i}
          ref={(g) => {
            slots.current[i] = g;
          }}
          visible={false}

            <sphereGeometry args={[0.15, 10, 10]} />
          </mesh>
          <mesh material={shellMat}>
            <sphereGeometry args={[0.25, 10, 10]} />
          </mesh>
          <mesh material={haloMat}>
            <sphereGeometry args={[0.4, 8, 8]} />
          </mesh>
          {/* vertical streak above the falling shell */}
          <mesh material={streakMat} position={[0, 0.95, 0]}>
            <cylinderGeometry args={[0.06, 0.17, 1.7, 6, 1, true]} />
          </mesh>
          <mesh material={streakCoreMat} position={[0, 0.7, 0]}>
            <cylinderGeometry args={[0.032, 0.09, 1.1, 6, 1, true]} />
          </mesh>
        </group>
      ))}
    </group>
  );
}
