"use client";

import { useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import { AdditiveBlending } from "three";
import type { Mesh, MeshStandardMaterial } from "three";
import type { Vec3 } from "@/protocol/generated";
import { COMMON } from "./theme";

interface FrameBarData {
  position: [number, number, number];
  size: [number, number, number];
}

// Thin flat frame around a 1.6-wide square on top of the base pad. The bars
// along X span the full width; the bars along Z span the gap between them so
// the corners do not double-glow. 0.08 tall, resting on the base (top at
// y = 0.2). The pillar sits on the same top surface.
const BASE_TOP_Y = 0.2;
const FRAME_WIDTH = 1.6;
const FRAME_THICKNESS = 0.08;
const FRAME_HEIGHT = 0.08;

const OPEN_PILLAR_HEIGHT = 80;
const OPEN_PILLAR_CORE_RADIUS = 0.35;
const OPEN_PILLAR_BEAM_RADIUS = 0.9;
const OPEN_PILLAR_CENTER_Y = BASE_TOP_Y + OPEN_PILLAR_HEIGHT / 2;
const CORE_EMISSIVE = 4;

const CLOSED_PILLAR_HEIGHT = 8;
const CLOSED_PILLAR_RADIUS = 0.45;
const CLOSED_PILLAR_CENTER_Y = BASE_TOP_Y + CLOSED_PILLAR_HEIGHT / 2;

const FRAME_EMISSIVE_OPEN = 1.5;
const FRAME_EMISSIVE_CLOSED = 0.35;

// Outer beam pulse (open state only): opacity 0.28..0.42, radius 1.0..1.12.
const BEAM_BASE_OPACITY = 0.28;
const BEAM_OPACITY_RANGE = 0.14;
const BEAM_SCALE_RANGE = 0.12;

const FRAME_BARS: ReadonlyArray<FrameBarData> = [
  {
    position: [0, BASE_TOP_Y + FRAME_HEIGHT / 2, FRAME_WIDTH / 2 - FRAME_THICKNESS / 2],
    size: [FRAME_WIDTH, FRAME_HEIGHT, FRAME_THICKNESS],
  },
  {
    position: [0, BASE_TOP_Y + FRAME_HEIGHT / 2, -(FRAME_WIDTH / 2 - FRAME_THICKNESS / 2)],
    size: [FRAME_WIDTH, FRAME_HEIGHT, FRAME_THICKNESS],
  },
  {
    position: [FRAME_WIDTH / 2 - FRAME_THICKNESS / 2, BASE_TOP_Y + FRAME_HEIGHT / 2, 0],
    size: [FRAME_THICKNESS, FRAME_HEIGHT, FRAME_WIDTH - FRAME_THICKNESS * 2],
  },
  {
    position: [-(FRAME_WIDTH / 2 - FRAME_THICKNESS / 2), BASE_TOP_Y + FRAME_HEIGHT / 2, 0],
    size: [FRAME_THICKNESS, FRAME_HEIGHT, FRAME_WIDTH - FRAME_THICKNESS * 2],
  },
];

export function Beacon(props: { position: Vec3; open: boolean }): JSX.Element {
  const beamMeshRef = useRef<Mesh>(null);
  const beamMaterialRef = useRef<MeshStandardMaterial>(null);

  useFrame((state) => {
    if (!props.open) {
      return;
    }
    const beam = beamMeshRef.current;
    const material = beamMaterialRef.current;
    if (beam === null || material === null) {
      return;
    }
    const pulse = Math.sin(state.clock.elapsedTime * 2.4) * 0.5 + 0.5;
    const scale = 1 + pulse * BEAM_SCALE_RANGE;
    beam.scale.x = scale;
    beam.scale.z = scale;
    material.opacity = BEAM_BASE_OPACITY + pulse * BEAM_OPACITY_RANGE;
  });

  const open = props.open;
  const frameColor = open ? COMMON.beacon.beam : COMMON.beacon.closed;
  const frameEmissive = open ? FRAME_EMISSIVE_OPEN : FRAME_EMISSIVE_CLOSED;

  return (
    <group position={[props.position.x, props.position.y, props.position.z]}>
      <mesh position={[0, 0.1, 0]}>
        <boxGeometry args={[1.4, 0.2, 1.4]} />
        <meshStandardMaterial
          color={open ? COMMON.beacon.beam : COMMON.beacon.closed}
          emissive={open ? COMMON.beacon.beam : COMMON.beacon.closed}
          emissiveIntensity={open ? 0.6 : 0.05}
        />
      </mesh>
      {FRAME_BARS.map((bar, index) => (
        <mesh key={index} position={bar.position}>
          <boxGeometry args={bar.size} />
          <meshStandardMaterial
            color={frameColor}
            emissive={frameColor}
            emissiveIntensity={frameEmissive}
          />
        </mesh>
      ))}
      {open ? (
        <>
          <mesh position={[0, OPEN_PILLAR_CENTER_Y, 0]}>
            <cylinderGeometry
              args={[OPEN_PILLAR_CORE_RADIUS, OPEN_PILLAR_CORE_RADIUS, OPEN_PILLAR_HEIGHT, 12]}
            />
            <meshStandardMaterial
              color={COMMON.beacon.core}
              emissive={COMMON.beacon.core}
              emissiveIntensity={CORE_EMISSIVE}
              toneMapped={false}
              fog={false}
            />
          </mesh>
          <mesh ref={beamMeshRef} position={[0, OPEN_PILLAR_CENTER_Y, 0]}>
            <cylinderGeometry
              args={[
                OPEN_PILLAR_BEAM_RADIUS,
                OPEN_PILLAR_BEAM_RADIUS,
                OPEN_PILLAR_HEIGHT,
                16,
                1,
                true,
              ]}
            />
            <meshStandardMaterial
              ref={beamMaterialRef}
              color={COMMON.beacon.beam}
              emissive={COMMON.beacon.beam}
              emissiveIntensity={1.2}
              transparent
              opacity={BEAM_BASE_OPACITY + BEAM_OPACITY_RANGE * 0.5}
              blending={AdditiveBlending}
              depthWrite={false}
              fog={false}
            />
          </mesh>
        </>
      ) : (
        <mesh position={[0, CLOSED_PILLAR_CENTER_Y, 0]}>
          <cylinderGeometry
            args={[CLOSED_PILLAR_RADIUS, CLOSED_PILLAR_RADIUS, CLOSED_PILLAR_HEIGHT, 12]}
          />
          <meshStandardMaterial
            color={COMMON.beacon.closed}
            transparent
            opacity={0.25}
            depthWrite={false}
          />
        </mesh>
      )}
    </group>
  );
}

"use client";

import { useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import { AdditiveBlending } from "three";
import type { Group, Mesh, MeshStandardMaterial } from "three";
import type { Vec3 } from "@/protocol/generated";
import { COMMON } from "./theme";

const SHARD_OFFSETS: ReadonlyArray<readonly [number, number, number]> = [
  [0.95, 0.25, 0],
  [-0.95, 0.25, 0],
  [0, 0.95, 0.35],
  [0, -0.95, -0.35],
];

const CORE_EMISSIVE = 2.2;
const SHARD_EMISSIVE = 0.8;
const AURA_RADIUS = 1.3;
const AURA_BASE_OPACITY = 0.12;
const AURA_SCALE_RANGE = 0.06;

export function Chaser(props: { position: Vec3 }): JSX.Element {
  const coreRef = useRef<Mesh>(null);
  const shardsRef = useRef<Group>(null);
  const auraMeshRef = useRef<Mesh>(null);
  const auraMaterialRef = useRef<MeshStandardMaterial>(null);

  useFrame((state, delta) => {
    const core = coreRef.current;
    if (core !== null) {
      core.rotation.y += delta * 2.5;
    }
    const shards = shardsRef.current;
    if (shards !== null) {
      shards.rotation.y -= delta * 1.8;
      shards.rotation.x = Math.sin(state.clock.elapsedTime * 1.3) * 0.35;
    }
    const aura = auraMeshRef.current;
    if (aura !== null) {
      const scale = 1 + Math.sin(state.clock.elapsedTime * 2.1) * AURA_SCALE_RANGE;
      aura.scale.x = scale;
      aura.scale.y = scale;
      aura.scale.z = scale;
    }
    const auraMaterial = auraMaterialRef.current;
    if (auraMaterial !== null) {
      auraMaterial.opacity =
        AURA_BASE_OPACITY +
        Math.sin(state.clock.elapsedTime * 2.1) * AURA_SCALE_RANGE * 0.5;
    }
  });

  return (
    <group position={[props.position.x, props.position.y, props.position.z]}>
      <mesh ref={coreRef}>
        <octahedronGeometry args={[0.7]} />
        <meshStandardMaterial
          color={COMMON.enemy.body}
          emissive={COMMON.enemy.glow}
          emissiveIntensity={CORE_EMISSIVE}
          toneMapped={false}
        />
      </mesh>
      <group ref={shardsRef}>
        {SHARD_OFFSETS.map((offset, index) => (
          <mesh key={index} position={offset as [number, number, number]}>
            <octahedronGeometry args={[0.14]} />
            <meshStandardMaterial
              color={COMMON.enemy.body}
              emissive={COMMON.enemy.glow}
              emissiveIntensity={SHARD_EMISSIVE}
              toneMapped={false}
            />
          </mesh>
        ))}
      </group>
      <mesh ref={auraMeshRef}>
        <sphereGeometry args={[AURA_RADIUS, 16, 16]} />
        <meshStandardMaterial
          ref={auraMaterialRef}
          color={COMMON.enemy.glow}
          transparent
          opacity={AURA_BASE_OPACITY}
          blending={AdditiveBlending}
          depthWrite={false}
        />
      </mesh>
    </group>
  );
}

import type { CSSProperties, JSX } from "react";

import { giftsLabel } from "./format";
import { COMMON } from "../scene/theme";

const COUNTER_TEXT_SHADOW: CSSProperties["textShadow"] =
  "0 2px 10px rgba(0, 0, 0, 0.9), 0 0 4px rgba(0, 0, 0, 0.9)";

type GameHudProps = {
  level: number;
  giftsRemaining: number;
  giftsTotal: number;
};

export function GameHud(props: GameHudProps): JSX.Element {
  const { level, giftsRemaining, giftsTotal } = props;
  const beaconOpen: boolean = giftsRemaining === 0 && giftsTotal > 0;
  const collected: number = Math.min(
    Math.max(giftsTotal - giftsRemaining, 0),
    giftsTotal,
  );

  return (
    <div className="pointer-events-none absolute inset-0 font-mono">
      <div className="absolute left-1/2 top-3 -translate-x-1/2">
        <div className="flex items-center justify-center gap-5 rounded-md border border-white/15 bg-black/60 px-5 py-2">
          <div className="whitespace-nowrap text-sm tracking-widest text-white/70">
            LEVEL{" "}
            <span data-testid="level-number" className="text-xl font-bold text-white">
              {level}
            </span>
          </div>
        </div>
        {beaconOpen ? (
          <div
            data-testid="beacon-open"
            className="mt-1 text-center text-sm font-bold tracking-widest text-[#38e8ff]"
          >
            BEACON OPEN
          </div>
        ) : null}
      </div>
      <div className="absolute bottom-10 left-1/2 flex -translate-x-1/2 items-center gap-3">
        <span
          aria-hidden="true"
          className="inline-block h-4 w-4"
          style={{ backgroundColor: COMMON.gift.body }}
        />
        <span
          data-testid="gift-counter"
          className="whitespace-nowrap text-5xl font-bold"
          style={{ color: COMMON.ui.counter, textShadow: COUNTER_TEXT_SHADOW }}
        >
          {collected} / {giftsTotal}
        </span>
      </div>
      <p data-testid="gifts-remaining" className="sr-only">
        {giftsLabel(giftsRemaining, giftsTotal)}
      </p>
    </div>
  );
}
