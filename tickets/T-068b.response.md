# Map chunk format (binding for the catalog and the assembler)

One file: `design/chunks.json`:

    { "version": 1, "chunks": [ CHUNK, ... ] }

CHUNK:

    {
      "id": "ch05_stair_of_embers",          // unique, [a-z0-9_]
      "name": "Stair of Embers",
      "role": "hub" | "path" | "junction" | "dead_end",
      "min_level": 1, "difficulty": 1,      // 1..3
      "tags": ["ramp", "climb"],
      "platforms": [ { "kind": "NORMAL" | "CRUMBLE" | "JUMP_PAD" | "RAMP_PX" | "RAMP_NX" | "RAMP_PZ" | "RAMP_NZ", "min": [x, y, z], "max": [x, y, z] } ],
      "sockets": [ { "id": "a", "pos": [x, y, z], "facing": "N" | "E" | "S" | "W", "width": 4 } ],
      "gift_slots": [ { "pos": [x, y, z], "elevated": false } ],
      "enemy_slots": [ [x, y, z] ],          // hub only: exactly 8, around the beacon
      "altar_slots": [ [x, y, z] ],
      "beacon": [x, y, z]                    // hub only: beacon dais top centre
    }

Coordinates (units = metres, player 0.8 x 1.8 x 0.8):
- Chunk-local, right-handed, Y up. Facing: N = +Z, E = +X, S = -Z, W = -X (direction pointing OUT of the chunk).
- The chunk's footprint is the XZ bounding box of its platforms. A socket's `pos` lies exactly on that footprint's
  edge on the side it faces; `pos.y` is the walking height (a platform top) at the socket; a platform whose top is
  at `pos.y` touches that edge across the full socket `width` (>= 3) centred on `pos`, so two joined chunks form a
  continuous walkway.
- Joining: chunk B is rotated by a multiple of 90 degrees about Y (sockets, slots and platforms rotate with it) and
  translated so that B's socket `pos` equals A's socket `pos`, with opposite facings (A faces N, B's socket faces S).
- Platforms are axis-aligned boxes (after 90 degree rotations they stay axis-aligned); ramps are axis-aligned wedges
  (see Ramps). Walk surface = `max.y` for boxes; ramps walk on the sloped line (see Ramps).
  Typical thickness 0.5-1.0 u; pillars are tall thin NORMAL boxes under walkways (decorative support, still solid).
- Widths: walkways >= 3 u across the direction of travel; hubs/junctions >= 6 u. A ramp's rise over its run must not
  exceed 0.5 (about 26.6 degrees), its run must be at least 2.0 u, and its width across the direction of travel must be
  at least 3 u. Stairs no longer exist anywhere in the game.
- Jumps inside a chunk must stay inside the design hop-envelope table (M3_DESIGN 4.1, "design max gap").
- JUMP_PAD: a NORMAL-sized box (>= 2 x 2 u) that launches the player to ~4 u above its top.
- Gift slots: `pos.y = platform top + 0.3`, on NORMAL platforms only (never CRUMBLE/JUMP_PAD), >= 2.0 u apart,
  >= 0.5 u from platform edges. Altar slots: on NORMAL tops, >= 1.5 u from edges. Gift slots line the walkways
  (about every 2.0-2.6 u) so the assembler can fill up to the level's gift count.
- Stacking: any two platforms of different layers that overlap in XZ must have >= 4.0 u vertical clearance between
  the lower one's top and the upper one's bottom (the same rule applies between chunks).

Ramps:
- Four kinds: `RAMP_PX` low at min.x high at max.x; `RAMP_NX` low at max.x high at min.x;
  `RAMP_PZ` low at min.z high at max.z; `RAMP_NZ` low at max.z high at min.z.
- `bounds.min.y` is the surface height at the LOW edge; `bounds.max.y` is the surface height at the HIGH edge.
- The walk surface is a straight line between them across the slope axis: surface height at (x, z) is linear along
  that axis from `bounds.min.y` (low edge) to `bounds.max.y` (high edge).
- The solid volume is everything below that surface. The low end has no thickness; the high end is
  `max.y - min.y` tall.
- Ramps use the same socket, stacking-clearance and hop-envelope rules as boxes. Ramp limits (rise/run <= 0.5,
  run >= 2.0 u, width >= 3 u) are stated under Widths above.

/** Phase offset per platform id so shakes/blinks do not line up. */
function phaseForId(id: number): number {
  return (id * 1.61803398875) % (Math.PI * 2);
}

  localScale: Vec3Tuple;

/** Compose scale * (optional Y rotation) into scratchMatrix and place it at (x, y, z). */
function composeInstanceMatrix(localScale: Vec3Tuple, spin: number, x: number, y: number, z: number): void {
  scratchMatrix.makeScale(localScale[0], localScale[1], localScale[2]);
  if (spin !== 0) {
    scratchRotation.makeRotationY(spin);
    scratchMatrix.premultiply(scratchRotation);
  }
  scratchMatrix.setPosition(x, y, z);
}

/**
 * Unit wedge: 1x1 footprint, flat bottom at y = -0.5, zero thickness at the low edge
 * (z = -0.5), 1 u tall at the high edge (z = +0.5). Non-indexed with computed flat
 * normals; scale per instance by the platform's size and rotate per ramp kind.
 */
function createWedgeGeometry(): BufferGeometry {
  // Corner shorthand: L* = low edge, H* = high edge, B* = bottom at high edge.
  const L0: Vec3Tuple = [-0.5, -0.5, -0.5];
  const L1: Vec3Tuple = [0.5, -0.5, -0.5];
  const B0: Vec3Tuple = [-0.5, -0.5, 0.5];
  const B1: Vec3Tuple = [0.5, -0.5, 0.5];
  const H0: Vec3Tuple = [-0.5, 0.5, 0.5];
  const H1: Vec3Tuple = [0.5, 0.5, 0.5];
  const triangles: Array<[Vec3Tuple, Vec3Tuple, Vec3Tuple]> = [
    // Bottom face (normal -Y).
    [L0, B1, B0],
    [L0, L1, B1],
    // Sloped top face (normal 0/1/-1 up toward the low edge).
    [L0, H1, L1],
    [H0, H1, L0],
    // High end (normal +Z). The low end has zero thickness, so no face.
    [B0, B1, H1],
    [B0, H1, H0],
    // Side faces (normals +/-X).
    [L1, H1, B1],
    [L0, B0, H0],
  ];
  const positions = new Float32Array(triangles.length * 9);
  const uvs = new Float32Array(triangles.length * 6);
  let p = 0;
  let u = 0;
  for (const [a, b, c] of triangles) {
    for (const v of [a, b, c]) {
      positions[p] = v[0];
      positions[p + 1] = v[1];
      positions[p + 2] = v[2];
      p += 3;
      // Planar UVs sized to the unit wedge; the surface shader checkers in world space.
      uvs[u] = v[0] + 0.5;
      uvs[u + 1] = Math.abs(v[1] + 0.5) > Math.abs(v[2] + 0.5) ? v[1] + 0.5 : v[2] + 0.5;
      u += 2;
    }
  }
  const geometry = new BufferGeometry();
  geometry.setAttribute("position", new BufferAttribute(positions, 3));
  geometry.setAttribute("uv", new BufferAttribute(uvs, 2));
  geometry.computeVertexNormals();
  return geometry;
}

/** Resolve the tile palette to concrete colors, applying the ice tint when frozen. */
function resolveSurfacePalette(theme: Theme, icy: boolean): SurfacePalette {
  return {
    light: icy ? iceTint(theme.tile.light) : theme.tile.light,
    side: icy ? iceTint(theme.tile.side || SIDE_TINT) : theme.tile.side || SIDE_TINT,
    dark: icy ? iceTint(theme.tile.dark) : theme.tile.dark,
    edge: icy ? iceTint(theme.tile.edge) : theme.tile.edge,
  };
}

function injectVertex(shader: { vertexShader: string }): void {
  shader.vertexShader = shader.vertexShader
    .replace("#include <common>", buildVertexHeader())
    .replace("#include <begin_vertex>", VERTEX_BEGIN);
}

function createSurfaceMaterial(
  kind: number,
  palette: SurfacePalette,
  icy: boolean,
  timeUniform: { value: number },
): MeshStandardMaterial {
  const isCrumble = kind === PLATFORM_CRUMBLE;
  const mat = new MeshStandardMaterial({
    color: "#ffffff",
    roughness: icy ? ICE_ROUGHNESS : TOP_ROUGHNESS,
    metalness: 0,
    emissive: "#000000",
  });
  // The fragment header is generated from this exact map, so every uniform
  // assigned on shader.uniforms is declared in the shader.
  const uniforms = buildSurfaceUniforms(palette, icy, isCrumble, timeUniform);
  mat.onBeforeCompile = (shader) => {
    Object.assign(shader.uniforms, uniforms);
    injectVertex(shader);
    shader.fragmentShader = shader.fragmentShader
      .replace("#include <common>", buildFragmentHeader(uniforms))
      .replace("#include <color_fragment>", SURFACE_FRAGMENT_COLOR)
      .replace("#include <emissivemap_fragment>", SURFACE_FRAGMENT_EMISSIVE);
  };
  mat.customProgramCacheKey = () => (isCrumble ? "vw-platform-crumble" : "vw-platform-normal");
  return mat;
}

function createPadMaterial(timeUniform: { value: number }): MeshStandardMaterial {
  const mat = new MeshStandardMaterial({
    color: "#ffffff",
    roughness: PAD_ROUGHNESS,
    metalness: 0.35,
    emissive: "#000000",
  });
  const uniforms = buildPadUniforms(timeUniform);
  mat.onBeforeCompile = (shader) => {
    Object.assign(shader.uniforms, uniforms);
    injectVertex(shader);
    shader.fragmentShader = shader.fragmentShader

}

function buildGroups(platforms: readonly Platform[], theme: Theme, icy: boolean): BuiltPlatforms {
  const timeUniform = { value: 0 };
  const holder = new Group();
  const records: InstanceRecord[] = [];
  const palette = resolveSurfacePalette(theme, icy);

  for (const kind of [PLATFORM_NORMAL, PLATFORM_CRUMBLE, PLATFORM_JUMP_PAD]) {
    const list = platforms.filter((p) => p.kind === kind);
    const geometry = new BoxGeometry(1, 1, 1);
    const material =
      kind === PLATFORM_JUMP_PAD
        ? createPadMaterial(timeUniform)
        : createSurfaceMaterial(kind, palette, icy, timeUniform);
    const count = Math.max(list.length, 1);
    const mesh = new InstancedMesh(geometry, material, count);
    mesh.count = list.length;
    mesh.frustumCulled = false; // instances are spread across the map; bounding sphere of a unit box is wrong
    mesh.instanceMatrix.setUsage(DynamicDrawUsage);
    const attr = new InstancedBufferAttribute(new Float32Array(count * 4), 4);
    attr.setUsage(DynamicDrawUsage);
    geometry.setAttribute("aState", attr);

    list.forEach((p, i) => {
      const size = platformSize(p);
      const center = platformCenter(p);
      composeInstanceMatrix(size, 0, center[0], center[1], center[2]);
      mesh.setMatrixAt(i, scratchMatrix);
      const pillar = isPillar(size);
      attr.setXYZW(i, 0, 0, 1, pillar ? 1 : 0);
      records.push({
        id: p.id,
        center,
        size,
        localScale: size,
        spin: 0,
        phase: phaseForId(p.id),
        pillar,
        mesh,
        attr,
        index: i,
        touched: false,
        activeFrame: false,
      });
    });
    mesh.instanceMatrix.needsUpdate = true;
    attr.needsUpdate = true;
    holder.add(mesh);
  }

  // Ramps: one shared InstancedMesh with a unit wedge geometry, rotated per ramp kind.
  // Ramps use the same non-crumble surface material as NORMAL platforms and participate in
  // the same per-instance tile-state animation (shake / sink), so a ramp with WARN/GONE
  // tiles shakes and sinks exactly like a box.
  const rampList = platforms.filter((p) => rampInfo(p.kind).isRamp);
  const rampGeometry = createWedgeGeometry();
  const rampMaterial = createSurfaceMaterial(PLATFORM_NORMAL, palette, icy, timeUniform);
  const rampCount = Math.max(rampList.length, 1);
  const rampMesh = new InstancedMesh(rampGeometry, rampMaterial, rampCount);
  rampMesh.count = rampList.length;
  rampMesh.frustumCulled = false;
  rampMesh.instanceMatrix.setUsage(DynamicDrawUsage);
  const rampAttr = new InstancedBufferAttribute(new Float32Array(rampCount * 4), 4);
  rampAttr.setUsage(DynamicDrawUsage);
  rampGeometry.setAttribute("aState", rampAttr);

  rampList.forEach((p, i) => {
    const size = platformSize(p);
    const center = platformCenter(p);
    const info = rampInfo(p.kind);
    const alongX = info.axis === "x";
    const spin = info.sign >= 0 ? (alongX ? Math.PI / 2 : 0) : alongX ? -Math.PI / 2 : Math.PI;
    // The wedge's run lies on local z; for X-axis ramps we scale local z by the world x extent.
    const localScale: Vec3Tuple = alongX ? [size[2], size[1], size[0]] : size;
    composeInstanceMatrix(localScale, spin, center[0], center[1], center[2]);
    rampMesh.setMatrixAt(i, scratchMatrix);
    const pillar = isPillar(size);
    rampAttr.setXYZW(i, 0, 0, 1, pillar ? 1 : 0);
    records.push({
      id: p.id,
      center,
      size,
      localScale,
      spin,
      phase: phaseForId(p.id),
      pillar,
      mesh: rampMesh,
      attr: rampAttr,
      index: i,
      touched: false,
      activeFrame: false,
    });
  });
  rampMesh.instanceMatrix.needsUpdate = true;
  rampAttr.needsUpdate = true;
  holder.add(rampMesh);

  return { holder, records, timeUniform };
}

function disposeBuilt(built: BuiltPlatforms): void {
  for (const child of [...built.holder.children]) {
    if (child instanceof InstancedMesh) {
      child.geometry.dispose();
      const material = child.material;
      if (Array.isArray(material)) {
        for (const m of material) m.dispose();
      } else {
        material.dispose();
      }
    }
  }
}

function resetRecord(rec: InstanceRecord): void {
  composeInstanceMatrix(rec.localScale, rec.spin, rec.center[0], rec.center[1], rec.center[2]);
  rec.mesh.setMatrixAt(rec.index, scratchMatrix);
  rec.attr.setXYZW(rec.index, 0, 0, 1, rec.pillar ? 1 : 0);
  rec.mesh.instanceMatrix.needsUpdate = true;
  rec.attr.needsUpdate = true;
  rec.touched = false;
}

/**
 * Platforms at scale: four InstancedMeshes (NORMAL, CRUMBLE, JUMP_PAD boxes, plus one
 * shared wedge mesh for all ramp kinds), world-space procedural checkerboard tops, glowing
 * rim on the top face, dark-stone shader path for tall thin platforms (pillars). Tile
 * states animate via per-instance matrix/attribute writes — no geometry rebuilds.
 */
export function Platforms(props: {
  platforms: readonly Platform[];
  theme: Theme;
  tiles: readonly TileState[];
  icy: boolean;
}): JSX.Element {
  const { platforms, theme, tiles, icy } = props;
  const built = useMemo(() => buildGroups(platforms, theme, icy), [platforms, theme, icy]);
  const builtRef = useRef<BuiltPlatforms | null>(null);
  useEffect(() => {
    builtRef.current = built;
  }, [built]);
  useEffect(() => () => disposeBuilt(built), [built]);

  const tilesRef = useRef(tiles);
  const tileMapRef = useRef(new Map<number, number>());
  const goneStartRef = useRef(new Map<number, number>());
  useEffect(() => {
    tilesRef.current = tiles;
  }, [tiles]);
  useEffect(() => {
    goneStartRef.current.clear();
  }, [built]);

  useFrame((state) => {
    const current = builtRef.current;
    if (!current) return;
    const now = state.clock.elapsedTime;
    current.timeUniform.value = now;
    const tileMap = tileMapRef.current;
    tileMap.clear();
    for (const t of tilesRef.current) {
      tileMap.set(t.id, t.state);
      if (t.state === TILE_GONE && !goneStartRef.current.has(t.id)) {
        goneStartRef.current.set(t.id, now);
      }
    }

    for (const rec of current.records) {
      // Per-instance animation bookkeeping on the three.js records built once per level; mutated only here in
      // useFrame (never during render), which is the intended R3F pattern.
      // eslint-disable-next-line react-hooks/immutability
      rec.activeFrame = false;
      const t = tileMap.get(rec.id);
      if (t !== TILE_WARN && t !== TILE_FRETTED && t !== TILE_GONE) continue;

      let ox = 0;
      let oz = 0;
      let sink = 0;
      let fade = 1;
      let hide = false;
      if (t === TILE_WARN) {
        ox = Math.sin(now * 34 + rec.phase) * SHAKE_AMP;
        oz = Math.cos(now * 27 + rec.phase) * SHAKE_AMP;
      }
      if (t === TILE_GONE) {
        const start = goneStartRef.current.get(rec.id) ?? now;
        const k = Math.min(1, Math.max(0, (now - start) / GONE_SINK_TIME));
        sink = k * k * (GONE_SINK_DEPTH + rec.size[1]);
        fade = 1 - k;
        if (k >= 1) {
          fade = 0;
          hide = true;
        }
      }

      if (hide) {
        composeInstanceMatrix([0.0001, 0.0001, 0.0001], rec.spin, rec.center[0] + ox, rec.center[1] - sink, rec.center[2] + oz);
      } else {
        composeInstanceMatrix(rec.localScale, rec.spin, rec.center[0] + ox, rec.center[1] - sink, rec.center[2] + oz);
      }
      rec.mesh.setMatrixAt(rec.index, scratchMatrix);
      rec.attr.setXYZW(rec.index, t === TILE_WARN ? 1 : 0, t === TILE_FRETTED ? 1 : 0, fade, rec.pillar ? 1 : 0);
      rec.mesh.instanceMatrix.needsUpdate = true;
      rec.attr.needsUpdate = true;
      rec.touched = true;
      rec.activeFrame = true;
    }
    for (const rec of current.records) {
      if (rec.touched && !rec.activeFrame) resetRecord(rec);
    }
  });

  return <primitive object={built.holder} />;
}

// Pure platform math helpers — no three.js imports (unit-testable).
import type { Platform } from "@/protocol/generated";

export type Vec3Tuple = [number, number, number];

// Ramp kinds — mirror web/src/protocol/generated.ts (redeclared so this module stays runtime-dependency-free for tests).
const PLATFORM_RAMP_PX = 3;
const PLATFORM_RAMP_NX = 4;
const PLATFORM_RAMP_PZ = 5;
const PLATFORM_RAMP_NZ = 6;

export function platformSize(platform: Platform): Vec3Tuple {
  const { min, max } = platform.bounds;
  return [max.x - min.x, max.y - min.y, max.z - min.z];
}

export function platformCenter(platform: Platform): Vec3Tuple {
  const { min, max } = platform.bounds;
  return [(min.x + max.x) / 2, (min.y + max.y) / 2, (min.z + max.z) / 2];
}

/**
 * Ticket T-057b1: a platform is a "pillar" when its height exceeds 3x its
 * larger horizontal extent. Pillars render as dark stone, without the
 * checkerboard top.
 */
export function isPillar(size: Vec3Tuple): boolean {
  const [w, h, d] = size;
  return h > 3 * Math.max(w, d);
}

export type RampAxis = "x" | "z";

export interface RampInfo {
  /** True for the four ramp kinds. */
  isRamp: boolean;
  /** Slope axis for ramps, else null. */
  axis: RampAxis | null;
  /** Sign along the axis where the surface rises: +1 rises toward +axis, -1 rises toward -axis, 0 for non-ramps. */
  sign: 1 | -1 | 0;
}

/**
 * Reports whether a platform kind is a ramp and, for ramps, the slope axis and
 * the direction the surface rises in. Renderer and decor placement share this
 * so they agree on what a ramp is and which way it slopes.
 */
export function rampInfo(kind: number): RampInfo {
  switch (kind) {
    case PLATFORM_RAMP_PX:
      return { isRamp: true, axis: "x", sign: 1 };
    case PLATFORM_RAMP_NX:
      return { isRamp: true, axis: "x", sign: -1 };
    case PLATFORM_RAMP_PZ:
      return { isRamp: true, axis: "z", sign: 1 };
    case PLATFORM_RAMP_NZ:
      return { isRamp: true, axis: "z", sign: -1 };
    default:
      return { isRamp: false, axis: null, sign: 0 };
  }
}

/**
 * Walk-surface height of a platform at world (x, z). For ramps this is the
 * sloped line between the low edge (bounds.min.y) and the high edge
 * (bounds.max.y); for other kinds it is bounds.max.y. (x, z) are clamped to
 * the platform's footprint.
 */
export function rampSurfaceHeight(platform: Platform, x: number, z: number): number {
  const info = rampInfo(platform.kind);
  if (!info.isRamp || info.axis === null || info.sign === 0) return platform.bounds.max.y;
  const { min, max } = platform.bounds;
  const span = info.axis === "x" ? max.x - min.x : max.z - min.z;
  const coord = info.axis === "x" ? x - min.x : z - min.z;
  const t = span > 0 ? Math.min(1, Math.max(0, coord / span)) : 0;
  const k = info.sign === 1 ? t : 1 - t;
  return min.y + (max.y - min.y) * k;
}

//     so its y is the surface height at (x, z), not bounds.max.y.
// walkwayUnits estimates path length as the sum of (width + depth) of non-JUMP_PAD platforms.
import type { Platform } from "@/protocol/generated";
import { rampInfo, rampSurfaceHeight } from "./platformUtils";

// Mirrors web/src/protocol/generated.ts (redeclared so this module stays runtime-dependency-free for tests).
const PLATFORM_NORMAL = 0;
const PLATFORM_JUMP_PAD = 2;

export const MAX_VINES = 180;
export const MAX_LANTERNS = 24;
export const LANTERN_SPACING_UNITS = 40;
const VINE_KEEP_PROB = 0.55;
const VINE_MIN_LEN = 0.8;
const VINE_MAX_LEN = 2.3;
const VINE_INSET = 0.25;
const VINE_ANCHOR_SPACING = 2.4;
const LANTERN_INSET = 0.32;
const LANTERN_MIN_SPAN = 2.2;
const LANTERN_CANDIDATE_PROB = 0.6;

export interface VinePlacement {
  platformId: number;
  /** Anchor point on the platform's underside edge. */
  x: number;
  y: number;
  z: number;
  length: number;
  /** 0..2π, used to vary strand thickness and to allow future sway. */
  phase: number;
}

export interface LanternPlacement {
  platformId: number;
  x: number;
  /** Surface y at (x, z) — the post stands on the deck; for ramps this is the ramp surface height. */
  y: number;
  z: number;
}

export interface DecorPlacement {
  vines: VinePlacement[];
  lanterns: LanternPlacement[];
}

/** 32-bit integer hash of (a, b). Stable across runs and platforms. */
export function hash32(a: number, b: number): number {
  let h = (a >>> 0) ^ Math.imul((b >>> 0) + 0x9e3779b9, 0x85ebca6b);
  h = Math.imul(h ^ (h >>> 13), 0xc2b2ae35);
  h ^= h >>> 16;
  return h >>> 0;
}

/** mulberry32 PRNG — small, fast, deterministic. */
export function mulberry32(seed: number): () => number {
  let s = seed >>> 0;
  return () => {
    s = (s + 0x6d2b79f5) | 0;
    let t = Math.imul(s ^ (s >>> 15), 1 | s);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

export function platformFootprint(p: Platform): {
  w: number;
  h: number;
  d: number;
  cx: number;
  cz: number;
  top: number;
  bottom: number;
} {
  const w = p.bounds.max.x - p.bounds.min.x;
  const h = p.bounds.max.y - p.bounds.min.y;
  const d = p.bounds.max.z - p.bounds.min.z;
  return {
    w,
    h,
    d,
    cx: (p.bounds.min.x + p.bounds.max.x) / 2,
    cz: (p.bounds.min.z + p.bounds.max.z) / 2,
    top: p.bounds.max.y,
    bottom: p.bounds.min.y,
  };
}

export function isLanternEligible(p: Platform): boolean {
  // Ramps are stable footing like NORMAL decks; CRUMBLE collapses and JUMP_PAD tops launch.
  if (p.kind !== PLATFORM_NORMAL && !rampInfo(p.kind).isRamp) return false;
  const { w, d } = platformFootprint(p);
  return Math.min(w, d) >= LANTERN_MIN_SPAN;
}

/** Sum of (w + d) over non-JUMP_PAD platforms; approximates walkway length in units. */
export function walkwayUnits(platforms: readonly Platform[]): number {
  let u = 0;
  for (const p of platforms) {
    if (p.kind === PLATFORM_JUMP_PAD) continue;
    const { w, d } = platformFootprint(p);
    u += w + d;
  }
  return u;
}

/** Max number of lantern lights for a level: one per 40 u of walkway, capped. */
export function lanternBudget(platforms: readonly Platform[]): number {
  return Math.min(MAX_LANTERNS, Math.floor(walkwayUnits(platforms) / LANTERN_SPACING_UNITS));
}

export function computeDecor(platforms: readonly Platform[], seed: number): DecorPlacement {
  const vines: VinePlacement[] = [];
  const lanternCandidates: Array<{ p: Platform; sort: number }> = [];

  for (const p of platforms) {
    const { bottom } = platformFootprint(p);
    const rng = mulberry32(hash32(seed, p.id + 1));

    // Hanging vines under the four underside edges (JUMP_PAD bottoms are allowed —
    // pads survive collapse, their undersides are still static geometry).
    // For ramps the underside is the flat plane at bounds.min.y (the wedge bottom),
    // which is exactly what `bottom` reports.
    const edges: Array<{ axis: "x" | "z"; fixed: number; from: number; to: number }> = [
      { axis: "x", fixed: p.bounds.min.z, from: p.bounds.min.x, to: p.bounds.max.x },
      { axis: "x", fixed: p.bounds.max.z, from: p.bounds.min.x, to: p.bounds.max.x },
      { axis: "z", fixed: p.bounds.min.x, from: p.bounds.min.z, to: p.bounds.max.z },
      { axis: "z", fixed: p.bounds.max.x, from: p.bounds.min.z, to: p.bounds.max.z },
    ];
    for (const e of edges) {
      const len = e.to - e.from;
      if (len < 1.6) continue;
      const anchors = Math.floor(len / VINE_ANCHOR_SPACING);
      for (let i = 1; i <= anchors; i += 1) {
        if (vines.length >= MAX_VINES) break;
        if (rng() > VINE_KEEP_PROB) continue;
        const along = (len * i) / (anchors + 1);
        const jitter = (rng() - 0.5) * 1.1;
        const at = Math.min(Math.max(e.from + along + jitter, e.from + VINE_INSET), e.to - VINE_INSET);
        const length = VINE_MIN_LEN + rng() * (VINE_MAX_LEN - VINE_MIN_LEN);
        const phase = rng() * Math.PI * 2;
        if (e.axis === "x") {
          vines.push({ platformId: p.id, x: at, y: bottom, z: e.fixed, length, phase });
        } else {
          vines.push({ platformId: p.id, x: e.fixed, y: bottom, z: at, length, phase });
        }
      }
      if (vines.length >= MAX_VINES) break;
    }

    // Lantern candidates: stable, spacious, plain platforms and ramps.
    if (isLanternEligible(p) && rng() < LANTERN_CANDIDATE_PROB) {
      lanternCandidates.push({ p, sort: hash32(seed ^ 0x51ab, p.id + 7) });
    }
  }

  // Even spread: pick the lowest-hash candidates until the walkway budget is spent.
  lanternCandidates.sort((a, b) => a.sort - b.sort);
  const budget = lanternBudget(platforms);
  const lanterns: LanternPlacement[] = [];
  for (const c of lanternCandidates) {
    if (lanterns.length >= budget) break;
    const p = c.p;
    const fp = platformFootprint(p);
    const cornerRng = mulberry32(hash32(seed ^ 0x2b1, p.id + 13));
    const corner = Math.floor(cornerRng() * 4);
    const sx = corner % 2 === 0 ? -1 : 1;
    const sz = corner < 2 ? -1 : 1;
    const x = fp.cx + sx * (fp.w / 2 - LANTERN_INSET);
    const z = fp.cz + sz * (fp.d / 2 - LANTERN_INSET);
    lanterns.push({
      platformId: p.id,
      x,
      y: rampSurfaceHeight(p, x, z),
      z,
    });
  }

  return { vines, lanterns };
}
