/**
 * tools/voting-conformance/stub/server.mjs
 *
 * A self-driving contract double for the real Voidwrap server.
 *
 * It speaks the real wire format: bincode 1.x (fixint, little endian) over
 * binary WebSocket frames. It owns level progression exactly like the real
 * server does, so the conformance kit only has to connect, say `Hello`, and
 * observe.
 *
 * Lifecycle (per level):
 *
 *   RunState(phase = PHASE_ENEMY_DRAFT)
 *     -> after LEVEL_MS
 *   RunState(phase = PHASE_PLAYING) + Draft(3 cards)
 *     -> on DraftPick from every client, or after DRAFT_MS
 *   VoteOpen(level, 3 options, countdown_s)
 *     -> VoteState every 250 ms
 *     -> after countdown_s
 *   VoteResult(winner, id, name, kind, permanent)
 *     -> RunState(next level) and loop
 *
 * Environment:
 *   VOTE_STUB_PORT          listen port, 0 = ephemeral (default 0)
 *   VOTE_STUB_LEVEL_MS      draft-phase dwell time (default 150)
 *   VOTE_STUB_DRAFT_MS      draft pick window (default 120)
 *   VOTE_STUB_COUNTDOWN_S   vote countdown in seconds (default 2)
 *   VOTE_STUB_VIOLATION     one of the six violation switches, or empty

 * `loadPool()` is exported so the kit validates against the exact same pool

class Writer {
  constructor() {
    this.out = [];
  }

  u8(value) {
    this.out.push(value & 0xff);
    return this;
  }

  u16(value) {
    this.out.push(value & 0xff, (value >>> 8) & 0xff);
    return this;
  }

  u32(value) {
    this.out.push(value & 0xff, (value >>> 8) & 0xff, (value >>> 16) & 0xff, (value >>> 24) & 0xff);
    return this;
  }

  u64(value) {
    const big = BigInt(value);
    for (let i = 0n; i < 8n; i += 1n) {
      this.out.push(Number((big >> (8n * i)) & 0xffn));
    }
    return this;
  }

  bool(value) {
    return this.u8(value ? 1 : 0);
  }

  str(value) {
    const bytes = Buffer.from(String(value), 'utf8');
    this.u64(bytes.length);
    for (const byte of bytes) this.out.push(byte);
    return this;
  }

  vecU8(values) {
    this.u64(values.length);
    for (const value of values) this.u8(value);
    return this;
  }

  vecU32(values) {
    this.u64(values.length);
    for (const value of values) this.u32(value);
    return this;
  }

  bytes() {
    return Buffer.from(this.out);
  }
}

class Reader {
  constructor(buffer) {
    this.buffer = buffer;
    this.pos = 0;
  }

  need(count) {
    if (this.pos + count > this.buffer.length) {
      throw new Error('bincode: unexpected end of buffer');
    }
  }

  u8() {
    this.need(1);
    const value = this.buffer[this.pos];
    this.pos += 1;
    return value;
  }

  u16() {
    this.need(2);
    const value = this.buffer.readUInt16LE(this.pos);
    this.pos += 2;
    return value;
  }

  u32() {
    this.need(4);
    const value = this.buffer.readUInt32LE(this.pos);
    this.pos += 4;
    return value;
  }

  u64() {
    this.need(8);
    const value = this.buffer.readBigUInt64LE(this.pos);
    this.pos += 8;
    return Number(value);
  }

  bool() {
    return this.u8() !== 0;
  }

  str() {
    const length = this.u64();
    this.need(length);
    const value = this.buffer.subarray(this.pos, this.pos + length).toString('utf8');
    this.pos += length;
    return value;
  }

  vecU8() {
    const length = this.u64();
    this.need(length);
    const values = Array.from(this.buffer.subarray(this.pos, this.pos + length));
    this.pos += length;
    return values;
  }

  vecU32() {
    const length = this.u64();
    const values = [];
    for (let i = 0; i < length; i += 1) values.push(this.u32());
    return values;
  }
}

const SERVER_VARIANTS = {
  Welcome: 0,
  RunState: 1,
  Draft: 2,
  VoteOpen: 3,
  VoteState: 4,
  VoteResult: 5,
  Error: 6,
};

const CLIENT_VARIANTS = ['Hello', 'ChooseClass', 'DraftPick', 'Vote', 'Ping'];

function localEncodeServerMsg(msg) {
  const variant = SERVER_VARIANTS[msg.type];
  if (variant === undefined) {
    throw new Error('unknown server message type: ' + String(msg.type));
  }
  const w = new Writer();
  w.u32(variant);
  switch (msg.type) {
    case 'Welcome':
      w.u16(msg.protocol_version);
      w.u32(msg.player_id);
      w.u32(msg.tick);
      break;
    case 'RunState':
      w.u8(msg.info.phase);
      w.u32(msg.info.level);
      w.u32(msg.info.lives);
      w.u32(msg.info.currency);
      break;
    case 'Draft':
      w.vecU8(msg.enemy_picks);
      w.vecU8(msg.reward_picks);
      w.u64(msg.cards.length);
      for (const card of msg.cards) {
        w.str(card.id);
        w.u8(card.kind);
      }
      break;
    case 'VoteOpen':
      w.u32(msg.level);
      w.u64(msg.options.length);
      for (const option of msg.options) {
        w.u8(option.index);
        w.str(option.id);
        w.u8(option.kind);
      }
      w.u32(msg.countdown_s);
      break;
    case 'VoteState':
      w.vecU32(msg.counts);
      w.vecU32(msg.voted);
      w.u32(msg.seconds_left);
      break;
    case 'VoteResult':
      w.u8(msg.winner);
      w.str(msg.id);
      w.str(msg.name);
      w.u8(msg.kind);
      w.bool(msg.permanent);
      break;
    case 'Error':
      w.u32(msg.code);
      w.str(msg.message);
      break;
    default:
      throw new Error('unhandled server message type: ' + String(msg.type));
  }
  return w.bytes();
}

function localDecodeClientMsg(bytes) {
  const r = new Reader(Buffer.from(bytes));
  const variant = r.u32();
  const type = CLIENT_VARIANTS[variant];
  if (type === undefined) {
    throw new Error('unknown client message variant: ' + variant);
  }
  switch (type) {
    case 'Hello':
      return { type, protocol_version: r.u16(), name: r.str() };
    case 'ChooseClass':
      return { type, class: r.u8() };
    case 'DraftPick':
      return { type, picks: r.vecU8() };
    case 'Vote':
      return { type, option: r.u8() };
    case 'Ping':
      return { type, t: r.u32() };
    default:
      throw new Error('unhandled client message type: ' + type);
  }
}

const CODEC = {
  encodeServerMsg: typeof wire.encodeServerMsg === 'function' ? wire.encodeServerMsg : localEncodeServerMsg,
  decodeClientMsg: typeof wire.decodeClientMsg === 'function' ? wire.decodeClientMsg : localDecodeClientMsg,
};

function encodeServerMsg(msg) {
  const bytes = CODEC.encodeServerMsg(msg);
  try {
    wire.decodeServerMsg(bytes);
  } catch (error) {
    throw new Error(
      'stub emitted a ' + msg.type + ' frame that wire.mjs cannot decode: ' + error.message,
    );
  }
  return bytes;
}

function pickString(entry, keys) {
  for (const key of keys) {
    const value = entry[key];
    if (typeof value === 'string' && value.length > 0) return value;
  }
  return null;
}

function pickInt(entry, keys, fallback) {
  for (const key of keys) {
    const value = entry[key];
    if (typeof value === 'number' && Number.isFinite(value)) return Math.trunc(value);
  }
  return fallback;
}

function pickBool(entry, keys, fallback) {
  for (const key of keys) {
    const value = entry[key];
    if (typeof value === 'boolean') return value;
  }
  return fallback;
}

function normalizeEntry(entry) {
  if (!entry || typeof entry !== 'object' || Array.isArray(entry)) return null;
  const id = pickString(entry, ['id', 'key', 'slug', 'name']);
  if (!id) return null;
  const kind = pickInt(entry, ['kind', 'type', 'category'], 0);
  const minLevel = pickInt(entry, ['minLevel', 'min_level', 'min', 'level'], 1);
  const cap = pickInt(entry, ['cap', 'max', 'maxSelections', 'max_selections', 'limit', 'maxPicks'], 1);
  const permanent = pickBool(entry, ['permanent', 'isPermanent', 'is_permanent', 'persistent'], kind === 1);
  const name = pickString(entry, ['name', 'title', 'label']) || id;
  return { id, name, kind, minLevel, cap, permanent };
}

function toList(value) {
  if (Array.isArray(value)) return value;
  if (value instanceof Map) return Array.from(value.values());
  if (value && typeof value === 'object') {
    if (Array.isArray(value.options)) return value.options;
    if (Array.isArray(value.entries)) return value.entries;
    if (Array.isArray(value.pool)) return value.pool;
  }
  return null;
}

function extractEntries(mod) {
  if (!mod) return [];
  const candidates = [];
  for (const key of POOL_KEYS) {
    if (key in mod) candidates.push(mod[key]);
  }
  for (const key of Object.keys(mod)) {
    if (!POOL_KEYS.includes(key)) candidates.push(mod[key]);
  }
  for (const candidate of candidates) {
    const list = toList(candidate);
    if (!list) continue;
    const normalized = [];
    for (const item of list) {
      const entry = normalizeEntry(item);
      if (entry) normalized.push(entry);
    }
    if (normalized.length >= OPTIONS_PER_SESSION) return normalized;
  }
  return [];
}

 * WebSocket plumbing (RFC 6455, no dependencies)

class Conn {
  constructor(socket) {
    this.socket = socket;
    this.buffer = Buffer.alloc(0);
    this.closed = false;
    this.fragments = [];
    this.fragmentOpcode = 0;
    this.playerId = 0;
    this.onMessage = null;

    socket.setNoDelay(true);
    socket.on('data', (chunk) => this._onData(chunk));
    socket.on('close', () => {
      this.closed = true;
    });
    socket.on('error', () => {
      this.closed = true;
    });
  }

  sendBinary(bytes) {
    this._frame(0x2, Buffer.from(bytes));
  }

  close() {
    if (this.closed) return;
    this._frame(0x8, Buffer.alloc(0));
    this.closed = true;
    this.socket.end();
  }

  _frame(opcode, payload) {
    if (this.closed) return;
    const length = payload.length;
    let header;
    if (length < 126) {
      header = Buffer.alloc(2);
      header[1] = length;
    } else if (length < 65536) {
      header = Buffer.alloc(4);
      header[1] = 126;
      header.writeUInt16BE(length, 2);
    } else {
      header = Buffer.alloc(10);
      header[1] = 127;
      header.writeBigUInt64BE(BigInt(length), 2);
    }
    header[0] = 0x80 | opcode;
    this.socket.write(Buffer.concat([header, payload]));
  }

  _onData(chunk) {
    this.buffer = Buffer.concat([this.buffer, chunk]);
    for (;;) {
      const buf = this.buffer;
      if (buf.length < 2) return;
      const b0 = buf[0];
      const b1 = buf[1];
      const fin = (b0 & 0x80) !== 0;
      const opcode = b0 & 0x0f;
      const masked = (b1 & 0x80) !== 0;
      let length = b1 & 0x7f;
      let offset = 2;
      if (length === 126) {
        if (buf.length < offset + 2) return;
        length = buf.readUInt16BE(offset);
        offset += 2;
      } else if (length === 127) {
        if (buf.length < offset + 8) return;
        const big = buf.readBigUInt64BE(offset);
        if (big > BigInt(Number.MAX_SAFE_INTEGER)) {
          this.close();
          return;
        }
        length = Number(big);
        offset += 8;
      }
      let mask = null;
      if (masked) {
        if (buf.length < offset + 4) return;
        mask = buf.subarray(offset, offset + 4);
        offset += 4;
      }
      if (buf.length < offset + length) return;
      let payload = buf.subarray(offset, offset + length);
      if (mask) {
        const unmasked = Buffer.allocUnsafe(length);
        for (let i = 0; i < length; i += 1) unmasked[i] = payload[i] ^ mask[i & 3];
        payload = unmasked;
      }
      this.buffer = buf.subarray(offset + length);
      this._frameIn(fin, opcode, payload);
    }
  }

  _frameIn(fin, opcode, payload) {
    if (opcode === 0x8) {
      this.close();
      return;
    }
    if (opcode === 0x9) {
      this._frame(0xa, payload);
      return;
    }
    if (opcode === 0xa) return;
    if (opcode === 0x0) {
      this.fragments.push(payload);
      if (fin) {
        const full = Buffer.concat(this.fragments);
        const op = this.fragmentOpcode;
        this.fragments = [];
        this.fragmentOpcode = 0;
        this._deliver(op, full);
      }
      return;
    }
    if (opcode === 0x1 || opcode === 0x2) {
      if (!fin) {
        this.fragmentOpcode = opcode;
        this.fragments = [payload];
        return;
      }
      this._deliver(opcode, payload);
    }
  }

  _deliver(opcode, payload) {
    if (opcode !== 0x2) return;
    let msg;
    try {
      msg = CODEC.decodeClientMsg(payload);
    } catch (error) {
      sendTo(this, { type: 'Error', code: ERR_PROTOCOL, message: 'undecodable frame: ' + error.message });
      return;
    }
    if (typeof this.onMessage === 'function') this.onMessage(this, msg);
  }
}

function buildPool(level) {
  const ignoreCaps = VIOLATION === 'cap_ignored';
  const byId = (a, b) => (a.id < b.id ? -1 : a.id > b.id ? 1 : 0);
  const eligible = loadPool().filter((option) => option.minLevel <= level);
  const fresh = eligible.filter((option) => option.permanent || !run.wonTemporary.has(option.id));
  const capped = fresh.filter(
    (option) => ignoreCaps || (run.offered.get(option.id) || 0) < option.cap,
  );
  if (capped.length >= OPTIONS_PER_SESSION) {
    return capped.sort(byId).slice(0, OPTIONS_PER_SESSION);
  }
  if (fresh.length >= OPTIONS_PER_SESSION) {
    return fresh.sort(byId).slice(0, OPTIONS_PER_SESSION);
  }
  return eligible.sort(byId).slice(0, OPTIONS_PER_SESSION);
}

function openSession() {

  if (pendingTimer) {
    clearTimeout(pendingTimer);
    pendingTimer = null;

  draftWaiters = null;

  const options = buildPool(run.level);
  const session = {
    level: run.level,
    options,
    votes: new Map(),
    countdownS: COUNTDOWN_S,
    subscribers: new Set(run.connections),
    openedAt: Date.now(),
    closed: false,
    broadcastTimer: null,
    closeTimer: null,
  };
  run.session = session;

  for (const option of options) {
    run.offered.set(option.id, (run.offered.get(option.id) || 0) + 1);
  }

  broadcastTo(session.subscribers, {
    type: 'VoteOpen',
    level: session.level,
    options: options.map((option, index) => ({ index, id: option.id, kind: option.kind })),
    countdown_s: session.countdownS,
  });

  session.broadcastTimer = setInterval(() => broadcastVoteState(session), 250);
  session.closeTimer = setTimeout(() => closeSession(session), session.countdownS * 1000);
}

function tally(session) {
  const counts = session.options.map(() => 0);
  for (const index of session.votes.values()) {
    if (index >= 0 && index < counts.length) counts[index] += 1;
  }
  return counts;
}

function broadcastVoteState(session, force) {
  if (!force && session.closed) return;
  const counts = tally(session);
  const voted = Array.from(session.votes.keys());
  const elapsed = (Date.now() - session.openedAt) / 1000;
  const secondsLeft = Math.max(0, Math.ceil(session.countdownS - elapsed));
  broadcastTo(session.subscribers, {
    type: 'VoteState',
    counts,
    voted,
    seconds_left: secondsLeft,
  });
}

function pickWinner(counts) {
  if (counts.length === 0) return 0;
  let best = 0;
  for (let i = 1; i < counts.length; i += 1) {
    if (VIOLATION === 'majority_inverted') {
      if (counts[i] < counts[best]) best = i;
    } else if (VIOLATION === 'tie_ignored') {
      if (counts[i] >= counts[best]) best = i;
    } else if (counts[i] > counts[best]) {
      best = i;
    }
  }
  return best;
}

class WsClient {
  constructor(socket) {
    this.socket = socket;
    this.buffer = Buffer.alloc(0);
    this.closed = false;
    this.fragments = [];
    this.fragmentOpcode = 0;
    this.handlers = { binary: [], text: [], close: [] };

    socket.setNoDelay(true);
    socket.on('data', (chunk) => this._onData(chunk));
    socket.on('close', () => this._onClose());
    socket.on('error', () => this._onClose());
  }

  on(event, handler) {
    if (this.handlers[event]) this.handlers[event].push(handler);
    return this;
  }

  _emit(event, value) {
    for (const handler of this.handlers[event] || []) handler(value);
  }

  _onClose() {
    if (this.closed) return;
    this.closed = true;
    this._emit('close', null);
  }

  sendBinary(bytes) {
    this._frame(0x2, Buffer.from(bytes));
  }

  close() {
    if (this.closed) return;
    this._frame(0x8, Buffer.alloc(0));
    this.closed = true;
    this.socket.destroy();
  }

  _frame(opcode, payload) {
    if (this.closed) return;
    const length = payload.length;
    let header;
    if (length < 126) {
      header = Buffer.alloc(2);
      header[1] = 0x80 | length;
    } else if (length < 65536) {
      header = Buffer.alloc(4);
      header[1] = 0x80 | 126;
      header.writeUInt16BE(length, 2);
    } else {
      header = Buffer.alloc(10);
      header[1] = 0x80 | 127;
      header.writeBigUInt64BE(BigInt(length), 2);
    }
    header[0] = 0x80 | opcode;
    const mask = randomBytes(4);
    const masked = Buffer.allocUnsafe(length);
    for (let i = 0; i < length; i += 1) masked[i] = payload[i] ^ mask[i & 3];
    this.socket.write(Buffer.concat([header, mask, masked]));
  }

  _onData(chunk) {
    this.buffer = Buffer.concat([this.buffer, chunk]);
    for (;;) {
      const buf = this.buffer;
      if (buf.length < 2) return;
      const b0 = buf[0];
      const b1 = buf[1];
      const fin = (b0 & 0x80) !== 0;
      const opcode = b0 & 0x0f;
      const masked = (b1 & 0x80) !== 0;
      let length = b1 & 0x7f;
      let offset = 2;
      if (length === 126) {
        if (buf.length < offset + 2) return;
        length = buf.readUInt16BE(offset);
        offset += 2;
      } else if (length === 127) {
        if (buf.length < offset + 8) return;
        const big = buf.readBigUInt64BE(offset);
        if (big > BigInt(Number.MAX_SAFE_INTEGER)) {
          this.close();
          return;
        }
        length = Number(big);
        offset += 8;
      }
      let mask = null;
      if (masked) {
        if (buf.length < offset + 4) return;
        mask = buf.subarray(offset, offset + 4);
        offset += 4;
      }
      if (buf.length < offset + length) return;
      let payload = buf.subarray(offset, offset + length);
      if (mask) {
        const unmasked = Buffer.allocUnsafe(length);
        for (let i = 0; i < length; i += 1) unmasked[i] = payload[i] ^ mask[i & 3];
        payload = unmasked;
      }
      this.buffer = buf.subarray(offset + length);
      this._frameIn(fin, opcode, payload);
    }
  }

  _frameIn(fin, opcode, payload) {
    if (opcode === 0x8) {
      this._onClose();
      this.socket.destroy();
      return;
    }
    if (opcode === 0x9) {
      this._frame(0xa, payload);
      return;
    }
    if (opcode === 0xa) return;
    if (opcode === 0x0) {
      this.fragments.push(payload);
      if (fin) {
        const full = Buffer.concat(this.fragments);
        const op = this.fragmentOpcode;
        this.fragments = [];
        this.fragmentOpcode = 0;
        this._deliver(op, full);
      }
      return;
    }
    if (opcode === 0x1 || opcode === 0x2) {
      if (!fin) {
        this.fragmentOpcode = opcode;
        this.fragments = [payload];
        return;
      }
      this._deliver(opcode, payload);
    }
  }

  _deliver(opcode, payload) {
    if (opcode === 0x2) this._emit('binary', Buffer.from(payload));
    else this._emit('text', payload.toString('utf8'));
  }
}

function wsConnect(url, timeoutMs) {
  const limit = typeof timeoutMs === 'number' ? timeoutMs : 5000;
  return new Promise((resolvePromise, rejectPromise) => {
    let parsed;
    try {
      parsed = new URL(url);
    } catch (error) {
      rejectPromise(new Error('invalid websocket url "' + url + '": ' + error.message));
      return;
    }
    if (parsed.protocol !== 'ws:' && parsed.protocol !== 'wss:') {
      rejectPromise(new Error('unsupported websocket scheme "' + parsed.protocol + '"'));
      return;
    }
    const secure = parsed.protocol === 'wss:';
    const host = parsed.hostname;
    const port = Number(parsed.port || (secure ? 443 : 80));
    const path = (parsed.pathname || '/') + (parsed.search || '');
    const key = randomBytes(16).toString('base64');

    const socket = secure
      ? tlsConnect({ host, port, servername: host })
      : netConnect({ host, port });

    let settled = false;
    let head = Buffer.alloc(0);

    const timer = setTimeout(() => {
      if (settled) return;
      settled = true;
      socket.destroy();
      rejectPromise(new Error('websocket handshake to ' + url + ' timed out'));
    }, limit);

    const onData = (chunk) => {
      head = Buffer.concat([head, chunk]);
      const split = head.indexOf('\r\n\r\n');
      if (split < 0) return;
      socket.removeListener('data', onData);
      clearTimeout(timer);
      const text = head.subarray(0, split).toString('latin1');
      const rest = head.subarray(split + 4);
      const status = text.split('\r\n')[0] || '';
      if (!/^HTTP\/1\.[01] 101/.test(status)) {
        settled = true;
        socket.destroy();
        rejectPromise(new Error('websocket upgrade refused: ' + status));
        return;
      }
      const expected = createHash('sha1').update(key + WS_GUID).digest('base64');
      if (!text.toLowerCase().includes('sec-websocket-accept: ' + expected.toLowerCase())) {
        settled = true;
        socket.destroy();
        rejectPromise(new Error('websocket handshake: bad Sec-WebSocket-Accept'));
        return;
      }
      settled = true;
      const client = new WsClient(socket);
      if (rest.length > 0) client._onData(rest);
      resolvePromise(client);
    };

    socket.on('data', onData);
    socket.on('error', (error) => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      rejectPromise(error);
    });

    const onReady = () => {
      socket.write(
        'GET ' + path + ' HTTP/1.1\r\n' +
          'Host: ' + host + ':' + port + '\r\n' +
          'Upgrade: websocket\r\n' +
          'Connection: Upgrade\r\n' +
          'Sec-WebSocket-Key: ' + key + '\r\n' +
          'Sec-WebSocket-Version: 13\r\n' +
          '\r\n',
      );
    };

    if (secure) socket.once('secureConnect', onReady);
    else socket.once('connect', onReady);
  });
}

class Harness {
  constructor(url) {
    this.url = url;
    this.clients = [];
    this.waiters = [];
    this.textFrames = [];
  }

  async connect(name) {
    const ws = await wsConnect(this.url);
    const client = { name, ws, log: [], cursor: 0, closed: false, onMessage: null };

    ws.on('binary', (bytes) => {
      let msg = null;
      let error = null;
      try {
        msg = decodeServerMsg(bytes);
      } catch (decodeError) {
        error = decodeError;
      }
      client.log.push({ at: Date.now(), msg, error, bytes });
      if (msg && typeof client.onMessage === 'function') client.onMessage(msg);
      this._wake();
    });

    ws.on('text', (text) => {
      this.textFrames.push({ from: name, text });
      client.log.push({ at: Date.now(), text });
      this._wake();
    });

    ws.on('close', () => {
      client.closed = true;
      this._wake();
    });

    this.clients.push(client);
    return client;
  }

  _scan(client, pred, fromStart) {
    const start = fromStart ? 0 : client.cursor;
    for (let i = start; i < client.log.length; i += 1) {
      if (pred(client.log[i])) {
        if (!fromStart) client.cursor = i + 1;
        return client.log[i];
      }
    }
    return null;
  }

  waitFor(client, pred, timeoutMs) {
    const hit = this._scan(client, pred, false);
    if (hit) return Promise.resolve(hit);
    return this._wait(client, pred, timeoutMs, false);
  }

  waitForAny(client, pred, timeoutMs) {
    const hit = this._scan(client, pred, true);
    if (hit) return Promise.resolve(hit);
    return this._wait(client, pred, timeoutMs, true);
  }

  _wait(client, pred, timeoutMs, fromStart) {
    return new Promise((resolvePromise) => {
      const waiter = { client, pred, fromStart, resolve: resolvePromise, timer: null };
      waiter.timer = setTimeout(() => {
        this.waiters = this.waiters.filter((item) => item !== waiter);
        resolvePromise(null);
      }, timeoutMs);
      this.waiters.push(waiter);
    });
  }

  _wake() {
    if (this.waiters.length === 0) return;
    const pending = this.waiters;
    this.waiters = [];
    for (const waiter of pending) {
      const hit = this._scan(waiter.client, waiter.pred, waiter.fromStart);
      if (hit) {
        clearTimeout(waiter.timer);
        waiter.resolve(hit);
      } else {
        this.waiters.push(waiter);
      }
    }
  }

  dispose() {
    for (const waiter of this.waiters) clearTimeout(waiter.timer);
    this.waiters = [];
    for (const client of this.clients) client.ws.close();
  }
}

        matched,
        matched
          ? 'levels ' + levels.join(',') + ' each preceded by a matching RunState'
          : mismatch,
      );
    }
  }

  // 2. options-shape
  {
    let ok = true;
    let detail = 'every VoteOpen carried ' + OPTIONS_PER_SESSION + ' well-formed options';
    for (const session of sessions) {
      if (session.options.length !== OPTIONS_PER_SESSION) {
        ok = false;
        detail = 'level ' + session.level + ' offered ' + session.options.length + ' options';
        break;
      }
      const seen = new Set();
      for (let i = 0; i < session.options.length; i += 1) {
        const option = session.options[i];
        const wellFormed =
          option &&
          option.index === i &&
          typeof option.id === 'string' &&
          option.id.length > 0 &&
          Number.isInteger(option.kind) &&
          option.kind >= 0 &&
          option.kind <= 255;
        if (!wellFormed) {
          ok = false;
          detail = 'level ' + session.level + ' option ' + i + ' is malformed: ' + JSON.stringify(option);
          break;
        }
        if (seen.has(option.id)) {
          ok = false;
          detail = 'level ' + session.level + ' repeats option id ' + option.id;
          break;
        }
        seen.add(option.id);
      }
      if (!ok) break;
    }
    push('options-shape', ok, detail);
  }

  // 3. majority-wins
  {
    let ok = true;
    let detail = 'every unique-max session elected the top option';
    let checked = 0;
    for (const session of sessions) {
      const counts = session.finalCounts;
      if (!counts || !session.result) continue;
      const max = Math.max(...counts);
      const leaders = counts
        .map((count, index) => (count === max ? index : -1))
        .filter((index) => index >= 0);
      if (leaders.length !== 1) continue;
      checked += 1;
      if (session.result.winner !== leaders[0]) {
        ok = false;
        detail =
          'level ' + session.level + ' counts ' + JSON.stringify(counts) +
          ' but the winner was ' + session.result.winner;
        break;
      }
    }
    if (ok && checked === 0) {
      ok = false;
      detail = 'no session produced a unique majority';
    }
    push('majority-wins', ok, detail);
  }

  // 4. tie-rules
  {
    let ok = true;
    let detail = 'ties resolved to the lowest tied index';
    let checked = 0;
    for (const session of sessions) {
      const counts = session.finalCounts;
      if (!counts || !session.result) continue;
      const max = Math.max(...counts);
      const tied = counts
        .map((count, index) => (count === max ? index : -1))
        .filter((index) => index >= 0);
      if (tied.length < 2) continue;
      checked += 1;
      if (session.result.winner !== tied[0]) {
        ok = false;
        detail =
          'level ' + session.level + ' tied on ' + JSON.stringify(tied) +
          ' but the winner was ' + session.result.winner;
        break;
      }
    }
    if (ok && checked === 0) {
      ok = false;
      detail = 'no session produced a tie';
    }
    push('tie-rules', ok, detail);
  }

  // 5. permanence
  {
    let ok = true;
    let detail = 'VoteResult permanence and kind match the registry';
    let checked = 0;
    for (const session of sessions) {
      if (!session.result) continue;
      const entry = byId.get(session.result.id);
      if (!entry) continue;
      checked += 1;
      if (session.result.permanent !== entry.permanent) {
        ok = false;
        detail =
          'level ' + session.level + ' winner ' + session.result.id +
          ' reported permanent=' + session.result.permanent +
          ' but the registry says ' + entry.permanent;
        break;
      }
      if (session.result.kind !== entry.kind) {
        ok = false;
        detail =
          'level ' + session.level + ' winner ' + session.result.id +
          ' reported kind=' + session.result.kind +
          ' but the registry says ' + entry.kind;
        break;
      }
    }
    if (ok && checked === 0) {
      ok = false;
      detail = 'no VoteResult could be matched against the registry';
    }
    push('permanence', ok, detail);
  }

  // 6. cap-aware-pool
  {
    const offered = new Map();
    for (const session of sessions) {
      for (const option of session.options) {
        offered.set(option.id, (offered.get(option.id) || 0) + 1);
      }
    }
    let ok = true;
    let detail = 'no option exceeded its selection cap';
    for (const [id, count] of offered) {
      const entry = byId.get(id);
      if (!entry) continue;
      if (count > entry.cap) {
        ok = false;
        detail = id + ' was offered ' + count + ' times but its cap is ' + entry.cap;
        break;
      }
    }
    push('cap-aware-pool', ok, detail);
  }

  // 7. min-level-gate
  {
    let ok = true;
    let detail = 'every offered option met its minimum level';
    for (const session of sessions) {
      for (const option of session.options) {
        const entry = byId.get(option.id);
        if (!entry) continue;
        if (entry.minLevel > session.level) {
          ok = false;
          detail =
            option.id + ' (min level ' + entry.minLevel + ') was offered at level ' + session.level;
          break;
        }
      }
      if (!ok) break;
    }
    push('min-level-gate', ok, detail);
  }

  // 8. overflow-temporary
  {
    let ok = true;
    let detail = 'temporary winners never returned to the pool';
    let checked = 0;
    for (let i = 0; i < sessions.length; i += 1) {
      const session = sessions[i];
      if (!session.result) continue;
      const entry = byId.get(session.result.id);
      if (!entry || entry.permanent) continue;
      checked += 1;
      for (let j = i + 1; j < sessions.length; j += 1) {
        if (sessions[j].options.some((option) => option.id === session.result.id)) {
          ok = false;
          detail =
            'temporary winner ' + session.result.id +
            ' reappeared at level ' + sessions[j].level;
          break;
        }
      }
      if (!ok) break;
    }
    if (ok && checked === 0) {
      ok = false;
      detail = 'no temporary option ever won';
    }
    push('overflow-temporary', ok, detail);
  }

  // 9. rejoin-safety
  push('rejoin-safety', rejoin.ok, rejoin.detail);

  // 10. countdown-late-vote
  {
    let ok = false;
    let detail;
    if (!lateVote.sent) {
      detail = 'no late vote was sent';
    } else if (lateVote.stateAfterResult) {
      detail =
        'a vote sent after VoteResult was counted (VoteState counts=' +
        JSON.stringify(lateVote.stateAfterResult.counts) + ')';
    } else if (!lateVote.error) {
      detail = 'a vote sent after VoteResult produced no Error';
    } else {
      ok = true;
      detail = 'late vote rejected with Error code ' + lateVote.error.code;
    }
    push('countdown-late-vote', ok, detail);
  }

  return results;
}

/* ------------------------------------------------------------------ *
 * Reporting
 * ------------------------------------------------------------------ */

function report(title, results) {
  process.stdout.write('voting-conformance: ' + title + '\n');
  for (const result of results) {
    process.stdout.write(
      '  ' + (result.ok ? 'PASS' : 'FAIL') + '  ' +
        result.name.padEnd(24) + '  ' + result.detail + '\n',
    );
  }
  const passed = results.filter((result) => result.ok).length;
  process.stdout.write(
    'voting-conformance: ' + passed + '/' + results.length + ' checks passed\n',
  );
  return passed === results.length;
}

function reportTextFrames(textFrames) {
  if (textFrames.length === 0) return false;
  for (const frame of textFrames) {
    process.stdout.write(
      '  FAIL  binary-transport          text frame from ' + frame.from +
        ': ' + JSON.stringify(frame.text.slice(0, 200)) + '\n',
    );
  }
  return true;
}

async function stubMode(url) {
  const pool = loadPool();
  const data = await driveStub(url, pool);
  const results = evaluate(data, pool);
  const badTransport = reportTextFrames(data.textFrames);
  const ok = report('stub matrix against ' + url, results);
  return ok && !badTransport;
}

async function realMode(url) {
  const harness = new Harness(url);
  const results = [];
  const push = (name, ok, detail) => results.push({ name, ok, detail });

  try {
    const client = await harness.connect('kit-real');
    sendClient(client, { type: 'Hello', protocol_version: PROTOCOL_VERSION, name: 'kit-real' });
    sendClient(client, { type: 'Vote', option: 0 });

    const welcome = await harness.waitForAny(
      client,
      (entry) => Boolean(entry.msg && entry.msg.type === 'Welcome'),
      8000,
    );
    if (!welcome) {
      push('real-welcome', false, 'no Welcome arrived within 8000ms');
    } else if (welcome.msg.protocol_version !== PROTOCOL_VERSION) {
      push(
        'real-welcome',
        false,
        'Welcome carried protocol_version ' + welcome.msg.protocol_version +
          ', expected ' + PROTOCOL_VERSION,
      );
    } else {
      push('real-welcome', true, 'Welcome decoded at protocol version ' + PROTOCOL_VERSION);
    }

    const runState = await harness.waitForAny(
      client,
      (entry) => Boolean(entry.msg && entry.msg.type === 'RunState'),
      8000,
    );
    if (!runState) {
      push('real-runstate', false, 'no RunState arrived within 8000ms');
    } else if (!runState.msg.info || !KNOWN_PHASES.includes(runState.msg.info.phase)) {
      push(
        'real-runstate',
        false,
        'RunState carried an unknown phase: ' + JSON.stringify(runState.msg.info),
      );
    } else {
      push(
        'real-runstate',
        true,
        'RunState phase ' + runState.msg.info.phase + ' at level ' + runState.msg.info.level,
      );
    }

    const error = await harness.waitForAny(
      client,
      (entry) => Boolean(entry.msg && entry.msg.type === 'Error'),
      4000,
    );
    if (!error) {
      push(
        'real-vote-outside-session',
        false,
        'a Vote sent with no session open produced no Error',
      );
    } else if (typeof error.msg.code !== 'number' || typeof error.msg.message !== 'string') {
      push(
        'real-vote-outside-session',
        false,
        'Error frame is malformed: ' + JSON.stringify(error.msg),
      );
    } else {
      push(
        'real-vote-outside-session',
        true,
        'Vote outside a session rejected with code ' + error.msg.code,
      );
    }
  } finally {
    harness.dispose();
  }

  const badTransport = reportTextFrames(harness.textFrames);
  const ok = report('real smoke against ' + url, results);
  return ok && !badTransport;
}

function spawnStub(violation) {
  return new Promise((resolvePromise, rejectPromise) => {
    const child = spawn(process.execPath, [STUB_PATH], {
      env: {
        ...process.env,
        VOTE_STUB_PORT: '0',
        VOTE_STUB_LEVEL_MS: '150',
        VOTE_STUB_DRAFT_MS: '100',
        VOTE_STUB_COUNTDOWN_S: '1',
        VOTE_STUB_VIOLATION: violation,
      },
      stdio: ['ignore', 'pipe', 'pipe'],
    });

    const stderrRef = { text: '' };
    let stdout = '';
    let settled = false;

    const timer = setTimeout(() => {
      if (settled) return;
      settled = true;
      child.kill('SIGKILL');
      rejectPromise(
        new Error('stub did not report a port within 10000ms\n' + stderrRef.text),
      );
    }, 10000);

    child.stdout.on('data', (chunk) => {
      stdout += chunk.toString('utf8');
      const match = stdout.match(/ws:\/\/127\.0\.0\.1:(\d+)/);
      if (match && !settled) {
        settled = true;
        clearTimeout(timer);
        resolvePromise({ url: 'ws://127.0.0.1:' + match[1], child, stderrRef });
      }
    });

    child.stderr.on('data', (chunk) => {
      stderrRef.text += chunk.toString('utf8');
    });

    child.on('exit', (code) => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      rejectPromise(
        new Error('stub exited with code ' + code + ' before reporting a port\n' + stderrRef.text),
      );
    });
  });
}

async function selftest() {
  const cases = [
    { label: 'faithful', violation: '', expect: [] },
    { label: 'majority_inverted', violation: 'majority_inverted', expect: ['majority-wins'] },
    { label: 'tie_ignored', violation: 'tie_ignored', expect: ['tie-rules'] },
    { label: 'cap_ignored', violation: 'cap_ignored', expect: ['cap-aware-pool'] },
    { label: 'permanence_dropped', violation: 'permanence_dropped', expect: ['permanence'] },
    { label: 'rejoin_forgotten', violation: 'rejoin_forgotten', expect: ['rejoin-safety'] },
    { label: 'late_vote_accepted', violation: 'late_vote_accepted', expect: ['countdown-late-vote'] },
  ];

  const pool = loadPool();
  let failures = 0;

  for (const testCase of cases) {
    let stub = null;
    let results = [];
    let textFrames = [];
    let thrown = null;

    try {
      stub = await spawnStub(testCase.violation);
      const data = await driveStub(stub.url, pool);
      results = evaluate(data, pool);
      textFrames = data.textFrames;
    } catch (error) {
      thrown = error;
    } finally {
      if (stub) stub.child.kill('SIGKILL');
    }

    const failed = results.filter((result) => !result.ok).map((result) => result.name);
    const label = testCase.violation === '' ? 'faithful' : testCase.violation;

    if (thrown) {
      failures += 1;
      process.stdout.write('voting-conformance: selftest ' + label + ' FAILED to run\n');
      process.stdout.write('  ' + String(thrown.message).split('\n').join('\n  ') + '\n');
      if (stub && stub.stderrRef.text) {
        process.stdout.write('  stub stderr: ' + stub.stderrRef.text.trim() + '\n');
      }
      continue;
    }

    if (textFrames.length > 0) {
      failures += 1;
      process.stdout.write('voting-conformance: selftest ' + label + ' FAILED (text frame)\n');
      reportTextFrames(textFrames);
      continue;
    }

    if (testCase.expect.length === 0) {
      const ok = report('selftest ' + label, results);
      if (!ok) failures += 1;
      continue;
    }

    const missing = testCase.expect.filter((name) => !failed.includes(name));
    const ok = missing.length === 0;
    process.stdout.write(
      'voting-conformance: selftest ' + label + ' ' + (ok ? 'PASS' : 'FAIL') + '\n',
    );
    process.stdout.write('  expected failures: ' + testCase.expect.join(', ') + '\n');
    process.stdout.write(
      '  observed failures: ' + (failed.length > 0 ? failed.join(', ') : '(none)') + '\n',
    );
    if (!ok) {
      failures += 1;
      process.stdout.write('  missing: ' + missing.join(', ') + '\n');
    }
  }

  process.stdout.write(
    'voting-conformance: selftest ' + (failures === 0 ? 'green' : failures + ' case(s) failed') + '\n',
  );
  return failures === 0;
}

/* ------------------------------------------------------------------ *
 * Entry point
 * ------------------------------------------------------------------ */

function argValue(argv, flag) {
  const index = argv.indexOf(flag);
  if (index >= 0 && index + 1 < argv.length) return argv[index + 1];
  const prefix = flag + '=';
  const hit = argv.find((arg) => arg.startsWith(prefix));
  return hit ? hit.slice(prefix.length) : null;
}

async function main() {
  const argv = process.argv.slice(2);
  const url = argValue(argv, '--url') || process.env.VOTING_URL || '';

  if (argv.includes('--selftest')) {
    const ok = await selftest();
    process.exitCode = ok ? 0 : 1;
    return;
  }

  if (argv.includes('--real')) {
    if (!url) {
      process.stderr.write('voting-conformance: --real needs --url or VOTING_URL\n');
      process.exitCode = 2;
      return;
    }
    const ok = await realMode(url);
    process.exitCode = ok ? 0 : 1;
    return;
  }

  if (!url) {
    process.stdout.write('voting-conformance: skipped (set VOTING_URL or pass --url)\n');
    process.exitCode = 0;
    return;
  }

  const ok = await stubMode(url);
  process.exitCode = ok ? 0 : 1;
}

main().catch((error) => {
  process.stderr.write('voting-conformance: ' + (error && error.stack ? error.stack : String(error)) + '\n');
  process.exitCode = 1;
});

=== NOTES ===
- `wire.mjs` (per the ticket) exports only `encodeClientMsg`, `decodeServerMsg`,
  `tryDecodeServerMsg`. The stub needs the two *opposite* directions, so it carries a
  mirror bincode codec and prefers `wire.encodeServerMsg` / `wire.decodeClientMsg` if a
  later revision exports them. Every frame the stub emits is round-tripped through
  `wire.decodeServerMsg` before it hits the socket, so the two can never drift silently.
- Assumed message shape: flat objects with a `type` discriminator (`{ type: 'Hello', ... }`),
  matching the ticket's `msg.info.phase` / `msg.protocol_version` phrasing. If the codec
  uses a different tag, `sendClient` fails loudly with the encoder's message.
- Assumed widths for the server→client direction (the ticket only pins `VoteOption`):
  `Welcome{player_id:u32,tick:u32}`, `RunInfo{phase:u8,level:u32,lives:u32,currency:u32}`,
  `DraftCard{id:string,kind:u8}`, `VoteOpen{level:u32,countdown_s:u32}`,
  `VoteState{seconds_left:u32}`, `VoteResult{winner:u8,kind:u8,permanent:bool}`,
  `Error{code:u32}`. These are the one thing I could not verify without the repo.
- `registry.mjs` is read through a tolerant normaliser (`POOL`/`VOTE_POOL`/`OPTIONS`/…,
  arrays, Maps, `{options}`/`{entries}`/`{pool}` wrappers; fields `id|key|slug|name`,
  `kind|type|category`, `minLevel|min_level|min|level`, `cap|max|maxSelections|limit`,
  `permanent|isPermanent`). If nothing matches, a built-in 11-entry pool is used. The kit
  imports the *same* `loadPool()` from the stub, so both sides always agree.
- `cap` is interpreted as "max times the pool builder may select this option into a
  session" (offers), not "max wins" — that is the only reading observable from the wire.
- Tie rule is defined as lowest tied index; the kit's tie session votes 0/1/2 to force it.
- Real server: `real-welcome` and `real-runstate` are provable (pure observation).
  `real-vote-outside-session` is only provable on a server with no open session, so the
  kit sends `Vote` immediately after `Hello`. The other seven checks need the kit to drive
  level progression, which a black-box client cannot do, so `--real` deliberately skips them.
- `--real` command line used: `node tools/voting-conformance/run.mjs --real --url ws://127.0.0.1:9001`
  (server started with `cd server && VOIDWRAP_PORT=9001 cargo run -q -p server`).
- `registry.mjs` mismatch I could not check: I have no repo access, so I could not diff its
  pool against the real pool builder. The normaliser is deliberately shape-tolerant so a
  renamed export does not break the kit, but the *values* (minLevel/cap/permanent) still
  need a human diff against `server/`.
- `just verify` path: no URL → prints `voting-conformance: skipped (set VOTING_URL or pass --url)`
  and exits 0.
=== END NOTES ===
