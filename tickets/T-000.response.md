/**
 * Clamp a numeric value to the inclusive range [lo, hi].
 *
 * @param v  Value to clamp.
 * @param lo Lower bound (inclusive).
 * @param hi Upper bound (inclusive).
 * @returns `v` limited to [lo, hi]; `NaN` for `v` yields `lo`.
 * @throws {RangeError} If `lo > hi`.
 */
export function clamp(v: number, lo: number, hi: number): number {
  if (lo > hi) {
    throw new RangeError(`clamp: lo (${lo}) must not be greater than hi (${hi})`);
  }

  if (Number.isNaN(v)) {
    return lo;
  }

  if (v < lo) {
    return lo;
  }

  if (v > hi) {
    return hi;
  }

  return v;
}
