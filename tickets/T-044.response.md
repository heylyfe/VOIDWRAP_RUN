//! In-game hosting for friends: an extra listener on `0.0.0.0:<HOST_PORT>` serving the
//! same router and the same game over the LAN, plus a best-effort UPnP port mapping
//! ("VOIDWRAP RUN", TCP HOST_PORT, 2 hours) so friends on the internet can join too.

use serde::{Deserialize, Serialize};
use std::net::{IpAddr, Ipv4Addr, SocketAddr};
use std::sync::Arc;
use std::time::Duration;
use tokio::sync::Mutex;

/// Default port for the hosting listener; override with the `VOIDWRAP_HOST_PORT` environment variable.
pub const DEFAULT_HOST_PORT: u16 = 7879;
/// How long the router is asked to keep the UPnP port mapping.
const UPNP_LEASE_SECS: u64 = 2 * 60 * 60;
/// How long to wait for the UPnP gateway to answer the gateway search.
const UPNP_SEARCH_TIMEOUT: Duration = Duration::from_secs(3);
/// How long to wait for the router to remove the UPnP mapping when hosting stops or the program exits.
const UPNP_REMOVE_TIMEOUT: Duration = Duration::from_secs(5);
/// How long to wait for the hosting listener to close before aborting it.
const HOST_STOP_TIMEOUT: Duration = Duration::from_secs(2);
/// Description attached to the UPnP port mapping (shown in the router's port-forwarding table).
const UPNP_DESCRIPTION: &str = "VOIDWRAP RUN";
/// Message used when the server itself listens on all interfaces (`VOIDWRAP_HOST=0.0.0.0`).
const FORCED_OPEN_MESSAGE: &str = "Hosting is on because VOIDWRAP_HOST=0.0.0.0 was set when the server started. Friends can join with the addresses below.";

/// UPnP state shown to the player: "off" | "trying" | "open" | "failed".
#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
#[serde(rename_all = "lowercase")]
pub enum UpnpStatus {
    /// Hosting is off, or UPnP has not been attempted.
    Off,
    /// Asking the router to open the port.
    Trying,
    /// Router opened the port; friends anywhere can join.
    Open,
    /// Router refused or was unreachable; LAN friends can still join.
    Failed,
}

impl UpnpStatus {
    /// Plain-words sentence for the player (for non-technical players).
    pub fn message(self) -> &'static str {
        match self {
            UpnpStatus::Off => "Hosting is off. Turn it on so friends can join.",
            UpnpStatus::Trying => "Asking your router to open the door for friends...",
            UpnpStatus::Open => "Friends anywhere can join with the address below.",
            UpnpStatus::Failed => "Your router did not allow automatic setup. Friends on your Wi-Fi can still join; for internet play see HOW-TO-PLAY.md.",
        }
    }
}

/// JSON body returned by `GET /api/host` and `POST /api/host`.
#[derive(Clone, Debug, Serialize)]
pub struct HostStatus {
    /// True while the hosting listener is active (or the server was started open).
    pub hosting: bool,
    /// Port friends connect to.
    pub port: u16,
    /// This machine's private IPv4 addresses (LAN), deduplicated and sorted.
    pub lan: Vec<String>,
    /// Public address friends on the internet can use, if the router reported one.
    pub public: Option<String>,
    /// UPnP state.
    pub upnp: UpnpStatus,
    /// Plain-words message for the player.
    pub message: String,
}

/// JSON body accepted by `POST /api/host`.
#[derive(Clone, Copy, Debug, Deserialize)]
pub struct HostRequest {
    /// True to start hosting, false to stop.
    pub enable: bool,
}

/// Parse the hosting port from `VOIDWRAP_HOST_PORT`.
/// Missing, empty, zero or unparseable values fall back to [`DEFAULT_HOST_PORT`].
pub fn parse_host_port(raw: Option<&str>) -> u16 {
    raw.and_then(|v| v.trim().parse::<u16>().ok())
        .filter(|&p| p != 0)
        .unwrap_or(DEFAULT_HOST_PORT)
}

/// True when the peer is on this machine (127.0.0.0/8 or `[::1]`).
/// Used to keep `/api/host` local-only, so friends can never toggle hosting.
pub fn is_loopback_peer(peer: &SocketAddr) -> bool {
    peer.ip().is_loopback()
}

/// True for RFC 1918 private IPv4 addresses (10/8, 172.16/12, 192.168/16).
pub fn is_private_ipv4(ip: Ipv4Addr) -> bool {
    let o = ip.octets();
    o[0] == 10 || (o[0] == 172 && (16..=31).contains(&o[1])) || (o[0] == 192 && o[1] == 168)
}

/// Filter to private addresses, deduplicate and sort them (stable output, testable).
pub fn private_ipv4_strings(addrs: &[Ipv4Addr]) -> Vec<String> {
    let mut list: Vec<String> = addrs
        .iter()
        .filter(|ip| is_private_ipv4(**ip))
        .map(|ip| ip.to_string())
        .collect();
    list.sort();
    list.dedup();
    list
}

/// Local IPv4 address this machine would use to reach `target`
/// (a UDP `connect` selects the route; no packet is ever sent).
fn outbound_ipv4(target: Ipv4Addr) -> Option<Ipv4Addr> {
    let socket = std::net::UdpSocket::bind((Ipv4Addr::UNSPECIFIED, 0)).ok()?;
    socket.connect((target, 80)).ok()?;
    match socket.local_addr().ok()?.ip() {
        IpAddr::V4(v4) => Some(v4),
        IpAddr::V6(_) => None,
    }
}

/// This machine's private IPv4 addresses, best effort. Sends no traffic.
pub fn lan_ipv4_addresses() -> Vec<String> {
    // Two well-known public resolvers usually route via the primary interface; the
    // connect only picks a route, it does not send a packet.
    let probes = [Ipv4Addr::new(8, 8, 8, 8), Ipv4Addr::new(1, 1, 1, 1)];
    let found: Vec<Ipv4Addr> = probes.into_iter().filter_map(outbound_ipv4).collect();
    private_ipv4_strings(&found)
}

/// Shared hosting state for the HTTP routes and the server. Clone into `Arc`s.
pub struct HostManager {
    main_port: u16,
    host_port: u16,
    forced_open: bool,
    inner: Mutex<HostInner>,
}

struct HostInner {
    /// Shutdown signal for the hosting listener, present while it runs.
    shutdown: Option<tokio::sync::oneshot::Sender<()>>,
    /// Handle of the hosting listener task, present while it runs.
    listener: Option<tokio::task::JoinHandle<()>>,
    /// UPnP gateway connection, kept so the mapping can be removed on stop/exit.
    gateway: Option<igd_next::aio::Gateway<igd_next::aio::tokio::Tokio>>,
    /// Current UPnP state.
    upnp: UpnpStatus,
    /// Public IP as reported by the router, if any.
    public: Option<IpAddr>,
}

impl HostManager {
    /// Create a manager. `forced_open` must be true when the server itself already
    /// listens on all interfaces (`VOIDWRAP_HOST=0.0.0.0` or `PORT` set).
    pub fn new(main_port: u16, host_port: u16, forced_open: bool) -> Self {
        Self {
            main_port,
            host_port,
            forced_open,
            inner: Mutex::new(HostInner {
                shutdown: None,
                listener: None,
                gateway: None,
                upnp: UpnpStatus::Off,
                public: None,
            }),
        }
    }

    /// Port reported to friends.
    fn reported_port(&self) -> u16 {
        if self.forced_open {
            self.main_port
        } else {
            self.host_port
        }
    }

    /// Build the JSON status from a locked inner state (synchronous).
    fn snapshot(&self, inner: &HostInner) -> HostStatus {
        let hosting = self.forced_open || inner.listener.is_some();
        let upnp = if hosting { inner.upnp } else { UpnpStatus::Off };
        let message = if self.forced_open {
            FORCED_OPEN_MESSAGE.to_string()
        } else {
            upnp.message().to_string()
        };
        HostStatus {
            hosting,
            port: self.reported_port(),
            lan: lan_ipv4_addresses(),
            public: if hosting {
                inner.public.map(|ip| ip.to_string())
            } else {
                None
            },
            upnp,
            message,
        }
    }

    /// Build the JSON status on failure to open the hosting port.
    fn bind_failure(&self, e: &std::io::Error) -> HostStatus {
        HostStatus {
            hosting: false,
            port: self.host_port,
            lan: lan_ipv4_addresses(),
            public: None,
            upnp: UpnpStatus::Off,
            message: format!("Could not open port {} so friends can join ({e}).", self.host_port),
        }
    }

    /// Current hosting status.
    pub async fn status(&self) -> HostStatus {
        let inner = self.inner.lock().await;
        self.snapshot(&inner)
    }

    /// Start the hosting listener (no-op when already hosting or when forced open).
    /// `app` must serve the same router as the main listener, with the same game.
    pub async fn start_hosting(self: &Arc<Self>, app: axum::Router) -> HostStatus {
        if self.forced_open {
            return self.status().await;
        }
        {
            let inner = self.inner.lock().await;
            if inner.listener.is_some() {
                return self.snapshot(&inner);
            }
        }
        let addr = SocketAddr::new(IpAddr::V4(Ipv4Addr::UNSPECIFIED), self.host_port);
        let listener = match tokio::net::TcpListener::bind(addr).await {
            Ok(listener) => listener,
            Err(e) => {
                tracing::error!("cannot start hosting on {addr}: {e}");
                return self.bind_failure(&e);
            }
        };
        let (shutdown_tx, shutdown_rx) = tokio::sync::oneshot::channel::<()>();
        let task = tokio::spawn(async move {
            let server = axum::serve(listener, app.into_make_service_with_connect_info::<SocketAddr>())
                .with_graceful_shutdown(async move {
                    let _ = shutdown_rx.await;
                });
            if let Err(e) = server.await {
                tracing::error!("hosting listener error: {e}");
            }
        });
        {
            let mut inner = self.inner.lock().await;
            inner.shutdown = Some(shutdown_tx);
            inner.listener = Some(task);
            inner.upnp = UpnpStatus::Trying;
            inner.public = None;
        }
        tracing::info!(
            "hosting listener on 0.0.0.0:{} (LAN); asking the router for a UPnP mapping",
            self.host_port
        );
        let manager = Arc::clone(self);
        tokio::spawn(async move {
            manager.upnp_try().await;
        });
        self.status().await
    }

    /// Stop hosting: close the listener and remove the UPnP mapping (best effort).
    pub async fn stop_hosting(&self) -> HostStatus {
        if self.forced_open {
            // Keep today's behaviour: the main listener stays open; only the report changes.
            return self.status().await;
        }
        let (shutdown, task, gateway) = {
            let mut inner = self.inner.lock().await;
            inner.upnp = UpnpStatus::Off;
            inner.public = None;
            (
                inner.shutdown.take(),
                inner.listener.take(),
                inner.gateway.take(),
            )
        };
        if let Some(tx) = shutdown {
            let _ = tx.send(());
        }
        if let Some(mut handle) = task {
            if tokio::time::timeout(HOST_STOP_TIMEOUT, &mut handle).await.is_err() {
                // A friend still has an open connection; close anyway.
                handle.abort();
                let _ = handle.await;
            }
        }
        if let Some(gateway) = gateway {
            let _ = tokio::time::timeout(UPNP_REMOVE_TIMEOUT, upnp_close(gateway, self.host_port)).await;
        }
        self.status().await
    }

    /// Ask the router to forward our hosting port to this machine's LAN address.
    /// Runs in the background; on success friends anywhere can join with `public`.
    async fn upnp_try(self: Arc<Self>) {
        let port = self.host_port;
        let result = upnp_open(port).await;
        let mut inner = self.inner.lock().await;
        if inner.listener.is_none() {
            // Hosting was stopped while we were asking the router; undo the mapping.
            drop(inner);
            if let Ok((_, gateway)) = result {
                let _ =
                    tokio::time::timeout(UPNP_REMOVE_TIMEOUT, upnp_close(gateway, port)).await;
            }
            return;
        }
        match result {
            Ok((external, gateway)) => {
                inner.upnp = UpnpStatus::Open;
                inner.public = Some(external);
                inner.gateway = Some(gateway);
            }
            Err(e) => {
                tracing::warn!("UPnP setup failed: {e}");
                inner.upnp = UpnpStatus::Failed;
                inner.public = None;
            }
        }
    }
}

/// One UPnP setup: find the gateway (3 s timeout), map TCP `port` for 2 hours
/// ("VOIDWRAP RUN") and read our public IP. The router maps the port to the device
/// that asked for the mapping, which is this machine's LAN address (the address used
/// to reach the gateway). Isolated in one function so the igd-next calls are easy to audit.
async fn upnp_open(
    port: u16,
) -> Result<(IpAddr, igd_next::aio::Gateway<igd_next::aio::tokio::Tokio>), String> {
    let mut options = igd_next::SearchOptions::default();
    options.timeout = Some(UPNP_SEARCH_TIMEOUT);
    let gateway = igd_next::aio::tokio::search_gateway(options)
        .await
        .map_err(|e| format!("gateway search failed: {e}"))?;
    let external = gateway
        .get_external_ip()
        .await
        .map_err(|e| format!("could not read the external IP: {e}"))?;
    gateway
        .add_port(
            Duration::from_secs(UPNP_LEASE_SECS),
            port,
            port,
            igd_next::PortMappingProtocol::TCP,
            UPNP_DESCRIPTION,
        )
        .await
        .map_err(|e| format!("router refused the port mapping: {e}"))?;
    Ok((external, gateway))
}

/// Remove the UPnP port mapping (best effort).
async fn upnp_close(gateway: igd_next::aio::Gateway<igd_next::aio::tokio::Tokio>, port: u16) {
    if let Err(e) = gateway
        .remove_port(port, igd_next::PortMappingProtocol::TCP)
        .await
    {
        tracing::warn!("could not remove the UPnP mapping for port {port}: {e}");
    }
}

//! WebSocket transport. Validates framing and decodes client messages at the boundary;
//! everything past this point only sees well-formed `ClientMsg` values.
//!
//! Also serves the loopback-only HTTP API for hosting for friends (`GET`/`POST /api/host`);
//! see `host.rs` for the hosting logic and the JSON shape.

use crate::game::{ConnId, GameEvent, OUTBOX_CAPACITY};

use axum::{
    Json, Router,
    extract::{
        ConnectInfo, State, WebSocketUpgrade,

};

static NEXT_CONN: AtomicU64 = AtomicU64::new(1);

/// State shared by every route: the game inbox, the web client directory and the hosting controller.
#[derive(Clone)]
pub struct AppState {
    /// Sender into the game task.
    pub game_tx: mpsc::Sender<GameEvent>,
    /// Built web client, reused when the hosting listener serves the game.
    pub web_dir: Option<PathBuf>,
    /// Hosting controller behind `/api/host`.
    pub host: Arc<HostManager>,
}

/// Build the router: the WebSocket game, the health check, the host API and the static web client.
pub fn router(game_tx: mpsc::Sender<GameEvent>, web_dir: Option<PathBuf>, host: Arc<HostManager>) -> Router {
    let state = AppState {
        game_tx,
        web_dir: web_dir.clone(),
        host,
    };
    let app = Router::new()
        .route("/ws", get(ws_handler))
        .route("/health", get(|| async { "ok" }))
        .route("/api/host", get(host_get).post(host_post))
        .with_state(state);
    match web_dir {
        // Serve the built web client from the same address, so one program is the whole game.
        Some(dir) => {
            let not_found = ServeFile::new(dir.join("404.html"));
            app.fallback_service(
                ServeDir::new(dir)
                    .append_index_html_on_directories(true)
                    .fallback(not_found),
            )
        }
        None => app,
    }
}

/// Locate the built web client: $VOIDWRAP_WEB_DIR, then `web/` next to the executable (release download),
/// then `web/out` in the source tree (after `pnpm --dir web build`).
pub fn find_web_dir() -> Option<PathBuf> {
    let has_index = |d: &PathBuf| d.join("index.html").is_file();
    if let Some(dir) = std::env::var_os("VOIDWRAP_WEB_DIR").map(PathBuf::from) {
        return has_index(&dir).then_some(dir);

        .ok()

    candidates.into_iter().flatten().find(has_index)
}

/// `GET /api/host`: current hosting status. Loopback only; friends can never call this.
async fn host_get(
    State(state): State<AppState>,
    ConnectInfo(peer): ConnectInfo<SocketAddr>,
) -> Response {
    if !host::is_loopback_peer(&peer) {
        tracing::warn!("refusing GET /api/host from non-loopback peer {peer}");
        return StatusCode::FORBIDDEN.into_response();
    }
    Json(state.host.status().await).into_response()
}

/// `POST /api/host` with `{ "enable": bool }`: start or stop hosting. Loopback only.
async fn host_post(
    State(state): State<AppState>,
    ConnectInfo(peer): ConnectInfo<SocketAddr>,
    Json(request): Json<HostRequest>,
) -> Response {
    if !host::is_loopback_peer(&peer) {
        tracing::warn!("refusing POST /api/host from non-loopback peer {peer}");
        return StatusCode::FORBIDDEN.into_response();
    }
    let status = if request.enable {
        // The hosting listener serves a fresh copy of the same router and the same game.
        let app = router(
            state.game_tx.clone(),
            state.web_dir.clone(),
            Arc::clone(&state.host),
        );
        state.host.start_hosting(app).await
    } else {
        state.host.stop_hosting().await
    };
    tracing::info!(hosting = status.hosting, port = status.port, "host status: {}", status.message);
    Json(status).into_response()
}

async fn handle_socket(socket: WebSocket, game_tx: mpsc::Sender<GameEvent>) {
    let id: ConnId = NEXT_CONN.fetch_add(1, Ordering::Relaxed);
    let (mut sink, mut stream) = socket.split();
    let (out_tx, mut out_rx) = mpsc::channel::<Vec<u8>>(OUTBOX_CAPACITY);

    if game_tx.send(GameEvent::Connected { id, out: out_tx }).await.is_err() {
        return;
    }

    // Writer: drains this connection's bounded outbox. Ends when the game drops the sender.
    let writer = tokio::spawn(async move {
        while let Some(bytes) = out_rx.recv().await {
            if sink.send(Message::Binary(bytes.into())).await.is_err() {
                break;
            }
        }
        let _ = sink.close().await;
    });

    // Reader: decode at the boundary, forward to the game task.
    while let Some(Ok(msg)) = stream.next().await {
        match msg {
            Message::Binary(bytes) => match protocol::decode_client_msg(&bytes) {
                Ok(m) => {
                    if game_tx.send(GameEvent::Message { id, msg: m }).await.is_err() {
                        break;
                    }
                }
                Err(e) => {
                    tracing::warn!(conn = id, "dropping connection: bad message: {e}");
                    break;
                }
            },
            Message::Close(_) => break,
            // Pings are answered by axum; text frames are not part of the protocol.
            Message::Text(_) => {
                tracing::warn!(conn = id, "dropping connection: text frame");
                break;
            }
            Message::Ping(_) | Message::Pong(_) => {}
        }
    }
    let _ = game_tx.send(GameEvent::Disconnected { id }).await;
    // The game task drops our outbox sender on Disconnected, which ends the writer.
    let _ = writer.await;
}

//! Native desktop window for the game (enabled by the `desktop` cargo feature).
//!
//! The server runs on a tokio runtime in a background thread; the main thread owns the

use std::time::{Duration, Instant};

/// Title of the native window.
pub const WINDOW_TITLE: &str = "VOIDWRAP RUN";
/// Window width in logical pixels.
const WINDOW_WIDTH: f64 = 1280.0;
/// Window height in logical pixels.
const WINDOW_HEIGHT: f64 = 800.0;

/// Decide whether to run in a native window. The window is disabled by `--browser` and
/// `--headless`, and when `PORT` is set (hosting platforms, background agents).
pub fn should_open_window(args: &[String], port_env_set: bool) -> bool {
    if port_env_set {
        return false;
    }
    !args.iter().any(|a| a == "--browser" || a == "--headless")
}

/// Wait until the server accepts TCP connections on `port`, or `timeout` passes
/// (in which case the window is opened anyway).
pub fn wait_for_server(port: u16, timeout: Duration) {
    let deadline = Instant::now() + timeout;
    loop {
        if std::net::TcpStream::connect((std::net::Ipv4Addr::LOCALHOST, port)).is_ok() {
            return;
        }
        if Instant::now() >= deadline {
            tracing::warn!(
                "server did not accept connections within {} s; opening the window anyway",
                timeout.as_secs()
            );
            return;
        }
        std::thread::sleep(Duration::from_millis(50));
    }
}

/// Open the native window and block until the user closes it.
/// Sends `shutdown` to the server when the window closes (best effort).
pub fn open_window(port: u16, shutdown: tokio::sync::oneshot::Sender<()>) {
    use tao::dpi::LogicalSize;
    use tao::event::{Event, WindowEvent};
    use tao::event_loop::{ControlFlow, EventLoop};
    use tao::window::WindowBuilder;

    let event_loop = match EventLoop::<()>::new() {
        Ok(event_loop) => event_loop,
        Err(e) => {
            tracing::error!("could not start the window event loop: {e}");
            let _ = shutdown.send(());
            return;
        }
    };
    let window = match WindowBuilder::new()
        .with_title(WINDOW_TITLE)
        .with_inner_size(LogicalSize::new(WINDOW_WIDTH, WINDOW_HEIGHT))
        .with_resizable(true)
        .build(&event_loop)
    {
        Ok(window) => window,
        Err(e) => {
            tracing::error!("could not create the window: {e}");
            let _ = shutdown.send(());
            return;
        }
    };
    let url = format!("http://localhost:{port}/");
    // `window` and `_webview` must stay alive for the whole event loop.
    let _webview = match wry::WebViewBuilder::new().with_url(&url).build(&window) {
        Ok(webview) => webview,
        Err(e) => {
            tracing::error!("could not create the web view ({e}); open {url} in a browser instead");
            let _ = shutdown.send(());
            return;
        }
    };
    let mut shutdown = Some(shutdown);
    let _ = event_loop.run(move |event, _, control_flow| {
        *control_flow = ControlFlow::Wait;
        if let Event::WindowEvent {
            event: WindowEvent::CloseRequested,
            ..
        } = event
        {
            if let Some(tx) = shutdown.take() {
                let _ = tx.send(());
            }
            *control_flow = ControlFlow::Exit;
        }
    });
}

//! VOIDWRAP RUN authoritative game server.
//!
//! Architecture: one game task owns all simulation state and runs the fixed 20 Hz
//! tick loop. Each WebSocket connection is a pair of small tasks that only exchange
//! bounded channel messages with the game task — no shared mutable state, no locks.
//!
//! Desktop builds (`--features desktop`) open a native window instead of a browser,
//! unless `--browser`, `--headless` or `PORT` are used; see `desktop.rs`.

mod content;
#[cfg(feature = "desktop")]
mod desktop;
mod game;

mod run;

/// Inbound queue from all connections to the game task.
const GAME_INBOX_CAPACITY: usize = 1024;

/// Everything `main` needs before choosing between the windowed and the normal path.
struct Startup {
    /// Address the normal listener binds.
    addr: SocketAddr,
    /// Built web client directory, if found.
    web_dir: Option<PathBuf>,
    /// First level of a new run (dev/QA aid).
    start_level: u32,
    /// True when the server itself listens on all interfaces (`VOIDWRAP_HOST=0.0.0.0` or `PORT` set).
    forced_open: bool,
    /// Port the hosting listener uses (`VOIDWRAP_HOST_PORT`, default 7879).
    host_port: u16,
}

/// Read the environment once and compute the listen address, web directory and ports.
fn startup() -> Startup {
    // Where to listen. Hosting platforms set PORT and need every interface; locally the default is this
    // computer only. VOIDWRAP_HOST=0.0.0.0 opens the game to other computers (LAN multiplayer).
    let hosted_port: Option<u16> = std::env::var("PORT").ok().and_then(|p| p.parse().ok());
    let port: u16 = hosted_port
        .or_else(|| std::env::var("VOIDWRAP_PORT").ok().and_then(|p| p.parse().ok()))
        .unwrap_or(7878);
    let default_host = if hosted_port.is_some() {
        "0.0.0.0"
    } else {
        "127.0.0.1"
    };
    let host: std::net::IpAddr = std::env::var("VOIDWRAP_HOST")
        .ok()
        .and_then(|h| h.parse().ok())
        .unwrap_or_else(|| {
            default_host
                .parse()
                .unwrap_or(std::net::IpAddr::from([127, 0, 0, 1]))
        });
    let addr = SocketAddr::from((host, port));
    let forced_open = host.is_unspecified();
    // Dev/QA aid: start every run at a later level (e.g. to review each colour theme).
    let start_level: u32 = std::env::var("VOIDWRAP_START_LEVEL")
        .ok()
        .and_then(|v| v.parse().ok())
        .unwrap_or(1)
        .clamp(1, 10_000);
    let host_port = host::parse_host_port(std::env::var("VOIDWRAP_HOST_PORT").ok().as_deref());
    Startup {
        addr,
        web_dir: net::find_web_dir(),
        start_level,
        forced_open,
        host_port,
    }
}

fn main() {
    init_logging();
    let startup = startup();

    // Desktop window mode: server on a background thread, tao/wry window on the main thread.
    #[cfg(feature = "desktop")]
    {
        let args: Vec<String> = std::env::args().skip(1).collect();
        let port_env_set = std::env::var_os("PORT").is_some();
        if desktop::should_open_window(&args, port_env_set) {
            run_desktop(startup);
            return;
        }
    }

    let runtime = match tokio::runtime::Builder::new_multi_thread().enable_all().build() {
        Ok(runtime) => runtime,
        Err(e) => {
            tracing::error!("could not create the tokio runtime: {e}");
            std::process::exit(1);
        }
    };
    if let Err(e) = runtime.block_on(run_server(startup, None)) {
        tracing::error!("fatal: {e}");
        std::process::exit(1);
    }
}

/// Desktop path: server on a background thread; window on the main thread; closing the window
/// shuts the server down (best effort) and then exits.
#[cfg(feature = "desktop")]
fn run_desktop(startup: Startup) {
    let port = startup.addr.port();
    let (shutdown_tx, shutdown_rx) = tokio::sync::oneshot::channel::<()>();
    let server_thread = std::thread::Builder::new()
        .name("voidwrap-server".to_string())
        .spawn(move || {
            let runtime = match tokio::runtime::Builder::new_multi_thread().enable_all().build() {
                Ok(runtime) => runtime,
                Err(e) => {
                    tracing::error!("could not create the tokio runtime: {e}");
                    return;
                }
            };
            if let Err(e) = runtime.block_on(run_server(startup, Some(shutdown_rx))) {
                tracing::error!("server stopped with an error: {e}");
            }
        });
    let server_thread = match server_thread {
        Ok(thread) => Some(thread),
        Err(e) => {
            tracing::error!("could not start the server thread: {e}");
            None
        }
    };
    desktop::wait_for_server(port, std::time::Duration::from_secs(10));
    desktop::open_window(port, shutdown_tx);
    // Best effort: let the server finish graceful shutdown (removes any UPnP mapping).
    if let Some(thread) = server_thread {
        let _ = thread.join();
    }
}

/// Run the server until the shutdown signal (Ctrl-C, or the desktop window closing).
/// Returns an error message when startup fails.
async fn run_server(
    startup: Startup,
    window_closed: Option<tokio::sync::oneshot::Receiver<()>>,
) -> Result<(), String> {
    let Startup {
        addr,
        web_dir,
        start_level,
        forced_open,
        host_port,
    } = startup;

    if start_level != 1 {
        tracing::warn!("VOIDWRAP_START_LEVEL={start_level}: runs start at level {start_level} (dev/QA only)");
    }
    let (game_tx, game_rx) = mpsc::channel(GAME_INBOX_CAPACITY);
    // Seed draft offers and level layouts from the clock: every run is different.
    let seed = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .map(|d| d.subsec_nanos() ^ (d.as_secs() as u32))
        .unwrap_or(0x5eed);
    let state = game::Game::starting_at(start_level, seed).map_err(|e| format!("cannot start: {e}"))?;
    let game = tokio::spawn(game::run(game_rx, state));

    let listener = tokio::net::TcpListener::bind(addr)
        .await
        .map_err(|e| format!("cannot bind {addr}: {e}"))?;
    tracing::info!("VOIDWRAP RUN server listening on ws://{addr}/ws");
    match &web_dir {
        Some(dir) => {
            let shown = if addr.ip().is_unspecified() || addr.ip().is_loopback() {
                "localhost".to_string()
            } else {
                addr.ip().to_string()
            };
            tracing::info!("serving the game from {}", dir.display());
            tracing::info!(">>> open http://{shown}:{} in your browser to play <<<", addr.port());
            if std::env::var_os("VOIDWRAP_OPEN_BROWSER").is_some() {
                open_browser(&format!("http://localhost:{}", addr.port()));
            }
        }
        None => tracing::info!("no built web client found (dev mode): run the web dev server separately"),
    }

    let host_manager = Arc::new(host::HostManager::new(addr.port(), host_port, forced_open));
    {
        let status = host_manager.status().await;
        if let Ok(json) = serde_json::to_string(&status) {
            tracing::info!("host status at startup: {json}");
        }
        if status.hosting {
            tracing::info!(
                "hosting is ON: friends can reach this computer on port {} (started with VOIDWRAP_HOST=0.0.0.0)",
                status.port
            );
        } else {
            tracing::info!("hosting is off: only this computer can play. Turn it on in the game.");
        }
    }

    let app = net::router(game_tx, web_dir, Arc::clone(&host_manager));
    let serve = axum::serve(listener, app.into_make_service_with_connect_info::<SocketAddr>())
        .with_graceful_shutdown(shutdown_signal(window_closed));
    if let Err(e) = serve.await {
        tracing::error!("server error: {e}");
    }
    // Best effort: close the hosting listener and remove any UPnP mapping before exit.
    let status = host_manager.stop_hosting().await;
    if let Ok(json) = serde_json::to_string(&status) {
        tracing::info!("host status at shutdown: {json}");
    }
    game.abort();
    Ok(())
}

/// Wait for Ctrl-C, or for the desktop window to close.
async fn shutdown_signal(window_closed: Option<tokio::sync::oneshot::Receiver<()>>) {
    match window_closed {
        Some(rx) => {
            tokio::select! {
                res = tokio::signal::ctrl_c() => {
                    if let Err(e) = res {
                        tracing::warn!("ctrl-c handler failed: {e}");
                    } else {
                        tracing::info!("shutting down (Ctrl-C)");
                    }
                }
                _ = rx => tracing::info!("the window was closed; shutting down"),
            }
        }
        None => {
            let _ = tokio::signal::ctrl_c().await;
            tracing::info!("shutting down");
        }
    }
}

/// `std::io::Write` adapter around a shared file handle, for tracing's `with_writer`.
#[cfg(all(feature = "desktop", target_os = "windows"))]
struct SharedFileWriter(std::sync::Arc<std::sync::Mutex<std::fs::File>>);

#[cfg(all(feature = "desktop", target_os = "windows"))]
impl std::io::Write for SharedFileWriter {
    fn write(&mut self, buf: &[u8]) -> std::io::Result<usize> {
        match self.0.lock() {
            Ok(mut file) => std::io::Write::write(&mut *file, buf),
            Err(_) => Err(std::io::Error::other("voidwrap.log lock poisoned")),
        }
    }
    fn flush(&mut self) -> std::io::Result<()> {
        match self.0.lock() {
            Ok(mut file) => std::io::Write::flush(&mut *file),
            Err(_) => Err(std::io::Error::other("voidwrap.log lock poisoned")),
        }
    }
}

=== NOTES ===
- Architecture: main() is a plain fn. Non-desktop builds create a multi-thread tokio runtime and block_on (same behaviour as #[tokio::main]); desktop builds run the server in a background thread and tao/wry on the main thread; window close sends a oneshot that feeds graceful shutdown, then the thread is joined.
- UPnP mapping is removed on stop AND on exit: run_server always calls host_manager.stop_hosting() after serve returns (2 s graceful for the hosting listener, then abort; UPnP remove has a 5 s timeout). A UPnP try completing after hosting stopped is undone immediately.
- Exact wgt/wry/tao/igd-next calls used (NONE verified by compilation — unverified API use, isolated in desktop.rs::open_window and host.rs::upnp_open/upnp_close):

State type Gateway<igd_next::aio::tokio::Tokio>.
