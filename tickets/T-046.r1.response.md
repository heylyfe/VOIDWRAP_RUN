function traceRoundRect(
  ctx: CanvasRenderingContext2D,
  x: number,
  y: number,
  w: number,
  h: number,
  r: number,
): void {
  ctx.beginPath();
  ctx.moveTo(x + r, y);
  ctx.arcTo(x + w, y, x + w, y + h, r);
  ctx.arcTo(x + w, y + h, x, y + h, r);
  ctx.arcTo(x, y + h, x, y, r);
  ctx.arcTo(x, y, x + w, y, r);
  ctx.closePath();
}

/** Canvas-drawn "E" key prompt; created once, shared by every altar. */
function createPromptTexture(): CanvasTexture | null {
  if (typeof document === "undefined") return null;
  const canvas = document.createElement("canvas");
  canvas.width = 128;
  canvas.height = 128;
  const ctx = canvas.getContext("2d");
  if (!ctx) return null;
  ctx.clearRect(0, 0, 128, 128);
  ctx.fillStyle = "rgba(8, 8, 14, 0.72)";
  traceRoundRect(ctx, 14, 18, 100, 92, 18);
  ctx.fill();
  ctx.strokeStyle = "rgba(255, 255, 255, 0.85)";
  ctx.lineWidth = 5;
  traceRoundRect(ctx, 14, 18, 100, 92, 18);
  ctx.stroke();
  ctx.fillStyle = "#ffffff";
  ctx.font = "bold 64px ui-monospace, Menlo, monospace";
  ctx.textAlign = "center";
  ctx.textBaseline = "middle";
  ctx.fillText("E", 64, 66);
  const tex = new CanvasTexture(canvas);
  tex.colorSpace = SRGBColorSpace;
  return tex;
}

import { useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import { AdditiveBlending } from "three";
import type { Group, MeshBasicMaterial } from "three";
import { CLASS_BLINKER } from "@/protocol/generated";
import type { GameSession } from "@/game/session";
import { classColor } from "./classColors";

const MAX_FLASHES = 4;
const FLASH_TIME = 0.45;

interface FlashSlot {
  x: number;
  y: number;
  z: number;
  t0: number;
  active: boolean;
}

/**
 * Blinker ability: when the server's abilityActiveFrames jumps upward (a blink just
 * started), spawn a brief expanding afterimage sphere at the position the player occupied
 * on the previous frame (the blink origin). Small fixed pool, no per-frame allocation.
 */
export function BlinkFlash({ session }: { session: GameSession }): JSX.Element {
  const meshes = useRef<(Group | null)[]>(Array.from({ length: MAX_FLASHES }, () => null));
  const slots = useRef<FlashSlot[]>(Array.from({ length: MAX_FLASHES }, () => ({ x: 0, y: 0, z: 0, t0: 0, active: false })));
  const prevFrames = useRef(0);
  const lastPos = useRef({ x: 0, y: 0, z: 0, ok: false });
  const flashMats = useRef<(MeshBasicMaterial | null)[]>(
    Array.from({ length: MAX_FLASHES }, () => null),
  );
  const color = classColor(CLASS_BLINKER);

  useFrame((state) => {
    const t = state.clock.elapsedTime;
    const c = session.playerController();
    if (c) {
      const prev = prevFrames.current;
      if (c.class === CLASS_BLINKER && c.abilityActiveFrames > prev && lastPos.current.ok) {
        let idx = -1;
        for (let i = 0; i < MAX_FLASHES; i += 1) {
          const sl = slots.current[i];
          if (sl && !sl.active) {
            idx = i;
            break;
          }
        }
        if (idx === -1) idx = 0;
        const target = slots.current[idx];
        if (target) {
          target.x = lastPos.current.x;
          target.y = lastPos.current.y;
          target.z = lastPos.current.z;
          target.t0 = t;
          target.active = true;
        }
      }
      prevFrames.current = c.abilityActiveFrames;
      const lp = lastPos.current;
      lp.x = c.pos.x;
      lp.y = c.pos.y;
      lp.z = c.pos.z;
      lp.ok = true;
    }
    for (let i = 0; i < MAX_FLASHES; i += 1) {
      const slot = slots.current[i];
      const g = meshes.current[i];
      if (!slot || !g) continue;
      if (!slot.active) {
        g.visible = false;
        continue;
      }
      const age = t - slot.t0;
      if (age >= FLASH_TIME) {
        slot.active = false;
        g.visible = false;
        continue;
      }
      const sc = 0.5 + age * 4.5;
      g.visible = true;
      g.position.set(slot.x, slot.y + 0.9, slot.z);
      g.scale.set(sc, sc, sc);
      const mat = flashMats.current[i];
      if (mat) mat.opacity = 0.85 * (1 - age / FLASH_TIME);
    }
  });

  return (
    <group>
      {Array.from({ length: MAX_FLASHES }, (_, i) => (
        <group
          key={i}
          ref={(g) => {
            meshes.current[i] = g;
          }}
          visible={false}
        >
          <mesh>
            <sphereGeometry args={[0.5, 12, 12]} />
            <meshBasicMaterial
              ref={(m) => {
                flashMats.current[i] = m;
              }}
              color={color}
              transparent
              opacity={0.85}
              blending={AdditiveBlending}
              depthWrite={false}
              toneMapped={false}
            />
          </mesh>
        </group>
      ))}
    </group>
  );
}

const SCRATCH_FROM = new Vector3();
const SCRATCH_TO = new Vector3();
const SCRATCH_MID = new Vector3();
const SCRATCH_DIR = new Vector3();
const SCRATCH_Q = new Quaternion();

/**
 * Grappler ability: a thin glowing line from the local player's chest to the grapple
 * target point. The server sends the target point in ControllerState.abilityDir while
 * the ability is active, so no local simulation of the rope is needed.
 */
export function GrappleLine({ session }: { session: GameSession }): JSX.Element {
  const groupRef = useRef<Group>(null);
  const coreMatRef = useRef<MeshBasicMaterial>(null);
  const color = classColor(CLASS_GRAPPLER);

  useFrame((state) => {
    const group = groupRef.current;
    if (!group) return;
    const c = session.playerController();
    const show = !!c && c.class === CLASS_GRAPPLER && c.abilityActiveFrames > 0;
    group.visible = show;
    if (!c || !show) return;
    SCRATCH_FROM.set(c.pos.x, c.pos.y + 1.1, c.pos.z);
    SCRATCH_TO.set(c.abilityDir.x, c.abilityDir.y, c.abilityDir.z);
    SCRATCH_MID.addVectors(SCRATCH_FROM, SCRATCH_TO).multiplyScalar(0.5);
    SCRATCH_DIR.subVectors(SCRATCH_TO, SCRATCH_FROM);
    const len = SCRATCH_DIR.length();
    if (len < 0.05) {
      group.visible = false;
      return;
    }
    group.position.copy(SCRATCH_MID);
    SCRATCH_Q.setFromUnitVectors(UP, SCRATCH_DIR.normalize());
    group.quaternion.copy(SCRATCH_Q);
    group.scale.set(1, len, 1);
    const coreMat = coreMatRef.current;
    if (coreMat) coreMat.opacity = 0.8 + 0.2 * Math.sin(state.clock.elapsedTime * 18);
  });

  return (
    <group ref={groupRef} visible={false}>
      <mesh>
        <cylinderGeometry args={[0.028, 0.028, 1, 6, 1, true]} />
        <meshBasicMaterial
          ref={coreMatRef}
          color={color}
          transparent
          opacity={0.95}
          blending={AdditiveBlending}
          depthWrite={false}
          toneMapped={false}
        />
      </mesh>

        <cylinderGeometry args={[0.075, 0.075, 1, 6, 1, true]} />
        <meshBasicMaterial
          color={color}
          transparent
          opacity={0.22}
          blending={AdditiveBlending}
          depthWrite={false}
          toneMapped={false}

const EDGE_BAR_THICKNESS = 0.08;

const EDGE_BAR_OVERHANG = 0.04;
const EDGE_BAR_EMISSIVE = 1.4;

// Four thin bars along the top perimeter. The bars along X straddle the
// top edge (overhang EDGE_BAR_OVERHANG on each side); the bars along Z
// span exactly the gap between them so nothing overlaps at the corners.
function toEdgeBars(size: readonly [number, number, number]): EdgeBarData[] {
  const [width, height, depth] = size;
  const halfWidth = width / 2;
  const halfDepth = depth / 2;
  const barY = height / 2 + EDGE_BAR_HEIGHT / 2;
  const longLength = width + EDGE_BAR_OVERHANG * 2;
  const shortLength = depth - EDGE_BAR_OVERHANG * 2;
  return [
    {
      size: [longLength, EDGE_BAR_HEIGHT, EDGE_BAR_THICKNESS],
      position: [0, barY, halfDepth],
    },
    {
      size: [longLength, EDGE_BAR_HEIGHT, EDGE_BAR_THICKNESS],
      position: [0, barY, -halfDepth],
    },
    {
      size: [EDGE_BAR_THICKNESS, EDGE_BAR_HEIGHT, shortLength],
      position: [halfWidth, barY, 0],
    },
    {
      size: [EDGE_BAR_THICKNESS, EDGE_BAR_HEIGHT, shortLength],
      position: [-halfWidth, barY, 0],
    },
  ];
}

function toBoxData(platform: Platform): PlatformBoxData {
  const { min, max } = platform.bounds;
  const size: [number, number, number] = [
    max.x - min.x,
    max.y - min.y,
    max.z - min.z,
  ];
  const center: [number, number, number] = [
    (min.x + max.x) / 2,
    (min.y + max.y) / 2,
    (min.z + max.z) / 2,
  ];
  return {
    id: platform.id,
    kind: platform.kind,
    center,
    size,
    edgeBars: toEdgeBars(size),
  };
}

/** Multiply a hex colour by `factor` (used to darken crumble tops so they read as fragile). */
function shadeColor(hex: string, factor: number): string {
  const c = new Color(hex);
  c.multiplyScalar(factor);
  return `#${c.getHexString()}`;
}

function makeTopMaterial(baseHex: string, map: Texture | null, icy: boolean): MeshStandardMaterial {
  if (map) {
    return new MeshStandardMaterial({

      color: icy ? iceTint("#ffffff") : "#ffffff",
      roughness: icy ? ICE_ROUGHNESS : TOP_ROUGHNESS,
      emissive: icy ? ICE_EMISSIVE : "#000000",
      emissiveIntensity: icy ? ICE_EMISSIVE_INTENSITY : 0,

  return new MeshStandardMaterial({

/** Deterministic cracked-stone canvas texture for crumbling platform tops (built once). */
function createCrackTexture(baseHex: string): CanvasTexture | null {
  if (typeof document === "undefined") return null;
  const size = 256;
  const canvas = document.createElement("canvas");
  canvas.width = size;
  canvas.height = size;
  const ctx = canvas.getContext("2d");
  if (!ctx) return null;
  // Deterministic LCG so every client sees the same cracks.
  let seed = 9001;
  const rnd = (): number => {
    seed = (Math.imul(seed, 1664525) + 1013904223) >>> 0;
    return seed / 4294967296;
  };
  ctx.fillStyle = baseHex;
  ctx.fillRect(0, 0, size, size);
  for (let i = 0; i < 40; i += 1) {
    ctx.fillStyle = `rgba(0, 0, 0, ${0.08 + rnd() * 0.12})`;
    ctx.beginPath();
    ctx.arc(rnd() * size, rnd() * size, 12 + rnd() * 42, 0, Math.PI * 2);
    ctx.fill();
  }
  for (let i = 0; i < 14; i += 1) {
    ctx.strokeStyle = "rgba(12, 9, 8, 0.9)";
    ctx.lineWidth = 1.5 + rnd() * 2.5;
    ctx.beginPath();
    let x = rnd() * size;
    let y = rnd() * size;
    let angle = rnd() * Math.PI * 2;
    ctx.moveTo(x, y);
    const segments = 5 + Math.floor(rnd() * 5);
    for (let j = 0; j < segments; j += 1) {
      angle += (rnd() - 0.5) * 1.4;
      const len = 18 + rnd() * 30;
      x += Math.cos(angle) * len;
      y += Math.sin(angle) * len;
      ctx.lineTo(x, y);
    }
    ctx.stroke();
  }
  const tex = new CanvasTexture(canvas);
  tex.wrapS = RepeatWrapping;
  tex.wrapT = RepeatWrapping;
  tex.colorSpace = SRGBColorSpace;
  tex.anisotropy = 4;
  return tex;
}

/** Materials shared by every platform of a theme (textures are module-cached). */
function createThemeMaterials(theme: Theme, icy: boolean): ThemeMaterials {
  const textures = themeTextures(theme);
  const top = makeTopMaterial(theme.tile.light, textures.top ?? null, icy);
  const side = new MeshStandardMaterial(
    textures.side
      ? { map: textures.side, color: icy ? iceTint(SIDE_TINT) : SIDE_TINT, roughness: SIDE_ROUGHNESS }
      : { color: icy ? iceTint(theme.tile.side) : theme.tile.side, roughness: SIDE_ROUGHNESS },
  );
  const bottom = new MeshStandardMaterial({
    color: icy ? iceTint(theme.tile.dark) : theme.tile.dark,
    roughness: BOTTOM_ROUGHNESS,
  });
  const edgeTrim = new MeshStandardMaterial({
    color: icy ? iceTint(theme.tile.edge) : theme.tile.edge,
    emissive: icy ? iceTint(theme.tile.edge) : theme.tile.edge,
    emissiveIntensity: EDGE_BAR_EMISSIVE,
    roughness: icy ? 0.2 : 0.4,
  });
  return {
    // BoxGeometry face order: +x, -x, +y, -y, +z, -z.
    byFace: [side, side, top, bottom, side, side],
    side,
    bottom,
    edgeTrim,
    crumbleBase: shadeColor(theme.tile.light, CRUMBLE_SHADE),
    ghostColor: theme.tile.edge,
    all: [top, side, bottom, edgeTrim],
  };
}

function disposeMaterials(materials: ThemeMaterials): void {
  for (const material of materials.all) {
    material.dispose();
  }
}

function PlatformMesh(props: {
  box: PlatformBoxData;
  materials: ThemeMaterials;
  tileState: number | null;
  icy: boolean;
}): JSX.Element {
  const { box, materials, tileState, icy } = props;
  const crumble = box.kind === PLATFORM_CRUMBLE;
  const warn = tileState === TILE_WARN;
  const gone = tileState === TILE_GONE;
  // World-scaled UVs are memoised per platform box; disposed on change/unmount.
  const geometry = useMemo(
    () => worldScaledBoxGeometry(box.size[0], box.size[1], box.size[2]),
    [box.size],
  );
  useEffect(() => () => geometry.dispose(), [geometry]);

  // Crumble platforms get their own cracked top material so TILE_WARN can flash it.
  const ownTop = useMemo(() => {
    if (!crumble) return null;
    const base = icy ? iceTint(materials.crumbleBase) : materials.crumbleBase;
    const map = createCrackTexture(base);
    const mat = new MeshStandardMaterial({
      color: map ? "#ffffff" : base,
      map,
      roughness: icy ? ICE_ROUGHNESS : CRUMBLE_ROUGHNESS,
      emissive: WARN_EMISSIVE,
      emissiveIntensity: CRUMBLE_IDLE_EMISSIVE,
    });
    return { mat, map };
  }, [crumble, icy, materials.crumbleBase]);
  useEffect(
    () => () => {
      ownTop?.mat.dispose();
      ownTop?.map?.dispose();
    },
    [ownTop],
  );
  // The crack material is hook-created once per identity change; relay it through a
  // ref (assigned in an effect) so per-frame emissive tweaks never touch a hook value.
  const ownTopMatRef = useRef<MeshStandardMaterialType | null>(null);
  useEffect(() => {
    ownTopMatRef.current = ownTop?.mat ?? null;
    return () => {
      ownTopMatRef.current = null;
    };
  }, [ownTop]);
  const byFace = useMemo(() => {
    if (!ownTop) return materials.byFace;
    const faces: [Material, Material, Material, Material, Material, Material] = [
      materials.side,
      materials.side,
      ownTop.mat,
      materials.bottom,
      materials.side,
      materials.side,
    ];
    return faces;
  }, [ownTop, materials]);

  const innerRef = useRef<Group>(null);
  const phase = useMemo(() => (box.id * 1.61803398875) % (Math.PI * 2), [box.id]);

  useFrame((state) => {
    if (!crumble && !warn) return;
    const t = state.clock.elapsedTime;
    const group = innerRef.current;
    if (group) {
      if (warn) {
        group.position.set(
          box.center[0] + Math.sin(t * 34 + phase) * SHAKE_AMP,
          box.center[1],
          box.center[2] + Math.cos(t * 27 + phase) * SHAKE_AMP,
        );
      } else if (
        group.position.x !== box.center[0] ||
        group.position.y !== box.center[1] ||
        group.position.z !== box.center[2]
      ) {
        group.position.set(box.center[0], box.center[1], box.center[2]);
      }
    }
    const ownTopMat = ownTopMatRef.current;
    if (ownTopMat) {
      ownTopMat.emissiveIntensity = warn
        ? 1.05 + Math.sin(t * 9 + phase) * 0.9
        : CRUMBLE_IDLE_EMISSIVE;
    }
  });

  return (
    <>
      <group ref={innerRef} position={box.center} visible={!gone}>
        <mesh geometry={geometry} material={byFace} />
        {box.edgeBars.map((bar, index) => (
          <EdgeTrimBar key={index} bar={bar} material={materials.edgeTrim} />
        ))}
      </group>
      {/* Faint ghost outline where a crumbled platform used to be. */}
      <mesh geometry={geometry} position={box.center} visible={gone}>
        <meshBasicMaterial
          color={materials.ghostColor}
          wireframe
          transparent
          opacity={GONE_GHOST_OPACITY}
          depthWrite={false}
        />
      </mesh>
    </>
  );
}

function EdgeTrimBar(props: { bar: EdgeBarData; material: Material }): JSX.Element {

  );
