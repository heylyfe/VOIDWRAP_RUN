#!/usr/bin/env node
// Validates design content (enemies, upgrades, curses, classes, shop, events, altars, screen grades) against the
// effect vocabulary and the playability limits. Used by `just verify` on design/content.json and by OVERSEER on
// subordinate drafts.
//   node tools/validate-content.mjs [path=design/content.json]
import { readFileSync } from "node:fs";

const path = process.argv[2] ?? "design/content.json";
const c = JSON.parse(readFileSync(path, "utf8"));
const errors = [];
const err = (m) => errors.push(m);

const MUL = ["run_speed_mul", "jump_mul", "gravity_mul", "dash_speed_mul", "dash_cooldown_mul", "gift_radius_mul",
  "enemy_speed_mul", "enemy_aggro_mul", "enemy_grace_mul", "enemy_size_mul", "fog_density_mul", "beacon_radius_mul"];
const ADD = ["air_jumps_add", "extra_lives_add", "extra_enemies_add"];
// Movement floors/ceilings that keep every level completable (1.5-unit steps, 4-unit gaps).
const PLAYABLE = { run_speed_mul: [0.8, 2], jump_mul: [0.85, 2], gravity_mul: [0.4, 1.25] };
const THEMES = ["ashfall", "fluxbloom", "slagstone", "tidegrave"];
const PLAYER_SPEED = 9;
const ID = /^[a-z][a-z0-9_]{1,31}$/;
const ids = new Set();

function checkId(where, id) {
  if (!ID.test(id ?? "")) err(`${where}: bad id ${JSON.stringify(id)}`);
  if (ids.has(id)) err(`${where}: duplicate id ${id}`);
  ids.add(id);
}
function checkText(where, field, v, max) {
  if (typeof v !== "string" || v.trim() === "") err(`${where}.${field}: missing`);
  else if (v.length > max) err(`${where}.${field}: ${v.length} > ${max} chars`);
}
function checkMods(where, mods, greater) {
  if (!mods || typeof mods !== "object" || Object.keys(mods).length === 0) return err(`${where}: no mods`);
  const [lo, hi] = greater ? [0.4, 2.0] : [0.5, 1.6];
  for (const [k, v] of Object.entries(mods)) {
    if (MUL.includes(k)) {
      if (typeof v !== "number" || v < lo || v > hi) err(`${where}.${k}=${v} outside [${lo}, ${hi}]`);
      const p = PLAYABLE[k];
      if (p && (v < p[0] || v > p[1])) err(`${where}.${k}=${v} breaks playability [${p[0]}, ${p[1]}]`);
    } else if (ADD.includes(k)) {
      if (!Number.isInteger(v) || v < 0 || v > 3) err(`${where}.${k}=${v} must be an integer 0..3`);
    } else err(`${where}: unknown mod key ${k}`);
  }
}

// --- enemies ---------------------------------------------------------------------------------------------------
// M3 roster. The array order IS the network kind index, so the ids are pinned in place. `params.behavior` repeats
// the id (one state machine per enemy); the M2 archetypes and their tuning fields (fire_interval_s,
// projectile_speed, reveal_radius, corrupt_interval_s) are gone - those numbers live in the sim now.
const ENEMY_IDS = ["pallbearer", "snapshrike", "moteling", "rift_warden", "gale_wight", "lantern_leech", "turret",
  "mimic", "corrupter"];
const ENEMY_PARAMS = ["speed", "aggro", "grace_s", "size", "behavior"];
const STATIONARY = ["turret", "mimic"]; // the only two that never roam

if (!Array.isArray(c.enemies) || c.enemies.length !== ENEMY_IDS.length) {
  err(`enemies: need exactly ${ENEMY_IDS.length} (have ${Array.isArray(c.enemies) ? c.enemies.length : 0})`);
}
for (const [i, e] of (c.enemies ?? []).entries()) {
  const w = `enemies[${i}]`;
  checkId(w, e.id);
  if (e.id !== ENEMY_IDS[i]) err(`${w}.id=${JSON.stringify(e.id)} must be ${ENEMY_IDS[i]} (array order is the network kind index)`);
  checkText(w, "name", e.name, 24);
  checkText(w, "flavor", e.flavor, 70);
  checkText(w, "behavior", e.behavior, 110);
  if (!Number.isInteger(e.threat) || e.threat < 1 || e.threat > 5) err(`${w}.threat`);
  const p = e.params ?? {};
  for (const k of Object.keys(p)) if (!ENEMY_PARAMS.includes(k)) err(`${w}.params.${k}: not an enemy param`);
  if (p.behavior !== e.id) err(`${w}.params.behavior=${JSON.stringify(p.behavior)} must equal the enemy id ${JSON.stringify(e.id)}`);
  if (!(typeof p.speed === "number" && p.speed >= 0 && p.speed <= 15)) err(`${w}.params.speed=${p.speed} must be in [0, 15]`);
  else if (p.speed === 0 && !STATIONARY.includes(e.id)) err(`${w}.params.speed=0 is only for ${STATIONARY.join(" / ")}`);
  if (!(typeof p.aggro === "number" && p.aggro >= 0 && p.aggro <= 40)) err(`${w}.params.aggro=${p.aggro} must be in [0, 40]`);
  if (p.grace_s !== 4.0) err(`${w}.params.grace_s=${p.grace_s} must be 4.0 (nothing damages anyone in the first 4 s)`);
  if (!(typeof p.size === "number" && p.size >= 0.3 && p.size <= 1.6)) err(`${w}.params.size=${p.size} must be in [0.3, 1.6]`);
}

// --- upgrades and curses ---------------------------------------------------------------------------------------
// Every mod key a shop item may use has to already exist on an upgrade or a curse.
const usedModKeys = new Set();
for (const [kind, list, min] of [["upgrades", c.upgrades, 8], ["curses", c.curses, 8]]) {
  if (!Array.isArray(list) || list.length < min) err(`need >= ${min} ${kind}`);
  for (const [i, k] of (list ?? []).entries()) {
    const w = `${kind}[${i}]`;
    checkId(w, k.id);
    checkText(w, "name", k.name, 28);
    checkText(w, "flavor", k.flavor, 70);
    checkText(w, "effect", k.effect, 60);
    if (!["common", "rare"].includes(k.rarity)) err(`${w}.rarity`);
    if (kind === "curses" && (!Number.isInteger(k.threat) || k.threat < 1 || k.threat > 5)) err(`${w}.threat`);
    checkMods(w, k.mods, kind === "curses" && k.greater === true);
    for (const key of Object.keys(k.mods ?? {})) usedModKeys.add(key);
  }
}
const greaterCount = (c.curses ?? []).filter((k) => k.greater === true).length;
if (greaterCount < 3) err(`need >= 3 greater curses (have ${greaterCount})`);
// Worst case: fastest enemy under the strongest enemy-speed curse must stay dash-escapable (dash 28 u/s).
const fastest = Math.max(...(c.enemies ?? []).map((e) => e.params?.speed ?? 0));
const worstMul = Math.max(1, ...(c.curses ?? []).map((k) => k.mods?.enemy_speed_mul ?? 1));
if (fastest * worstMul > 14) err(`fastest enemy x worst curse = ${(fastest * worstMul).toFixed(1)} u/s > 14 (cap)`);

// --- classes ---------------------------------------------------------------------------------------------------
// Order is fixed: the client indexes classes by position on the select screen. M3 replaced the Blinker with the
// Bulwark in slot 2; the old id must not survive anywhere in the file.
const CLASS_IDS = ["striker", "grappler", "bulwark", "glider"];
if (!Array.isArray(c.classes)) err("classes: missing");
else {

# Enemies - who hunts you

The lobby votes on which enemies join the run. Whatever gets voted in stays for the whole run: nothing new arrives
unannounced, and nothing you learn on the first level stops being true on the tenth.

Every enemy sleeps for the first 4 seconds of a level. Nothing can hurt you until it wakes.

Every deadly attack is announced first. A ring, a line, a wedge, a flare, a rising tone - the warning is drawn in
the world, it lasts between 0.6 and 2.5 seconds, and the same attack always uses the same shape. Learn the shapes
and the roster turns into a timetable.

Two of the nine cannot kill you at all. They shove you, slow you and take your abilities away; the void does the
rest. Falling is still falling.

Two attacks are heavy - the Gale Wight's volley and the Moteling's inhale. Only one heavy wind-up runs at a time,
so while one of them is charging, the other is waiting its turn.

## The Pallbearer - threat 3/5

**What it does.** It follows the trail you left a few seconds ago, not where you are now. When it reaches where you
actually stand it stops, rears back, and drives straight through that spot.

**The warning (what you see before it hurts).** It kneels for 1.2 seconds: the lantern on its prow flares white, a
ring appears under your feet, and a straight line is drawn from the coffin to you. The line is where it will go.

**How to survive.** Keep moving forward. It punishes backtracking, so a route that doubles back over old ground
feeds it. When it kneels, dash sideways off the line - it commits to the point it locked, not to you.

**Watch out with friends.** It picks one player and stays with them until the lunge is over. That player is the
guide whether they want to be or not; the rest of the team loots while it is busy.

**Lethal or not.** Lethal. Only the lunge kills. Bumping into it while it hunts just shoves you.

## Snapshrike - threat 4/5

**What it does.** It circles you at 4 to 7 metres, picks a line, then dashes down that line at 13.5 metres per
second for up to 11 metres. It cannot turn during the dash.

**The warning (what you see before it hurts).** A full-width red wedge along the dash line, 0.75 seconds long. It
tracks you for the first 0.4 seconds, then the line freezes for the last 0.35. When the line stops moving, that is
your cue.

**How to survive.** Sidestep the moment the line freezes. Put a pillar between you and it and the dash ends early,
stunning it for 0.6 seconds. At close range you can jump over it.

**Watch out with friends.** It goes for the nearest player, so aggro passes around the team. Call your dash. It
cannot hit anyone but its target.

**Lethal or not.** Lethal, during the dash only. Touching it while it circles shoves you.

## Moteling - threat 2/5

**What it does.** It drifts, then circles whichever player has been moving fastest, and every 6 seconds it begins
to inhale. When the inhale ends it shrieks at that one player.

**The warning (what you see before it hurts).** 2 seconds of inhale: the world goes quiet, an indigo ring closes
around the marked player, and the screen darkens at the edges. The ring tightens as the timer runs out.

**How to survive.** Stand still. If your horizontal speed is under 1 metre per second when the ring snaps, nothing
happens. Then move again.

**Watch out with friends.** Only the marked player has to freeze; everyone else keeps looting. It re-marks each
cycle, so the job moves around.

**Lethal or not.** Lethal, but only to the marked player, in a 4 metre burst.

## The Rift Warden - threat 2/5

**What it does.** A doorframe that drifts through walls toward the nearest player. Inside 4 metres it ignites, then
throws everything away from itself.

**The warning (what you see before it hurts).** 1.1 seconds of ignition: the storm-runes carved on its slabs blaze
white and a low groan builds.

**How to survive.** It does no damage. The fall does. Fight it on wide ground, step behind cover when the runes
light, and never let it catch you on a narrow bridge.

**Watch out with friends.** It shoves enemies too, so a well-timed pulse peels a Snapshrike off a teammate. One
player can leash it while the others loot.

**Lethal or not.** Not lethal. Displacement only: 8 metres per second away from it, plus a 0.9 second stagger.

## Gale Wight - threat 4/5

**What it does.** It cruises above the map and drops shells on marked spots. Two shells solo, up to three with a
full team, each aimed at a random living player.

**The warning (what you see before it hurts).** A circle appears on the floor and follows you for 0.9 seconds,
then locks for 0.6, then a streak falls for 1.1 with a whistle. That is 2.6 seconds from circle to impact.

**How to survive.** Leave the circle - 2.2 metres is enough. Shells break on the first platform top they hit, so
standing under a bridge or an overhang cancels them outright.

**Watch out with friends.** Circles spread across the team; call yours out. One player baiting draws shells off
whoever is running the gold.

**Lethal or not.** Lethal. The 2.1 metre blast kills; from 2.1 to 4.2 metres it knocks you back 7 metres per
second.

## Lantern Leech - threat 3/5

**What it does.** It drifts harmlessly until someone picks up a gift within 30 metres. Then it locks that spot,
glides to it at speed, and stings anything still standing there.

**The warning (what you see before it hurts).** 0.8 seconds before it arrives the destination flares and its path
is drawn across the air; then it rears back for 0.6 seconds before the sting.

**How to survive.** Step 2 metres away and let it gorge on empty air. It sits for 2 seconds, stings every second
while you stay close, then leaves for at least 3.

**Watch out with friends.** One player baits, the rest harvest. It only answers the newest pickup, so a second gift
does not stack a second leech.

**Lethal or not.** Lethal, a 1.5 metre sting, only while it is feeding.

## Bolt Sentry - threat 3/5

**What it does.** It is bolted to a platform and never moves. It tracks the nearest player it can see and fires a
bolt along a predicted line.

**The warning (what you see before it hurts).** 0.95 seconds of charge: a thin aiming line from the muzzle to
where it thinks you will be, a rising whine, and the lens flaring brighter as it fills.

**How to survive.** Break line of sight during the charge, or zigzag - the prediction assumes you run straight. It
fires at the last predicted point even if you vanish. Then it vents for 1.4 seconds and cannot shoot: that is your
window.

**Watch out with friends.** It only sees one of you. Two players peeking in turn keep it spinning.

**Lethal or not.** Lethal, a 0.35 metre bolt. Jump it - it flies flat at muzzle height.

## False Gift - threat 3/5

**What it does.** It sits as a gift. When a player comes within 4 metres it opens and lunges 5.1 metres at them,
then goes inert.

**The warning (what you see before it hurts).** Every 3 seconds it shimmers magenta for half a second and gives a
faint tearing sound. That is the only tell. Then 0.9 seconds of hissing and splitting before the lunge.

**How to survive.** Dash through suspicious gifts. The dash outruns the lunge and reveals it safely. Never walk
into a cluster of gifts at full speed.

**Watch out with friends.** Split the work: one grabs, one watches for the shimmer.

**Lethal or not.** Lethal, during the lunge.

## Crumble Wisp - threat 2/5

**What it does.** It drifts to the platform nearest its target and frets it. A fretted tile stays solid, but it
slows you 40% and locks your abilities for 8 seconds.

**The warning (what you see before it hurts).** A 2 metre aura around the wisp, then 1.5 seconds of black-green
veins crawling outward across the floor with a rasp.

**How to survive.** It never damages. Do not anchor on a fretted tile; cross fast and loot somewhere else.

**Watch out with friends.** It can only fret one spot at a time. Spread out and call the tiles.

**Lethal or not.** Not lethal. Affliction only.
