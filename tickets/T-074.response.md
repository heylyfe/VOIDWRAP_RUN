//! Enemy brains for the registry entries added in T-072.
//!
//! T-074 implements the first five: Rewind, Warden, Peal, Kite and Static. Each
//! enemy lives in its own module and exposes a phase enum the client can draw, a
//! brain struct holding the phase and its memory, a `step` function that advances
//! one 20 Hz tick, and a `relation` function carrying the registry's class
//! interactions.
//!
//! Everything here is deterministic: only `+ - * /`, `sqrt`, `abs`, `min`, `max`
//! and `clamp` touch floats, and no wall clock is read. The tick index comes from
//! [`StepContext::tick`].

pub mod peal;
pub mod rewind;
pub mod r#static;

/// Seconds per enemy brain tick (20 Hz, matching the existing enemy timers).
pub const ENEMY_DT: f32 = 0.05;

/// The four shipped classes, as far as enemy class interactions care.
///
/// The server maps its own class id onto this enum; the registry pins the set to
/// exactly these four (`design/ENEMY_REGISTRY.md` §6).
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum EnemyClass {
    /// Striker.
    Striker,
    /// Grappler.
    Grappler,
    /// Bulwark.
    Bulwark,
    /// Glider.
    Glider,
}

/// How an enemy treats a class, straight from `design/ENEMY_REGISTRY.md` §6.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ClassRelation {
    /// No counter, no immunity, no exclusion.
    Neutral,
    /// The class's base ability fully answers the signature threat.
    Counters,
    /// The signature effect is never applied to that class.
    Immune,
    /// The enemy never selects that class.
    Ignores,
}

/// One player as the enemy brains see them for a single tick.
#[derive(Debug, Clone, PartialEq)]
pub struct PlayerSnapshot {
    /// Player id, echoed back in [`Hit::player_id`].
    pub id: u32,
    /// Feet position, the same point `ControllerState.pos` holds.
    pub feet: Vec3,
    /// The player's class.
    pub class: EnemyClass,
    /// Persistent debuff state (OVERTUNED and friends).
    pub status: PlayerStatus,
    /// True while the player is inside a dash.
    pub dashing: bool,
}

/// Everything a brain may read about the world for one tick.
#[derive(Debug, Clone, Copy, Default)]
pub struct StepContext<'a> {
    /// Tick index since the level started, at 20 Hz.
    pub tick: u32,
    /// Players alive this tick.
    pub players: &'a [PlayerSnapshot],
    /// Gifts still on the floor this tick.
    pub gifts: &'a [Gift],
    /// True during the collapse phase.
    pub collapse: bool,
    /// True when `coil` is a permanent in the run (`peal_lethal_when_overtuned`).
    pub coil_permanent: bool,
}

/// A circular hit volume a client can draw as a floor decal.
#[derive(Debug, Clone, Copy, PartialEq)]
pub struct Ring {
    /// Centre of the ring, in world units.
    pub centre: Vec3,
    /// Radius of the ring, in world units.
    pub radius: f32,
}

/// What a brain wants applied to a player it caught.

/// One player caught by an enemy this tick.

/// What the server should apply after one brain tick.
#[derive(Debug, Clone, PartialEq)]

/// Map a registry id (`design/content.json` → `enemies[].id`) to its `BEHAVIOR_*` constant.
///
/// Returns `None` for ids this build does not implement; callers keep their
/// existing `BEHAVIOR_PALLBEARER` fallback for those.
pub fn behavior_for_id(id: &str) -> Option<u8> {
    let behavior = match id {
        "pallbearer" => crate::rules::BEHAVIOR_PALLBEARER,
        "snapshrike" => crate::rules::BEHAVIOR_SNAPSHRIKE,
        "moteling" => crate::rules::BEHAVIOR_MOTELING,
        "rift_warden" => crate::rules::BEHAVIOR_RIFT_WARDEN,
        "gale_wight" => crate::rules::BEHAVIOR_GALE_WIGHT,
        "lantern_leech" => crate::rules::BEHAVIOR_LANTERN_LEECH,
        "turret" => crate::rules::BEHAVIOR_TURRET,
        "mimic" => crate::rules::BEHAVIOR_MIMIC,
        "corrupter" => crate::rules::BEHAVIOR_CORRUPTER,
        "rewind" => crate::rules::BEHAVIOR_REWIND,
        "warden" => crate::rules::BEHAVIOR_WARDEN,
        "peal" => crate::rules::BEHAVIOR_PEAL,
        "kite" => crate::rules::BEHAVIOR_KITE,
        "static" => crate::rules::BEHAVIOR_STATIC,
        _ => return None,
    };
    Some(behavior)
}

/// The step from `from` toward `to` of at most `max_step`, returned as a delta.
pub(crate) fn step_toward(from: &Vec3, to: &Vec3, max_step: f32) -> Vec3 {
    let dx = to.x - from.x;
    let dy = to.y - from.y;
    let dz = to.z - from.z;
    let d = (dx * dx + dy * dy + dz * dz).sqrt();
    if d <= max_step || d <= 1e-9 {
        return Vec3::new(dx, dy, dz);
    }
    let scale = max_step / d;
    Vec3::new(dx * scale, dy * scale, dz * scale)
}

/// The player nearest to `pos`, skipping one class and one id when asked.
pub(crate) fn nearest_player_excluding<'a>(
    players: &'a [PlayerSnapshot],
    pos: &Vec3,
    skip_class: Option<EnemyClass>,
    skip_id: Option<u32>,
) -> Option<&'a PlayerSnapshot> {
    let mut best: Option<&'a PlayerSnapshot> = None;
    let mut best_d = f32::MAX;
    for p in players {
        if Some(p.class) == skip_class || Some(p.id) == skip_id {
            continue;
        }
        let d = distance(pos, &player_centre(&p.feet));
        if d < best_d {
            best_d = d;
            best = Some(p);
        }
    }
    best
}

/// The player nearest to `pos`, with no class or id filter.
pub(crate) fn nearest_player<'a>(
    players: &'a [PlayerSnapshot],
    pos: &Vec3,
) -> Option<&'a PlayerSnapshot> {
    nearest_player_excluding(players, pos, None, None)
}

/// The player farthest from their nearest team-mate; a lone player is their own group.
pub(crate) fn most_isolated_player(players: &[PlayerSnapshot]) -> Option<&PlayerSnapshot> {
    let mut best: Option<&PlayerSnapshot> = None;
    let mut best_d = -1.0_f32;
    for p in players {
        let mut nearest = 0.0_f32;
        let mut found = false;
        for q in players {
            if q.id == p.id {
                continue;
            }
            let d = distance(&p.feet, &q.feet);
            if !found || d < nearest {
                nearest = d;
                found = true;
            }
        }
        if nearest > best_d {
            best_d = nearest;
            best = Some(p);
        }
    }
    best
}

//! Rewind: replays the nearest player's path from 1.5 s ago.
//!
//! Tier-2 telegraph: the body trembles in place for 1.5 s with a floor mark on
//! the exact hit volume, then it starts moving. Touching it is lethal. A Striker
//! dash breaks the drag; a Grappler is immune.

use protocol::Vec3;

use crate::rules::{distance, player_centre, EnemyParams};

use super::{
    nearest_player, step_toward, ClassRelation, EnemyAction, EnemyClass, Hit, HitKind, Ring,
    StepContext, ENEMY_DT,
};

/// Ticks the tremble telegraph lasts (1.5 s at 20 Hz; tier 2 needs 1.0-2.0 s).
pub const REWIND_TREMBLE_TICKS: u32 = 30;
/// Ticks the replay chase lasts before the enemy fades (4.0 s).
pub const REWIND_REPLAY_TICKS: u32 = 80;
/// Ticks the harmless fade lasts (1.0 s).
pub const REWIND_FADE_TICKS: u32 = 20;
/// Player positions the brain remembers (1.5 s of history).
pub const REWIND_TRAIL_TICKS: usize = 30;
/// Replay speed, units/s.
pub const REWIND_SPEED: f32 = 7.0;
/// Fade drift speed, units/s.
pub const REWIND_FADE_SPEED: f32 = 2.0;
/// Tremble amplitude, units.
pub const REWIND_TREMBLE_AMPLITUDE: f32 = 0.25;
/// Ticks per tremble half-cycle (0.15 s).
pub const REWIND_TREMBLE_PERIOD: u32 = 3;
/// Radius around the enemy centre that kills on contact, units.
pub const REWIND_TOUCH_RADIUS: f32 = 0.9;

/// Rewind phases.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum RewindPhase {
    /// Shivers in place; the tier-2 floor mark sits on the body.
    Tremble,
    /// Follows the target's recorded path 1.5 s behind; lethal on contact.
    Replay,
    /// Harmless drift back toward the last recorded point.
    Fade,
}

/// Rewind brain state.
#[derive(Debug, Clone, PartialEq)]
pub struct RewindBrain {
    /// Current phase.
    pub phase: RewindPhase,
    /// Phase countdown in ticks.
    pub timer: u32,
    /// The player whose path is being replayed.
    pub target_id: Option<u32>,
    /// Recorded target feet positions, oldest first.
    pub trail: Vec<Vec3>,
    /// Where the tier-2 floor mark is drawn.
    pub mark: Vec3,
}

impl Default for RewindBrain {
    fn default() -> Self {
        Self {
            phase: RewindPhase::Tremble,
            timer: REWIND_TREMBLE_TICKS,
            target_id: None,
            trail: Vec::new(),
            mark: Vec3::new(0.0, 0.0, 0.0),
        }
    }
}

impl RewindBrain {
    /// The tier-2 floor mark: the exact hit volume, drawn for the whole tremble.
    pub fn telegraph(&self) -> Option<Ring> {
        match self.phase {
            RewindPhase::Tremble => Some(Ring {
                centre: self.mark,
                radius: REWIND_TOUCH_RADIUS,
            }),
            RewindPhase::Replay | RewindPhase::Fade => None,
        }
    }
}

/// Registry class interaction for Rewind: Striker counters, Grappler immune.
pub fn relation(class: EnemyClass) -> ClassRelation {
    match class {
        EnemyClass::Striker => ClassRelation::Counters,
        EnemyClass::Grappler => ClassRelation::Immune,
        EnemyClass::Bulwark | EnemyClass::Glider => ClassRelation::Neutral,
    }
}

/// Advance Rewind by one 20 Hz tick.
///
/// Contact is tested against `pos` as it was at the start of the tick; the server
/// applies [`EnemyAction::motion`] afterwards.
pub fn step(
    brain: &mut RewindBrain,
    pos: &Vec3,
    params: &EnemyParams,
    ctx: &StepContext<'_>,
) -> EnemyAction {
    let mut action = EnemyAction::idle();

    let target = nearest_player(ctx.players, pos);
    brain.target_id = target.map(|p| p.id);
    if let Some(p) = target {
        brain.trail.push(p.feet);
        if brain.trail.len() > REWIND_TRAIL_TICKS {
            brain.trail.remove(0);
        }
    }

    match brain.phase {
        RewindPhase::Tremble => {
            let sign = if (ctx.tick / REWIND_TREMBLE_PERIOD) % 2 == 0 {
                1.0
            } else {
                -1.0
            };
            action.motion = Vec3::new(
                sign * REWIND_TREMBLE_AMPLITUDE,
                0.0,
                sign * REWIND_TREMBLE_AMPLITUDE,
            );
            brain.mark = *pos;
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                brain.phase = RewindPhase::Replay;
                brain.timer = REWIND_REPLAY_TICKS;
            }
        }
        RewindPhase::Replay => {
            if let Some(goal) = brain.trail.first().copied() {
                brain.mark = goal;
                action.motion =
                    step_toward(pos, &goal, REWIND_SPEED * params.speed_mul * ENEMY_DT);
            }
            for p in ctx.players {
                if relation(p.class) == ClassRelation::Immune {
                    continue;
                }
                if p.class == EnemyClass::Striker && p.dashing {
                    continue;
                }
                if distance(&player_centre(&p.feet), pos) <= REWIND_TOUCH_RADIUS + params.size {
                    action.hits.push(Hit {
                        player_id: p.id,
                        kind: HitKind::Lethal,
                    });
                }
            }
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                brain.phase = RewindPhase::Fade;
                brain.timer = REWIND_FADE_TICKS;
            }
        }
        RewindPhase::Fade => {
            action.motion = step_toward(pos, &brain.mark, REWIND_FADE_SPEED * ENEMY_DT);
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                brain.phase = RewindPhase::Tremble;
                brain.timer = REWIND_TREMBLE_TICKS;
            }
        }
    }

    action
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::rules::enemies::PlayerSnapshot;
    use crate::rules::PlayerStatus;

    fn params() -> EnemyParams {
        EnemyParams {
            behavior: crate::rules::BEHAVIOR_REWIND,
            speed_mul: 1.0,
            size: 1.0,
            grace_ticks: 0,
        }
    }

    fn player(id: u32, x: f32, z: f32, class: EnemyClass) -> PlayerSnapshot {
        PlayerSnapshot {
            id,
            feet: Vec3::new(x, 0.0, z),
            class,
            status: PlayerStatus::default(),
            dashing: false,
        }
    }

    #[test]
    fn tremble_lasts_the_tier_two_minimum() {
        assert!(
            (20..=40).contains(&REWIND_TREMBLE_TICKS),
            "tier 2 needs 1.0-2.0 s of decal"
        );
    }

    #[test]
    fn trembles_before_it_moves() {
        let players = [player(1, 10.0, 0.0, EnemyClass::Bulwark)];
        let mut brain = RewindBrain::default();
        let mut pos = Vec3::new(0.0, 0.0, 0.0);
        for tick in 0..REWIND_TREMBLE_TICKS - 1 {
            let ctx = StepContext {
                tick,
                players: &players,
                ..StepContext::default()
            };
            let action = step(&mut brain, &pos, &params(), &ctx);
            assert_eq!(brain.phase, RewindPhase::Tremble);
            assert!(brain.telegraph().is_some(), "the floor mark is up");
            assert!(action.motion.x.abs() <= REWIND_TREMBLE_AMPLITUDE + 1e-4);
            pos.x += action.motion.x;
            pos.z += action.motion.z;
        }
        let ctx = StepContext {
            tick: REWIND_TREMBLE_TICKS,
            players: &players,
            ..StepContext::default()
        };
        step(&mut brain, &pos, &params(), &ctx);
        assert_eq!(brain.phase, RewindPhase::Replay);
        assert!(brain.telegraph().is_none(), "the mark is removed on resolve");
    }

    #[test]
    fn replays_the_delayed_trail() {
        let mut brain = RewindBrain::default();
        let mut pos = Vec3::new(0.0, 0.0, 0.0);
        let mut tick = 0_u32;
        while brain.phase == RewindPhase::Tremble {
            let players = [player(1, 20.0 + tick as f32 * 0.5, 0.0, EnemyClass::Bulwark)];
            let ctx = StepContext {
                tick,
                players: &players,
                ..StepContext::default()
            };
            let action = step(&mut brain, &pos, &params(), &ctx);
            pos.x += action.motion.x;
            pos.z += action.motion.z;
            tick += 1;
        }
        assert_eq!(brain.phase, RewindPhase::Replay);
        assert_eq!(brain.trail.len(), REWIND_TRAIL_TICKS);

        let players = [player(1, 100.0, 0.0, EnemyClass::Bulwark)];
        let ctx = StepContext {
            tick,
            players: &players,
            ..StepContext::default()
        };
        let action = step(&mut brain, &pos, &params(), &ctx);
        assert!(action.motion.x > 0.0, "replay walks the recorded path");
        assert!(brain.mark.x < 100.0, "the mark is the delayed point");
    }

    #[test]
    fn grappler_is_immune_and_striker_dash_breaks_the_drag() {
        let pos = Vec3::new(0.0, 0.0, 0.0);

        let mut brain = RewindBrain::default();
        brain.phase = RewindPhase::Replay;
        brain.timer = REWIND_REPLAY_TICKS;
        let grappler = [player(1, 0.0, 0.0, EnemyClass::Grappler)];
        let ctx = StepContext {
            tick: 0,
            players: &grappler,
            ..StepContext::default()
        };
        let action = step(&mut brain, &pos, &params(), &ctx);
        assert!(action.hits.is_empty(), "the grapple anchors");

        let striker = [player(2, 0.0, 0.0, EnemyClass::Striker)];

            tick: 0,
            players: &striker,
            ..StepContext::default()
        };
        let action = step(&mut brain, &pos, &params(), &ctx);
        assert_eq!(action.hits.len(), 1);
        assert_eq!(action.hits[0].kind, HitKind::Lethal);

        let mut dashing = player(3, 0.0, 0.0, EnemyClass::Striker);
        dashing.dashing = true;
        let striker_dash = [dashing];
        let ctx = StepContext {
            tick: 0,
            players: &striker_dash,
            ..StepContext::default()
        };
        let action = step(&mut brain, &pos, &params(), &ctx);
        assert!(action.hits.is_empty(), "a striker dash breaks the drag");
    }
}

//! Warden: guards the densest gift cluster and lunges at whoever takes a gift.
//!
//! Tier-2 telegraph: a ring around the guarded gift for 1.5 s before its reach
//! extends to the ring edge. Standing inside the ring when it closes is lethal.
//! A Grappler pulls the gift out of the ring; a Bulwark is immune.

use protocol::{Gift, Vec3};

use crate::rules::{distance, EnemyParams};

use super::{
    nearest_player, step_toward, ClassRelation, EnemyAction, EnemyClass, Hit, HitKind, Ring,
    StepContext, ENEMY_DT,

/// Ticks the guard ring winds up (1.5 s at 20 Hz; tier 2 needs 1.0-2.0 s).

/// Ticks the lunge lasts (0.5 s).
pub const WARDEN_LUNGE_TICKS: u32 = 10;
/// Ticks the recovery lasts (1.0 s).
pub const WARDEN_RECOVER_TICKS: u32 = 20;
/// Patrol speed, units/s.
pub const WARDEN_PATROL_SPEED: f32 = 3.5;
/// Frenzied patrol speed, units/s.
pub const WARDEN_FRENZY_SPEED: f32 = 5.5;

pub const WARDEN_LUNGE_SPEED: f32 = 14.0;
/// Radius used to count gifts into one cluster, units.
pub const WARDEN_CLUSTER_RADIUS: f32 = 6.0;
/// Radius of the guard ring, units.
pub const WARDEN_RING_RADIUS: f32 = 4.0;

/// Warden phases.

/// The gift with the most neighbours inside [`WARDEN_CLUSTER_RADIUS`]; ties keep content order.
fn densest_gift(gifts: &[Gift]) -> Option<Vec3> {
    let mut best: Option<Vec3> = None;
    let mut best_count = 0_usize;
    for g in gifts {
        let mut count = 0_usize;
        for other in gifts {
            if distance(&g.pos, &other.pos) <= WARDEN_CLUSTER_RADIUS {
                count += 1;
            }
        }
        if count > best_count {
            best_count = count;
            best = Some(g.pos);
        }
    }
    best
}

/// Advance Warden by one 20 Hz tick.
pub fn step(
    brain: &mut WardenBrain,
    pos: &Vec3,
    params: &EnemyParams,
    ctx: &StepContext<'_>,
) -> EnemyAction {
    let mut action = EnemyAction::idle();
    let speed = if ctx.collapse {
        WARDEN_FRENZY_SPEED
    } else {
        WARDEN_PATROL_SPEED
    } * params.speed_mul;

    match brain.phase {
        WardenPhase::Patrol => {
            if brain.guarded.is_none() {
                brain.guarded = densest_gift(ctx.gifts);
            }
            if let Some(guarded) = brain.guarded {
                let still_there = ctx
                    .gifts
                    .iter()
                    .any(|g| distance(&g.pos, &guarded) <= 0.01);
                if !still_there {
                    brain.ring_centre = guarded;
                    brain.target_id = nearest_player(ctx.players, &guarded).map(|p| p.id);
                    brain.phase = WardenPhase::Wind;
                    brain.timer = if ctx.collapse {
                        WARDEN_FRENZY_WIND_TICKS
                    } else {
                        WARDEN_WIND_TICKS
                    };
                    return action;
                }
                action.motion = step_toward(pos, &guarded, speed * ENEMY_DT);
            }
        }
        WardenPhase::Wind => {
            let pulled = ctx.players.iter().any(|p| {
                p.class == EnemyClass::Grappler
                    && distance(&p.feet, &brain.ring_centre) <= WARDEN_GRAPPLE_REACH
            });
            if pulled {
                brain.phase = WardenPhase::Patrol;
                brain.guarded = None;
                brain.timer = 0;
                return action;
            }
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                for p in ctx.players {
                    if relation(p.class) == ClassRelation::Immune {
                        continue;
                    }
                    if distance(&p.feet, &brain.ring_centre) <= WARDEN_RING_RADIUS {
                        action.hits.push(Hit {
                            player_id: p.id,
                            kind: HitKind::Lethal,
                        });
                    }
                }
                brain.locked = match brain.target_id {
                    Some(id) => ctx
                        .players
                        .iter()
                        .find(|p| p.id == id)
                        .map_or(brain.ring_centre, |p| p.feet),
                    None => brain.ring_centre,
                };
                brain.phase = WardenPhase::Lunge;
                brain.timer = WARDEN_LUNGE_TICKS;
            }
        }
        WardenPhase::Lunge => {
            action.motion = step_toward(
                pos,
                &brain.locked,
                WARDEN_LUNGE_SPEED * params.speed_mul * ENEMY_DT,
            );
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                brain.phase = WardenPhase::Recover;
                brain.timer = WARDEN_RECOVER_TICKS;
            }
        }
        WardenPhase::Recover => {
            action.motion = step_toward(pos, &brain.ring_centre, speed * ENEMY_DT);
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                brain.phase = WardenPhase::Patrol;
                brain.guarded = None;
            }
        }
    }

    action
}

//! Peal: teleports next to the closest player every 6 s and rings three times.
//!
//! Tier-1 telegraph: the bell body opens for 0.6 s before each ring. The ring is
//! harmless unless the target already carries OVERTUNED, in which case it kills.
//! With `coil` permanent the final ring is lethal outright. A Bulwark is immune;
//! a Glider above the ring crosses over it.

use protocol::Vec3;

use crate::rules::{distance, flat_direction, EnemyParams};

use super::{
    nearest_player, ClassRelation, EnemyAction, EnemyClass, Hit, HitKind, Ring, StepContext,
};

/// Ticks between Peal cycles (6.0 s at 20 Hz).
pub const PEAL_CYCLE_TICKS: u32 = 120;
/// Ticks the bell body stays open before a ring (0.6 s; tier 1 needs 0.35-0.95 s).
pub const PEAL_CHIME_TICKS: u32 = 12;
/// Ticks of rest after the final ring (1.0 s).
pub const PEAL_REST_TICKS: u32 = 20;
/// Radii of the three rings, units; the third is the final one.
pub const PEAL_RING_RADII: [f32; 3] = [3.0, 4.0, 5.0];
/// Horizontal distance from the target at which Peal reappears, units.
pub const PEAL_TELEPORT_DISTANCE: f32 = 3.0;
/// A player this far above or below the ring centre is outside it, units.
pub const PEAL_RING_HEIGHT: f32 = 2.5;
/// Ticks of OVERTUNED one ring applies (6.0 s).
pub const PEAL_OVERTUNED_TICKS: u32 = 120;

/// Peal phases.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum PealPhase {
    /// Waiting out the 6 s cycle.
    Idle,
    /// The bell body is open; the tier-1 chime telegraph.
    Chime,
    /// The ring resolved this tick.
    Ring,
    /// Harmless rest after the final ring.
    Rest,
}

/// Peal brain state.
#[derive(Debug, Clone, PartialEq)]
pub struct PealBrain {
    /// Current phase.
    pub phase: PealPhase,
    /// Phase countdown in ticks.
    pub timer: u32,
    /// Which of the three rings is being rung (0..=2).
    pub ring_index: usize,
    /// Centre of the ring being charged or resolved.
    pub centre: Vec3,
    /// The player Peal teleported next to.
    pub target_id: Option<u32>,
}

impl Default for PealBrain {
    fn default() -> Self {
        Self {
            phase: PealPhase::Idle,
            timer: PEAL_CYCLE_TICKS,
            ring_index: 0,
            centre: Vec3::new(0.0, 0.0, 0.0),
            target_id: None,
        }
    }
}

impl PealBrain {
    /// The tier-1 chime: the ring that will resolve, drawn while the bell body is open.
    pub fn telegraph(&self) -> Option<Ring> {
        match self.phase {
            PealPhase::Chime => Some(Ring {
                centre: self.centre,
                radius: PEAL_RING_RADII[self.ring_index],
            }),
            PealPhase::Idle | PealPhase::Ring | PealPhase::Rest => None,
        }
    }
}

/// Registry class interaction for Peal: Bulwark immune, Glider counters.
pub fn relation(class: EnemyClass) -> ClassRelation {
    match class {
        EnemyClass::Bulwark => ClassRelation::Immune,
        EnemyClass::Glider => ClassRelation::Counters,
        EnemyClass::Striker | EnemyClass::Grappler => ClassRelation::Neutral,
    }
}

/// Advance Peal by one 20 Hz tick.
pub fn step(
    brain: &mut PealBrain,
    pos: &Vec3,
    _params: &EnemyParams,
    ctx: &StepContext<'_>,
) -> EnemyAction {
    let mut action = EnemyAction::idle();

    match brain.phase {
        PealPhase::Idle => {
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                let target = nearest_player(ctx.players, pos);
                brain.target_id = target.map(|p| p.id);
                let centre = match target {
                    Some(p) => {
                        let away = flat_direction(&p.feet, pos);
                        Vec3::new(
                            p.feet.x + away.x * PEAL_TELEPORT_DISTANCE,
                            p.feet.y,
                            p.feet.z + away.z * PEAL_TELEPORT_DISTANCE,
                        )
                    }
                    None => *pos,
                };
                brain.centre = centre;
                brain.ring_index = 0;
                brain.phase = PealPhase::Chime;
                brain.timer = PEAL_CHIME_TICKS;
                action.teleport = Some(centre);
            }
        }
        PealPhase::Chime => {
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                let radius = PEAL_RING_RADII[brain.ring_index];
                let final_ring = brain.ring_index + 1 == PEAL_RING_RADII.len();
                for p in ctx.players {
                    if relation(p.class) == ClassRelation::Immune {
                        continue;
                    }
                    if (p.feet.y - brain.centre.y).abs() > PEAL_RING_HEIGHT {
                        continue;
                    }
                    if distance(&p.feet, &brain.centre) > radius {
                        continue;
                    }
                    let kind = if ctx.coil_permanent && final_ring {
                        HitKind::Lethal
                    } else if p.status.is_overtuned() {
                        HitKind::Lethal
                    } else {
                        HitKind::Overtuned
                    };
                    action.hits.push(Hit {
                        player_id: p.id,
                        kind,
                    });
                }
                brain.phase = PealPhase::Ring;
                brain.timer = 1;
            }
        }
        PealPhase::Ring => {
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                if brain.ring_index + 1 < PEAL_RING_RADII.len() {
                    brain.ring_index += 1;
                    brain.phase = PealPhase::Chime;
                    brain.timer = PEAL_CHIME_TICKS;
                } else {
                    brain.phase = PealPhase::Rest;
                    brain.timer = PEAL_REST_TICKS;
                }
            }
        }
        PealPhase::Rest => {
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                brain.phase = PealPhase::Idle;
                brain.timer = PEAL_CYCLE_TICKS;
            }
        }
    }

    action
}

//! Kite: locks the most isolated player and detonates a tracked strike marker.
//!
//! Tier-2 telegraph: the tether brightens and the marker tracks the target for
//! 1.4 s, then freezes for 0.4 s before the blast rings resolve. A Striker or a

use protocol::Vec3;

use crate::rules::{distance, EnemyParams};

use super::{
    most_isolated_player, step_toward, ClassRelation, EnemyAction, EnemyClass, Hit, HitKind, Ring,
    StepContext, ENEMY_DT,

/// Ticks the tether tracks the target before it freezes (1.4 s).
pub const KITE_LOCK_TICKS: u32 = 28;
/// Ticks the frozen marker waits before detonating (0.4 s).
pub const KITE_STRIKE_TICKS: u32 = 8;
/// Ticks of recovery after a detonation (2.0 s).
pub const KITE_RECOVER_TICKS: u32 = 40;
/// Ticks of hunting between locks (1.0 s).
pub const KITE_HUNT_TICKS: u32 = 20;
/// Blast radius of the marker, units.
pub const KITE_BLAST_RADIUS: f32 = 3.5;
/// A Striker or Grappler this close to the Kite cuts the tether, units.
pub const KITE_CUT_RANGE: f32 = 3.0;
/// Height above the Kite at which the tether breaks, units.
pub const KITE_LOS_HEIGHT: f32 = 6.0;
/// Drift speed while hunting, units/s.
pub const KITE_HUNT_SPEED: f32 = 4.0;

/// Kite phases.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum KitePhase {
    /// Picks the most isolated player.
    Hunt,
    /// The tether brightens and the marker tracks the target (tier-2 decal).
    Lock,
    /// The marker is frozen; the blast resolves at the end.
    Strike,
    /// Harmless drift after a detonation.
    Recover,
}

/// Kite brain state.
#[derive(Debug, Clone, PartialEq)]
pub struct KiteBrain {
    /// Current phase.
    pub phase: KitePhase,
    /// Phase countdown in ticks.
    pub timer: u32,
    /// The locked player.
    pub target_id: Option<u32>,
    /// The strike marker (blast centre).
    pub marker: Vec3,
}

impl Default for KiteBrain {
    fn default() -> Self {
        Self {
            phase: KitePhase::Hunt,
            timer: KITE_HUNT_TICKS,
            target_id: None,
            marker: Vec3::new(0.0, 0.0, 0.0),
        }
    }
}

impl KiteBrain {
    /// The tier-2 strike marker: the exact blast volume, drawn from lock to detonation.
    pub fn telegraph(&self) -> Option<Ring> {
        match self.phase {
            KitePhase::Lock | KitePhase::Strike => Some(Ring {
                centre: self.marker,
                radius: KITE_BLAST_RADIUS,
            }),
            KitePhase::Hunt | KitePhase::Recover => None,
        }
    }
}

/// Registry class interaction for Kite: Striker and Grappler both counter it.
pub fn relation(class: EnemyClass) -> ClassRelation {
    match class {
        EnemyClass::Striker | EnemyClass::Grappler => ClassRelation::Counters,
        EnemyClass::Bulwark | EnemyClass::Glider => ClassRelation::Neutral,
    }
}

/// Advance Kite by one 20 Hz tick.
pub fn step(
    brain: &mut KiteBrain,
    pos: &Vec3,
    params: &EnemyParams,
    ctx: &StepContext<'_>,
) -> EnemyAction {
    let mut action = EnemyAction::idle();

    match brain.phase {
        KitePhase::Hunt => {
            if let Some(p) = most_isolated_player(ctx.players) {
                action.motion =
                    step_toward(pos, &p.feet, KITE_HUNT_SPEED * params.speed_mul * ENEMY_DT);
            }
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                match most_isolated_player(ctx.players) {
                    Some(p) => {
                        brain.target_id = Some(p.id);
                        brain.marker = p.feet;
                        brain.phase = KitePhase::Lock;
                        brain.timer = KITE_LOCK_TICKS;
                    }
                    None => brain.timer = KITE_HUNT_TICKS,
                }
            }
        }
        KitePhase::Lock => {
            if let Some(p) = brain
                .target_id
                .and_then(|id| ctx.players.iter().find(|p| p.id == id))
            {
                brain.marker = p.feet;
                if p.feet.y - pos.y > KITE_LOS_HEIGHT {
                    brain.phase = KitePhase::Recover;
                    brain.timer = KITE_RECOVER_TICKS;
                    return action;
                }
            }
            let cut = ctx.players.iter().any(|p| {
                matches!(p.class, EnemyClass::Striker | EnemyClass::Grappler)
                    && distance(&p.feet, pos) <= KITE_CUT_RANGE
            });
            if cut {
                brain.phase = KitePhase::Recover;
                brain.timer = KITE_RECOVER_TICKS;
                return action;
            }
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                brain.phase = KitePhase::Strike;
                brain.timer = KITE_STRIKE_TICKS;
            }
        }
        KitePhase::Strike => {
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                for p in ctx.players {
                    if distance(&p.feet, &brain.marker) <= KITE_BLAST_RADIUS {
                        action.hits.push(Hit {
                            player_id: p.id,
                            kind: HitKind::Lethal,
                        });
                    }
                }
                brain.phase = KitePhase::Recover;
                brain.timer = KITE_RECOVER_TICKS;
            }
        }
        KitePhase::Recover => {
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                brain.phase = KitePhase::Hunt;
                brain.timer = KITE_HUNT_TICKS;
            }
        }
    }

    action
}

//! Static: fog-bound dasher that vanishes after every strike.
//!
//! Tier-3 telegraph: no per-attack tell. The fog patch around the body is the
//! persistent cue and it is also the hit volume, so a player can walk out of it
//! at any time. A Glider is never selected; a Bulwark is immune to the strike.

use protocol::Vec3;

use crate::rules::{distance, flat_direction, EnemyParams};

use super::{
    nearest_player_excluding, step_toward, ClassRelation, EnemyAction, EnemyClass, Hit, HitKind,
    Ring, StepContext, ENEMY_DT,
};

/// Ticks the fog zone grows before the dash (2.0 s of visible cue).
pub const STATIC_TRACK_TICKS: u32 = 40;
/// Ticks the dash lasts (0.4 s).
pub const STATIC_DASH_TICKS: u32 = 8;
/// Ticks the enemy stays invisible after a strike (1.0 s).
pub const STATIC_VANISH_TICKS: u32 = 20;
/// Fog radius at the start of a track, units.
pub const STATIC_FOG_MIN: f32 = 2.0;
/// Fog radius at the end of a track, units.
pub const STATIC_FOG_MAX: f32 = 5.0;
/// Dash speed, units/s.
pub const STATIC_DASH_SPEED: f32 = 18.0;
/// Track drift speed, units/s.
pub const STATIC_TRACK_SPEED: f32 = 3.0;
/// Distance from the next target at which Static reappears, units.
pub const STATIC_TELEPORT_DISTANCE: f32 = 4.0;

/// Static phases.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum StaticPhase {
    /// Fog thickens around it while it walks the target down (the tier-3 cue).
    Track,
    /// Committed dash at the locked point.
    Dash,
    /// The strike resolves; only players still inside the fog are hit.
    Strike,
    /// Invisible and harmless; teleports to the next target at the end.
    Vanish,
}

/// Static brain state.
#[derive(Debug, Clone, PartialEq)]
pub struct StaticBrain {
    /// Current phase.
    pub phase: StaticPhase,
    /// Phase countdown in ticks.
    pub timer: u32,
    /// The player being tracked.
    pub target_id: Option<u32>,
    /// Centre of the fog patch.
    pub fog_centre: Vec3,
    /// Radius of the fog patch this tick.
    pub fog_radius: f32,
    /// Point the dash commits to.
    pub locked: Vec3,
    /// True while the body is invisible to clients.
    pub hidden: bool,
}

impl Default for StaticBrain {
    fn default() -> Self {
        Self {
            phase: StaticPhase::Track,
            timer: STATIC_TRACK_TICKS,
            target_id: None,
            fog_centre: Vec3::new(0.0, 0.0, 0.0),
            fog_radius: STATIC_FOG_MIN,
            locked: Vec3::new(0.0, 0.0, 0.0),
            hidden: false,
        }
    }
}

impl StaticBrain {
    /// The tier-3 ambient cue: the fog patch around the body, gone while it is away.
    pub fn telegraph(&self) -> Option<Ring> {
        if self.hidden {
            None
        } else {
            Some(Ring {
                centre: self.fog_centre,
                radius: self.fog_radius,
            })
        }
    }
}

/// Registry class interaction for Static: Glider ignores, Bulwark immune.
pub fn relation(class: EnemyClass) -> ClassRelation {
    match class {
        EnemyClass::Glider => ClassRelation::Ignores,
        EnemyClass::Bulwark => ClassRelation::Immune,
        EnemyClass::Striker | EnemyClass::Grappler => ClassRelation::Neutral,
    }
}

/// Advance Static by one 20 Hz tick.
pub fn step(
    brain: &mut StaticBrain,
    pos: &Vec3,
    params: &EnemyParams,
    ctx: &StepContext<'_>,
) -> EnemyAction {
    let mut action = EnemyAction::idle();

    match brain.phase {
        StaticPhase::Track => {
            brain.hidden = false;
            let target = nearest_player_excluding(ctx.players, pos, Some(EnemyClass::Glider), None);
            brain.target_id = target.map(|p| p.id);
            if let Some(p) = target {
                action.motion =
                    step_toward(pos, &p.feet, STATIC_TRACK_SPEED * params.speed_mul * ENEMY_DT);
            }
            brain.fog_centre = *pos;
            let span = (STATIC_TRACK_TICKS - 1) as f32;
            let elapsed = (STATIC_TRACK_TICKS - brain.timer) as f32;
            brain.fog_radius =
                STATIC_FOG_MIN + (STATIC_FOG_MAX - STATIC_FOG_MIN) * (elapsed / span);
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                brain.locked = target.map_or(*pos, |p| p.feet);
                brain.phase = StaticPhase::Dash;
                brain.timer = STATIC_DASH_TICKS;
            }
        }
        StaticPhase::Dash => {
            action.motion =
                step_toward(pos, &brain.locked, STATIC_DASH_SPEED * params.speed_mul * ENEMY_DT);
            brain.fog_centre = *pos;
            brain.fog_radius = STATIC_FOG_MAX;
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                brain.phase = StaticPhase::Strike;
                brain.timer = 1;
            }
        }
        StaticPhase::Strike => {
            for p in ctx.players {
                if relation(p.class) == ClassRelation::Immune {
                    continue;
                }
                if distance(&p.feet, &brain.fog_centre) <= brain.fog_radius {
                    action.hits.push(Hit {
                        player_id: p.id,
                        kind: HitKind::Lethal,
                    });
                }
            }
            brain.phase = StaticPhase::Vanish;
            brain.timer = STATIC_VANISH_TICKS;
            brain.hidden = true;
        }
        StaticPhase::Vanish => {
            brain.hidden = true;
            brain.timer = brain.timer.saturating_sub(1);
            if brain.timer == 0 {
                let next =
                    nearest_player_excluding(ctx.players, pos, Some(EnemyClass::Glider), brain.target_id)
                        .or_else(|| {
                            nearest_player_excluding(
                                ctx.players,
                                pos,
                                Some(EnemyClass::Glider),
                                None,
                            )
                        });
                let centre = match next {
                    Some(p) => {
                        let away = flat_direction(&p.feet, pos);
                        Vec3::new(
                            p.feet.x + away.x * STATIC_TELEPORT_DISTANCE,
                            p.feet.y,
                            p.feet.z + away.z * STATIC_TELEPORT_DISTANCE,
                        )
                    }
                    None => *pos,
                };
                brain.fog_centre = centre;
                brain.fog_radius = STATIC_FOG_MIN;
                brain.target_id = next.map(|p| p.id);
                brain.phase = StaticPhase::Track;
                brain.timer = STATIC_TRACK_TICKS;
                brain.hidden = false;
                action.teleport = Some(centre);
            }
        }
    }

    action
}

/// Persistent per-player debuff state that enemy brains read.
///
/// The server owns one of these per player and ticks it once per enemy tick.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
pub struct PlayerStatus {
    /// Ticks left of OVERTUNED; 0 means the player is not overtuned.
    pub overtuned_ticks: u32,
}

impl PlayerStatus {
    /// Ticks of OVERTUNED one Peal ring applies (6.0 s at 20 Hz).
    pub const OVERTUNED_TICKS: u32 = 120;

    /// Apply OVERTUNED for `ticks`, keeping the longer of the two timers.
    pub fn apply_overtuned(&mut self, ticks: u32) {
        self.overtuned_ticks = self.overtuned_ticks.max(ticks);
    }

    /// Advance the timers by one enemy tick.
    pub fn tick(&mut self) {
        self.overtuned_ticks = self.overtuned_ticks.saturating_sub(1);
    }

    /// True while the player carries OVERTUNED.
    pub fn is_overtuned(&self) -> bool {
        self.overtuned_ticks > 0
    }
}

/// Per-enemy state machine payload; the variant matches `params.behavior`.
#[derive(Debug, Clone, PartialEq)]
pub enum Brain {
    Pallbearer(PallbearerBrain),
    Snapshrike(SnapshrikeBrain),
    Moteling(MotelingBrain),
    RiftWarden(RiftWardenBrain),
    GaleWight(GaleWightBrain),
    LanternLeech(LanternLeechBrain),
    Turret(TurretBrain),
    Mimic(MimicBrain),
    Corrupter(CorrupterBrain),
    /// Rewind: replays a player's path from 1.5 s ago.
    Rewind(RewindBrain),
    /// Warden: guards the densest gift cluster.
    Warden(WardenBrain),
    /// Peal: teleports and rings; lethal only to OVERTUNED players.
    Peal(PealBrain),
    /// Kite: locks the most isolated player and detonates a marker.
    Kite(KiteBrain),
    /// Static: fog-bound dasher that vanishes after each strike.
    Static(StaticBrain),
}

impl Brain {
    /// Default brain for a behaviour tag; unknown behaviours fall back to pallbearer.
    pub fn for_behavior(behavior: u8) -> Self {
        match behavior {
            BEHAVIOR_SNAPSHRIKE => Brain::Snapshrike(SnapshrikeBrain::default()),
            BEHAVIOR_MOTELING => Brain::Moteling(MotelingBrain::default()),
            BEHAVIOR_RIFT_WARDEN => Brain::RiftWarden(RiftWardenBrain::default()),
            BEHAVIOR_GALE_WIGHT => Brain::GaleWight(GaleWightBrain::default()),
            BEHAVIOR_LANTERN_LEECH => Brain::LanternLeech(LanternLeechBrain::default()),
            BEHAVIOR_TURRET => Brain::Turret(TurretBrain::default()),
            BEHAVIOR_MIMIC => Brain::Mimic(MimicBrain::default()),
            BEHAVIOR_CORRUPTER => Brain::Corrupter(CorrupterBrain::default()),
            BEHAVIOR_REWIND => Brain::Rewind(RewindBrain::default()),
            BEHAVIOR_WARDEN => Brain::Warden(WardenBrain::default()),
            BEHAVIOR_PEAL => Brain::Peal(PealBrain::default()),
            BEHAVIOR_KITE => Brain::Kite(KiteBrain::default()),
            BEHAVIOR_STATIC => Brain::Static(StaticBrain::default()),
            _ => Brain::Pallbearer(PallbearerBrain::default()),
        }
    }

    /// Advance this brain by one 20 Hz tick and report what the server should apply.
    ///
    /// Only the brains implemented in [`crate::rules::enemies`] are stepped here;
    /// the nine M3 brains keep their server-side step and return
    /// [`EnemyAction::idle`]. The caller applies [`EnemyAction::motion`] or
    /// [`EnemyAction::teleport`] to the enemy position and resolves
    /// [`EnemyAction::hits`] against the players.
    pub fn step(
        &mut self,
        pos: &Vec3,
        params: &EnemyParams,
        ctx: &StepContext<'_>,
    ) -> EnemyAction {
        match self {
            Brain::Rewind(b) => enemies::rewind::step(b, pos, params, ctx),
            Brain::Warden(b) => enemies::warden::step(b, pos, params, ctx),
            Brain::Peal(b) => enemies::peal::step(b, pos, params, ctx),
            Brain::Kite(b) => enemies::kite::step(b, pos, params, ctx),
            Brain::Static(b) => enemies::r#static::step(b, pos, params, ctx),
            _ => EnemyAction::idle(),
        }
    }
}

/// Euclidean distance between two points.
pub(crate) fn distance(a: &Vec3, b: &Vec3) -> f32 {
    let dx = a.x - b.x;
    let dy = a.y - b.y;
    let dz = a.z - b.z;
    (dx * dx + dy * dy + dz * dz).sqrt()
}

/// Horizontal (XZ) distance between two points.
pub(crate) fn horizontal_distance(a: &Vec3, b: &Vec3) -> f32 {
    let dx = a.x - b.x;
    let dz = a.z - b.z;
    (dx * dx + dz * dz).sqrt()
}

/// The player's centre point, derived from their feet position.
pub(crate) fn player_centre(player_feet: &Vec3) -> Vec3 {
    Vec3::new(player_feet.x, player_feet.y + 0.9, player_feet.z)
}

/// Move `pos` toward `target` by at most `max_step` (3D), never overshooting.
pub(crate) fn move_toward(pos: &mut Vec3, target: &Vec3, max_step: f32) {
    let dx = target.x - pos.x;
    let dy = target.y - pos.y;
    let dz = target.z - pos.z;
    let d = (dx * dx + dy * dy + dz * dz).sqrt();
    if d <= max_step {
        *pos = *target;
        return;
    }
    if d <= 1e-9 {
        return;
    }
    let scale = max_step / d;
    pos.x += dx * scale;
    pos.y += dy * scale;
    pos.z += dz * scale;
}

/// Normalised 3D direction from `from` to `to`; falls back to +X when they coincide.
pub(crate) fn dir_normalized(from: &Vec3, to: &Vec3) -> Vec3 {
    let dx = to.x - from.x;
    let dy = to.y - from.y;
    let dz = to.z - from.z;
    let d = (dx * dx + dy * dy + dz * dz).sqrt();
    if d <= 1e-9 {
        return Vec3::new(1.0, 0.0, 0.0);
    }
    Vec3::new(dx / d, dy / d, dz / d)
}

/// Normalised horizontal (XZ) direction from `from` to `to`; falls back to +X when coincident.
pub(crate) fn flat_direction(from: &Vec3, to: &Vec3) -> Vec3 {
    let dx = to.x - from.x;
    let dz = to.z - from.z;
    let d = (dx * dx + dz * dz).sqrt();
    if d <= 1e-9 {
        return Vec3::new(1.0, 0.0, 0.0);
    }
    Vec3::new(dx / d, 0.0, dz / d)
}

/// Squared distance from a point to an AABB (0 when the point is inside).
pub(crate) fn aabb_distance(p: &Vec3, b: &Aabb) -> f32 {
    let dx = (b.min.x - p.x).max(0.0).max(p.x - b.max.x);
    let dy = (b.min.y - p.y).max(0.0).max(p.y - b.max.y);
    let dz = (b.min.z - p.z).max(0.0).max(p.z - b.max.z);
    (dx * dx + dy * dy + dz * dz).sqrt()
}

/// True when the segment `a`..`b` intersects (or touches) the AABB `bounds`.
pub(crate) fn segment_intersects_aabb(a: &Vec3, b: &Vec3, bounds: &Aabb) -> bool {
    let mut t_min = 0.0_f32;
    let mut t_max = 1.0_f32;
    if !clip_axis(a.x, b.x - a.x, bounds.min.x, bounds.max.x, &mut t_min, &mut t_max) {
        return false;
    }
    if !clip_axis(a.y, b.y - a.y, bounds.min.y, bounds.max.y, &mut t_min, &mut t_max) {
        return false;
    }
    if !clip_axis(a.z, b.z - a.z, bounds.min.z, bounds.max.z, &mut t_min, &mut t_max) {
        return false;
    }
    true
}

/// Clip the slab `[min, max]` on one axis into the `[t_min, t_max]` window; false = no hit.
fn clip_axis(origin: f32, dir: f32, min: f32, max: f32, t_min: &mut f32, t_max: &mut f32) -> bool {
    if dir.abs() <= 1e-9 {
        return origin >= min && origin <= max;
    }
    let inv = 1.0 / dir;
    let mut t0 = (min - origin) * inv;
    let mut t1 = (max - origin) * inv;
    if t0 > t1 {
        std::mem::swap(&mut t0, &mut t1);
    }
    if t0 > *t_min {
        *t_min = t0;
    }
    if t1 < *t_max {
        *t_max = t1;
    }
    *t_min <= *t_max
}

/// True when a sphere at `centre` with `radius` overlaps the player's collision box.
pub(crate) fn sphere_hits_player_box(centre: &Vec3, radius: f32, feet: &Vec3) -> bool {
    let b = player_box(feet);
    let dx = centre.x - centre.x.clamp(b.min.x, b.max.x);
    let dy = centre.y - centre.y.clamp(b.min.y, b.max.y);
    let dz = centre.z - centre.z.clamp(b.min.z, b.max.z);
    dx * dx + dy * dy + dz * dz <= radius * radius
}

/// True when the player's centre is within [`GIFT_RADIUS`] of the gift centre.
pub fn gift_touched(player_feet: &Vec3, gift: &Gift) -> bool {
    gift_touched_within(player_feet, gift, GIFT_RADIUS)
}

/// True when the player's centre is within `radius` of the gift centre.
pub fn gift_touched_within(player_feet: &Vec3, gift: &Gift, radius: f32) -> bool {
    distance(&player_centre(player_feet), &gift.pos) <= radius
}

/// True when the player stands on the beacon: close horizontally and roughly level.
pub fn beacon_reached(player_feet: &Vec3, beacon: &Vec3) -> bool {
    beacon_reached_within(player_feet, beacon, BEACON_RADIUS)
}

/// True when the player is within `radius` horizontally of the beacon and roughly
/// level with it. The vertical tolerance stays [`BEACON_HEIGHT`].
pub fn beacon_reached_within(player_feet: &Vec3, beacon: &Vec3, radius: f32) -> bool {
    let dx = player_feet.x - beacon.x;
    let dz = player_feet.z - beacon.z;
    let horizontal = (dx * dx + dz * dz).sqrt();
    horizontal <= radius && (player_feet.y - beacon.y).abs() <= BEACON_HEIGHT
}

/// Create an enemy sitting at `pos`, with `pos` as its home, default params and a
/// full default spawn grace (pallbearers wait 4.5 s).
pub fn spawn_enemy(id: u32, kind: u8, pos: Vec3) -> Enemy {
    spawn_enemy_with(id, kind, pos, EnemyParams::default())
}

/// Create an enemy with explicit [`EnemyParams`]; `active_in` starts at
/// `params.grace_ticks`, with a 4.5 s floor for pallbearers.
pub fn spawn_enemy_with(id: u32, kind: u8, pos: Vec3, params: EnemyParams) -> Enemy {
    let active_in = if params.behavior == BEHAVIOR_PALLBEARER {
        params.grace_ticks.max(PALLBEARER_WAKE_TICKS)
    } else {
        params.grace_ticks
    };
    Enemy {
        id,
        kind,
        home: pos,
        pos,
        active_in,
        brain: Brain::for_behavior(params.behavior),
        knock_vel: Vec3::new(0.0, 0.0, 0.0),
        knock_ticks: 0,
        params,
    }
}

/// Fraction of the drawn body radius that actually kills for `enemy_touches`.
pub fn enemy_hit_radius(enemy: &Enemy) -> f32 {
    enemy.params.size * ENEMY_HIT_FORGIVENESS
}

/// True when the enemy's hit sphere overlaps the player's collision box.
pub fn enemy_touches(enemy: &Enemy, player_feet: &Vec3) -> bool {
    sphere_hits_player_box(&enemy.pos, enemy_hit_radius(enemy), player_feet)
}
