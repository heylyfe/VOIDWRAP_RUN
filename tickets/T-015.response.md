"use client";

import { useMemo } from "react";
import type { JSX } from "react";
import type { Platform } from "@/protocol/generated";
import { PALETTE, platformColor } from "./palette";

interface EdgeBarData {
  size: [number, number, number];
  position: [number, number, number];
}

interface PlatformBoxData {
  id: number;
  center: [number, number, number];
  size: [number, number, number];
  top: string;
  accent: string;
  edgeBars: EdgeBarData[];
}

const TOP_EMISSIVE = 0.08;
const EDGE_BAR_THICKNESS = 0.08;
const EDGE_BAR_HEIGHT = 0.08;
const EDGE_BAR_OVERHANG = 0.04;
const EDGE_BAR_EMISSIVE = 1.0;

// Four thin bars along the top perimeter. The bars along X straddle the
// top edge (overhang EDGE_BAR_OVERHANG on each side); the bars along Z
// span exactly the gap between them so nothing overlaps at the corners.
function toEdgeBars(size: readonly [number, number, number]): EdgeBarData[] {
  const [width, height, depth] = size;
  const halfWidth = width / 2;
  const halfDepth = depth / 2;
  const barY = height / 2 + EDGE_BAR_HEIGHT / 2;
  const longLength = width + EDGE_BAR_OVERHANG * 2;
  const shortLength = depth - EDGE_BAR_OVERHANG * 2;
  return [
    {
      size: [longLength, EDGE_BAR_HEIGHT, EDGE_BAR_THICKNESS],
      position: [0, barY, halfDepth],
    },
    {
      size: [longLength, EDGE_BAR_HEIGHT, EDGE_BAR_THICKNESS],
      position: [0, barY, -halfDepth],
    },
    {
      size: [EDGE_BAR_THICKNESS, EDGE_BAR_HEIGHT, shortLength],
      position: [halfWidth, barY, 0],
    },
    {
      size: [EDGE_BAR_THICKNESS, EDGE_BAR_HEIGHT, shortLength],
      position: [-halfWidth, barY, 0],
    },
  ];
}

function toBoxData(platform: Platform): PlatformBoxData {
  const { min, max } = platform.bounds;
  const size: [number, number, number] = [
    max.x - min.x,
    max.y - min.y,
    max.z - min.z,
  ];
  const center: [number, number, number] = [
    (min.x + max.x) / 2,
    (min.y + max.y) / 2,
    (min.z + max.z) / 2,
  ];
  return {
    id: platform.id,
    center,
    size,
    top: platformColor(platform.id),
    accent:
      platform.id % 2 === 0 ? PALETTE.edgeGlowCyan : PALETTE.edgeGlowMagenta,
    edgeBars: toEdgeBars(size),
  };
}

export function Platforms(props: {
  platforms: readonly Platform[];
}): JSX.Element {
  const boxes = useMemo(() => props.platforms.map(toBoxData), [props.platforms]);

  return (
    <group>
      {boxes.map((box) => (
        <group key={box.id} position={box.center}>
          <mesh>
            <boxGeometry args={box.size} />
            <meshStandardMaterial
              color={box.top}
              emissive={box.top}
              emissiveIntensity={TOP_EMISSIVE}
            />
          </mesh>
          {box.edgeBars.map((bar, index) => (
            <mesh key={index} position={bar.position}>
              <boxGeometry args={bar.size} />
              <meshStandardMaterial
                color={box.accent}
                emissive={box.accent}
                emissiveIntensity={EDGE_BAR_EMISSIVE}
              />
            </mesh>
          ))}
        </group>
      ))}
    </group>
  );
}

"use client";

import { useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import type { Group, Mesh } from "three";
import type { Vec3 } from "@/protocol/generated";
import { PALETTE } from "./palette";

const SHARD_OFFSETS: ReadonlyArray<readonly [number, number, number]> = [
  [0.95, 0.25, 0],
  [-0.95, 0.25, 0],
  [0, 0.95, 0.35],
  [0, -0.95, -0.35],
];

const CORE_COLOR = PALETTE.chaserRed;
const SHARD_COLOR = "#c81e1e";
const EMISSIVE = "#ff1a1a";

export function Chaser(props: { position: Vec3 }): JSX.Element {
  const coreRef = useRef<Mesh>(null);
  const shardsRef = useRef<Group>(null);

  useFrame((state, delta) => {
    const core = coreRef.current;
    if (core !== null) {
      core.rotation.y += delta * 2.5;
    }
    const shards = shardsRef.current;
    if (shards !== null) {
      shards.rotation.y -= delta * 1.8;
      shards.rotation.x = Math.sin(state.clock.elapsedTime * 1.3) * 0.35;
    }
  });

  return (
    <group position={[props.position.x, props.position.y, props.position.z]}>
      <mesh ref={coreRef}>
        <octahedronGeometry args={[0.7]} />
        <meshStandardMaterial
          color={CORE_COLOR}
          emissive={EMISSIVE}
          emissiveIntensity={0.6}
          toneMapped={false}
        />
      </mesh>
      <group ref={shardsRef}>
        {SHARD_OFFSETS.map((offset, index) => (
          <mesh key={index} position={offset as [number, number, number]}>
            <octahedronGeometry args={[0.14]} />
            <meshStandardMaterial
              color={SHARD_COLOR}
              emissive={EMISSIVE}
              emissiveIntensity={0.8}
              toneMapped={false}
            />
          </mesh>
        ))}
      </group>
    </group>
  );
}

"use client";

import { useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import type { MeshStandardMaterial } from "three";
import type { Vec3 } from "@/protocol/generated";
import { PALETTE } from "./palette";

interface FrameBarData {
  position: [number, number, number];
  size: [number, number, number];
}

// Thin flat frame around a 1.6-wide square on top of the base pad. The bars
// along X span the full width; the bars along Z span the gap between them so
// the corners do not double-glow. 0.08 tall, resting on the base (top at
// y = 0.2).
const BASE_TOP_Y = 0.2;
const FRAME_WIDTH = 1.6;
const FRAME_THICKNESS = 0.08;
const FRAME_HEIGHT = 0.08;
const FRAME_EMISSIVE_OPEN = 1.5;
const FRAME_EMISSIVE_CLOSED = 0.5;
const CLOSED_PILLAR_EMISSIVE = 0.12;

const FRAME_BARS: ReadonlyArray<FrameBarData> = [
  {
    position: [0, BASE_TOP_Y + FRAME_HEIGHT / 2, FRAME_WIDTH / 2 - FRAME_THICKNESS / 2],
    size: [FRAME_WIDTH, FRAME_HEIGHT, FRAME_THICKNESS],
  },
  {
    position: [0, BASE_TOP_Y + FRAME_HEIGHT / 2, -(FRAME_WIDTH / 2 - FRAME_THICKNESS / 2)],
    size: [FRAME_WIDTH, FRAME_HEIGHT, FRAME_THICKNESS],
  },
  {
    position: [FRAME_WIDTH / 2 - FRAME_THICKNESS / 2, BASE_TOP_Y + FRAME_HEIGHT / 2, 0],
    size: [FRAME_THICKNESS, FRAME_HEIGHT, FRAME_WIDTH - FRAME_THICKNESS * 2],
  },
  {
    position: [-(FRAME_WIDTH / 2 - FRAME_THICKNESS / 2), BASE_TOP_Y + FRAME_HEIGHT / 2, 0],
    size: [FRAME_THICKNESS, FRAME_HEIGHT, FRAME_WIDTH - FRAME_THICKNESS * 2],
  },
];

export function Beacon(props: { position: Vec3; open: boolean }): JSX.Element {
  const pillarMaterialRef = useRef<MeshStandardMaterial>(null);

  useFrame((state) => {
    const material = pillarMaterialRef.current;
    if (material === null) {
      return;
    }
    material.emissiveIntensity = props.open
      ? 1.2 + Math.sin(state.clock.elapsedTime * 3) * 0.6
      : CLOSED_PILLAR_EMISSIVE;
  });

  const accent = props.open ? PALETTE.beaconOpen : PALETTE.beaconClosed;
  const frameEmissive = props.open ? FRAME_EMISSIVE_OPEN : FRAME_EMISSIVE_CLOSED;

  return (
    <group position={[props.position.x, props.position.y, props.position.z]}>
      <mesh position={[0, 0.1, 0]}>
        <boxGeometry args={[1.4, 0.2, 1.4]} />
        <meshStandardMaterial
          color={accent}
          emissive={accent}
          emissiveIntensity={props.open ? 0.6 : 0.1}
        />
      </mesh>
      {FRAME_BARS.map((bar, index) => (
        <mesh key={index} position={bar.position}>
          <boxGeometry args={bar.size} />
          <meshStandardMaterial
            color={PALETTE.beaconOpen}
            emissive={PALETTE.beaconOpen}
            emissiveIntensity={frameEmissive}
          />
        </mesh>
      ))}
      <mesh position={[0, 3.2, 0]}>
        <boxGeometry args={[0.8, 6, 0.8]} />
        <meshStandardMaterial
          ref={pillarMaterialRef}
          color={accent}
          emissive={PALETTE.beaconOpen}
          emissiveIntensity={props.open ? 1.2 : CLOSED_PILLAR_EMISSIVE}
          transparent
          depthWrite={false}
          opacity={props.open ? 0.4 : 0.28}
        />
      </mesh>
    </group>
  );
}

"use client";

import { useEffect, useState } from "react";

import type { JSX } from "react";

const FADE_MS = 1500;
const HIDE_MS = 2000;

type BannerPhase = "shown" | "faded" | "hidden";

type LevelBannerProps = {
  level: number;
};

export function LevelBanner(props: LevelBannerProps): JSX.Element | null {
  const { level } = props;
  const [shownFor, setShownFor] = useState<number>(level);
  const [phase, setPhase] = useState<BannerPhase>("shown");

  // Adjust state while rendering (recommended React pattern): when the level
  // prop changes, re-arm the banner synchronously. The effect below never
  // calls setState synchronously, only inside its timers.
  if (level !== shownFor) {
    setShownFor(level);
    setPhase("shown");
  }

  useEffect(() => {
    const fadeTimer = window.setTimeout(() => setPhase("faded"), FADE_MS);
    const hideTimer = window.setTimeout(() => setPhase("hidden"), HIDE_MS);
    return () => {
      window.clearTimeout(fadeTimer);
      window.clearTimeout(hideTimer);
    };
  }, [shownFor]);

  if (phase === "hidden") {
    return null;
  }

  return (
    <div
      data-testid="level-banner"
      className={`pointer-events-none absolute left-1/2 top-[22%] -translate-x-1/2 whitespace-nowrap rounded-full bg-black/70 px-6 py-3 font-mono text-4xl font-bold tracking-widest text-white transition-opacity duration-500 [text-shadow:0_0_24px_#ff3db8,0_0_6px_#07060d] sm:text-5xl ${
        phase === "faded" ? "opacity-0" : "opacity-100"
      }`}
    >
      LEVEL {level}
    </div>
  );
}
