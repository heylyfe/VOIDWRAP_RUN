use protocol::{Aabb, LevelData};

/// Collision geometry the controller moves through.
#[derive(Debug, Clone, Default, PartialEq)]
pub struct World {
    pub solids: Vec<Aabb>,
    /// Anything whose feet fall below this height is lost to the void.
    pub kill_y: f32,
}

impl World {
    pub fn from_level(level: &LevelData) -> Self {
        World {
            solids: level.platforms.iter().map(|p| p.bounds.clone()).collect(),
            kill_y: level.kill_y,
        }
    }
}

/// Strict overlap test (touching faces do not overlap).
pub fn overlaps(a: &Aabb, b: &Aabb) -> bool {
    a.min.x < b.max.x
        && a.max.x > b.min.x
        && a.min.y < b.max.y
        && a.max.y > b.min.y
        && a.min.z < b.max.z
        && a.max.z > b.min.z
}

//! Movement tuning. All durations are in physics frames (1 / PHYSICS_HZ seconds).

use protocol::PHYSICS_HZ;

/// Fixed physics timestep in seconds.
pub const DT: f32 = 1.0 / PHYSICS_HZ as f32;

/// Player collision box half-width (x and z). `ControllerState.pos` is the feet point.
pub const PLAYER_HALF_WIDTH: f32 = 0.4;
/// Player collision box height.
pub const PLAYER_HEIGHT: f32 = 1.8;

/// Ground run speed, units/s. Applied instantly (no acceleration mush).
pub const RUN_SPEED: f32 = 9.0;
/// Gravity, units/s².
pub const GRAVITY: f32 = 32.0;
/// Initial jump velocity, units/s (apex ≈ v²/2g ≈ 2.2 units).
pub const JUMP_VELOCITY: f32 = 12.0;
/// Terminal fall speed, units/s.
pub const MAX_FALL_SPEED: f32 = 40.0;

//! Kinematic character controller.
//!
//! M0 bootstrap version: instant run speed, gravity, edge-triggered ground jump,
//! axis-separated AABB collision. Ticket T-010 replaces this with the full M1
//! controller (coyote time, jump buffering, class ability). The public signature
//! [`step`] is the contract shared by server and WASM client and must not change
//! without updating both.

use crate::tuning::*;
use crate::world::{World, overlaps};
use protocol::{Aabb, BTN_JUMP, ControllerState, InputFrame, Vec3};

/// A fresh controller standing at `pos`.
pub fn spawn_state(pos: Vec3) -> ControllerState {
    ControllerState {
        pos,
        vel: Vec3::ZERO,
        grounded: false,
        coyote_frames: 0,
        jump_buffer_frames: 0,
        prev_buttons: 0,
        air_jumps_used: 0,
        ability_cooldown_frames: 0,
        ability_active_frames: 0,
        ability_dir: Vec3::ZERO,
    }
}

/// The player's collision box for a given feet position.
pub fn player_box(pos: &Vec3) -> Aabb {
    Aabb {
        min: Vec3::new(pos.x - PLAYER_HALF_WIDTH, pos.y, pos.z - PLAYER_HALF_WIDTH),
        max: Vec3::new(
            pos.x + PLAYER_HALF_WIDTH,
            pos.y + PLAYER_HEIGHT,
            pos.z + PLAYER_HALF_WIDTH,
        ),
    }
}

/// Normalised wish direction from the quantised input (length ≤ 1).
pub fn wish_dir(input: &InputFrame) -> (f32, f32) {
    // Clamp -128 to -127 so the range is symmetric.
    let x = f32::from(input.move_x.max(-127)) / 127.0;
    let z = f32::from(input.move_z.max(-127)) / 127.0;
    let len_sq = x * x + z * z;
    if len_sq > 1.0 {
        let len = len_sq.sqrt();
        (x / len, z / len)
    } else {
        (x, z)
    }
}

/// Advance one fixed physics frame of [`DT`] seconds.
pub fn step(state: &mut ControllerState, input: &InputFrame, world: &World) {
    let (wx, wz) = wish_dir(input);
    state.vel.x = wx * RUN_SPEED;
    state.vel.z = wz * RUN_SPEED;

    let jump_pressed = input.buttons & BTN_JUMP != 0 && state.prev_buttons & BTN_JUMP == 0;
    if jump_pressed && state.grounded {
        state.vel.y = JUMP_VELOCITY;
        state.grounded = false;
    }
    state.vel.y = (state.vel.y - GRAVITY * DT).max(-MAX_FALL_SPEED);
    state.prev_buttons = input.buttons;

    let landed = move_axis(state, world, Axis::Y, state.vel.y * DT);
    state.grounded = landed;
    move_axis(state, world, Axis::X, state.vel.x * DT);
    move_axis(state, world, Axis::Z, state.vel.z * DT);
}

#[derive(Clone, Copy)]
enum Axis {
    X,
    Y,
    Z,
}

fn get(v: &Vec3, a: Axis) -> f32 {
    match a {
        Axis::X => v.x,
        Axis::Y => v.y,
        Axis::Z => v.z,
    }
}

fn set(v: &mut Vec3, a: Axis, value: f32) {
    match a {
        Axis::X => v.x = value,
        Axis::Y => v.y = value,
        Axis::Z => v.z = value,
    }
}

/// Moves along one axis, stopping flush against the first solid hit.
/// Returns true if the move was a downward move that landed on something.
fn move_axis(state: &mut ControllerState, world: &World, axis: Axis, delta: f32) -> bool {
    if delta == 0.0 {
        return false;
    }
    let mut pos = state.pos.clone();
    let target = get(&pos, axis) + delta;
    set(&mut pos, axis, target);
    let mut hit = false;
    for solid in &world.solids {
        let b = player_box(&pos);
        if !overlaps(&b, solid) {
            continue;
        }
        hit = true;
        // Push back flush against the face we entered through.
        let p = get(&pos, axis);
        let resolved = match axis {
            Axis::Y if delta < 0.0 => solid.max.y,
            Axis::Y => solid.min.y - PLAYER_HEIGHT,
            _ if delta < 0.0 => get(&solid.max, axis) + PLAYER_HALF_WIDTH,
            _ => get(&solid.min, axis) - PLAYER_HALF_WIDTH,
        };
        // Only ever push backwards along the motion, never forwards.
        let clamped = if delta < 0.0 { resolved.max(p) } else { resolved.min(p) };
        set(&mut pos, axis, clamped);
    }
    if hit {
        set(&mut state.vel, axis, 0.0);
    }
    state.pos = pos;
    hit && matches!(axis, Axis::Y) && delta < 0.0
}

## Task
Replace the M0 controller with the M1 controller. Required behaviour (frames are 1/60 s):

1. **Run**: horizontal velocity = wish direction × `RUN_SPEED` (9.0), applied instantly on ground AND in air
   (snappy, no acceleration), except while dashing.
2. **Gravity** `GRAVITY` 32.0, terminal fall `MAX_FALL_SPEED` 40.0. `JUMP_VELOCITY` 12.0.
3. **Coyote time** `COYOTE_FRAMES = 7` (~120 ms): while grounded `coyote_frames = 7`; each airborne step
   decrements it (saturating). A jump is allowed iff `grounded || coyote_frames > 0`. Jumping sets
   `coyote_frames = 0` so it cannot be used twice. Walking off a ledge and pressing jump on airborne step 1..=7
   must still jump; on step 8 it must not.
4. **Jump buffer** `JUMP_BUFFER_FRAMES = 6` (~100 ms): a jump *press edge* (BTN_JUMP set now, not set in
   `prev_buttons`) sets `jump_buffer_frames = 6`. Every step, if `jump_buffer_frames > 0` and a jump is allowed,
   jump and clear the buffer; otherwise decrement it (saturating). Pressing jump up to 6 steps before landing
   must jump on the first step at which a jump is allowed (the landing step or the one right after it).
5. **Jump cut** (variable height): if BTN_JUMP is NOT held and `vel.y > 0` and not dashing, set
   `vel.y = min(vel.y, JUMP_CUT_VELOCITY)` with `JUMP_CUT_VELOCITY = 4.0`.
6. **Dash ability** (BTN_ABILITY press edge, only when `ability_cooldown_frames == 0` and not already dashing):
   direction = normalised wish direction; if no wish input, the normalised horizontal velocity; if that is zero
   too, `(0, 0, -1)`. Sets `ability_active_frames = DASH_FRAMES (9)`, `ability_cooldown_frames = DASH_COOLDOWN_FRAMES (45)`,
   `ability_dir` = direction. While `ability_active_frames > 0`: horizontal velocity = dir × `DASH_SPEED` (28.0),
   `vel.y = 0`, gravity not applied, jump-cut not applied; decrement `ability_active_frames` once per step.
   Cooldown decrements once per step (saturating) whether or not dashing. A jump during a dash is allowed if
   the jump rules allow it and ends the dash immediately (`ability_active_frames = 0`).
7. **Collision**: keep the existing axis-separated AABB approach (move Y, then X, then Z; resolve flush against
   the face entered; zero that velocity component on hit). `grounded` = this step's downward Y move hit something,
   OR the player is resting exactly on a surface (feet within 0.001 of a solid's top while overlapping it in x/z
   and `vel.y <= 0`). The second condition matters because a dash sets `vel.y = 0`, making the Y move zero.
8. Order inside `step` (document it in a comment): timers & edge detection → dash start → jump (buffer/coyote)
   → horizontal velocity → vertical velocity (gravity / jump cut / dash) → collide Y, X, Z → grounded & coyote
   refresh → `prev_buttons = input.buttons`.
9. Keep these public items with the same signatures: `spawn_state(pos: Vec3) -> ControllerState`,
   `player_box(pos: &Vec3) -> Aabb`, `wish_dir(input: &InputFrame) -> (f32, f32)`,
   `step(state: &mut ControllerState, input: &InputFrame, world: &World)`. Tuning constants live in `tuning.rs`
   as `pub const` (keep `DT`, `PLAYER_HALF_WIDTH`, `PLAYER_HEIGHT` and their values).

## Acceptance criteria (each needs a test)
- [ ] Coyote: walk off the bootstrap slab edge; jump press on airborne step 7 jumps (vel.y > 0 after the step); a separate run pressing on step 8 does not.
- [ ] Buffer: falling onto the slab, jump pressed 5 steps before landing → the player is airborne with vel.y > 0 within 1 step of touching down; pressed 8 steps before → no jump on landing.
- [ ] Holding jump never re-jumps after landing (edge-triggered); a new press does.
- [ ] Jump cut: tapping jump (held 1 step) yields a lower apex than holding it (measure max y).
- [ ] Dash: from standing with wish +X, dash moves ≥ 4.0 and ≤ 4.4 units in x over its 9 frames (28 × 9/60 = 4.2); vel.y stays 0 during the dash even in mid-air; a second press within 45 frames does nothing; after the cooldown it works again.
- [ ] Dash with no input and zero velocity goes toward -Z.
- [ ] No tunnelling: falling at terminal speed onto a 1-unit-thick slab lands on top.
- [ ] Determinism: the same 600-step scripted input sequence run twice from the same start yields identical states (`assert_eq!` on the whole ControllerState).
- [ ] Wall: running into a tall solid wall stops flush at the wall face (pos.x == wall.min.x - PLAYER_HALF_WIDTH) and does not set grounded.
