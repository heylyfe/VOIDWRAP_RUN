// Pure colour helper for the ice event: blends a platform colour towards a pale blue
// and lifts every channel so icy platforms read as glossy/specular.
const ICE = { r: 186, g: 224, b: 244 } as const;
const BLEND = 0.5;
const LIFT = 24;

function channel(v: number): number {
  return Math.min(255, Math.max(0, Math.round(v)));
}

function toHex(v: number): string {
  return channel(v).toString(16).padStart(2, "0");
}

export function iceTint(baseColor: string): string {
  const raw = baseColor.trim().replace(/^#/, "");
  const norm = raw.length === 3 ? raw.split("").map((c) => c + c).join("") : raw;
  if (!/^[0-9a-fA-F]{6}$/.test(norm)) {
    throw new Error(`iceTint: expected a hex colour like "#rrggbb", got "${baseColor}"`);
  }
  const r = parseInt(norm.slice(0, 2), 16);
  const g = parseInt(norm.slice(2, 4), 16);
  const b = parseInt(norm.slice(4, 6), 16);
  const mix = (c: number, t: number) => channel(c + (t - c) * BLEND + LIFT);
  return `#${toHex(mix(r, ICE.r))}${toHex(mix(g, ICE.g))}${toHex(mix(b, ICE.b))}`;
}

// Pure routing logic shared by the enemy pool: which archetype a wire `kind` maps to,
// and which "readability mode" an enemy is in (dormant / hunting / disguised).
import { ENEMY_DISGUISED, ENEMY_DORMANT } from "../protocol/generated";
import { ENEMIES } from "../content";
import type { EnemyBehavior } from "../content";

export type EnemyMode = "dormant" | "hunting" | "disguised";

/** `kind` indexes design/content.json enemies; unknown/missing kinds behave as chasers. */
export function enemyBehaviorFor(kind: number | undefined): EnemyBehavior {
  if (kind === undefined) return "chaser";
  const card = ENEMIES[kind];
  return card?.params.behavior ?? "chaser";
}

/** Map the wire ENEMY_* state to the visual readability mode. Missing info reads as hunting. */
export function modeForState(state: number | undefined): EnemyMode {
  if (state === ENEMY_DORMANT) return "dormant";
  if (state === ENEMY_DISGUISED) return "disguised";
  return "hunting";
}

/** Scratch vector reused every frame (no allocation). */
const SCRATCH = new Vector3();
const PULSE_SPEED = 1.6;
const DORMANT_PULSE_SPEED = 0.7;

/**
 * Squat pylon with a single eye. Tracks the local player (yaw only) and pulses slowly;
 * the pulse brightens when hunting, dims + slows when dormant.
 */
export function Turret(props: { position: Vec3; mode: { current: EnemyMode }; session: GameSession }): JSX.Element {
  const headRef = useRef<Group>(null);
  const bodyMat = useMemo(
    () =>
      new MeshStandardMaterial({
        color: "#2b2130",
        emissive: COMMON.enemy.glow,
        emissiveIntensity: 0.3,
        roughness: 0.6,
      }),
    [],
  );
  const eyeMat = useMemo(
    () =>
      new MeshStandardMaterial({
        color: "#ff9ad9",
        emissive: COMMON.enemy.glow,
        emissiveIntensity: 2.2,
        toneMapped: false,
      }),
    [],
  );
  useEffect(() => () => bodyMat.dispose(), [bodyMat]);
  useEffect(() => () => eyeMat.dispose(), [eyeMat]);

  useFrame((state) => {
    const dormant = props.mode.current === "dormant";
    const head = headRef.current;
    const ctl = props.session.playerController();
    if (head && ctl) {
      head.getWorldPosition(SCRATCH);
      const dx = ctl.pos.x - SCRATCH.x;
      const dz = ctl.pos.z - SCRATCH.z;
      if (dx * dx + dz * dz > 0.01) head.rotation.y = Math.atan2(dx, dz);
    }
    const t = state.clock.elapsedTime;
    const pulse = 0.5 + 0.5 * Math.sin(t * (dormant ? DORMANT_PULSE_SPEED : PULSE_SPEED));
    eyeMat.emissiveIntensity = dormant ? 0.5 + pulse * 0.6 : 1.5 + pulse * 1.5;
    bodyMat.emissiveIntensity = dormant ? 0.1 : 0.3;
  });

  return (
    <group position={[props.position.x, props.position.y, props.position.z]}>
      <mesh position={[0, -0.28, 0]} material={bodyMat}>
        <cylinderGeometry args={[0.62, 0.88, 0.42, 8]} />
      </mesh>
      <mesh position={[0, 0.12, 0]} material={bodyMat}>
        <cylinderGeometry args={[0.34, 0.5, 0.72, 8]} />
      </mesh>
      <group ref={headRef} position={[0, 0.55, 0]}>
        <mesh material={bodyMat} scale={[1, 0.72, 1]}>

import { useMemo, useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import type { Group } from "three";
import type { Vec3 } from "@/protocol/generated";
import type { GameSession } from "@/game/session";
import { Chaser } from "./Chaser";
import { Turret } from "./Turret";
import { Mimic } from "./Mimic";
import { Corrupter } from "./Corrupter";
import { enemyBehaviorFor, modeForState } from "./enemyBehavior";
import type { EnemyMode } from "./enemyBehavior";

export const MAX_ENEMIES = 8;

const ORIGIN: Vec3 = { x: 0, y: 0, z: 0 };

interface SlotRefs {
  outer: Group | null;
  chaser: Group | null;
  turret: Group | null;
  mimic: Group | null;
  corrupter: Group | null;
}

function emptySlot(): SlotRefs {
  return { outer: null, chaser: null, turret: null, mimic: null, corrupter: null };
}

function setVisible(g: Group | null, visible: boolean): void {
  if (g) g.visible = visible;
}

/**
 * Pooled enemy slots (fixed size, visibility toggled — never re-created).
 * Every slot pre-builds all four archetype models and switches which one is
 * visible per frame from the server's ENEMY_* state and the content kind.
 * The slot is scaled by the server's hit-sphere radius; a disguised mimic is
 * scaled to 1 so it is indistinguishable from a normal gift.
 */
export function EnemyField({ session }: { session: GameSession }): JSX.Element {
  const slots = useRef<SlotRefs[]>([]);
  if (slots.current.length === 0) {
    slots.current = Array.from({ length: MAX_ENEMIES }, emptySlot);
  }
  const modes = useMemo(() => Array.from({ length: MAX_ENEMIES }, () => ({ current: "hunting" as EnemyMode })), []);

  useFrame(() => {
    const now = performance.now();
    let i = 0;
    for (const [id, o] of session.enemyPositions(now)) {
      if (i >= MAX_ENEMIES) break;
      const slot = slots.current[i];
      if (!slot) {
        i += 1;
        continue;
      }
      const info = session.enemyInfo.get(id);
      const mode = modeForState(info?.state);
      const behavior = enemyBehaviorFor(info?.kind);
      const modeRef = modes[i];
      if (modeRef) modeRef.current = mode;
      const outer = slot.outer;
      if (outer) {
        outer.visible = true;
        outer.position.set(o.x, o.y, o.z);
        const disguised = behavior === "mimic" && mode === "disguised";
        outer.scale.setScalar(disguised ? 1 : session.enemyRadius.get(id) ?? 0.5);
      }
      setVisible(slot.chaser, behavior === "chaser");
      setVisible(slot.turret, behavior === "turret");
      setVisible(slot.mimic, behavior === "mimic");
      setVisible(slot.corrupter, behavior === "corrupter");
      i += 1;
    }
    for (; i < MAX_ENEMIES; i += 1) {
      const slot = slots.current[i];
      setVisible(slot?.outer ?? null, false);
    }
  });

  return (
    <group>
      {Array.from({ length: MAX_ENEMIES }, (_, i) => {
        const mode = modes[i] ?? { current: "hunting" as EnemyMode };
        return (
          <group
            key={i}
            ref={(g) => {
              const slot = slots.current[i];
              if (slot) slot.outer = g;
            }}
            visible={false}
          >
            <group
              ref={(g) => {
                const slot = slots.current[i];
                if (slot) slot.chaser = g;
              }}
            >
              <Chaser position={ORIGIN} mode={mode} />
            </group>
            <group
              ref={(g) => {
                const slot = slots.current[i];
                if (slot) slot.turret = g;
              }}
            >
              <Turret position={ORIGIN} mode={mode} session={session} />
            </group>
            <group
              ref={(g) => {
                const slot = slots.current[i];
                if (slot) slot.mimic = g;
              }}
            >
              <Mimic position={ORIGIN} mode={mode} />
            </group>
            <group
              ref={(g) => {
                const slot = slots.current[i];
                if (slot) slot.corrupter = g;
              }}
            >
              <Corrupter position={ORIGIN} mode={mode} />
            </group>
          </group>
        );
      })}
    </group>
  );
}

  y: number,
  w: number,
  h: number,

  ctx.beginPath();

          new MeshBasicMaterial({

            transparent: true,
            opacity: 0.85,
            blending: AdditiveBlending,
            depthWrite: false,
            toneMapped: false,
          }),

    if (c) {

      if (c.class === CLASS_BLINKER && c.abilityActiveFrames > prev && lastPos.current.ok) {

          const sl = slots.current[i];

          target.y = lastPos.current.y;
          target.z = lastPos.current.z;
          target.t0 = t;

// Four thin bars along the top perimeter. The bars along X straddle the
// top edge (overhang EDGE_BAR_OVERHANG on each side); the bars along Z
// span exactly the gap between them so nothing overlaps at the corners.
function toEdgeBars(size: readonly [number, number, number]): EdgeBarData[] {
  const [width, height, depth] = size;
  const halfWidth = width / 2;
  const halfDepth = depth / 2;
  const barY = height / 2 + EDGE_BAR_HEIGHT / 2;
  const longLength = width + EDGE_BAR_OVERHANG * 2;
  const shortLength = depth - EDGE_BAR_OVERHANG * 2;
  return [
    {
      size: [longLength, EDGE_BAR_HEIGHT, EDGE_BAR_THICKNESS],
      position: [0, barY, halfDepth],
    },
    {
      size: [longLength, EDGE_BAR_HEIGHT, EDGE_BAR_THICKNESS],
      position: [0, barY, -halfDepth],
    },
    {
      size: [EDGE_BAR_THICKNESS, EDGE_BAR_HEIGHT, shortLength],
      position: [halfWidth, barY, 0],
    },
    {
      size: [EDGE_BAR_THICKNESS, EDGE_BAR_HEIGHT, shortLength],
      position: [-halfWidth, barY, 0],
    },
  ];
}

function toBoxData(platform: Platform): PlatformBoxData {
  const { min, max } = platform.bounds;
  const size: [number, number, number] = [
    max.x - min.x,
    max.y - min.y,
    max.z - min.z,
  ];
  const center: [number, number, number] = [
    (min.x + max.x) / 2,
    (min.y + max.y) / 2,
    (min.z + max.z) / 2,
  ];
  return {
    id: platform.id,
    kind: platform.kind,
    center,
    size,
    edgeBars: toEdgeBars(size),
  };
}

/** Deterministic cracked-stone canvas texture for crumbling platform tops (built once). */
function createCrackTexture(baseHex: string): CanvasTexture | null {
  if (typeof document === "undefined") return null;
  const size = 256;
  const canvas = document.createElement("canvas");
  canvas.width = size;
  canvas.height = size;
  const ctx = canvas.getContext("2d");
  if (!ctx) return null;
  // Deterministic LCG so every client sees the same cracks.
  let seed = 9001;
  const rnd = (): number => {
    seed = (Math.imul(seed, 1664525) + 1013904223) >>> 0;
    return seed / 4294967296;
  };
  ctx.fillStyle = baseHex;
  ctx.fillRect(0, 0, size, size);
  for (let i = 0; i < 40; i += 1) {
    ctx.fillStyle = `rgba(0, 0, 0, ${0.08 + rnd() * 0.12})`;
    ctx.beginPath();
    ctx.arc(rnd() * size, rnd() * size, 12 + rnd() * 42, 0, Math.PI * 2);
    ctx.fill();
  }
  for (let i = 0; i < 14; i += 1) {
    ctx.strokeStyle = "rgba(12, 9, 8, 0.9)";
    ctx.lineWidth = 1.5 + rnd() * 2.5;
    ctx.beginPath();
    let x = rnd() * size;
    let y = rnd() * size;
    let angle = rnd() * Math.PI * 2;
    ctx.moveTo(x, y);
    const segments = 5 + Math.floor(rnd() * 5);
    for (let j = 0; j < segments; j += 1) {
      angle += (rnd() - 0.5) * 1.4;
      const len = 18 + rnd() * 30;
      x += Math.cos(angle) * len;
      y += Math.sin(angle) * len;
      ctx.lineTo(x, y);
    }
    ctx.stroke();
  }
  const tex = new CanvasTexture(canvas);
  tex.wrapS = RepeatWrapping;
  tex.wrapT = RepeatWrapping;
  tex.colorSpace = SRGBColorSpace;
  tex.anisotropy = 4;
  return tex;
}

/** Materials shared by every platform of a theme (textures are module-cached). */
function createThemeMaterials(theme: Theme, icy: boolean): ThemeMaterials {
  const textures = themeTextures(theme);
  const top = makeTopMaterial(theme.tile.light, textures.top ?? null, icy);
  const side = new MeshStandardMaterial(
    textures.side
      ? { map: textures.side, color: icy ? iceTint(SIDE_TINT) : SIDE_TINT, roughness: SIDE_ROUGHNESS }
      : { color: icy ? iceTint(theme.tile.side) : theme.tile.side, roughness: SIDE_ROUGHNESS },
  );
  const bottom = new MeshStandardMaterial({
    color: icy ? iceTint(theme.tile.dark) : theme.tile.dark,
    roughness: BOTTOM_ROUGHNESS,
  });
  const edgeTrim = new MeshStandardMaterial({
    color: icy ? iceTint(theme.tile.edge) : theme.tile.edge,
    emissive: icy ? iceTint(theme.tile.edge) : theme.tile.edge,
    emissiveIntensity: EDGE_BAR_EMISSIVE,
    roughness: icy ? 0.2 : 0.4,
  });
  return {
    // BoxGeometry face order: +x, -x, +y, -y, +z, -z.
    byFace: [side, side, top, bottom, side, side],
    side,
    bottom,
    edgeTrim,
    crumbleBase: shadeColor(theme.tile.light, CRUMBLE_SHADE),
    ghostColor: theme.tile.edge,
    all: [top, side, bottom, edgeTrim],
  };
}

function disposeMaterials(materials: ThemeMaterials): void {
  for (const material of materials.all) {
    material.dispose();
  }
}

function PlatformMesh(props: {
  box: PlatformBoxData;
  materials: ThemeMaterials;
  tileState: number | null;
  icy: boolean;
}): JSX.Element {

  const crumble = box.kind === PLATFORM_CRUMBLE;
  const warn = tileState === TILE_WARN;
  const gone = tileState === TILE_GONE;
  // World-scaled UVs are memoised per platform box; disposed on change/unmount.
  const geometry = useMemo(
    () => worldScaledBoxGeometry(box.size[0], box.size[1], box.size[2]),
    [box.size],
  );
  useEffect(() => () => geometry.dispose(), [geometry]);

  // Crumble platforms get their own cracked top material so TILE_WARN can flash it.
  const ownTop = useMemo(() => {
    if (!crumble) return null;
    const base = icy ? iceTint(materials.crumbleBase) : materials.crumbleBase;
    const map = createCrackTexture(base);
    const mat = new MeshStandardMaterial({
      color: map ? "#ffffff" : base,
      map,
      roughness: icy ? ICE_ROUGHNESS : CRUMBLE_ROUGHNESS,
      emissive: WARN_EMISSIVE,
      emissiveIntensity: CRUMBLE_IDLE_EMISSIVE,
    });
    return { mat, map };
  }, [crumble, icy, materials.crumbleBase]);
  useEffect(
    () => () => {
      ownTop?.mat.dispose();
      ownTop?.map?.dispose();
    },
    [ownTop],
  );
  const byFace = useMemo(() => {
    if (!ownTop) return materials.byFace;
    const faces: [Material, Material, Material, Material, Material, Material] = [
      materials.side,
      materials.side,
      ownTop.mat,
      materials.bottom,
      materials.side,
      materials.side,
    ];
    return faces;
  }, [ownTop, materials]);

  const innerRef = useRef<Group>(null);
  const phase = useMemo(() => (box.id * 1.61803398875) % (Math.PI * 2), [box.id]);

  useFrame((state) => {
    if (!crumble && !warn) return;
    const t = state.clock.elapsedTime;
    const group = innerRef.current;
    if (group) {
      if (warn) {
        group.position.set(
          box.center[0] + Math.sin(t * 34 + phase) * SHAKE_AMP,
          box.center[1],
          box.center[2] + Math.cos(t * 27 + phase) * SHAKE_AMP,
        );
      } else if (
        group.position.x !== box.center[0] ||
        group.position.y !== box.center[1] ||
        group.position.z !== box.center[2]
      ) {
        group.position.set(box.center[0], box.center[1], box.center[2]);
      }
    }
    if (ownTop) {
      ownTop.mat.emissiveIntensity = warn
        ? 1.05 + Math.sin(t * 9 + phase) * 0.9
        : CRUMBLE_IDLE_EMISSIVE;
    }
  });

  return (
    <>
      <group ref={innerRef} position={box.center} visible={!gone}>
        <mesh geometry={geometry} material={byFace} />
        {box.edgeBars.map((bar, index) => (
          <EdgeTrimBar key={index} bar={bar} material={materials.edgeTrim} />
        ))}
      </group>
      {/* Faint ghost outline where a crumbled platform used to be. */}
      <mesh geometry={geometry} position={box.center} visible={gone}>
        <meshBasicMaterial
          color={materials.ghostColor}
          wireframe
          transparent
          opacity={GONE_GHOST_OPACITY}
          depthWrite={false}
        />
      </mesh>
    </>
  );
}

function EdgeTrimBar(props: { bar: EdgeBarData; material: Material }): JSX.Element {
  const [x, y, z] = props.bar.size;
  // World-scaled UVs do not matter for the flat trim bars, so a plain box does.
  const geometry = useMemo(() => new BoxGeometry(x, y, z), [x, y, z]);
  useEffect(() => () => geometry.dispose(), [geometry]);
  return (
    <mesh geometry={geometry} position={props.bar.position} material={props.material} />
  );
}

export function Platforms(props: {
  platforms: readonly Platform[];
  theme: Theme;
  tiles: readonly TileState[];
  icy: boolean;
}): JSX.Element {
  const boxes = useMemo(() => props.platforms.map(toBoxData), [props.platforms]);
  const tileById = useMemo(() => {
    const m = new Map<number, number>();
    for (const tile of props.tiles) m.set(tile.id, tile.state);
    return m;
  }, [props.tiles]);
  // Materials are shared between all platforms and memoised per theme.
  const materials = useMemo(
    () => createThemeMaterials(props.theme, props.icy),
    [props.theme, props.icy],
  );
  useEffect(() => () => disposeMaterials(materials), [materials]);

  return (
    <group>
      {boxes.map((box) => (
        <PlatformMesh
          key={box.id}
          box={box}
          materials={materials}
          tileState={tileById.get(box.id) ?? null}
          icy={props.icy}
        />
      ))}
    </group>
  );
}

export default function GameClient() {
  const [session, setSession] = useState<GameSession | null>(null);
  const [level, setLevel] = useState<LevelData | null>(null);
  const [loadError, setLoadError] = useState<string | null>(null);
  const container = useRef<HTMLDivElement>(null);
  const { status, pointerLocked, runOver, giftsTaken, runInfo, draft, myMode, activeIds } = useGameStore();
  const [following, setFollowing] = useState<number | null>(null);
  const drafting = !!draft && runInfo?.phase === PHASE_ENEMY_DRAFT;
  // The lobby hub is shown in the colours of the level about to be played.
  const runLevel = runInfo?.level ?? level?.level ?? 1;
  const normalIds = useMemo(() => (level ? level.gifts.filter((g) => g.kind !== GIFT_GOLDEN).map((g) => g.id) : []), [level]);
  const normalTotal = normalIds.length;
  const normalLeft = normalIds.filter((id) => !giftsTaken.includes(id)).length;
  const playing = runInfo ? runInfo.phase === PHASE_PLAYING : true;

  useEffect(() => {
    let disposed = false;
    let s: GameSession | null = null;
    WasmStepper.load()
      .then((stepper) => {
        if (disposed) return;
        s = new GameSession(stepper, defaultServerUrl(), "blocky");
        s.onLevel = setLevel;
        setSession(s);
      })
      .catch((e: unknown) => setLoadError(String(e)));
    return () => {
      disposed = true;
      s?.dispose();
    };
  }, []);

  // Free the mouse for the run summary and the lobby draft.
  useEffect(() => {
    if ((runOver || drafting) && document.pointerLockElement) document.exitPointerLock();
  }, [runOver, drafting]);

  useEffect(() => {
    if (!session) return;
    const key = (down: boolean) => (e: KeyboardEvent) => {
      const st = useGameStore.getState();
      const spectatingNow = st.runInfo?.phase === PHASE_PLAYING && st.myMode !== PLAYER_ACTIVE;
      if (down && spectatingNow && !e.repeat && (e.code === "KeyF" || e.code === "KeyG")) {
        session.followIndex = e.code === "KeyF" ? nextFollow(session.followIndex, st.activeIds.length) : null;
        setFollowing(session.followIndex);
        e.preventDefault();
        return;
      }
      const b = bindingFor(e.code);
      if (!b) return;
      session.keys[b] = down;
      e.preventDefault();
    };
    const onDown = key(true);
    const onUp = key(false);
    const onMove = (e: MouseEvent) => {
      if (document.pointerLockElement !== container.current) return;
      session.yaw -= e.movementX * MOUSE_SENSITIVITY;
      session.pitch = Math.min(1.3, Math.max(-0.2, session.pitch + e.movementY * MOUSE_SENSITIVITY));
    };
    const onMouse = (down: boolean) => (e: MouseEvent) => {
      if (e.button === 2) session.rightMouse = down;
    };
    const onMouseDown = onMouse(true);
    const onMouseUp = onMouse(false);
    const onLock = () => useGameStore.getState().set({ pointerLocked: document.pointerLockElement === container.current });
    const onBlur = () => {
      for (const k of Object.keys(session.keys) as (keyof typeof session.keys)[]) session.keys[k] = false;
      session.rightMouse = false;
    };
    window.addEventListener("keydown", onDown);
    window.addEventListener("keyup", onUp);
    window.addEventListener("mousemove", onMove);
    window.addEventListener("mousedown", onMouseDown);
    window.addEventListener("mouseup", onMouseUp);
    window.addEventListener("blur", onBlur);
    document.addEventListener("pointerlockchange", onLock);
    return () => {
      window.removeEventListener("keydown", onDown);
      window.removeEventListener("keyup", onUp);
      window.removeEventListener("mousemove", onMove);
      window.removeEventListener("mousedown", onMouseDown);
      window.removeEventListener("mouseup", onMouseUp);
      window.removeEventListener("blur", onBlur);
      document.removeEventListener("pointerlockchange", onLock);
    };
  }, [session]);

  const restart = useCallback(() => session?.restart(), [session]);
  const submitDraft = useCallback((picks: number[]) => session?.submitDraft(picks), [session]);

  return (
    <div
      ref={container}
      className="relative h-screen w-screen overflow-hidden bg-[#07060d]"
      onClick={() => {
        const st = useGameStore.getState();
        const inDraft = !!st.draft && st.runInfo?.phase === PHASE_ENEMY_DRAFT;
        if (!st.runOver && !inDraft) void container.current?.requestPointerLock();
      }}
      onContextMenu={(e) => e.preventDefault()}
    >
      {session && level ? (
        <Canvas camera={{ fov: 70, near: 0.1, far: 1000 }} dpr={[1, 2]}>
          <World session={session} level={level} themeLevel={runLevel} />
        </Canvas>
      ) : (
        <div className="flex h-full items-center justify-center font-mono text-white/70">
          {loadError ?? "entering the void…"}
        </div>
      )}
      {level && playing && normalTotal > 0 && (
        <>
          <GameHud level={runLevel} giftsRemaining={normalLeft} giftsTotal={normalTotal} />
          <LevelBanner level={runLevel} />
        </>
      )}
      {playing && myMode !== PLAYER_ACTIVE && !runOver && (
        <SpectatorHud
          reason={myMode === PLAYER_FINISHED ? "beacon" : "dead"}
          playersLeft={activeIds.length}
          following={following === null ? null : `Player ${activeIds[following % Math.max(activeIds.length, 1)] ?? "?"}`}
        />
      )}
      {runInfo && playing && (
        <RunStatus
          currency={runInfo.currency}
          lives={runInfo.lives}
          enemies={runInfo.enemies}
          curses={runInfo.curses}
          upgrades={runInfo.upgrades}
        />
      )}
      {drafting && draft && runInfo && (
        <DraftScreen
          cards={draft.cards}
          enemyPicks={draft.enemyPicks}
          rewardPicks={draft.rewardPicks}
          nextLevel={runInfo.level}
          currency={runInfo.currency}
          lives={runInfo.lives}
          waiting={draft.submitted}
          onSubmit={submitDraft}
        />
      )}
      {!pointerLocked && status === "connected" && !runOver && !drafting && (
        <div className="pointer-events-none absolute bottom-8 left-1/2 -translate-x-1/2 rounded bg-black/60 px-4 py-2 font-mono text-sm text-white">
          click to aim · WASD move · Space jump · Shift dash
        </div>
      )}
      {runOver && (
        <RunSummary
          levelReached={runOver.levelReached}
          giftsCollected={runOver.giftsCollected}
          cause={causeOf(runOver.cause)}
          onRestart={restart}
        />
      )}
      <DebugPanel />
    </div>
  );
}
