/** Deterministic little gift texture: body colour with a ribbon cross and a bow dot. */
function createGiftTexture(body: string, ribbon: string, bow: string): CanvasTexture | null {
  if (typeof document === "undefined") return null;
  const size = 128;
  const canvas = document.createElement("canvas");
  canvas.width = size;
  canvas.height = size;
  const ctx = canvas.getContext("2d");
  if (!ctx) return null;
  ctx.fillStyle = body;
  ctx.fillRect(0, 0, size, size);
  ctx.fillStyle = "rgba(0, 0, 0, 0.18)";
  ctx.fillRect(0, size * 0.72, size, size * 0.28);
  const band = Math.floor(size * 0.2);
  ctx.fillStyle = ribbon;
  ctx.fillRect((size - band) / 2, 0, band, size);
  ctx.fillRect(0, (size - band) / 2, size, band);
  ctx.fillStyle = bow;
  ctx.beginPath();
  ctx.arc(size / 2, size / 2, band * 0.45, 0, Math.PI * 2);
  ctx.fill();
  const tex = new CanvasTexture(canvas);
  tex.wrapS = RepeatWrapping;
  tex.wrapT = RepeatWrapping;
  tex.colorSpace = SRGBColorSpace;
  tex.anisotropy = 4;
  return tex;
}

function createGiftAssets(): GiftAssets {
  const normalTex = createGiftTexture(COMMON.gift.body, COMMON.gift.ribbon, COMMON.gift.glow);
  const goldenTex = createGiftTexture(GOLD_BODY, GOLD_RIBBON, GOLD_BOW);
  const box = new BoxGeometry(GIFT_SIZE, GIFT_SIZE, GIFT_SIZE);
  const sparkle = new OctahedronGeometry(SPARKLE_RADIUS);
  const normalMaterial = new MeshStandardMaterial({
    map: normalTex ?? undefined,
    color: normalTex ? "#ffffff" : COMMON.gift.body,
    emissive: COMMON.gift.glow,
    emissiveIntensity: NORMAL_EMISSIVE_INTENSITY,
    roughness: 0.55,
    metalness: 0.1,
    toneMapped: false,
  });
  const goldenMaterial = new MeshStandardMaterial({
    map: goldenTex ?? undefined,
    color: goldenTex ? "#ffffff" : GOLD_BODY,
    emissive: GOLD_EMISSIVE,
    emissiveIntensity: GOLD_EMISSIVE_INTENSITY,
    roughness: 0.32,
    metalness: 0.35,
    toneMapped: false,
  });
  const sparkleMaterial = new MeshBasicMaterial({
    color: SPARKLE_COLOR,
    transparent: true,
    opacity: 0.9,
    blending: AdditiveBlending,
    depthWrite: false,
    toneMapped: false,
    fog: false,
  });
  return {
    box,
    sparkle,
    normalMaterial,
    goldenMaterial,
    sparkleMaterial,
    dispose() {
      box.dispose();
      sparkle.dispose();
      normalMaterial.dispose();
      goldenMaterial.dispose();
      sparkleMaterial.dispose();
      normalTex?.dispose();
      goldenTex?.dispose();
    },
  };

/** The platform supporting a gift: highest platform top below the gift whose XZ contains it. */
function platformUnder(pos: Vec3, platforms: readonly Platform[]): number | null {
  let best: Platform | null = null;
  for (const p of platforms) {
    if (pos.x < p.bounds.min.x || pos.x > p.bounds.max.x) continue;
    if (pos.z < p.bounds.min.z || pos.z > p.bounds.max.z) continue;
    const top = p.bounds.max.y;
    if (top > pos.y + 0.01) continue;
    if (best === null || top > best.bounds.max.y) best = p;
  }
  return best === null ? null : best.id;
}

/**
 * All gifts on the level, drawn as three instanced meshes (normal, golden, sparkle).
 * Phase 1: untaken normal gifts. Phase 2 (levelPhase != LEVEL_COLLECT): one golden gift at
 * every collected normal gift position, unless its id is in store.goldenTaken or its platform
 * is TILE_GONE (golden gifts stay collectable during TILE_WARN, like on the server).
 */
export function Gifts({
  level,
  tiles,
}: {
  level: LevelData;
  tiles: readonly TileState[];
}): JSX.Element {
  const assets = useMemo(() => createGiftAssets(), []);
  useEffect(
    () => () => {
      assets.dispose();
    },
    [assets],
  );

  const scratchRef = useRef<Scratch | null>(null);

  const giftsTaken = useGameStore((s) => s.giftsTaken);
  const goldenTaken = useGameStore((s) => s.goldenTaken);
  const levelPhase = useGameStore((s) => s.levelPhase);
  const phase2 = levelPhase !== LEVEL_COLLECT;

  const normalGifts = useMemo(
    () => level.gifts.filter((g) => g.kind !== GIFT_GOLDEN),
    [level],
  );
  // Resolved once per level; only TILE_GONE re-filters the visible golden set.
  const giftPlatform = useMemo(
    () => normalGifts.map((g) => platformUnder(g.pos, level.platforms)),
    [normalGifts, level.platforms],
  );
  const taken = useMemo(() => new Set<number>(giftsTaken), [giftsTaken]);
  const goldTaken = useMemo(() => new Set<number>(goldenTaken), [goldenTaken]);
  const gone = useMemo(() => {
    const s = new Set<number>();
    for (const t of tiles) {
      if (t.state === TILE_GONE) s.add(t.id);
    }
    return s;
  }, [tiles]);

  const visibleNormal = useMemo(() => {
    const out: number[] = [];
    if (phase2) return out;
    for (let i = 0; i < normalGifts.length; i += 1) {
      const g = normalGifts[i];
      if (!g || taken.has(g.id)) continue;
      out.push(i);
    }
    return out;
  }, [phase2, normalGifts, taken]);

  const visibleGolden = useMemo(() => {
    const out: number[] = [];
    if (!phase2) return out;
    for (let i = 0; i < normalGifts.length; i += 1) {
      const g = normalGifts[i];
      if (!g) continue;
      if (!taken.has(g.id)) continue;
      if (goldTaken.has(g.id)) continue;
      const p = giftPlatform[i];
      if (p !== undefined && p !== null && gone.has(p)) continue;
      out.push(i);
    }
    return out;
  }, [phase2, normalGifts, taken, goldTaken, giftPlatform, gone]);

  const normalRef = useRef<InstancedMesh>(null);
  const goldenRef = useRef<InstancedMesh>(null);
  const sparkleRef = useRef<InstancedMesh>(null);
  useEffect(() => {
    for (const m of [normalRef.current, goldenRef.current, sparkleRef.current]) {
      if (m) {
        m.instanceMatrix.setUsage(DynamicDrawUsage);
        m.count = 0;
        m.instanceMatrix.needsUpdate = true;
      }
    }
  }, [assets, normalGifts.length]);

  useFrame((state) => {
    let s = scratchRef.current;
    if (s === null) {
      s = {
        mat: new Matrix4(),
        pos: new Vector3(),
        quat: new Quaternion(),
        scale: new Vector3(1, 1, 1),
        axis: new Vector3(0, 1, 0),
      };
      scratchRef.current = s;
    }
    const t = state.clock.elapsedTime;

    const normalMesh = normalRef.current;
    if (normalMesh) {
      let n = 0;
      for (const gi of visibleNormal) {
        const g = normalGifts[gi];
        if (!g) continue;
        const phase = (gi * 1.61803398875) % (Math.PI * 2);
        s.pos.set(g.pos.x, g.pos.y + Math.sin(t * BOB_HZ + phase) * BOB_AMP, g.pos.z);
        s.quat.setFromAxisAngle(s.axis, t * SPIN_RAD_PER_S + phase);
        s.scale.set(1, 1, 1);
        s.mat.compose(s.pos, s.quat, s.scale);
        normalMesh.setMatrixAt(n, s.mat);
        n += 1;
      }
      normalMesh.count = n;
      normalMesh.instanceMatrix.needsUpdate = true;
    }

    const goldenMesh = goldenRef.current;
    const sparkleMesh = sparkleRef.current;
    if (goldenMesh) {
      let n = 0;
      for (const gi of visibleGolden) {
        const g = normalGifts[gi];
        if (!g) continue;
        const phase = (gi * 2.39996322973) % (Math.PI * 2);
        s.pos.set(g.pos.x, g.pos.y + Math.sin(t * BOB_HZ + phase) * BOB_AMP, g.pos.z);
        s.quat.setFromAxisAngle(s.axis, t * SPIN_RAD_PER_S * 1.5 + phase);
        s.scale.set(1, 1, 1);
        s.mat.compose(s.pos, s.quat, s.scale);
        goldenMesh.setMatrixAt(n, s.mat);
        if (sparkleMesh) {
          const pulse = 0.75 + 0.45 * Math.sin(t * 3.1 + phase * 2);
          s.pos.set(
            g.pos.x,
            g.pos.y + SPARKLE_LIFT + Math.sin(t * 2.2 + phase) * 0.06,
            g.pos.z,
          );
          s.quat.setFromAxisAngle(s.axis, t * 2.4 + phase);
          s.scale.setScalar(Math.max(0.2, pulse));
          s.mat.compose(s.pos, s.quat, s.scale);
          sparkleMesh.setMatrixAt(n, s.mat);
        }
        n += 1;
      }
      goldenMesh.count = n;
      goldenMesh.instanceMatrix.needsUpdate = true;
      if (sparkleMesh) {
        sparkleMesh.count = n;
        sparkleMesh.instanceMatrix.needsUpdate = true;
      }
    }
  });

  const capacity = Math.max(1, normalGifts.length);
  return (
    <group>
      <instancedMesh
        ref={normalRef}
        args={[assets.box, assets.normalMaterial, capacity]}
        frustumCulled={false}
        dispose={null}
      />
      <instancedMesh
        ref={goldenRef}
        args={[assets.box, assets.goldenMaterial, capacity]}
        frustumCulled={false}
        dispose={null}
      />
      <instancedMesh
        ref={sparkleRef}
        args={[assets.sparkle, assets.sparkleMaterial, capacity]}
        frustumCulled={false}
        dispose={null}
      />
    </group>
  );
}

/**
 * One stand-alone gift with exactly the look and size of the instanced gifts (used by the False Gift
 * enemy while it is disguised, so it is indistinguishable from a real gift).
 */
export function GiftBox(props: { position: Vec3; taken: boolean }): JSX.Element {
  const assets = useMemo(() => createGiftAssets(), []);
  useEffect(() => () => assets.dispose(), [assets]);
  return (
    <mesh
      position={[props.position.x, props.position.y, props.position.z]}
      visible={!props.taken}
      geometry={assets.box}
      material={assets.normalMaterial}
    />
  );
}

  float warn = vStateF.x;
  float fret = vStateF.y;
  if (warn > 0.5) {

    totalEmissiveRadiance += mix(uWarnAmber, uWarnRed, b) * (1.1 + 1.6 * b);
  }
  if (fret > 0.5) {

    float v2 = 1.0 - smoothstep(0.0, 0.16, abs(sin(fp.y * 1.3 + sin(fp.x * 1.9) * 1.6)));

    float crawl = 0.65 + 0.35 * sin(uTime * 0.9 + fp.x * 0.6 + fp.y * 0.4);
    totalEmissiveRadiance += vec3(0.10, 1.0, 0.55) * vein * (0.5 + 0.5 * crawl) * 0.9;

function injectVertex(shader: { vertexShader: string }): void {
  shader.vertexShader = shader.vertexShader
    .replace("#include <common>", VERTEX_COMMON)
    .replace("#include <begin_vertex>", VERTEX_BEGIN);
}

function createSurfaceMaterial(
  kind: number,
  theme: Theme,
  icy: boolean,
  timeUniform: { value: number },
): MeshStandardMaterial {
  const isCrumble = kind === PLATFORM_CRUMBLE;
  const mat = new MeshStandardMaterial({
    color: "#ffffff",
    roughness: icy ? ICE_ROUGHNESS : TOP_ROUGHNESS,
    metalness: 0,
    emissive: "#000000",
  });
  mat.onBeforeCompile = (shader) => {
    shader.uniforms.uTime = timeUniform;
    Object.assign(shader.uniforms, commonUniforms(theme, icy));
    shader.uniforms.uCrumble = { value: isCrumble ? 1 : 0 };
    injectVertex(shader);
    shader.fragmentShader = shader.fragmentShader
      .replace("#include <common>", FRAGMENT_COMMON)
      .replace("#include <color_fragment>", SURFACE_FRAGMENT_COLOR)
      .replace("#include <emissivemap_fragment>", SURFACE_FRAGMENT_EMISSIVE);
  };
  mat.customProgramCacheKey = () => (isCrumble ? "vw-platform-crumble" : "vw-platform-normal");
  return mat;
}

function createPadMaterial(timeUniform: { value: number }): MeshStandardMaterial {

function buildGroups(platforms: readonly Platform[], theme: Theme, icy: boolean): BuiltPlatforms {
  const timeUniform = { value: 0 };
  const holder = new Group();
  const records: InstanceRecord[] = [];

  for (const kind of [PLATFORM_NORMAL, PLATFORM_CRUMBLE, PLATFORM_JUMP_PAD]) {
    const list = platforms.filter((p) => p.kind === kind);
    const geometry = new BoxGeometry(1, 1, 1);
    const material =
      kind === PLATFORM_JUMP_PAD
        ? createPadMaterial(timeUniform)
        : createSurfaceMaterial(kind, theme, icy, timeUniform);
    const count = Math.max(list.length, 1);
    const mesh = new InstancedMesh(geometry, material, count);
    mesh.count = list.length;
    mesh.frustumCulled = false; // instances are spread across the map; bounding sphere of a unit box is wrong
    mesh.instanceMatrix.setUsage(DynamicDrawUsage);
    const attr = new InstancedBufferAttribute(new Float32Array(count * 4), 4);
    attr.setUsage(DynamicDrawUsage);
    geometry.setAttribute("aState", attr);

    list.forEach((p, i) => {
      const size = platformSize(p);
      const center = platformCenter(p);
      scratchMatrix.makeScale(size[0], size[1], size[2]);
      scratchMatrix.setPosition(center[0], center[1], center[2]);
      mesh.setMatrixAt(i, scratchMatrix);
      const pillar = isPillar(size);
      attr.setXYZW(i, 0, 0, 1, pillar ? 1 : 0);
      records.push({
        id: p.id,
        center,
        size,
        phase: phaseForId(p.id),
        pillar,
        mesh,
        attr,
        index: i,
        touched: false,
        activeFrame: false,
      });
    });
    mesh.instanceMatrix.needsUpdate = true;
    attr.needsUpdate = true;
    holder.add(mesh);

}

function disposeBuilt(built: BuiltPlatforms): void {
  for (const child of [...built.holder.children]) {
    if (child instanceof InstancedMesh) {
      child.geometry.dispose();
      const material = child.material;
      if (Array.isArray(material)) {
        for (const m of material) m.dispose();
      } else {
        material.dispose();
      }
    }
  }
}

      rec.activeFrame = true;

    }
  });

// Deterministic from (platforms, seed). Density capped:
//   * at most MAX_VINES vines, hung from underside edges of walkable platforms,
//   * at most one lantern per 40 u of walkway (floor(walkwayUnits / 40)), capped at MAX_LANTERNS,
//   * lanterns only on NORMAL platforms with min horizontal span >= 2.2 u
//     (posts need stable footing; CRUMBLE collapses, JUMP_PAD tops are launch surfaces).
// walkwayUnits estimates path length as the sum of (width + depth) of non-JUMP_PAD platforms.
import type { Platform } from "@/protocol/generated";

// Mirrors web/src/protocol/generated.ts (redeclared so this module stays runtime-dependency-free for tests).
const PLATFORM_NORMAL = 0;
const PLATFORM_JUMP_PAD = 2;

export const MAX_VINES = 180;
export const MAX_LANTERNS = 24;
export const LANTERN_SPACING_UNITS = 40;
const VINE_KEEP_PROB = 0.55;
const VINE_MIN_LEN = 0.8;
const VINE_MAX_LEN = 2.3;
const VINE_INSET = 0.25;
const VINE_ANCHOR_SPACING = 2.4;
const LANTERN_INSET = 0.32;
const LANTERN_MIN_SPAN = 2.2;
const LANTERN_CANDIDATE_PROB = 0.6;

export interface VinePlacement {
  platformId: number;
  /** Anchor point on the platform's underside edge. */
  x: number;
  y: number;
  z: number;
  length: number;
  /** 0..2π, used to vary strand thickness and to allow future sway. */
  phase: number;
}

export interface LanternPlacement {
  platformId: number;
  x: number;
  /** Platform top y (the post stands on the deck). */
  y: number;
  z: number;

export interface DecorPlacement {

/** 32-bit integer hash of (a, b). Stable across runs and platforms. */
export function hash32(a: number, b: number): number {
  let h = (a >>> 0) ^ Math.imul((b >>> 0) + 0x9e3779b9, 0x85ebca6b);
  h = Math.imul(h ^ (h >>> 13), 0xc2b2ae35);
  h ^= h >>> 16;
  return h >>> 0;
}

/** mulberry32 PRNG — small, fast, deterministic. */
export function mulberry32(seed: number): () => number {
  let s = seed >>> 0;
  return () => {
    s = (s + 0x6d2b79f5) | 0;
    let t = Math.imul(s ^ (s >>> 15), 1 | s);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

export function platformFootprint(p: Platform): {
  w: number;
  h: number;
  d: number;
  cx: number;
  cz: number;
  top: number;
  bottom: number;
} {
  const w = p.bounds.max.x - p.bounds.min.x;
  const h = p.bounds.max.y - p.bounds.min.y;
  const d = p.bounds.max.z - p.bounds.min.z;
  return {
    w,
    h,
    d,
    cx: (p.bounds.min.x + p.bounds.max.x) / 2,
    cz: (p.bounds.min.z + p.bounds.max.z) / 2,
    top: p.bounds.max.y,
    bottom: p.bounds.min.y,
  };
}

export function isLanternEligible(p: Platform): boolean {
  if (p.kind !== PLATFORM_NORMAL) return false;
  const { w, d } = platformFootprint(p);
  return Math.min(w, d) >= LANTERN_MIN_SPAN;
}

/** Sum of (w + d) over non-JUMP_PAD platforms; approximates walkway length in units. */
export function walkwayUnits(platforms: readonly Platform[]): number {
  let u = 0;
  for (const p of platforms) {
    if (p.kind === PLATFORM_JUMP_PAD) continue;
    const { w, d } = platformFootprint(p);
    u += w + d;
  }
  return u;
}

/** Max number of lantern lights for a level: one per 40 u of walkway, capped. */
export function lanternBudget(platforms: readonly Platform[]): number {
  return Math.min(MAX_LANTERNS, Math.floor(walkwayUnits(platforms) / LANTERN_SPACING_UNITS));
}

export function computeDecor(platforms: readonly Platform[], seed: number): DecorPlacement {
  const vines: VinePlacement[] = [];
  const lanternCandidates: Array<{ p: Platform; sort: number }> = [];

  for (const p of platforms) {
    const { bottom } = platformFootprint(p);
    const rng = mulberry32(hash32(seed, p.id + 1));

    // Hanging vines under the four underside edges (JUMP_PAD bottoms are allowed —
    // pads survive collapse, their undersides are still static geometry).
    const edges: Array<{ axis: "x" | "z"; fixed: number; from: number; to: number }> = [
      { axis: "x", fixed: p.bounds.min.z, from: p.bounds.min.x, to: p.bounds.max.x },
      { axis: "x", fixed: p.bounds.max.z, from: p.bounds.min.x, to: p.bounds.max.x },
      { axis: "z", fixed: p.bounds.min.x, from: p.bounds.min.z, to: p.bounds.max.z },
      { axis: "z", fixed: p.bounds.max.x, from: p.bounds.min.z, to: p.bounds.max.z },
    ];
    for (const e of edges) {
      const len = e.to - e.from;
      if (len < 1.6) continue;
      const anchors = Math.floor(len / VINE_ANCHOR_SPACING);
      for (let i = 1; i <= anchors; i += 1) {
        if (vines.length >= MAX_VINES) break;
        if (rng() > VINE_KEEP_PROB) continue;
        const along = (len * i) / (anchors + 1);
        const jitter = (rng() - 0.5) * 1.1;
        const at = Math.min(Math.max(e.from + along + jitter, e.from + VINE_INSET), e.to - VINE_INSET);
        const length = VINE_MIN_LEN + rng() * (VINE_MAX_LEN - VINE_MIN_LEN);
        const phase = rng() * Math.PI * 2;
        if (e.axis === "x") {
          vines.push({ platformId: p.id, x: at, y: bottom, z: e.fixed, length, phase });
        } else {
          vines.push({ platformId: p.id, x: e.fixed, y: bottom, z: at, length, phase });
        }
      }
      if (vines.length >= MAX_VINES) break;
    }

    // Lantern candidates: stable, spacious, plain platforms.
    if (isLanternEligible(p) && rng() < LANTERN_CANDIDATE_PROB) {
      lanternCandidates.push({ p, sort: hash32(seed ^ 0x51ab, p.id + 7) });
    }
  }

  // Even spread: pick the lowest-hash candidates until the walkway budget is spent.
  lanternCandidates.sort((a, b) => a.sort - b.sort);
  const budget = lanternBudget(platforms);
  const lanterns: LanternPlacement[] = [];
  for (const c of lanternCandidates) {
    if (lanterns.length >= budget) break;
    const p = c.p;
    const fp = platformFootprint(p);
    const cornerRng = mulberry32(hash32(seed ^ 0x2b1, p.id + 13));
    const corner = Math.floor(cornerRng() * 4);
    const sx = corner % 2 === 0 ? -1 : 1;
    const sz = corner < 2 ? -1 : 1;
    lanterns.push({
      platformId: p.id,
      x: fp.cx + sx * (fp.w / 2 - LANTERN_INSET),
      y: fp.top,
      z: fp.cz + sz * (fp.d / 2 - LANTERN_INSET),
    });
  }

  return { vines, lanterns };
}
