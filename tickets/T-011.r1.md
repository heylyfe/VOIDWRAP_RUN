=== FILE: web/src/hud/format.ts ===
export type DeathCause = "void" | "enemy" | "disconnect";

const CAUSE_TEXT: Record<DeathCause, string> = {
  void: "Fell into the void",
  enemy: "Caught by an enemy",
  disconnect: "Lost connection",
};

export function causeText(cause: DeathCause): string {
  return CAUSE_TEXT[cause];
}

export function giftsLabel(remaining: number, total: number): string {
  const safeTotal = Number.isFinite(total) && total > 0 ? Math.floor(total) : 0;
  const safeRemaining = Number.isFinite(remaining) ? Math.floor(remaining) : 0;
  const clamped = Math.min(Math.max(safeRemaining, 0), safeTotal);
  return `${clamped} / ${safeTotal}`;
}

=== FILE: web/src/hud/LevelBanner.tsx ===
"use client";

import { useEffect, useState } from "react";

import type { JSX } from "react";

const FADE_MS = 1500;
const HIDE_MS = 2000;

type LevelBannerProps = {
  level: number;
};

export function LevelBanner(props: LevelBannerProps): JSX.Element | null {
  const { level } = props;
  const [visible, setVisible] = useState<boolean>(true);
  const [faded, setFaded] = useState<boolean>(false);

  useEffect(() => {
    setVisible(true);
    setFaded(false);
    const fadeTimer = window.setTimeout(() => setFaded(true), FADE_MS);
    const hideTimer = window.setTimeout(() => setVisible(false), HIDE_MS);
    return () => {
      window.clearTimeout(fadeTimer);
      window.clearTimeout(hideTimer);
    };
  }, [level]);

  if (!visible) {
    return null;
  }

  return (
    <div
      data-testid="level-banner"
      className={`pointer-events-none absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 whitespace-nowrap font-mono text-6xl font-bold tracking-widest text-white transition-opacity duration-500 [text-shadow:0_0_24px_#ff3db8,0_0_6px_#07060d] ${
        faded ? "opacity-0" : "opacity-100"
      }`}
    >
      LEVEL {level}
    </div>
  );
}

"use client";

import Link from "next/link";
import { useEffect } from "react";

import type { JSX } from "react";

import { causeText } from "./format";
import type { DeathCause } from "./format";

type RunSummaryProps = {
  levelReached: number;
  giftsCollected: number;
  cause: DeathCause;
  onRestart: () => void;
};

export function RunSummary(props: RunSummaryProps): JSX.Element {
  const { levelReached, giftsCollected, cause, onRestart } = props;

  useEffect(() => {
    const handleKeyDown = (event: KeyboardEvent): void => {
      if (event.key === "Enter") {
        event.preventDefault();
        onRestart();
      }
    };
    window.addEventListener("keydown", handleKeyDown);
    return () => {
      window.removeEventListener("keydown", handleKeyDown);
    };
  }, [onRestart]);

  return (
    <div
      data-testid="run-summary"
      role="dialog"
      aria-modal="true"
      aria-labelledby="run-summary-title"
      className="pointer-events-auto absolute inset-0 z-50 flex items-center justify-center bg-black/70 font-mono"
    >
      <div className="w-[min(24rem,90vw)] rounded-lg border border-white/15 bg-[#120f1f] p-6">
        <h2
          id="run-summary-title"
          className="text-center text-3xl font-bold tracking-widest text-[#ff3db8]"
        >
          RUN OVER
        </h2>
        <dl className="mt-6 space-y-2 text-lg">
          <div className="flex items-baseline justify-between gap-4">
            <dt className="text-white/70">Level reached</dt>
            <dd className="text-xl font-bold text-white">{levelReached}</dd>
          </div>
          <div className="flex items-baseline justify-between gap-4">
            <dt className="text-white/70">Gifts collected</dt>
            <dd className="text-xl font-bold text-[#ffd23d]">{giftsCollected}</dd>
          </div>
          <div className="flex items-baseline justify-between gap-4">
            <dt className="text-white/70">Cause</dt>
            <dd className="text-right text-base font-bold text-[#ff4d4d]">
              {causeText(cause)}
            </dd>
          </div>
        </dl>
        <button
          type="button"
          data-testid="run-again"
          autoFocus
          onClick={onRestart}
          className="mt-8 block w-full rounded bg-[#ff3db8] px-4 py-3 text-lg font-bold text-black transition-colors hover:bg-[#ff6cd0]"
        >
          Run again
        </button>
        <Link
          href="/"
          className="mt-3 block text-center text-sm text-[#38e8ff] underline hover:text-white"
        >
          Back to title
        </Link>
      </div>
    </div>
  );
}
