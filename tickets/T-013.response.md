  edgeGlowMagenta: "#ff3db8",
  giftGold: "#ffd23d",

  playerLegs: "#2b2b4a",
  playerFace: "#07060d",
} as const;

const TOP_COLOR_COUNT = PLATFORM_TOP_COLORS.length;

/**
 * Deterministically pick a platform top colour from the platform id.
 * Safe for negative ids: the result of `id % 4` is normalised into [0, 3].
 */
export function platformColor(id: number): string {
  const normalized = ((id % TOP_COLOR_COUNT) + TOP_COLOR_COUNT) % TOP_COLOR_COUNT;
  return PALETTE.platformTopColors[normalized] ?? PLATFORM_TOP_COLORS[0];
}

"use client";

import { useMemo } from "react";
import type { JSX } from "react";
import type { Platform } from "@/protocol/generated";
import { PALETTE, platformColor } from "./palette";

interface PlatformBoxData {
  id: number;
  center: [number, number, number];
  size: [number, number, number];
  top: string;
  accent: string;
}

function toBoxData(platform: Platform): PlatformBoxData {
  const { min, max } = platform.bounds;
  const size: [number, number, number] = [
    max.x - min.x,
    max.y - min.y,
    max.z - min.z,
  ];
  const center: [number, number, number] = [
    (min.x + max.x) / 2,
    (min.y + max.y) / 2,
    (min.z + max.z) / 2,
  ];
  return {
    id: platform.id,
    center,
    size,
    top: platformColor(platform.id),
    accent:
      platform.id % 2 === 0 ? PALETTE.edgeGlowCyan : PALETTE.edgeGlowMagenta,
  };
}

export function Platforms(props: {
  platforms: readonly Platform[];
}): JSX.Element {
  const boxes = useMemo(() => props.platforms.map(toBoxData), [props.platforms]);

  return (
    <group>
      {boxes.map((box) => (
        <group key={box.id} position={box.center}>
          <mesh>
            <boxGeometry args={box.size} />
            <meshStandardMaterial
              color={box.top}
              emissive={box.top}
              emissiveIntensity={0.15}
            />
          </mesh>
          <mesh position={[0, box.size[1] / 2 + 0.015, 0]}>
            <boxGeometry
              args={[
                box.size[0] + 0.06,
                0.06,
                box.size[2] + 0.06,
              ]}
            />
            <meshStandardMaterial
              color={box.accent}
              emissive={box.accent}
              emissiveIntensity={1.2}
            />
          </mesh>
        </group>
      ))}
    </group>
  );
}

import { useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import type { Group } from "three";
import type { Vec3 } from "@/protocol/generated";
import { PALETTE } from "./palette";

export function GiftBox(props: {
  position: Vec3;
  taken: boolean;
}): JSX.Element | null {
  const groupRef = useRef<Group>(null);

  useFrame((state, delta) => {
    const group = groupRef.current;
    if (group === null) {
      return;
    }
    group.rotation.y += delta * 0.9;
    group.position.y =
      props.position.y + Math.sin(state.clock.elapsedTime * 1.6) * 0.15;

  if (props.taken) {
    return null;

  return (
    <group
      ref={groupRef}
      position={[props.position.x, props.position.y, props.position.z]}

"use client";

import { useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import type { MeshStandardMaterial } from "three";
import type { Vec3 } from "@/protocol/generated";
import { PALETTE } from "./palette";

export function Beacon(props: { position: Vec3; open: boolean }): JSX.Element {
  const pillarMaterialRef = useRef<MeshStandardMaterial>(null);

  useFrame((state) => {
    const material = pillarMaterialRef.current;
    if (material === null) {
      return;
    }
    if (!props.open) {
      material.emissiveIntensity = 0;
      return;
    }
    material.emissiveIntensity =
      1.2 + Math.sin(state.clock.elapsedTime * 3) * 0.6;
  });

  const accent = props.open ? PALETTE.beaconOpen : PALETTE.beaconClosed;

  return (
    <group position={[props.position.x, props.position.y, props.position.z]}>
      <mesh position={[0, 0.1, 0]}>
        <boxGeometry args={[1.4, 0.2, 1.4]} />
        <meshStandardMaterial
          color={accent}
          emissive={accent}
          emissiveIntensity={props.open ? 0.6 : 0.1}
        />
      </mesh>
      <mesh position={[0, 3.2, 0]}>
        <boxGeometry args={[0.8, 6, 0.8]} />
        <meshStandardMaterial
          ref={pillarMaterialRef}
          color={accent}
          emissive={accent}
          emissiveIntensity={props.open ? 1.2 : 0}
          transparent
          depthWrite={false}
          opacity={props.open ? 0.35 : 0.12}
        />
      </mesh>
    </group>
  );
}

const SHARD_OFFSETS: ReadonlyArray<readonly [number, number, number]> = [
  [0.95, 0.25, 0],
  [-0.95, 0.25, 0],
  [0, 0.95, 0.35],
  [0, -0.95, -0.35],
];

export function Chaser(props: { position: Vec3 }): JSX.Element {
  const coreRef = useRef<Mesh>(null);
  const shardsRef = useRef<Group>(null);

  useFrame((state, delta) => {
    const core = coreRef.current;
    if (core !== null) {
      core.rotation.y += delta * 2.5;
    }
    const shards = shardsRef.current;
    if (shards !== null) {
      shards.rotation.y -= delta * 1.8;
      shards.rotation.x = Math.sin(state.clock.elapsedTime * 1.3) * 0.35;
    }
  });

  return (
    <group position={[props.position.x, props.position.y, props.position.z]}>
      <mesh ref={coreRef}>
        <octahedronGeometry args={[0.6]} />
        <meshStandardMaterial
          color={PALETTE.chaserRed}
          emissive={PALETTE.chaserRed}
          emissiveIntensity={1.4}
        />
      </mesh>
      <group ref={shardsRef}>
        {SHARD_OFFSETS.map((offset, index) => (
          <mesh key={index} position={offset as [number, number, number]}>
            <octahedronGeometry args={[0.14]} />
            <meshStandardMaterial

              emissive={PALETTE.chaserRed}

"use client";

import { useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import type { Group } from "three";
import { PALETTE } from "./palette";

export const AVATAR_HEIGHT = 1.8;

export function BlockyAvatar(props: {
  bodyColor?: string;
  moving?: boolean;
}): JSX.Element {
  const leftArmRef = useRef<Group>(null);
  const rightArmRef = useRef<Group>(null);
  const leftLegRef = useRef<Group>(null);
  const rightLegRef = useRef<Group>(null);

  useFrame((state, delta) => {
    const ease = Math.min(1, delta * 8);
    const swing = Math.sin(state.clock.elapsedTime * 9) * 0.6;
    const leftArm = leftArmRef.current;
    const rightArm = rightArmRef.current;
    const leftLeg = leftLegRef.current;
    const rightLeg = rightLegRef.current;
    if (props.moving) {
      if (leftArm !== null) {
        leftArm.rotation.x = swing;
      }
      if (rightArm !== null) {
        rightArm.rotation.x = -swing;
      }
      if (leftLeg !== null) {
        leftLeg.rotation.x = -swing;
      }
      if (rightLeg !== null) {
        rightLeg.rotation.x = swing;
      }
    } else {
      if (leftArm !== null) {
        leftArm.rotation.x += (0 - leftArm.rotation.x) * ease;
      }
      if (rightArm !== null) {
        rightArm.rotation.x += (0 - rightArm.rotation.x) * ease;
      }
      if (leftLeg !== null) {
        leftLeg.rotation.x += (0 - leftLeg.rotation.x) * ease;
      }
      if (rightLeg !== null) {
        rightLeg.rotation.x += (0 - rightLeg.rotation.x) * ease;
      }
    }
  });

  const body = props.bodyColor ?? PALETTE.playerBody;

  return (
    <group>
      {/* Legs: pivot groups at the hips (y = 0.72), feet at y = 0 */}
      <group ref={leftLegRef} position={[0.13, 0.72, 0]}>
        <mesh position={[0, -0.36, 0]}>
          <boxGeometry args={[0.22, 0.72, 0.24]} />
          <meshStandardMaterial color={PALETTE.playerLegs} />
        </mesh>
      </group>
      <group ref={rightLegRef} position={[-0.13, 0.72, 0]}>
        <mesh position={[0, -0.36, 0]}>
          <boxGeometry args={[0.22, 0.72, 0.24]} />
          <meshStandardMaterial color={PALETTE.playerLegs} />
        </mesh>
      </group>

      {/* Torso */}
      <mesh position={[0, 1.01, 0]}>
        <boxGeometry args={[0.5, 0.58, 0.3]} />
        <meshStandardMaterial color={body} />
      </mesh>

      {/* Arms: pivot groups at the shoulders (y = 1.22), hands in skin tone */}
      <group ref={leftArmRef} position={[0.31, 1.22, 0]}>
        <mesh position={[0, -0.19, 0]}>
          <boxGeometry args={[0.14, 0.44, 0.14]} />
          <meshStandardMaterial color={body} />
        </mesh>
        <mesh position={[0, -0.48, 0]}>
          <boxGeometry args={[0.16, 0.16, 0.16]} />
          <meshStandardMaterial color={PALETTE.playerHeadHands} />
        </mesh>
      </group>
      <group ref={rightArmRef} position={[-0.31, 1.22, 0]}>
        <mesh position={[0, -0.19, 0]}>
          <boxGeometry args={[0.14, 0.44, 0.14]} />
          <meshStandardMaterial color={body} />
        </mesh>
        <mesh position={[0, -0.48, 0]}>
          <boxGeometry args={[0.16, 0.16, 0.16]} />
          <meshStandardMaterial color={PALETTE.playerHeadHands} />
        </mesh>
      </group>

      {/* Head (skin tone), y from 1.30 to 1.80 */}
      <mesh position={[0, 1.55, 0]}>
        <boxGeometry args={[0.5, 0.5, 0.42]} />
        <meshStandardMaterial color={PALETTE.playerHeadHands} />
      </mesh>

      {/* Face features on the -Z (forward) face of the head */}
      <mesh position={[0.12, 1.61, -0.215]}>
        <boxGeometry args={[0.07, 0.09, 0.03]} />
        <meshStandardMaterial color={PALETTE.playerFace} />
      </mesh>
      <mesh position={[-0.12, 1.61, -0.215]}>
        <boxGeometry args={[0.07, 0.09, 0.03]} />
        <meshStandardMaterial color={PALETTE.playerFace} />
      </mesh>
      <mesh position={[0, 1.45, -0.215]}>
        <boxGeometry args={[0.2, 0.05, 0.03]} />
        <meshStandardMaterial color={PALETTE.playerFace} />
      </mesh>
    </group>
  );
}

import { describe, expect, it } from "vitest";
import { PALETTE, platformColor } from "./palette";

describe("platformColor", () => {
  it("is deterministic for the same id", () => {
    for (let id = -120; id < 120; id++) {
      expect(platformColor(id)).toBe(platformColor(id));
    }
  });

  it("always returns one of the four platform top colours", () => {
    for (let id = -400; id < 400; id += 7) {
      expect(PALETTE.platformTopColors).toContain(platformColor(id));
    }
  });

  it("cycles with period 4", () => {
    expect(platformColor(0)).toBe(platformColor(4));
    expect(platformColor(1)).toBe(platformColor(5));
    expect(platformColor(2)).toBe(platformColor(6));
    expect(platformColor(3)).toBe(platformColor(7));
  });

  it("handles negative ids like their positive equivalents", () => {
    expect(platformColor(-1)).toBe(platformColor(3));
    expect(platformColor(-2)).toBe(platformColor(2));
    expect(platformColor(-3)).toBe(platformColor(1));
    expect(platformColor(-4)).toBe(platformColor(0));
  });

  it("handles large ids", () => {
    expect(platformColor(1_000_000_001)).toBe(platformColor(1));
    expect(platformColor(2 ** 40)).toBe(platformColor(0));
    expect(platformColor(Number.MAX_SAFE_INTEGER)).toBe(
      platformColor(Number.MAX_SAFE_INTEGER - 4),
    );
  });
});

=== NOTES ===
- `platformColor` uses `((id % 4) + 4) % 4` so negative ids map to the same colour as their positive equivalent; a `?? PLATFORM_TOP_COLORS[0]` fallback satisfies `noUncheckedIndexedAccess` without a non-null assertion.
- Platforms: box centre/size computed from `bounds.min`/`bounds.max` in a `useMemo` keyed on `platforms`; edge strip is a slightly-larger 0.06-tall emissive box overlapping the top face, colour alternating cyan/magenta by id parity (parity is well-defined for negatives).
- GiftBox/Beacon/Chaser/BlockyAvatar: all `useFrame` callbacks only mutate `.rotation`/`.position`/`.emissiveIntensity` on refs — no `new`, no object/array literals, no allocations per frame. Hooks are called unconditionally before GiftBox's `taken` early-return.
- Avatar: legs 0.72 + torso 0.58 + head 0.5 = AVATAR_HEIGHT 1.8; feet at group origin y=0; shoulders ±0.31 → total width 0.78 ≤ 0.8; max depth 0.42 ≤ 0.45; eyes/mouth thin boxes protruding on the −Z face. Limb swing ±0.6 rad around X at hip/shoulder pivots, arms in anti-phase to legs; when `moving` is false, rotations ease to 0 with `delta`-scaled lerp.
- Beacon drives `emissiveIntensity` via a material ref (pulsing when open, forced to 0 when closed); colour/opacity differences are declarative props so toggling `open` re-renders correctly.
- All seven files created; no other files touched, no new dependencies, no default exports, `"use client"` on every `.tsx`. Believed to satisfy all acceptance criteria; visual check left to the OVERSEER since I have no runtime here.
