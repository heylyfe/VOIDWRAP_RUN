```tsx
"use client";

import { useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import type { Group } from "three";
import { PALETTE } from "./palette";

export const AVATAR_HEIGHT = 1.8;

export function BlockyAvatar(props: {
  bodyColor?: string;
  moving?: boolean;
}): JSX.Element {
  const leftArmRef = useRef<Group>(null);
  const rightArmRef = useRef<Group>(null);
  const leftLegRef = useRef<Group>(null);
  const rightLegRef = useRef<Group>(null);

  useFrame((state, delta) => {
    const ease = Math.min(1, delta * 8);
    const swing = Math.sin(state.clock.elapsedTime * 9) * 0.6;
    const leftArm = leftArmRef.current;
    const rightArm = rightArmRef.current;
    const leftLeg = leftLegRef.current;
    const rightLeg = rightLegRef.current;
    if (props.moving) {
      if (leftArm !== null) {
        leftArm.rotation.x = swing;
      }
      if (rightArm !== null) {
        rightArm.rotation.x = -swing;
      }
      if (leftLeg !== null) {
        leftLeg.rotation.x = -swing;
      }
      if (rightLeg !== null) {
        rightLeg.rotation.x = swing;
      }
    } else {
      if (leftArm !== null) {
        leftArm.rotation.x += (0 - leftArm.rotation.x) * ease;
      }
      if (rightArm !== null) {
        rightArm.rotation.x += (0 - rightArm.rotation.x) * ease;
      }
      if (leftLeg !== null) {
        leftLeg.rotation.x += (0 - leftLeg.rotation.x) * ease;
      }
      if (rightLeg !== null) {
        rightLeg.rotation.x += (0 - rightLeg.rotation.x) * ease;
      }
    }
  });

  const body = props.bodyColor ?? PALETTE.playerBody;

  return (
    <group>
      {/* Legs: pivot groups at the hips (y = 0.72), feet at y = 0 */}
      <group ref={leftLegRef} position={[0.13, 0.72, 0]}>
        <mesh position={[0, -0.36, 0]}>
          <boxGeometry args={[0.22, 0.72, 0.24]} />
          <meshStandardMaterial color={PALETTE.playerLegs} />
        </mesh>
      </group>
      <group ref={rightLegRef} position={[-0.13, 0.72, 0]}>
        <mesh position={[0, -0.36, 0]}>
          <boxGeometry args={[0.22, 0.72, 0.24]} />
          <meshStandardMaterial color={PALETTE.playerLegs} />
        </mesh>
      </group>

      {/* Torso */}
      <mesh position={[0, 1.01, 0]}>
        <boxGeometry args={[0.5, 0.58, 0.3]} />
        <meshStandardMaterial color={body} />
      </mesh>

      {/* Arms: pivot groups at the shoulders (y = 1.22), hands in skin tone */}
      <group ref={leftArmRef} position={[0.31, 1.22, 0]}>
        <mesh position={[0, -0.19, 0]}>
          <boxGeometry args={[0.14, 0.44, 0.14]} />
          <meshStandardMaterial color={body} />
        </mesh>
        <mesh position={[0, -0.48, 0]}>
          <boxGeometry args={[0.16, 0.16, 0.16]} />
          <meshStandardMaterial color={PALETTE.playerHeadHands} />
        </mesh>
      </group>
      <group ref={rightArmRef} position={[-0.31, 1.22, 0]}>
        <mesh position={[0, -0.19, 0]}>
          <boxGeometry args={[0.14, 0.44, 0.14]} />
          <meshStandardMaterial color={body} />
        </mesh>
        <mesh position={[0, -0.48, 0]}>
          <boxGeometry args={[0.16, 0.16, 0.16]} />
          <meshStandardMaterial color={PALETTE.playerHeadHands} />
        </mesh>
      </group>

      {/* Head (skin tone), y from 1.30 to 1.80 */}
      <mesh position={[0, 1.55, 0]}>
        <boxGeometry args={[0.5, 0.5, 0.42]} />
        <meshStandardMaterial color={PALETTE.playerHeadHands} />
      </mesh>

      {/* Face features on the -Z (forward) face of the head */}
      <mesh position={[0.12, 1.61, -0.215]}>
        <boxGeometry args={[0.07, 0.09, 0.03]} />
        <meshStandardMaterial color={PALETTE.playerFace} />
      </mesh>
      <mesh position={[-0.12, 1.61, -0.215]}>
        <boxGeometry args={[0.07, 0.09, 0.03]} />
        <meshStandardMaterial color={PALETTE.playerFace} />
      </mesh>
      <mesh position={[0, 1.45, -0.215]}>
        <boxGeometry args={[0.2, 0.05, 0.03]} />
        <meshStandardMaterial color={PALETTE.playerFace} />
      </mesh>
    </group>
  );
}
```

## Task
1. New `web/src/scene/avatarPose.ts` (pure, no three/react imports):
   ```ts
   export interface AvatarMotion { speed: number; vy: number; grounded: boolean; dashing: boolean }
   export interface AvatarPose {
     leftArm: number; rightArm: number;          // pitch (rad), + = forward swing
     leftArmRoll: number; rightArmRoll: number;   // roll outward (rad), for jump/fall flail
     leftLeg: number; rightLeg: number;           // pitch (rad)
     torsoPitch: number;                          // lean forward (+)
     bodyY: number;                               // vertical offset of the whole body (bob / squash compensation)
     scaleY: number; scaleXZ: number;             // squash & stretch
   }
   export interface PoseState { phase: number; landTimer: number; wasGrounded: boolean; pose: AvatarPose }
   export function createPoseState(): PoseState;
   /** Advance animation by dt seconds toward the target pose for this motion; mutates and returns state.pose. */
   export function stepPose(state: PoseState, m: AvatarMotion, dt: number): AvatarPose;
   ```
   Behaviour (targets; blend current pose toward target with exponential smoothing, time constant ~60 ms; squash
   values blend faster):
   - Idle (grounded, speed < 0.5): arms/legs ~0, subtle breathing bob (bodyY +-0.02 at ~1.2 Hz).
   - Run (grounded, speed >= 0.5): cadence phase advances at ~(speed / 9) * 2.6 cycles/s; legs swing +-0.75 rad and
     arms +-0.6 rad in opposite phase, amplitude scaled by min(speed/9, 1); torsoPitch 0.12 at full speed; bodyY bob
     |sin| * 0.06 (two bounces per cycle).
   - Jump rising (not grounded, vy > 1): arms raised forward-up (~-2.4 rad pitch i.e. overhead) with slight outward
     roll, legs tucked (front leg +0.6, back leg -0.3), torsoPitch 0.05, slight stretch (scaleY 1.06, scaleXZ 0.96).
   - Apex / falling (not grounded, vy <= 1): arms out to the sides (roll ~1.1 rad) and flailing gently (+-0.25 rad at
     ~4 Hz), legs apart (+0.35 / -0.35); when vy < -14 add more flail amplitude.
   - Land: on the frame grounded becomes true after being false, start a 0.12 s squash: scaleY 0.82 -> 1, scaleXZ
     1.12 -> 1, bodyY dips -0.08 -> 0 (ease-out).
   - Dash (dashing true, any grounding): strong forward lean torsoPitch 0.55, both arms swept back (about 1.3 rad, i.e.
     behind the body, pick the sign consistent with your rig), legs trailing (-0.5 / -0.2), stretch scaleY 0.92,
     scaleXZ 1.1. Dash pose overrides others while active.
   All outputs finite for any finite input (clamp dt to [0, 0.1]).
