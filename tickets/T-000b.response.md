/**
 * Clamp a number to the inclusive range [lo, hi].
 *
 * - If `lo > hi`, throws a RangeError (invalid range).
 * - If `v` is NaN, returns `lo`.
 *
 * @param v - Value to clamp.
 * @param lo - Lower bound (inclusive).
 * @param hi - Upper bound (inclusive).
 * @returns `v` limited to [lo, hi].
 */
export function clamp(v: number, lo: number, hi: number): number {
  if (lo > hi) {
    throw new RangeError(
      `clamp: invalid range [${lo}, ${hi}]: lo must not be greater than hi`,
    );
  }
  if (Number.isNaN(v)) {
    return lo;
  }
  if (v < lo) {
    return lo;
  }
  if (v > hi) {
    return hi;
  }
  return v;
}
