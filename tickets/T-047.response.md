// Lobby shop helpers: resolve the wire offer against design/content.json and
// work out whether a purchase is allowed. Pure and unit-tested.
import type { ShopItem } from '../protocol/generated';
import { SHOP } from '../content';

export interface ResolvedShopItem {
  id: string;
  name: string;
  flavor: string;
  effect: string;
  /** Cost from the wire (server authoritative), not from the manifest. */
  cost: number;
  /** Stock from the wire (server authoritative). */
  stock: number;
}

/**
 * Merge the wire offer (id, cost, stock) with the content manifest (name,
 * flavor, effect). Unknown ids are skipped; wire cost and stock always win.
 */
export function resolveShopItems(items: readonly ShopItem[]): ResolvedShopItem[] {
  const resolved: ResolvedShopItem[] = [];
  for (const item of items) {
    const details = SHOP.find((entry) => entry.id === item.id);
    if (!details) {
      continue;
    }
    resolved.push({
      id: item.id,
      name: details.name,
      flavor: details.flavor,
      effect: details.effect,
      cost: item.cost,
      stock: item.stock,
    });
  }
  return resolved;
}

/**
 * Why the Buy button is disabled, as plain text; null when the purchase is
 * allowed. Sold out wins over an empty purse.
 */
export function buyBlockedReason(item: { cost: number; stock: number }, currency: number): string | null {
  if (item.stock <= 0) {
    return 'Sold out';
  }
  if (currency < item.cost) {
    return `Need ${item.cost - currency} more gold`;
  }
  return null;
}

// Pure notice helpers: what a notice says, and how visible it is right now.
// Unit-tested; never throws.
import type { Notice } from '../game/store';
import { NOTICE_ALTAR, NOTICE_BOUGHT, NOTICE_EVENT } from '../protocol/generated';
import { CARD_CURSE, CARD_UPGRADE, EVENTS, SHOP, lookupCard } from '../content';

export const NOTICE_LIFETIME_MS = 4000;
export const NOTICE_FADE_MS = 500;

export interface NoticeText {
  title: string;
  body: string;
}

/**
 * Resolve a notice to plain text. Returns null for unknown kinds, unknown
 * content ids, or malformed notices; never throws.
 */
export function noticeText(notice: Notice): NoticeText | null {
  try {
    if (notice.kind === NOTICE_ALTAR) {
      const [upgradeId, curseId] = notice.ids;
      if (!upgradeId || !curseId) {
        return null;
      }
      const upgrade = lookupCard(CARD_UPGRADE, upgradeId);
      const curse = lookupCard(CARD_CURSE, curseId);
      if (upgrade === null || upgrade.kind !== CARD_UPGRADE) {
        return null;
      }
      if (curse === null || curse.kind !== CARD_CURSE) {
        return null;
      }
      return {
        title: 'THE ALTAR',
        body: `The altar gave you ${upgrade.card.name} and ${curse.card.name}.`,
      };
    }
    if (notice.kind === NOTICE_BOUGHT) {
      const [itemId] = notice.ids;
      if (!itemId) {
        return null;
      }
      const item = SHOP.find((entry) => entry.id === itemId);
      if (!item) {
        return null;
      }
      return { title: 'SHOP', body: `Bought ${item.name}.` };
    }
    if (notice.kind === NOTICE_EVENT) {
      const [eventId] = notice.ids;
      if (!eventId) {
        return null;
      }
      const event = EVENTS.find((entry) => entry.id === eventId);
      if (!event) {
        return null;
      }
      return { title: event.name, body: event.banner };
    }
    return null;
  } catch {
    return null;
  }
}

/**
 * Opacity for a notice of the given age. Opaque for the first
 * NOTICE_LIFETIME_MS - NOTICE_FADE_MS, then a linear fade to 0. Future or
 * invalid ages are clamped; non-finite input hides the notice.
 */
export function noticeOpacity(ageMs: number): number {
  if (!Number.isFinite(ageMs)) {
    return 0;
  }
  if (ageMs <= NOTICE_LIFETIME_MS - NOTICE_FADE_MS) {
    return 1;
  }
  if (ageMs >= NOTICE_LIFETIME_MS) {
    return 0;
  }
  return 1 - (ageMs - (NOTICE_LIFETIME_MS - NOTICE_FADE_MS)) / NOTICE_FADE_MS;
}

  text: NoticeText;
  opacity: number;

  // which renders every notice fully opaque until the first frame ticks.

// Pure parsing for the join-a-friend address field. Accepts a bare address
// or host:port (scheme optional, default http), or an explicit http(s) URL.
// Ports default to the game port; output is always scheme://host:port.
export const DEFAULT_JOIN_PORT = 7879;

const SCHEME_PREFIX = /^([a-zA-Z][a-zA-Z0-9+.-]*):\/\//;
const SCHEME_LIKE = /^[a-zA-Z][a-zA-Z0-9+.-]*:/;
const HOST_LABEL = /^[a-zA-Z0-9]([a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?$/;
const IPV4 = /^\d+(\.\d+){3}$/;

function isValidHost(host: string): boolean {
  if (host.length === 0 || host.length > 253) {
    return false;
  }
  const labels = host.split('.');
  for (const label of labels) {
    if (!HOST_LABEL.test(label)) {
      return false;
    }
  }
  if (IPV4.test(host)) {
    for (const label of labels) {
      const octet = Number.parseInt(label, 10);
      if (octet > 255) {
        return false;
      }
    }
  }
  return true;
}

/**
 * Parse a friend's address into scheme://host:port, or null when it is not
 * an http(s) host[:port]. Surrounding whitespace is trimmed; any inner
 * whitespace, path, query, fragment, userinfo, IPv6 literal or non-http(s)
 * scheme is rejected. A missing port defaults to DEFAULT_JOIN_PORT.
 */
export function parseJoinAddress(text: string): string | null {
  const trimmed = text.trim();
  if (trimmed.length === 0) {
    return null;
  }
  if (/\s/.test(trimmed)) {
    return null;
  }

  let scheme = 'http';
  let rest = trimmed;
  const schemeMatch = SCHEME_PREFIX.exec(trimmed);
  if (schemeMatch) {
    const candidate = (schemeMatch[1] ?? '').toLowerCase();
    if (candidate !== 'http' && candidate !== 'https') {
      return null;
    }
    scheme = candidate;
    rest = trimmed.slice(schemeMatch[0].length);
  } else if (SCHEME_LIKE.test(trimmed)) {
    // Has a scheme-like prefix but is not http(s)://.
    return null;
  }

  if (rest.length === 0) {
    return null;
  }
  // No path, query, fragment, userinfo or bracketed literals.
  if (/[/?#@\\\[\]]/.test(rest)) {
    return null;
  }

  const colons = (rest.match(/:/g) ?? []).length;
  if (colons > 1) {
    return null;
  }

  let host = rest.toLowerCase();
  let port = DEFAULT_JOIN_PORT;
  if (colons === 1) {
    const splitAt = rest.lastIndexOf(':');
    host = rest.slice(0, splitAt).toLowerCase();
    const portText = rest.slice(splitAt + 1);
    if (!/^[0-9]{1,5}$/.test(portText)) {
      return null;
    }
    port = Number.parseInt(portText, 10);
    if (port < 1 || port > 65535) {
      return null;
    }
  }

  if (!isValidHost(host)) {
    return null;
  }
  if (IPV4.test(host)) {
    host = host
      .split('.')
      .map((label) => String(Number.parseInt(label, 10)))
      .join('.');
  }
  return `${scheme}://${host}:${port}`;
}

// Pure helpers for the hosting panel: the shape the game program answers on
// GET/POST /api/host, and how to show an address. Unit-tested.
export type UpnpState = 'off' | 'trying' | 'open' | 'failed';

export interface HostStatus {
  hosting: boolean;
  port: number;
  lan: string[];
  public: string | null;
  upnp: UpnpState;
  message: string;
}

/** Validate an unknown value as a HostStatus; null when anything is off. */
export function parseHostResponse(value: unknown): HostStatus | null {
  if (typeof value !== 'object' || value === null) {
    return null;
  }
  const record = value as Record<string, unknown>;
  const { hosting, port, lan, public: publicAddress, upnp, message } = record;

  if (typeof hosting !== 'boolean') {
    return null;
  }
  if (typeof port !== 'number' || !Number.isFinite(port) || port < 1 || port > 65535) {
    return null;
  }
  if (!Array.isArray(lan)) {
    return null;
  }
  const lanList: string[] = [];
  for (const item of lan) {
    if (typeof item !== 'string') {
      return null;
    }
    lanList.push(item);
  }
  if (!(typeof publicAddress === 'string' || publicAddress === null)) {
    return null;
  }
  if (upnp !== 'off' && upnp !== 'trying' && upnp !== 'open' && upnp !== 'failed') {
    return null;
  }
  if (typeof message !== 'string') {
    return null;
  }
  return { hosting, port, lan: lanList, public: publicAddress, upnp, message };
}

/**
 * Show an address the way a friend should type it. Bare addresses get the
 * http scheme and the given port; an already absolute http(s) URL is used
 * as-is. Trailing slashes are stripped.
 */
export function hostUrl(address: string, port: number): string {
  const trimmed = address.trim().replace(/\/+$/, '');
  if (/^https?:\/\//i.test(trimmed)) {
    return trimmed;
  }
  return `http://${trimmed}:${port}`;
}

async function copyText(text: string): Promise<boolean> {
  try {
    if (
      typeof navigator !== 'undefined' &&
      navigator.clipboard &&
      typeof navigator.clipboard.writeText === 'function'
    ) {
      await navigator.clipboard.writeText(text);
      return true;
    }
  } catch {
    // Fall through to the legacy path (plain http origins may lack clipboard access).
  }
  try {
    const textarea = document.createElement('textarea');
    textarea.value = text;
    textarea.setAttribute('readonly', '');
    textarea.style.position = 'fixed';
    textarea.style.top = '0';
    textarea.style.left = '0';
    textarea.style.opacity = '0';
    document.body.appendChild(textarea);
    textarea.select();
    const ok = document.execCommand('copy');
    document.body.removeChild(textarea);
    return ok;
  } catch {
    return false;
  }
}

  shop: ShopItem[] | null;
  onBuy: (itemId: string) => void;
  onSubmit: (picks: number[]) => void;
}

function DraftScreenInner(props: DraftScreenProps): JSX.Element {
  const {
    cards,
    enemyPicks,
    rewardPicks,
    nextLevel,
    currency,
    lives,
    waiting,
    classId,
    onChooseClass,
    shop,
    onBuy,
    onSubmit,
  } = props;

  const enemyCount = cards.filter((c) => c.kind === CARD_ENEMY).length;
  const rewardCount = cards.length - enemyCount;
  // Never allow more picks than there are cards in a group.
  const enemyLimit = Math.min(enemyPicks, enemyCount);
  const rewardLimit = Math.min(rewardPicks, rewardCount);

  const [selected, setSelected] = useState<readonly number[]>([]);

  const toggle = useCallback(
    (index: number) => {
      if (waiting || cards.length === 0) {
        return;
      }
      setSelected((prev) => toggleSelection(prev, index, cards, enemyLimit, rewardLimit));
    },
    [cards, enemyLimit, rewardLimit, waiting],
  );

  const complete = selectionComplete(selected, cards, enemyLimit, rewardLimit);

  const submit = useCallback(() => {
    if (waiting || !complete) {
      return;
    }
    onSubmit([...selected].sort((a, b) => a - b));
  }, [complete, onSubmit, selected, waiting]);

  useEffect(() => {
    if (waiting) {
      return;
    }
    const onKeyDown = (event: KeyboardEvent): void => {
      if (event.repeat) {
        return;
      }
      if (event.ctrlKey || event.metaKey || event.altKey) {
        return;
      }
      if (event.key === 'Enter') {
        // A focused Continue button handles its own click via Enter.
        if (event.target instanceof HTMLElement && event.target.tagName === 'BUTTON') {
          return;
        }
        submit();
        return;
      }
      const digit = Number.parseInt(event.key, 10);
      if (!Number.isNaN(digit) && digit >= 1 && digit <= 9) {
        toggle(digit - 1);
      }
    };
    window.addEventListener('keydown', onKeyDown);
    return () => {
      window.removeEventListener('keydown', onKeyDown);
    };
  }, [submit, toggle, waiting]);

  const enemySelected = selected.filter((i) => i >= 0 && i < cards.length && cards[i]?.kind === CARD_ENEMY).length;
  const rewardSelected = selected.length - enemySelected;
  const firstRound = cards.length === 0;

  return (
    <div
      data-testid="draft-screen"
      className="pointer-events-auto fixed inset-0 z-40 overflow-y-auto bg-black/55 font-mono backdrop-blur-sm"
    >
      <div className="mx-auto flex min-h-full w-full max-w-4xl flex-col gap-5 px-4 py-6 text-white">
        <header className="flex flex-wrap items-baseline justify-between gap-2 border-b border-white/10 pb-3">
          <h1 className="text-2xl font-bold tracking-[0.3em]">LEVEL {nextLevel}</h1>
          <div className="flex items-center gap-4 text-sm">
            <span className="flex items-center gap-1" title="currency">
              <span aria-hidden="true" style={{ color: GOLD }}>●</span>
              <span className="text-white/90">{currency}</span>
            </span>
            <span style={{ color: DANGER }}>❤ {lives}</span>
          </div>
        </header>

        {firstRound ? (
          <>
            <section data-testid="draft-first-round" className="flex flex-col gap-2 py-2 text-center">
              <h2 className="text-lg font-bold tracking-[0.25em] text-white/90">A CLEAN START</h2>
              <p className="text-sm text-white/75">
                No hunters, no upgrades, no curses this round. Collect every gift and reach the beacon.
              </p>
              <p className="text-xs tracking-widest" style={{ color: MUTED }}>
                After this level you vote on who hunts you. Pick a class, then press CONTINUE (or Enter).
              </p>
            </section>
            <ClassPicker selected={classId} onChoose={onChooseClass} disabled={waiting} />
          </>
        ) : (
          <p className="text-xs tracking-widest" style={{ color: MUTED }}>
            {enemySelected}/{enemyLimit} hunters · {rewardSelected}/{rewardLimit} rewards · keys 1-9 toggle · enter continues
          </p>
        )}

        {enemyCount > 0 && (
          <section className="flex flex-col gap-3">
            <h2 className="text-sm font-bold tracking-[0.25em] text-white/85">
              WHO HUNTS YOU <span style={{ color: DANGER }}>— choose {enemyPicks}</span>
            </h2>
            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-4">
              {cards.map((card, index) =>
                card.kind === CARD_ENEMY ? (
                  <span
                    key={`${index}:${card.id}`}
                    className={waiting ? 'block min-w-0 pointer-events-none opacity-60' : 'block min-w-0'}
                  >
                    <DraftCard
                      kind={card.kind}
                      id={card.id}
                      selected={selected.includes(index)}
                      hotkey={index + 1}
                      onToggle={() => toggle(index)}
                    />
                  </span>
                ) : null,
              )}
            </div>
          </section>
        )}

        {rewardCount > 0 && (
          <section className="flex flex-col gap-3">
            <h2 className="text-sm font-bold tracking-[0.25em] text-white/85">
              TAKE ONE FROM THE VOID <span style={{ color: MAGENTA }}>— choose {rewardPicks}</span>
            </h2>
            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-4">
              {cards.map((card, index) =>
                card.kind === CARD_ENEMY ? null : (
                  <span
                    key={`${index}:${card.id}`}
                    className={waiting ? 'block min-w-0 pointer-events-none opacity-60' : 'block min-w-0'}
                  >
                    <DraftCard
                      kind={card.kind}
                      id={card.id}
                      selected={selected.includes(index)}
                      hotkey={index + 1}
                      onToggle={() => toggle(index)}
                    />
                  </span>
                ),
              )}
            </div>
          </section>
        )}

        {shop !== null && (
          <ShopPanel items={shop} currency={currency} disabled={waiting} onBuy={onBuy} />
        )}

        <footer className="mt-auto flex flex-col items-center gap-2 pt-4">
          {waiting ? (
            <div
              data-testid="draft-waiting"
              className="rounded-lg border border-white/15 bg-[#120f1f] px-6 py-3 text-sm tracking-[0.2em] text-white/70"
            >
              Waiting for other players...
            </div>
          ) : (
            <button
              type="button"
              data-testid="draft-continue"
              disabled={!complete}
              onClick={submit}
              className={[
                'rounded-lg border px-8 py-3 text-sm font-bold tracking-[0.3em] transition',
                complete
                  ? 'border-[#ff3db8] text-[#ff3db8] hover:bg-[#ff3db8]/15 focus-visible:ring-2 focus-visible:ring-[#ff3db8]/60 focus:outline-none'
                  : 'cursor-not-allowed border-white/15 text-white/30',
              ].join(' ')}
            >
              CONTINUE
            </button>
          )}
        </footer>
      </div>
    </div>
  );
}

export function DraftScreen(props: DraftScreenProps): JSX.Element {
  // Key the inner component by the card manifest so a fresh draft (new cards
  // array with new content) resets the selection without setState in effects.
  const cardsKey = props.cards.map((c) => `${c.kind}:${c.id}`).join('|');
  return <DraftScreenInner key={`draft:${cardsKey}`} {...props} />;
}
