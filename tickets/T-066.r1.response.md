// Matchmaker (lobby browser) client: types, payload validation, cold-start
// timings and the version-filtered fetch. Pure and unit-tested; the React
// components live in MatchList.tsx. Nothing here touches the DOM.
import { parseJoinAddress } from './join';

/** Keep in sync with web/package.json "version"; the matchmaker filters entries exactly. */
export const CLIENT_VERSION = '0.4.0-alpha.1';

/** Slow first fetch: after this long without a response we show "Waking server…". */
export const MATCHMAKER_WAKE_AFTER_MS = 2_500;
/** No error is surfaced before this long; the matchmaker may be cold-starting. */
export const MATCHMAKER_ERROR_AFTER_MS = 90_000;

/** Shape of one entry in the frozen GET /matches?version=<str> contract. */
export interface MatchEntry {
  name: string;
  address: string;
  players: number;
  max_players: number;
  level: string;
  phase: string;
  joinable: boolean;
  version: string;
}

/**
 * Base URL of the matchmaker from NEXT_PUBLIC_MATCHMAKER_URL, or null when
 * it is unset/blank (the lobby browser then renders nothing and the manual
 * join form below it still works). Trailing slashes are stripped.
 */
export function matchmakerBaseUrl(): string | null {
  const raw = process.env.NEXT_PUBLIC_MATCHMAKER_URL;
  if (typeof raw !== 'string') {
    return null;
  }
  const trimmed = raw.trim().replace(/\/+$/, '');
  return trimmed.length === 0 ? null : trimmed;
}

function parseMatchEntry(value: unknown): MatchEntry | null {
  if (typeof value !== 'object' || value === null) {
    return null;
  }
  const record = value as Record<string, unknown>;
  const { name, address, players, max_players, level, phase, joinable, version } = record;
  if (typeof name !== 'string' || name.trim().length === 0) {
    return null;
  }
  if (typeof address !== 'string' || address.trim().length === 0) {
    return null;
  }
  if (typeof players !== 'number' || !Number.isFinite(players) || players < 0) {
    return null;
  }
  if (typeof max_players !== 'number' || !Number.isFinite(max_players) || max_players < 0) {
    return null;
  }
  if (typeof level !== 'string') {
    return null;
  }
  if (typeof phase !== 'string') {
    return null;
  }
  if (typeof joinable !== 'boolean') {
    return null;
  }
  if (typeof version !== 'string') {
    return null;
  }
  return { name, address, players, max_players, level, phase, joinable, version };
}

/**
 * Validate a /matches payload. Returns null when the body is not an array;
 * otherwise returns the well-formed entries and drops individual malformed
 * ones so one bad row cannot blank the whole list.
 */
export function parseMatchList(value: unknown): MatchEntry[] | null {
  if (!Array.isArray(value)) {
    return null;
  }
  const entries: MatchEntry[] = [];
  for (const item of value) {
    const entry = parseMatchEntry(item);
    if (entry !== null) {
      entries.push(entry);
    }
  }
  return entries;
}

/**
 * The join path for a listed address: the existing parseJoinAddress with
 * the /play/ suffix appended, exactly like the manual JOIN form navigates.
 * Returns null when the address is unusable.
 */
export function resolveJoinUrl(address: string): string | null {
  const parsed = parseJoinAddress(address);
  return parsed === null ? null : `${parsed}/play/`;
}

/**
 * GET <baseUrl>/matches?version=<version>. Rejects on HTTP errors and on
 * payloads that are not an array of entries. Pass an AbortSignal to cancel.
 */
export async function fetchMatches(
  baseUrl: string,
  version: string,
  signal?: AbortSignal,
  fetchImpl: typeof fetch = fetch,
): Promise<MatchEntry[]> {
  const url = `${baseUrl}/matches?version=${encodeURIComponent(version)}`;
  const response = await fetchImpl(url, { signal, headers: { Accept: 'application/json' } });
  if (!response.ok) {
    throw new Error(`matchmaker responded with HTTP ${response.status}`);
  }
  const body: unknown = await response.json();
  const entries = parseMatchList(body);
  if (entries === null) {
    throw new Error('matchmaker returned an unexpected payload');
  }
  return entries;
}

/**
 * Fetches /matches on mount and owns the timers. Refresh on reopen comes
 * from unmount/remount of the panel; no polling is ever scheduled.
 */
export function MatchList({ onJoin }: { onJoin: (address: string) => void }): JSX.Element | null {
  const baseUrl = matchmakerBaseUrl();
  const [state, setState] = useState<MatchListState>({ status: 'loading', waking: false });

  useEffect(() => {
    if (baseUrl === null) {
      return undefined;
    }
    const startedAt = Date.now();
    const controller = new AbortController();
    let failTimer: ReturnType<typeof setTimeout> | undefined;

    setState({ status: 'loading', waking: false });

    const wakeTimer = setTimeout(() => {
      setState((current) => (current.status === 'loading' ? { ...current, waking: true } : current));
    }, MATCHMAKER_WAKE_AFTER_MS);

    // If the fetch hangs, abort at the error threshold; the rejection funnels
    // into the same failure path and surfaces the error after ~90 s.
    const deadlineTimer = setTimeout(() => {
      controller.abort();
    }, MATCHMAKER_ERROR_AFTER_MS);

    const clearAll = (): void => {
      clearTimeout(wakeTimer);
      clearTimeout(deadlineTimer);
      if (failTimer !== undefined) {
        clearTimeout(failTimer);
      }
    };

    fetchMatches(baseUrl, CLIENT_VERSION, controller.signal).then(
      (entries) => {
        clearAll();
        setState({ status: 'ready', entries });
      },
      () => {
        clearAll();
        // Hold a hard floor of ~90 s from open before any error is shown;
        // until then show the cold-start state.
        setState((current) =>
          current.status === 'loading' ? { ...current, waking: true } : current,
        );
        const remaining = MATCHMAKER_ERROR_AFTER_MS - (Date.now() - startedAt);
        if (remaining <= 0) {
          setState({ status: 'error' });
          return;
        }
        failTimer = setTimeout(() => {
          setState({ status: 'error' });
        }, remaining);
      },
    );

    return () => {
      controller.abort();
      clearAll();
    };
  }, [baseUrl]);

  if (baseUrl === null) {
    return null;
  }
  return <MatchListView state={state} onJoin={onJoin} />;
}

import { afterEach, describe, expect, it } from 'vitest';
import {
  CLIENT_VERSION,
  MATCHMAKER_ERROR_AFTER_MS,
  MATCHMAKER_WAKE_AFTER_MS,
  fetchMatches,
  matchmakerBaseUrl,
  parseMatchList,
  resolveJoinUrl,
} from './matchmaker';

const ORIGINAL_ENV = process.env.NEXT_PUBLIC_MATCHMAKER_URL;

const validMatch = {
  name: 'Alpha Keep',
  address: '203.0.113.10:7879',
  players: 3,
  max_players: 8,
  level: 'Neon Ruins',
  phase: 'Lobby',
  joinable: true,
  version: CLIENT_VERSION,
};

afterEach(() => {
  if (ORIGINAL_ENV === undefined) {
    delete process.env.NEXT_PUBLIC_MATCHMAKER_URL;
  } else {
    process.env.NEXT_PUBLIC_MATCHMAKER_URL = ORIGINAL_ENV;
  }
});

describe('cold-start timings', () => {
  it('shows Waking server before any error and errors only after ~90 s', () => {
    expect(MATCHMAKER_WAKE_AFTER_MS).toBe(2_500);
    expect(MATCHMAKER_ERROR_AFTER_MS).toBe(90_000);
    expect(MATCHMAKER_WAKE_AFTER_MS).toBeLessThan(MATCHMAKER_ERROR_AFTER_MS);
  });
});

describe('matchmakerBaseUrl', () => {
  it('returns null when NEXT_PUBLIC_MATCHMAKER_URL is unset or blank', () => {
    delete process.env.NEXT_PUBLIC_MATCHMAKER_URL;
    expect(matchmakerBaseUrl()).toBeNull();
    process.env.NEXT_PUBLIC_MATCHMAKER_URL = '   ';
    expect(matchmakerBaseUrl()).toBeNull();
  });

  it('trims whitespace and trailing slashes from the configured URL', () => {
    process.env.NEXT_PUBLIC_MATCHMAKER_URL = '  https://matchmaker.example.com/  ';
    expect(matchmakerBaseUrl()).toBe('https://matchmaker.example.com');
  });
});

describe('parseMatchList', () => {
  it('accepts a well-formed list', () => {
    expect(parseMatchList([validMatch])).toEqual([validMatch]);
  });

  it('returns null when the payload is not an array', () => {
    expect(parseMatchList(null)).toBeNull();
    expect(parseMatchList({})).toBeNull();
    expect(parseMatchList('nope')).toBeNull();
  });

  it('drops malformed entries and keeps the valid ones, including joinable:false', () => {
    const listedButDown = { ...validMatch, joinable: false };
    const parsed = parseMatchList([
      listedButDown,
      { name: 42 },
      null,
      { ...validMatch, joinable: 'yes' },
      { ...validMatch, address: '   ' },
      { ...validMatch, max_players: 'eight' },
    ]);
    expect(parsed).toEqual([listedButDown]);
  });
});

describe('resolveJoinUrl (address handoff to the existing join path)', () => {
  it('feeds listed addresses through parseJoinAddress with the /play/ suffix', () => {
    expect(resolveJoinUrl('192.168.1.23:7879')).toBe('http://192.168.1.23:7879/play/');
    expect(resolveJoinUrl('https://games.example.com:8443')).toBe('https://games.example.com:8443/play/');
  });

  it('rejects unusable addresses', () => {
    expect(resolveJoinUrl('not a url')).toBeNull();
    expect(resolveJoinUrl('ftp://192.168.1.23:7879')).toBeNull();
  });
});

describe('fetchMatches', () => {
  it('requests /matches filtered by the client version', async () => {
    const calls: string[] = [];
    const fetchImpl: typeof fetch = async (input) => {
      calls.push(String(input));
      return new Response(JSON.stringify([validMatch]), { status: 200 });
    };
    const entries = await fetchMatches('https://matchmaker.example.com', CLIENT_VERSION, undefined, fetchImpl);
    expect(entries).toEqual([validMatch]);
    expect(calls).toEqual([
      `https://matchmaker.example.com/matches?version=${encodeURIComponent(CLIENT_VERSION)}`,
    ]);
  });

  it('rejects on HTTP failure', async () => {
    const fetchImpl: typeof fetch = async () => new Response('boom', { status: 503 });
    await expect(
      fetchMatches('https://matchmaker.example.com', CLIENT_VERSION, undefined, fetchImpl),
    ).rejects.toThrow();
  });

  it('rejects when the payload is not a match array', async () => {
    const fetchImpl: typeof fetch = async () =>
      new Response(JSON.stringify({ not: 'an array' }), { status: 200 });
    await expect(
      fetchMatches('https://matchmaker.example.com', CLIENT_VERSION, undefined, fetchImpl),
    ).rejects.toThrow(/unexpected payload/);
  });
});

'use client';

// Shown from the title page. The player pastes the address the host saw in
// their hosting panel; we navigate the window to <address>/play/. The lobby
// browser (MatchList) above the form lists open lobbies from the matchmaker;
// joinable entries navigate through the exact same path.
import { useMemo, useState } from 'react';
import type { FormEvent, JSX } from 'react';
import { parseJoinAddress } from './join';
import { MatchList } from './MatchList';
import { resolveJoinUrl } from './matchmaker';

const MAGENTA = '#ff3db8';
const CYAN = '#38e8ff';

export function JoinPanel({ onClose }: { onClose: () => void }): JSX.Element {
  const [value, setValue] = useState('');
  const [error, setError] = useState<string | null>(null);
  const parsed = useMemo(() => parseJoinAddress(value), [value]);

  // Join path for a lobby-browser entry: same parse + navigate as the form.
  const onJoinAddress = (address: string): void => {
    const target = resolveJoinUrl(address);
    if (target === null) {
      setError('That listing does not contain a usable address.');
      return;
    }
    window.location.assign(target);
  };

  const onSubmit = (event: FormEvent<HTMLFormElement>): void => {
    event.preventDefault();
    if (parsed === null) {
      setError('That does not look like an address. Try something like 192.168.1.23:7879');
      return;
    }
    window.location.assign(`${parsed}/play/`);
  };

  return (
    <div
      className="fixed inset-0 z-40 overflow-y-auto bg-black/60 font-mono backdrop-blur-sm"
      onClick={onClose}
    >
      <div
        role="dialog"
        aria-modal="true"
        aria-labelledby="join-panel-title"
        className="mx-auto my-6 flex w-full max-w-md flex-col gap-4 break-words rounded-xl border border-white/10 bg-[#120f1f] p-5 text-white"
        onClick={(event) => event.stopPropagation()}
      >
        <div className="flex items-center justify-between gap-4 border-b border-white/10 pb-3">
          <h2 id="join-panel-title" className="text-lg font-bold tracking-[0.25em] text-[#38e8ff]">
            JOIN A FRIEND
          </h2>
          <button
            type="button"
            onClick={onClose}
            className="rounded border border-white/15 px-3 py-1 text-[11px] tracking-widest text-white/70 transition hover:text-white focus:outline-none focus-visible:ring-2 focus-visible:ring-white/40"
          >
            CLOSE
          </button>
        </div>

        <p className="text-sm leading-snug text-white/80">
          Ask the player hosting the game for their address, then paste it here.
        </p>

        <MatchList onJoin={onJoinAddress} />

        <form onSubmit={onSubmit} className="flex flex-col gap-3">
          <label
            htmlFor="join-address"
            className="text-[11px] font-bold tracking-[0.25em] text-white/70"
          >
            FRIEND ADDRESS
          </label>
          <input
            id="join-address"
            name="address"
            type="text"
            inputMode="url"
            autoCapitalize="none"
            autoCorrect="off"
            spellCheck={false}
            placeholder="192.168.1.23:7879"
            value={value}
            onChange={(event) => {
              setValue(event.target.value);
              setError(null);
            }}
            aria-invalid={error !== null}
            aria-describedby="join-address-help"
            className="rounded border border-white/15 bg-black/40 px-3 py-2 text-sm text-white placeholder-white/30 focus:outline-none focus-visible:ring-2 focus-visible:ring-[#38e8ff]/60"
          />
          {error !== null ? (
            <p id="join-address-help" role="alert" className="text-xs text-[#ff4d6d]">
              {error}
            </p>
          ) : parsed !== null ? (
            <p id="join-address-help" className="text-xs" style={{ color: CYAN }}>
              Opens {parsed}/play/
            </p>
          ) : (
            <p id="join-address-help" className="text-xs text-white/55">
              Works with or without http:// - a bare address uses the game port 7879.
            </p>
          )}
          <button
            type="submit"
            className="rounded-lg border-2 px-6 py-3 text-center text-sm font-bold tracking-[0.3em] transition hover:text-black focus:outline-none focus-visible:ring-2 focus-visible:ring-[#ff3db8]/60"
            style={{ borderColor: MAGENTA, color: MAGENTA }}
          >
            JOIN
          </button>
        </form>
      </div>
    </div>
  );
}
