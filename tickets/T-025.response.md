export interface AvatarMotion {
  speed: number;
  vy: number;
  grounded: boolean;
  dashing: boolean;
}

export interface AvatarPose {
  leftArm: number;
  rightArm: number; // pitch (rad), + = forward swing
  leftArmRoll: number;
  rightArmRoll: number; // roll outward (rad), for jump/fall flail
  leftLeg: number;
  rightLeg: number; // pitch (rad)
  torsoPitch: number; // lean forward (+)
  bodyY: number; // vertical offset of the whole body (bob / squash compensation)
  scaleY: number;
  scaleXZ: number; // squash & stretch
}

export interface PoseState {
  phase: number;
  landTimer: number;
  wasGrounded: boolean;
  pose: AvatarPose;
}

const TAU = Math.PI * 2;
const BLEND_TAU = 0.06; // 60 ms time constant for rotations / bodyY
const SQUASH_TAU = 0.025; // squash values blend faster
const LAND_DURATION = 0.12;
const RUN_SPEED = 9;
const RUN_CADENCE = 2.6; // cycles/s at full run speed
const LEG_SWING = 0.75;
const ARM_SWING = 0.6;
const RUN_TORSO_PITCH = 0.12;
const RUN_BODY_BOB = 0.06;
const IDLE_BREATH_RATE = 1.2; // cycles/s
const IDLE_BREATH_Y = 0.02;
const JUMP_ARM_PITCH = -2.4;
const JUMP_ARM_ROLL = 0.3;
const JUMP_FRONT_LEG = 0.6;
const JUMP_BACK_LEG = -0.3;
const JUMP_TORSO_PITCH = 0.05;
const JUMP_STRETCH_Y = 1.06;
const JUMP_STRETCH_XZ = 0.96;
const FALL_ARM_ROLL = 1.1;
const FALL_FLAIL_START_VY = -14;
const FALL_FLAIL = 0.25;
const FALL_FLAIL_FAST = 0.45;
const FALL_FLAIL_RATE = 4; // cycles/s
const FALL_LEG_SPREAD = 0.35;
const FALL_TORSO_PITCH = 0.1;
const DASH_TORSO_PITCH = 0.55;
const DASH_ARM_PITCH = -1.3; // arms swept back behind the body
const DASH_FRONT_LEG = -0.5;
const DASH_BACK_LEG = -0.2;
const DASH_STRETCH_Y = 0.92;
const DASH_STRETCH_XZ = 1.1;
const LAND_SQUASH_Y = 0.82;
const LAND_SQUASH_XZ = 1.12;
const LAND_BODY_DIP = -0.08;
const MIN_RUN_SPEED = 0.5;

// Reusable scratch targets so stepPose allocates nothing per call. This module
// is used from a single animation loop per avatar; the scratch is fully
// overwritten at the start of every computeTarget call.
const target: AvatarPose = {
  leftArm: 0,
  rightArm: 0,
  leftArmRoll: 0,
  rightArmRoll: 0,
  leftLeg: 0,
  rightLeg: 0,
  torsoPitch: 0,
  bodyY: 0,
  scaleY: 1,
  scaleXZ: 1,
};

export function createPoseState(): PoseState {
  return {
    phase: 0,
    landTimer: 0,
    wasGrounded: true,
    pose: {
      leftArm: 0,
      rightArm: 0,
      leftArmRoll: 0,
      rightArmRoll: 0,
      leftLeg: 0,
      rightLeg: 0,
      torsoPitch: 0,
      bodyY: 0,
      scaleY: 1,
      scaleXZ: 1,
    },
  };
}

function computeTarget(m: AvatarMotion, state: PoseState): number {
  target.leftArm = 0;
  target.rightArm = 0;
  target.leftArmRoll = 0;
  target.rightArmRoll = 0;
  target.leftLeg = 0;
  target.rightLeg = 0;
  target.torsoPitch = 0;
  target.bodyY = 0;
  target.scaleY = 1;
  target.scaleXZ = 1;

  if (m.dashing) {
    // Dash overrides everything: strong lean, arms swept back, trailing legs.
    target.torsoPitch = DASH_TORSO_PITCH;
    target.leftArm = DASH_ARM_PITCH;
    target.rightArm = DASH_ARM_PITCH;
    target.leftLeg = DASH_FRONT_LEG;
    target.rightLeg = DASH_BACK_LEG;
    target.scaleY = DASH_STRETCH_Y;
    target.scaleXZ = DASH_STRETCH_XZ;
    return 0; // phase frozen while dashing
  }

  if (m.grounded) {
    if (m.speed >= MIN_RUN_SPEED) {
      const rate = (m.speed / RUN_SPEED) * RUN_CADENCE;
      const amp = Math.min(Math.abs(m.speed) / RUN_SPEED, 1);
      const s = Math.sin(state.phase * TAU);
      target.leftLeg = s * LEG_SWING * amp;
      target.rightLeg = -s * LEG_SWING * amp;
      target.leftArm = -s * ARM_SWING * amp;
      target.rightArm = s * ARM_SWING * amp;
      target.torsoPitch = RUN_TORSO_PITCH * amp;
      target.bodyY = Math.abs(s) * RUN_BODY_BOB * amp; // two bounces per cycle
      return rate;
    }
    // Idle: subtle breathing.
    target.bodyY = Math.sin(state.phase * TAU) * IDLE_BREATH_Y;
    return IDLE_BREATH_RATE * Math.sign(state.phase) * (state.phase < 0 ? -1 : 1);
  }

  // Airborne: the shared phase oscillator drives the fall flail rate.
  if (m.vy > 1) {
    // Jump rising: arms overhead, legs tucked, slight stretch.
    target.leftArm = JUMP_ARM_PITCH;
    target.rightArm = JUMP_ARM_PITCH;
    target.leftArmRoll = JUMP_ARM_ROLL;
    target.rightArmRoll = JUMP_ARM_ROLL;
    target.leftLeg = JUMP_FRONT_LEG;
    target.rightLeg = JUMP_BACK_LEG;
    target.torsoPitch = JUMP_TORSO_PITCH;
    target.scaleY = JUMP_STRETCH_Y;
    target.scaleXZ = JUMP_STRETCH_XZ;
  } else {
    // Apex / falling: arms out to the sides, flailing; more flail when fast.
    const flail = m.vy < FALL_FLAIL_START_VY ? FALL_FLAIL_FAST : FALL_FLAIL;
    const f = Math.sin(state.phase * TAU);
    target.leftArm = f * flail;
    target.rightArm = -f * flail;
    target.leftArmRoll = FALL_ARM_ROLL;
    target.rightArmRoll = FALL_ARM_ROLL;
    target.leftLeg = FALL_LEG_SPREAD;
    target.rightLeg = -FALL_LEG_SPREAD;
    target.torsoPitch = FALL_TORSO_PITCH;
  }
  return FALL_FLAIL_RATE;
}

/** Advance animation by dt seconds toward the target pose for this motion; mutates and returns state.pose. */
export function stepPose(state: PoseState, m: AvatarMotion, dtRaw: number): AvatarPose {
  const dt = Math.min(0.1, Math.max(0, dtRaw));

  // Detect the landing frame: grounded flips back to true after being false.
  if (m.grounded && !state.wasGrounded) {
    state.landTimer = LAND_DURATION;
  }
  state.wasGrounded = m.grounded;

  const phaseRate = computeTarget(m, state);
  state.phase += phaseRate * dt;

  const k = 1 - Math.exp(-dt / BLEND_TAU);
  const k2 = 1 - Math.exp(-dt / SQUASH_TAU);
  const p = state.pose;
  p.leftArm += (target.leftArm - p.leftArm) * k;
  p.rightArm += (target.rightArm - p.rightArm) * k;
  p.leftArmRoll += (target.leftArmRoll - p.leftArmRoll) * k;
  p.rightArmRoll += (target.rightArmRoll - p.rightArmRoll) * k;
  p.leftLeg += (target.leftLeg - p.leftLeg) * k;
  p.rightLeg += (target.rightLeg - p.rightLeg) * k;
  p.torsoPitch += (target.torsoPitch - p.torsoPitch) * k;
  p.bodyY += (target.bodyY - p.bodyY) * k;
  p.scaleY += (target.scaleY - p.scaleY) * k2;
  p.scaleXZ += (target.scaleXZ - p.scaleXZ) * k2;

  // Timed land squash, applied directly (not blended) so it always reads clearly.
  state.landTimer = Math.max(0, state.landTimer - dt);
  if (state.landTimer > 0 && !m.dashing) {
    const prog = 1 - state.landTimer / LAND_DURATION;
    const easeOut = 1 - (1 - prog) * (1 - prog);
    p.scaleY = LAND_SQUASH_Y + (1 - LAND_SQUASH_Y) * easeOut;
    p.scaleXZ = LAND_SQUASH_XZ + (1 - LAND_SQUASH_XZ) * easeOut;
    p.bodyY += LAND_BODY_DIP * (1 - easeOut);
  }

  return p;
}

export function BlockyAvatar(props: {
  bodyColor?: string;
  motion?: { current: AvatarMotion };
}): JSX.Element {
  const poseStateRef = useRef(createPoseState());
  const outerRef = useRef<Group>(null);
  const torsoRef = useRef<Group>(null);
  const leftArmRef = useRef<Group>(null);
  const rightArmRef = useRef<Group>(null);
  const leftLegRef = useRef<Group>(null);
  const rightLegRef = useRef<Group>(null);

  useFrame((_state, delta) => {
    const motion = props.motion?.current ?? IDLE_MOTION;
    const pose = stepPose(poseStateRef.current, motion, delta);

    const outer = outerRef.current;
    if (outer !== null) {
      // Scale around the feet origin (group sits at y = 0) so feet stay planted.
      outer.position.y = pose.bodyY;
      outer.scale.set(pose.scaleXZ, pose.scaleY, pose.scaleXZ);
    }

    const torso = torsoRef.current;
    if (torso !== null) {
      torso.rotation.x = pose.torsoPitch;
    }
    const leftArm = leftArmRef.current;

      leftArm.rotation.x = pose.leftArm;
      leftArm.rotation.z = pose.leftArmRoll;

      rightArm.rotation.x = pose.rightArm;
      rightArm.rotation.z = -pose.rightArmRoll; // mirrored: outward is -Z rot on -X side

      leftLeg.rotation.x = pose.leftLeg;

    const rightLeg = rightLegRef.current;
    if (rightLeg !== null) {
      rightLeg.rotation.x = pose.rightLeg;

"use client";

import { useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import { AdditiveBlending } from "three";
import type { Group, MeshBasicMaterial } from "three";
import type { AvatarMotion } from "./avatarPose";

const FADE_RATE = 50; // exponential approach; ~95% faded in/out within ~60 ms
const MIN_VISIBLE = 0.01;

export function DashStreak(props: {
  motion: { current: AvatarMotion };
  color: string;
}): JSX.Element {
  const groupRef = useRef<Group>(null);
  const material0Ref = useRef<MeshBasicMaterial>(null);
  const material1Ref = useRef<MeshBasicMaterial>(null);
  const material2Ref = useRef<MeshBasicMaterial>(null);
  const fadeRef = useRef(0);

  useFrame((_state, delta) => {
    const fade = fadeRef.current;
    const target = props.motion.current.dashing ? 1 : 0;
    const k = Math.min(1, delta * FADE_RATE);
    const next = fade + (target - fade) * k;
    fadeRef.current = next;

    const group = groupRef.current;
    if (group === null) {
      return;
    }
    group.visible = next > MIN_VISIBLE;

    const m0 = material0Ref.current;
    if (m0 !== null) {
      m0.opacity = 0.35 * next;
    }
    const m1 = material1Ref.current;
    if (m1 !== null) {
      m1.opacity = 0.2 * next;
    }
    const m2 = material2Ref.current;
    if (m2 !== null) {
      m2.opacity = 0.1 * next;
    }
  });

  return (
    <group ref={groupRef} visible={false}>
      {/* Streaks trail behind the avatar: local -Z is forward, so +Z is behind */}
      <mesh position={[0, 0.8, 0.5]}>
        <boxGeometry args={[0.6, 1.6, 0.3]} />
        <meshBasicMaterial
          ref={material0Ref}
          color={props.color}
          transparent
          opacity={0}
          blending={AdditiveBlending}
          depthWrite={false}
        />
      </mesh>
      <mesh position={[0, 0.8, 1.0]}>
        <boxGeometry args={[0.6, 1.6, 0.3]} />
        <meshBasicMaterial
          ref={material1Ref}
          color={props.color}
          transparent
          opacity={0}
          blending={AdditiveBlending}
          depthWrite={false}
        />
      </mesh>
      <mesh position={[0, 0.8, 1.5]}>
        <boxGeometry args={[0.6, 1.6, 0.3]} />
        <meshBasicMaterial
          ref={material2Ref}
          color={props.color}
          transparent
          opacity={0}
          blending={AdditiveBlending}
          depthWrite={false}
        />
      </mesh>
    </group>
  );
}

import { describe, expect, it } from "vitest";
import { createPoseState, stepPose } from "./avatarPose";
import type { AvatarMotion } from "./avatarPose";

const DT = 1 / 60;

function frames(n: number, step: (state: import("vitest").TaskState) => void): void {
  for (let i = 0; i < n; i++) {
    step({ i } as unknown as import("vitest").TaskState);
  }
}
void frames;

function motionOf(partial: Partial<AvatarMotion>): AvatarMotion {
  return { speed: 0, vy: 0, grounded: true, dashing: false, ...partial };
}

function stepN(state: ReturnType<typeof createPoseState>, m: AvatarMotion, n: number): void {
  for (let i = 0; i < n; i++) {
    stepPose(state, m, DT);
  }
}

describe("avatarPose", () => {
  it("advances run cadence phase faster at higher speed", () => {
    const slow = createPoseState();
    const fast = createPoseState();
    stepN(slow, motionOf({ speed: 4.5 }), 1);
    stepN(fast, motionOf({ speed: 9 }), 1);
    expect(fast.phase).toBeGreaterThan(slow.phase);
  });

  it("triggers land squash on landing and recovers after 0.2 s", () => {
    const state = createPoseState();
    stepN(state, motionOf({ speed: 0, vy: -10, grounded: false }), 1);
    stepN(state, motionOf({ speed: 0.6, vy: 0, grounded: true }), 1);
    expect(state.pose.scaleY).toBeLessThan(0.9);
    stepN(state, motionOf({ speed: 0.6, vy: 0, grounded: true }), 12);
    expect(state.pose.scaleY).toBeGreaterThan(0.95);
  });

  it("leanS forward strongly while dashing", () => {
    const state = createPoseState();
    stepN(state, motionOf({ speed: 28, vy: 0, grounded: true, dashing: true }), 6);
    expect(state.pose.torsoPitch).toBeGreaterThan(0.4);
  });

  it("raises the arms overhead while jump rising", () => {
    const state = createPoseState();
    stepN(state, motionOf({ speed: 0, vy: 12, grounded: false }), 6);
    expect(state.pose.leftArm).toBeLessThan(-1);
    expect(state.pose.rightArm).toBeLessThan(-1);
  });

  it("produces finite output for dt 0, huge speeds and huge dt", () => {
    const state = createPoseState();
    stepPose(state, motionOf({ speed: 0 }), 0);
    stepN(state, motionOf({ speed: 0 }), 1);
    stepPose(state, motionOf({ speed: 1e12, vy: -1e9, grounded: false }), 0.05);
    stepPose(state, motionOf({ speed: 1e12, vy: 1e9, grounded: true, dashing: true }), 1e9);
    const p = state.pose;
    const values = [
      p.leftArm,
      p.rightArm,
      p.leftArmRoll,
      p.rightArmRoll,
      p.leftLeg,
      p.rightLeg,
      p.torsoPitch,
      p.bodyY,
      p.scaleY,
      p.scaleXZ,
    ];
    for (const v of values) {
      expect(Number.isFinite(v)).toBe(true);
    }
  });
});
