```tsx
"use client";

import { useMemo } from "react";
import type { JSX } from "react";
import type { Platform } from "@/protocol/generated";
import { PALETTE, platformColor } from "./palette";

interface EdgeBarData {
  size: [number, number, number];
  position: [number, number, number];
}

interface PlatformBoxData {
  id: number;
  center: [number, number, number];
  size: [number, number, number];
  top: string;
  accent: string;
  edgeBars: EdgeBarData[];
}

const TOP_EMISSIVE = 0.08;
const EDGE_BAR_THICKNESS = 0.08;
const EDGE_BAR_HEIGHT = 0.08;
const EDGE_BAR_OVERHANG = 0.04;
const EDGE_BAR_EMISSIVE = 1.0;

// Four thin bars along the top perimeter. The bars along X straddle the
// top edge (overhang EDGE_BAR_OVERHANG on each side); the bars along Z
// span exactly the gap between them so nothing overlaps at the corners.
function toEdgeBars(size: readonly [number, number, number]): EdgeBarData[] {
  const [width, height, depth] = size;
  const halfWidth = width / 2;
  const halfDepth = depth / 2;
  const barY = height / 2 + EDGE_BAR_HEIGHT / 2;
  const longLength = width + EDGE_BAR_OVERHANG * 2;
  const shortLength = depth - EDGE_BAR_OVERHANG * 2;
  return [
    {
      size: [longLength, EDGE_BAR_HEIGHT, EDGE_BAR_THICKNESS],
      position: [0, barY, halfDepth],
    },
    {
      size: [longLength, EDGE_BAR_HEIGHT, EDGE_BAR_THICKNESS],
      position: [0, barY, -halfDepth],
    },
    {
      size: [EDGE_BAR_THICKNESS, EDGE_BAR_HEIGHT, shortLength],
      position: [halfWidth, barY, 0],
    },
    {
      size: [EDGE_BAR_THICKNESS, EDGE_BAR_HEIGHT, shortLength],
      position: [-halfWidth, barY, 0],
    },
  ];
}

function toBoxData(platform: Platform): PlatformBoxData {
  const { min, max } = platform.bounds;
  const size: [number, number, number] = [
    max.x - min.x,
    max.y - min.y,
    max.z - min.z,
  ];
  const center: [number, number, number] = [
    (min.x + max.x) / 2,
    (min.y + max.y) / 2,
    (min.z + max.z) / 2,
  ];
  return {
    id: platform.id,
    center,
    size,
    top: platformColor(platform.id),
    accent:
      platform.id % 2 === 0 ? PALETTE.edgeGlowCyan : PALETTE.edgeGlowMagenta,
    edgeBars: toEdgeBars(size),
  };
}

export function Platforms(props: {
  platforms: readonly Platform[];
}): JSX.Element {
  const boxes = useMemo(() => props.platforms.map(toBoxData), [props.platforms]);

  return (
    <group>
      {boxes.map((box) => (
        <group key={box.id} position={box.center}>
          <mesh>
            <boxGeometry args={box.size} />
            <meshStandardMaterial
              color={box.top}
              emissive={box.top}
              emissiveIntensity={TOP_EMISSIVE}
            />
          </mesh>
          {box.edgeBars.map((bar, index) => (
            <mesh key={index} position={bar.position}>
              <boxGeometry args={bar.size} />
              <meshStandardMaterial
                color={box.accent}
                emissive={box.accent}
                emissiveIntensity={EDGE_BAR_EMISSIVE}
              />
            </mesh>
          ))}
        </group>
      ))}
    </group>
  );
}
```

## Task
1. New `web/src/scene/checker.ts`:
   - `export function worldScaledBoxGeometry(width: number, height: number, depth: number): BoxGeometry` - a
     `new BoxGeometry(width, height, depth)` whose UVs are rescaled per face so that ONE UV unit equals ONE world unit
     (three's BoxGeometry face order is +x, -x, +y, -y, +z, -z with 4 vertices each; on +/-x faces u spans depth and
     v spans height, on +/-y faces u spans width and v spans depth, on +/-z faces u spans width and v spans height).
     Additionally offset the UVs of each face by half its extent in world units, so checks line up with the world grid
     for platforms whose edges sit on integer coordinates. Pure geometry math - must work in node.
   - `export function makeCheckerTexture(light: string, dark: string, seed: number): CanvasTexture | null` - returns
     null when `typeof document === "undefined"`. Otherwise draws a 256x256 canvas holding a 2x2 checker (each check 128px
     = one world unit; set `texture.repeat` to (0.5, 0.5)), RepeatWrapping on both
     axes, `colorSpace = SRGBColorSpace`, anisotropy 8, generateMipmaps on. Add grunge so it reads as worn stone, not a
     test pattern: deterministic per-pixel brightness noise of +/-6% from a seeded PRNG (mulberry32 on `seed`), plus 3-6
     faint darker marble veins (thin bezier strokes at ~10% opacity) and a 2px slightly darker groove along every check
     border. Deterministic for a given (light, dark, seed).
   - `export function themeTextures(theme: Theme): { top: CanvasTexture | null; side: CanvasTexture | null }` -
     memoised per `theme.id` (module-level Map): top = checker(tile.light, tile.dark), side = checker(tile.side,
     tile.dark).
2. `web/src/scene/Platforms.tsx`: new prop `theme: Theme` (keep `platforms`). Each platform is a single mesh using
   `worldScaledBoxGeometry` (memoised per platform; dispose on change/unmount) and a 6-entry material array in face
   order [side, side, top, bottom, side, side]: top = MeshStandardMaterial with `map` = top texture, roughness 0.85;
   sides = map side texture, color "#9a9a9a" (dimmer), roughness 0.9; bottom = plain `theme.tile.dark`. When textures
   are null (tests/SSR) fall back to plain colours (top tile.light, sides tile.side). Keep the four thin edge trim bars
   from the current version, but coloured `theme.tile.edge` with emissive intensity 1.4 (they will bloom later).
   Share the materials between platforms (memoise per theme), do not create one material per platform.
3. `web/src/scene/checker.test.ts` - vitest (node): UVs of a 4x1x2 box: the +y face spans exactly 4 units in u and 2 in
   v; the +x face spans 2 in u and 1 in v; `makeCheckerTexture` returns null in node; `themeTextures` is memoised
   (same object for the same theme id).
