// Pure voting state logic for the collapse-window pedestal vote and the lobby
// poll. Deterministic, fully typed, no protocol imports: the wire mapping
// lives in protocolAdapter.ts and components render from VoteViewState.
export type VoteKind = 'enemy' | 'curse' | 'theme' | 'difficulty';

export interface VoteOption {
  id: string;
  name: string;
  kind: VoteKind;
}

export type VotePhase = 'idle' | 'open' | 'revealed';

export interface VoteViewState {
  phase: VotePhase;
  options: VoteOption[];
  counts: number[];
  /** Index into options, or null when this player has not voted. */
  yourVote: number | null;
  /** Clamped to [0, durationMs]. The authoritative countdown comes from the server; local ticks keep it smooth. */
  remainingMs: number;
  durationMs: number;
  winnerIndex: number | null;
  error: string | null;
}

export type VoteEvent =
  | { type: 'reset' }
  // remainingMs is optional; when absent the full duration is used.
  | { type: 'opened'; options: VoteOption[]; durationMs: number; remainingMs?: number | undefined }
  | {
      type: 'progress';
      counts?: number[] | undefined;
      remainingMs?: number | undefined;
      /** -1 maps to null ("this player has no vote"). undefined = field not sent. */
      yourVote?: number | null | undefined;
    }
  | { type: 'cast'; index: number }
  | { type: 'result'; winnerIndex: number | null; counts?: number[] | undefined }
  | { type: 'tick'; deltaMs: number }
  | { type: 'failed'; message: string };

/** The collapse-window vote always lasts 20 s (T-078 server-side). */
export const VOTE_DURATION_MS = 20_000;

export function createInitialVoteState(): VoteViewState {
  return {
    phase: 'idle',
    options: [],
    counts: [],
    yourVote: null,
    remainingMs: 0,
    durationMs: VOTE_DURATION_MS,
    winnerIndex: null,
    error: null,
  };
}

export const initialVoteState: VoteViewState = createInitialVoteState();

export function secondsRemaining(remainingMs: number): number {
  return Math.max(0, Math.ceil(remainingMs / 1000));
}

export function isVotingOpen(state: VoteViewState): boolean {
  return state.phase === 'open' && state.remainingMs > 0;
}

function isFiniteNumber(value: unknown): value is number {
  return typeof value === 'number' && Number.isFinite(value);
}

function isValidIndex(index: unknown, length: number): index is number {
  return typeof index === 'number' && Number.isInteger(index) && index >= 0 && index < length;
}

/** Returns fallback unchanged when counts is undefined; null on length mismatch. */
function normalizeCounts(
  counts: readonly number[],
  length: number,
  fallback: readonly number[],
): number[] | null {
  if (counts.length !== length) return null;
  return counts.map((c) => (isFiniteNumber(c) && c > 0 ? Math.floor(c) : 0));
}

export function voteReducer(state: VoteViewState, event: VoteEvent): VoteViewState {
  switch (event.type) {
    case 'reset':
      return createInitialVoteState();

    case 'opened': {
      if (!Array.isArray(event.options) || event.options.length === 0) {
        return { ...createInitialVoteState(), error: 'Vote opened with no options.' };
      }
      const durationMs =
        isFiniteNumber(event.durationMs) && event.durationMs > 0
          ? Math.floor(event.durationMs)
          : VOTE_DURATION_MS;
      const remainingMs =
        event.remainingMs !== undefined && isFiniteNumber(event.remainingMs)
          ? Math.min(Math.max(0, Math.floor(event.remainingMs)), durationMs)
          : durationMs;
      const options = event.options.map((o) => ({ ...o }));
      return {
        phase: 'open',
        options,
        counts: options.map(() => 0),
        yourVote: null,
        remainingMs,
        durationMs,
        winnerIndex: null,
        error: null,
      };
    }

    case 'progress': {
      if (state.phase !== 'open') return state;
      let counts = state.counts;
      let error: string | null = state.error;
      if (event.counts !== undefined) {
        const normalized = normalizeCounts(event.counts, state.options.length, state.counts);
        if (normalized === null) {
          error = 'Vote update had the wrong number of counts.';
        } else {
          counts = normalized;
        }
      }
      let remainingMs = state.remainingMs;
      if (event.remainingMs !== undefined && isFiniteNumber(event.remainingMs)) {
        remainingMs = Math.min(Math.max(0, Math.floor(event.remainingMs)), state.durationMs);
      }
      let yourVote = state.yourVote;
      if (event.yourVote !== undefined) {
        if (event.yourVote === null) {
          yourVote = null;
        } else if (isValidIndex(event.yourVote, state.options.length)) {
          yourVote = event.yourVote;
        }
      }
      return { ...state, counts, remainingMs, yourVote, error };
    }

    case 'cast': {
      if (state.phase !== 'open') return { ...state, error: 'No vote is open.' };
      if (state.remainingMs <= 0) {
        return { ...state, error: 'The vote window has closed.' };
      }
      if (!isValidIndex(event.index, state.options.length)) {
        return { ...state, error: 'That option does not exist.' };
      }
      return { ...state, yourVote: event.index, error: null };
    }

    case 'tick': {
      if (state.phase !== 'open') return state;
      if (!isFiniteNumber(event.deltaMs) || event.deltaMs <= 0) return state;
      const remainingMs = Math.max(0, state.remainingMs - Math.floor(event.deltaMs));
      if (remainingMs === state.remainingMs) return state;
      return { ...state, remainingMs };
    }

    case 'result': {
      if (state.options.length === 0) {
        return { ...state, error: 'Vote result received with no open vote.' };
      }
      let counts = state.counts;
      let error: string | null = null;
      if (event.counts !== undefined) {
        const normalized = normalizeCounts(event.counts, state.options.length, state.counts);
        if (normalized === null) {
          error = 'Vote result had the wrong number of counts.';
        } else {
          counts = normalized;
        }
      }
      let winnerIndex: number | null = null;
      if (event.winnerIndex !== null && isValidIndex(event.winnerIndex, state.options.length)) {
        winnerIndex = event.winnerIndex;
      } else {
        error = error ?? 'Vote result did not name a valid winner.';
      }
      return { ...state, phase: 'revealed', counts, remainingMs: 0, winnerIndex, error };
    }

    case 'failed':
      return { ...state, error: event.message };

    default:
      return state;
  }
}

// The ONLY module that knows the wire shape. T-078 is not merged yet and its
// exact field names are not in this ticket, so the assumed shapes are listed
// here and validated at runtime; if the regenerated protocol differs, this
// file (and nothing else) absorbs the change.
//
// ASSUMED WIRE SHAPES (T-078):
//   VoteOpen   (server): { type: 'VoteOpen';
//                          options: [{ id: string; name: string;
//                                      kind: number | string }];
//                          durationMs?: number; remainingMs?: number }
//                kind: CARD_ENEMY(0) or CARD_CURSE(2), or the string
//                'enemy' | 'curse' | 'theme' | 'difficulty'.
//   VoteState  (server): { type: 'VoteState'; counts: number[];
//                          remainingMs?: number;
//                          yourVote?: number; you?: number }
//                yourVote/you: option index, or any negative number for "no vote".
//   VoteResult (server): { type: 'VoteResult'; winner?: number; winnerIndex?: number;
//                          counts?: number[] }
//   Vote       (client): { type: 'Vote'; option: number } — not encoded here;
//                the host encodes it via the send callback threaded into the HUD.
import { CARD_CURSE, CARD_ENEMY } from '../../protocol/generated';
import { VOTE_DURATION_MS } from './model';
import type { VoteEvent, VoteKind, VoteOption } from './model';

function asRecord(value: unknown): Record<string, unknown> | null {
  return typeof value === 'object' && value !== null ? (value as Record<string, unknown>) : null;
}

function isNonNegativeFinite(value: unknown): value is number {
  return typeof value === 'number' && Number.isFinite(value) && value >= 0;
}

function toCount(value: unknown): number {
  return typeof value === 'number' && Number.isFinite(value) && value > 0 ? Math.floor(value) : 0;
}

function toCounts(value: unknown): number[] | undefined {
  return Array.isArray(value) ? value.map(toCount) : undefined;
}

function kindFromWire(value: unknown): VoteKind | null {
  if (typeof value === 'string') {
    const v = value.toLowerCase();
    if (v === 'enemy' || v === 'curse' || v === 'theme' || v === 'difficulty') return v;
    return null;
  }
  if (value === CARD_ENEMY) return 'enemy';
  if (value === CARD_CURSE) return 'curse';
  return null;
}

function optionFromWire(value: unknown): VoteOption | null {
  const rec = asRecord(value);
  if (rec === null) return null;
  const { id, name } = rec;
  if (typeof id !== 'string' || id.length === 0) return null;
  if (typeof name !== 'string' || name.trim().length === 0) return null;
  const kind = kindFromWire(rec.kind);
  if (kind === null) return null;
  return { id, name: name.trim(), kind };
}

/**
 * Maps any incoming message to a VoteEvent, or null when the message is not a
 * vote message. Never throws: malformed vote messages become 'failed' events
 * so the UI can show an error instead of crashing the HUD.
 */
export function parseVoteMessage(message: unknown): VoteEvent | null {
  const rec = asRecord(message);
  if (rec === null || typeof rec.type !== 'string') return null;

  switch (rec.type) {
    case 'VoteOpen': {
      if (!Array.isArray(rec.options)) {
        return { type: 'failed', message: 'VoteOpen was missing its options.' };
      }
      const options: VoteOption[] = [];
      for (const item of rec.options) {
        const option = optionFromWire(item);
        if (option === null) {
          return { type: 'failed', message: 'VoteOpen contained a malformed option.' };
        }
        options.push(option);
      }
      if (options.length === 0) {
        return { type: 'failed', message: 'Vote opened with no options.' };
      }
      const durationMs = isNonNegativeFinite(rec.durationMs) && rec.durationMs > 0
        ? Math.floor(rec.durationMs)
        : VOTE_DURATION_MS;
      if (isNonNegativeFinite(rec.remainingMs)) {
        return { type: 'opened', options, durationMs, remainingMs: Math.floor(rec.remainingMs) };
      }
      return { type: 'opened', options, durationMs };
    }

    case 'VoteState': {
      const remainingMs = isNonNegativeFinite(rec.remainingMs)
        ? Math.floor(rec.remainingMs)
        : undefined;
      let yourVote: number | null | undefined;
      const youRaw = rec.yourVote !== undefined ? rec.yourVote : rec.you;
      if (typeof youRaw === 'number' && Number.isFinite(youRaw)) {
        yourVote = youRaw < 0 ? null : Number.isInteger(youRaw) ? Math.floor(youRaw) : undefined;
      }
      return { type: 'progress', counts: toCounts(rec.counts), remainingMs, yourVote };
    }

    case 'VoteResult': {
      const winnerRaw =
        typeof rec.winner === 'number' ? rec.winner
        : typeof rec.winnerIndex === 'number' ? rec.winnerIndex
        : undefined;
      if (winnerRaw === undefined || !Number.isFinite(winnerRaw)) {
        return { type: 'failed', message: 'VoteResult did not name a winner.' };
      }
      return { type: 'result', winnerIndex: Math.floor(winnerRaw), counts: toCounts(rec.counts) };
    }

    default:
      return null;
  }
}

// Hook bridging the pure reducer to React: local 200 ms countdown ticks
// (re-synced whenever the server sends remainingMs) and message ingestion.
import { useCallback, useEffect, useMemo, useReducer, useRef } from 'react';
import { initialVoteState, voteReducer } from './model';
import type { VoteViewState } from './model';
import { parseVoteMessage } from './protocolAdapter';

export interface VoteController {
  state: VoteViewState;
  /** Marks the vote locally and reports the choice; no server echo required. */
  sendVote: (index: number) => void;
  /** Feed any server message; non-vote messages are ignored. */
  ingestMessage: (message: unknown) => void;
  reset: () => void;
}

export function useVote(onVote?: ((optionIndex: number) => void) | undefined): VoteController {
  const [state, dispatch] = useReducer(voteReducer, initialVoteState);
  const lastTick = useRef<number | null>(null);

  useEffect(() => {
    const id = window.setInterval(() => {
      const now = Date.now();
      const previous = lastTick.current ?? now;
      lastTick.current = now;
      const delta = now - previous;
      if (delta > 0) dispatch({ type: 'tick', deltaMs: delta });
    }, 200);
    return () => window.clearInterval(id);
  }, []);

  const sendVote = useCallback(
    (index: number) => {
      const canCast =
        state.phase === 'open' &&
        state.remainingMs > 0 &&
        Number.isInteger(index) &&
        index >= 0 &&
        index < state.options.length;
      dispatch({ type: 'cast', index });
      if (canCast) onVote?.(index);
    },
    [state, onVote],
  );

  const ingestMessage = useCallback((message: unknown) => {
    const event = parseVoteMessage(message);
    if (event !== null) dispatch(event);
  }, []);

  const reset = useCallback(() => dispatch({ type: 'reset' }), []);

  return useMemo(
    () => ({ state, sendVote, ingestMessage, reset }),
    [state, sendVote, ingestMessage, reset],
  );
}

import type { JSX } from 'react';
import { secondsRemaining } from './model';

const RADIUS = 26;
const CIRCUMFERENCE = 2 * Math.PI * RADIUS;

/** Pure SVG countdown ring for the 20 s vote window. No timers of its own. */
export function CountdownRing({
  remainingMs,
  durationMs,
}: {
  remainingMs: number;
  durationMs: number;
}): JSX.Element {
  const safeDuration = durationMs > 0 ? durationMs : 1;
  const clamped = Math.min(Math.max(remainingMs, 0), safeDuration);
  const fraction = clamped / safeDuration;
  const seconds = secondsRemaining(clamped);
  const stroke = seconds <= 5 ? '#ff4d6d' : '#38e8ff';

  return (
    <svg
      width={72}
      height={72}
      viewBox="0 0 72 72"
      role="timer"
      aria-label={`Vote closes in ${seconds} seconds`}
      className="shrink-0"
    >
      <circle
        cx={36}
        cy={36}
        r={RADIUS}
        stroke="rgba(255,255,255,0.15)"
        strokeWidth={6}
        fill="none"
      />
      <circle
        cx={36}
        cy={36}
        r={RADIUS}
        stroke={stroke}
        strokeWidth={6}
        fill="none"
        strokeLinecap="round"
        strokeDasharray={CIRCUMFERENCE}
        strokeDashoffset={CIRCUMFERENCE * (1 - fraction)}
        transform="rotate(-90 36 36)"
      />
      <text
        x={36}
        y={42}
        textAnchor="middle"
        fontSize={20}
        fontWeight={700}
        fill="#ffffff"
        fontFamily="monospace"
      >
        {seconds}
      </text>
    </svg>
  );
}

          {state.error}
        </p>
      )}
    </div>
  );

// Pre-match poll, reusing the voting widget. Optional by design: one button to
// start, a SKIP button to dismiss, and nothing gates the run start. The server
// owns the effect (tag weights for the next generated level); this UI only
// reports the choice via onVote / tracks the server's VoteOpen..VoteResult flow.
import { useEffect, useRef, useState } from 'react';
import type { JSX } from 'react';
import { VoteWidget } from './vote/VoteWidget';
import { useVote } from './vote/useVote';

export interface LobbyPollProps {
  /** Host starts the poll server-side (protocol owned by T-078/host). */
  onStartPoll?: (() => void) | undefined;
  /** Host reports the chosen option index to the server. */
  onVote?: ((optionIndex: number) => void) | undefined;
  /** Called when the player dismisses the poll without running it. */
  onDismiss?: (() => void) | undefined;
  /** Server messages, e.g. a growing readonly array the HUD appends to. */
  messages?: readonly unknown[] | undefined;
  title?: string | undefined;
}

export function LobbyPoll({
  onStartPoll,
  onVote,
  onDismiss,
  messages,
  title = 'PRE-MATCH POLL',
}: LobbyPollProps): JSX.Element | null {
  const { state, sendVote, ingestMessage, reset } = useVote(onVote);
  const [dismissed, setDismissed] = useState(false);
  const appliedCount = useRef(0);

  useEffect(() => {
    if (messages === undefined) return;
    if (messages.length < appliedCount.current) appliedCount.current = 0;
    for (let i = appliedCount.current; i < messages.length; i += 1) {
      ingestMessage(messages[i]);
    }
    appliedCount.current = messages.length;
  }, [messages, ingestMessage]);

  if (dismissed) return null;

  return (
    <section
      aria-label={title}
      className="w-full max-w-md rounded-xl border border-white/10 bg-[#120f1f] p-4 font-mono text-white"
    >
      <div className="flex items-center justify-between gap-3 border-b border-white/10 pb-2">
        <h2 className="text-sm font-bold tracking-[0.25em] text-[#38e8ff]">{title}</h2>
        <button
          type="button"
          aria-label="Skip the poll"
          onClick={() => {
            setDismissed(true);
            onDismiss?.();
          }}
          className="rounded border border-white/15 px-3 py-1 text-[11px] tracking-widest text-white/70 transition hover:text-white focus:outline-none focus-visible:ring-2 focus-visible:ring-white/40"
        >
          SKIP
        </button>
      </div>

      <p className="mt-2 text-[11px] leading-snug text-white/60">
        Optional — the run can start without it. Theme and difficulty picks are reported and feed
        the pattern engine&apos;s tag weights for the next generated level.
      </p>

      {state.phase === 'idle' ? (
        <button
          type="button"
          onClick={() => onStartPoll?.()}
          className="mt-3 w-full rounded-lg border-2 border-[#ff3db8] px-4 py-3 text-sm font-bold tracking-[0.3em] text-[#ff3db8] transition hover:bg-[#ff3db8] hover:text-black focus:outline-none focus-visible:ring-2 focus-visible:ring-[#ff3db8]/60"
        >
          START POLL
        </button>
      ) : (
        <div className="mt-3">
          <VoteWidget state={state} onVote={sendVote} variant="poll" onDismissReveal={reset} />
        </div>
      )}
    </section>
  );
}

=== NOTES ===
Design decisions
- T-078 is not in the tree. The exact regenerated wire schema (field names, variants) is unknowable from this ticket, so ALL protocol assumptions are isolated in web/src/hud/vote/protocolAdapter.ts, validated at runtime, and documented in its header comment (VoteOpen/VoteState/VoteResult shapes incl. numeric CARD_* kinds; client Vote outbound). It imports only CARD_ENEMY/CARD_CURSE (already generated). If T-078's schema differs, only that file changes. Nothing references VoteOpen/VoteState/VoteResult types by name — impossible before regeneration and brittle after; the literal type tags are used instead.
- Pedestals are a bottom-anchored HUD overlay (pointer-events: none; buttons re-enabled), not in-world meshes: the scene files are not shown in this ticket and only minimal edits to existing HUD files were allowed for mounting. Bottom placement + semi-transparent backdrop keeps the beacon clear; distance readability via clamp(text-base..2xl), text-4xl counts, SVG ring with role="timer". True 3D pedestals need a scene edit (outside allowed paths).
- Poll options arrive via server messages / host-injected `messages` and `onStartPoll`/`onVote` callbacks: no poll-start protocol exists in the shown ticket/T-078 description, and the theme vocabulary lives in design/ (off-limits), so no theme names are invented. The UI reports the choice; the server owns tag weights.
- Local 200 ms countdown ticks re-sync to server VoteState.remainingMs; server owns counts and winner. NEW PERMANENT banner is pedestal-only; poll reveal shows a pattern-engine caption instead.
- Nothing was mounted into existing files: the in-run HUD root / lobby screen filenames are not visible in the ticket, JoinPanel is the title-page join dialog (not a lobby), and MatchList is off-limits for refactor. Mount sites (one line each): in-run HUD `<RunVoteOverlay messages={…} onVote={…} />`; lobby `<LobbyPoll onStartPoll={…} onVote={…} messages={…} />`. No matchmaker/JoinPanel/MatchList edits.
- Tests assume jsdom + @testing-library/react are already devDependencies (no new deps allowed); per-file `// @vitest-environment jsdom`. Timer logic is tested through the pure reducer (manual ticks), no fake timers.

Acceptance criteria
- Three options w/ live counts: met (VoteWidget + tests). Countdown reaching zero + disabled voting: met (reducer tick, ring shows 0 s, buttons disabled, test). Results reveal + winner: met. NEW PERMANENT banner naming the winner: met (data-testid, test). Change-vote until close: met (aria-pressed moves, test with spy). Lobby poll start (one button), optional/skip, reporting choice: met (LobbyPoll tests). Empty state (idle) and error state (failed event / malformed VoteOpen, role="alert"): met. Both `messages` arrays treated as append-only transcripts with an applied-count ref.

Could not do / risks
- tsc/eslint/vitest not executed (no tools); claims verified only by construction.
- Wire-shape assumptions unverified against T-078; adapter is the single point of adjustment.
- Countdown shows between server updates via local ticks — cosmetic only; server remains authoritative.
- If @testing-library/react or jsdom are absent from web/, the component tests fail to run; the reducer/adapter tests still pass in node.
