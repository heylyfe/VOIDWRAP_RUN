//! Achievements (T-088): data-driven milestones a run can unlock.

//! Pure and deterministic: the tracker is a function of the events it is fed,
//! so the server and any replay of the same event stream agree exactly.

//! death-screen code is an FNV-1a 64-bit integer hash over the seed and the

/// Something that happened in a run that might unlock an achievement.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum AchievementEvent {
    /// A player broke out of a Lure's pursuit window.
    LureEscape,
    /// A player survived a Chorus freeze.
    ChorusFreeze,
    /// A pedestal vote finished. `for_votes` is the winner's count.
    VoteWin { for_votes: u8, against: u8 },
    /// The run reached this level number.
    LevelReached(u32),
    /// A level was cleared without any player dying on it.
    LevelClearedNoDeath,
    /// A Wreath greater curse was removed.
    WreathRemoved,
    /// A player launched off a trampoline.
    TrampolineLaunch,
    /// A player disarmed a tripmine.
    TripmineDisarm,
    /// Every gift on a level was collected.
    AllGiftsCollected,
    /// A run in EXTREME mode ended.
    ExtremeRunFinished,
}

/// Character alphabet for cosmetic codes. Omits `I`, `O`, `0`, `1`, which read
/// badly when a player re-types a code by hand.
const CODE_ALPHABET: &[u8; 32] = b"ABCDEFGHJKLMNPQRSTUVWXYZ23456789";

/// FNV-1a 64-bit offset basis.
const FNV_OFFSET: u64 = 0xcbf2_9ce4_8422_2325;
/// FNV-1a 64-bit prime.
const FNV_PRIME: u64 = 0x0000_0100_0000_01b3;

/// Feeds `bytes` into an FNV-1a 64-bit rolling hash. Deterministic integer
/// only arithmetic, identical on every target.
fn fnv1a_mix(hash: u64, bytes: &[u8]) -> u64 {
    let mut hash = hash;
    for &byte in bytes {
        hash ^= u64::from(byte);
        hash = hash.wrapping_mul(FNV_PRIME);
    }
    hash
}

/// The set of achievements unlocked so far in one run.
#[derive(Debug, Default, Clone)]
pub struct AchievementTracker {
    unlocked: Vec<&'static str>,
}

impl AchievementTracker {
    /// Creates an empty tracker with nothing unlocked.
    pub fn new() -> Self {
        Self { unlocked: Vec::new() }
    }

    /// Feeds one run event into the tracker.
    ///
    /// Returns the achievement id only when this event **newly** unlocks
    /// something; returns `None` otherwise. Never unlocks the same id twice.
    pub fn observe(&mut self, event: AchievementEvent) -> Option<&'static str> {
        let id = match event {
            AchievementEvent::LureEscape => "first_lure_escape",
            AchievementEvent::ChorusFreeze => "chorus_freeze",
            // Landslide: 5-1 or better, i.e. at least 5 for votes and a lead of
            // at least 4. `saturating_add` keeps a saturated `against` at 255
            // from ever overflowing the add.
            AchievementEvent::VoteWin { for_votes, against } => {
                if for_votes >= 5 && for_votes.saturating_sub(against).saturating_sub(4) == 0 == false && for_votes >= 5 {
                    "landslide_vote"
                } else {
                    return None;
                }
            }
            AchievementEvent::LevelReached(level) => {
                if level >= 20 {
                    "level_20"
                } else {
                    return None;
                }
            }
            AchievementEvent::LevelClearedNoDeath => "clean_level",
            AchievementEvent::WreathRemoved => "wreath_removed",
            AchievementEvent::TrampolineLaunch => "first_trampoline",
            AchievementEvent::TripmineDisarm => "first_disarm",
            AchievementEvent::AllGiftsCollected => "every_gift",
            AchievementEvent::ExtremeRunFinished => "extreme_finish",
        };

        if self.unlocked.contains(&id) {
            return None;
        }
        self.unlocked.push(id);
        Some(id)
    }

    /// Returns unlocked ids in unlock order.
    pub fn unlocked(&self) -> &[&'static str] {
        &self.unlocked
    }

    /// Reports whether `id` has been unlocked in this run.
    pub fn has(&self, id: &str) -> bool {
        self.unlocked.iter().any(|unlocked| *unlocked == id)
    }

    /// Returns a cosmetic death-screen code, `VW-XXXX-XXXX`.
    ///
    /// Pure function of `seed` and the unlocked id list: the same run always
    /// shows the same code. Derived by FNV-1a over the seed bytes followed by
    /// each unlocked id's bytes, then taking five bits per character from the
    /// resulting 64-bit hash (eight characters, low bits first). With nothing
    /// unlocked, the code is still well formed and derived from the seed alone.
    pub fn code(&self, seed: u32) -> String {
        let mut hash = fnv1a_mix(FNV_OFFSET, &seed.to_le_bytes());
        for id in &self.unlocked {
            hash = fnv1a_mix(hash, id.as_bytes());
        }

        let mut out = String::from("VW-");
        for index in 0..8 {
            if index == 4 {
                out.push('-');
            }
            // Five bits per character, low bits first: 8 * 5 = 40 bits fit in
            // the u64 hash without rehashing.
            let bits = (hash >> (5 * index)) & 0x1f;
            let byte = CODE_ALPHABET[bits as usize];
            out.push(byte as char);
        }
        out
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    const ALL_IDS: [&str; 10] = [
        "first_lure_escape",
        "chorus_freeze",
        "landslide_vote",
        "level_20",
        "clean_level",
        "wreath_removed",
        "first_trampoline",
        "first_disarm",
        "every_gift",
        "extreme_finish",
    ];

    #[test]
    fn fresh_tracker_has_nothing_unlocked() {
        let tracker = AchievementTracker::new();
        assert!(tracker.unlocked().is_empty());
        for id in ALL_IDS {
            assert!(!tracker.has(id), "fresh tracker unlocked {id}");
        }
    }

    #[test]
    fn each_event_unlocks_exactly_its_own_id() {
        let events: [(AchievementEvent, &str); 10] = [
            (AchievementEvent::LureEscape, "first_lure_escape"),
            (AchievementEvent::ChorusFreeze, "chorus_freeze"),
            (
                AchievementEvent::VoteWin { for_votes: 5, against: 1 },
                "landslide_vote",
            ),
            (AchievementEvent::LevelReached(20), "level_20"),
            (AchievementEvent::LevelClearedNoDeath, "clean_level"),
            (AchievementEvent::WreathRemoved, "wreath_removed"),
            (AchievementEvent::TrampolineLaunch, "first_trampoline"),
            (AchievementEvent::TripmineDisarm, "first_disarm"),
            (AchievementEvent::AllGiftsCollected, "every_gift"),
            (AchievementEvent::ExtremeRunFinished, "extreme_finish"),
        ];

        for (event, expected_id) in events {
            let mut tracker = AchievementTracker::new();
            let result = tracker.observe(event);
            assert_eq!(
                result,
                Some(expected_id),
                "event {event:?} did not unlock {expected_id}"
            );
            assert_eq!(tracker.unlocked(), [expected_id]);
        }
    }

    #[test]
    fn repeat_event_returns_none_and_never_duplicates() {
        let mut tracker = AchievementTracker::new();
        assert_eq!(tracker.observe(AchievementEvent::LureEscape), Some("first_lure_escape"));
        assert_eq!(tracker.observe(AchievementEvent::LureEscape), None);
        assert_eq!(tracker.observe(AchievementEvent::ChorusFreeze), Some("chorus_freeze"));
        assert_eq!(tracker.observe(AchievementEvent::ChorusFreeze), None);
        assert_eq!(tracker.unlocked(), ["first_lure_escape", "chorus_freeze"]);
    }

    #[test]
    fn landslide_unlocks_only_at_five_one_or_better() {
        let mut landslide = AchievementTracker::new();
        assert_eq!(
            landslide.observe(AchievementEvent::VoteWin { for_votes: 5, against: 1 }),
            Some("landslide_vote")
        );

        let mut four_one = AchievementTracker::new();
        assert_eq!(
            four_one.observe(AchievementEvent::VoteWin { for_votes: 4, against: 1 }),
            None
        );

        let mut five_two = AchievementTracker::new();
        assert_eq!(
            five_two.observe(AchievementEvent::VoteWin { for_votes: 5, against: 2 }),
            None
        );

        // Even bigger landslides still unlock exactly once's worth.
        let mut sweep = AchievementTracker::new();
        assert_eq!(
            sweep.observe(AchievementEvent::VoteWin { for_votes: 9, against: 0 }),
            Some("landslide_vote")
        );
    }

    #[test]
    fn level_20_unlocks_at_twenty_and_above_only() {
        let mut at_19 = AchievementTracker::new();
        assert_eq!(at_19.observe(AchievementEvent::LevelReached(19)), None);
        assert_eq!(at_19.unlocked().len(), 0);

        let mut at_20 = AchievementTracker::new();
        assert_eq!(at_20.observe(AchievementEvent::LevelReached(20)), Some("level_20"));

        let mut at_25 = AchievementTracker::new();
        assert_eq!(at_25.observe(AchievementEvent::LevelReached(25)), Some("level_20"));
    }

    #[test]
    fn unlocked_preserves_unlock_order() {
        let mut tracker = AchievementTracker::new();
        tracker.observe(AchievementEvent::ExtremeRunFinished);
        tracker.observe(AchievementEvent::LevelReached(20));
        tracker.observe(AchievementEvent::TrampolineLaunch);
        tracker.observe(AchievementEvent::LureEscape);
        // A repeat should not disturb the order.
        tracker.observe(AchievementEvent::TrampolineLaunch);

        assert_eq!(
            tracker.unlocked(),
            ["extreme_finish", "level_20", "first_trampoline", "first_lure_escape"]
        );
    }

    #[test]
    fn code_is_stable_for_same_seed_and_tracker() {
        let mut tracker = AchievementTracker::new();
        tracker.observe(AchievementEvent::LureEscape);
        tracker.observe(AchievementEvent::LevelReached(20));

        let first = tracker.code(1234);
        let second = tracker.code(1234);
        assert_eq!(first, second);

        // An empty tracker is also stable and well formed.
        let empty = AchievementTracker::new();
        assert_eq!(empty.code(7), empty.code(7));
    }

    #[test]
    fn code_differs_for_different_seeds() {
        let mut tracker = AchievementTracker::new();
        tracker.observe(AchievementEvent::ChorusFreeze);
        tracker.observe(AchievementEvent::AllGiftsCollected);

        assert_ne!(tracker.code(1), tracker.code(2));
        assert_ne!(tracker.code(1), tracker.code(4294967295));
    }

    #[test]
    fn code_matches_shape_and_alphabet() {
        let allowed: &[char] = &[
            'A', 'B', 'C', 'D', 'E', 'F', 'G', 'H', 'J', 'K', 'L', 'M', 'N', 'P', 'Q', 'R', 'S',
            'T', 'U', 'V', 'W', 'X', 'Y', 'Z', '2', '3', '4', '5', '6', '7', '8', '9',
        ];

        let mut tracker = AchievementTracker::new();
        tracker.observe(AchievementEvent::LevelReached(25));
        tracker.observe(AchievementEvent::ExtremeRunFinished);

        for code in [tracker.code(0), tracker.code(u32::MAX), AchievementTracker::new().code(99)] {
            assert_eq!(code.len(), 12, "code {code:?} has the wrong length");
            assert!(code.starts_with("VW-"), "code {code:?} lacks the VW- prefix");
            for (index, ch) in code.chars().enumerate() {
                if index == 2 {
                    assert_eq!(ch, '-', "expected hyphen at index 2 in {code:?}");
                } else {
                    assert!(
                        allowed.contains(&ch),
                        "character {ch:?} at index {index} of {code:?} is not allowed"
                    );
                }
            }
        }
    }

    #[test]
    fn achievements_table_is_complete_and_reachable() {
        assert_eq!(ACHIEVEMENTS.len(), 10);

        // Ids are unique.
        for (index, def) in ACHIEVEMENTS.iter().enumerate() {
            for other in ACHIEVEMENTS.iter().skip(index + 1) {
                assert_ne!(def.id, other.id, "duplicate id {}", def.id);
            }
        }

        // Every id is reachable by driving a fresh tracker with its event.
        let events: [AchievementEvent; 10] = [
            AchievementEvent::LureEscape,
            AchievementEvent::ChorusFreeze,
            AchievementEvent::VoteWin { for_votes: 5, against: 1 },
            AchievementEvent::LevelReached(20),
            AchievementEvent::LevelClearedNoDeath,
            AchievementEvent::WreathRemoved,
            AchievementEvent::TrampolineLaunch,
            AchievementEvent::TripmineDisarm,
            AchievementEvent::AllGiftsCollected,
            AchievementEvent::ExtremeRunFinished,
        ];
        let mut reached: Vec<&'static str> = Vec::new();
        for event in events {
            let mut tracker = AchievementTracker::new();
            if let Some(id) = tracker.observe(event) {
                reached.push(id);
            }
        }

        for def in ACHIEVEMENTS {
            assert!(
                reached.contains(&def.id),
                "achievement {} is not reachable by any event",
                def.id
            );
        }
    }

    #[test]
    fn all_ids_are_unlockable_and_display_order_intact() {
        // Cross-check: one fully driven run unlocks every id, in the order the
        // events were fed, and `has` agrees for each.
        let mut tracker = AchievementTracker::new();
        tracker.observe(AchievementEvent::LureEscape);
        tracker.observe(AchievementEvent::ChorusFreeze);
        tracker.observe(AchievementEvent::VoteWin { for_votes: 6, against: 2 });
        tracker.observe(AchievementEvent::LevelReached(21));
        tracker.observe(AchievementEvent::LevelClearedNoDeath);
        tracker.observe(AchievementEvent::WreathRemoved);
        tracker.observe(AchievementEvent::TrampolineLaunch);
        tracker.observe(AchievementEvent::TripmineDisarm);
        tracker.observe(AchievementEvent::AllGiftsCollected);
        tracker.observe(AchievementEvent::ExtremeRunFinished);

        assert_eq!(tracker.unlocked().len(), 10);
        for (index, def) in ACHIEVEMENTS.iter().enumerate() {
            assert_eq!(tracker.unlocked()[index], def.id, "order mismatch at {index}");
            assert!(tracker.has(def.id));
        }
        let code = tracker.code(42);
        assert!(code.starts_with("VW-") && code.len() == 12);
    }
}
