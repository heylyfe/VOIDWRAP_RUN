//! Integration tests for `sim::hazards`: the nine reworked enemies, the heavy-attack
//! mutex, ward immunity rules, crumble/fretted tiles and determinism.
//!
//! Everything here runs against the public crate API; small levels are built inline.
//! Timing expectations come from the design (20 Hz server): 0.6 s = 12 ticks,
//! 0.75 s = 15 ticks, 0.9 s = 18 ticks, 1.1 s = 22 ticks, 2.0 s = 40 ticks,
//! 2.5 s = 50 ticks, 4.0 s = 80 ticks.

use protocol::{
    Aabb, DEATH_ENEMY, DEATH_PROJECTILE, ENEMY_ATTACK, ENEMY_DISGUISED, ENEMY_DORMANT,
    ENEMY_FLAG_HEAVY, ENEMY_HUNTING, ENEMY_RECOVER, ENEMY_TELEGRAPH, LevelData, Platform,
    PLATFORM_CRUMBLE, TILE_FRETTED, TILE_GONE, TILE_WARN, Vec3,
};
use sim::hazards::{HazardEffect, Hazards, PlayerSense};
use sim::rules::{
    spawn_enemy_with, Brain, Enemy, EnemyParams, LanternLeechBrain, LeechPhase, SnapshrikePhase,
    BEHAVIOR_CORRUPTER, BEHAVIOR_GALE_WIGHT, BEHAVIOR_LANTERN_LEECH, BEHAVIOR_MIMIC,
    BEHAVIOR_MOTELING, BEHAVIOR_PALLBEARER, BEHAVIOR_RIFT_WARDEN, BEHAVIOR_SNAPSHRIKE,
    BEHAVIOR_TURRET,
};
use sim::world::World;

fn v3(x: f32, y: f32, z: f32) -> Vec3 {
    Vec3::new(x, y, z)
}

fn assert_vec3_near(a: &Vec3, b: &Vec3) {
    assert!((a.x - b.x).abs() < 1e-4, "x mismatch: {} vs {}", a.x, b.x);
    assert!((a.y - b.y).abs() < 1e-4, "y mismatch: {} vs {}", a.y, b.y);
    assert!((a.z - b.z).abs() < 1e-4, "z mismatch: {} vs {}", a.z, b.z);
}

fn platform(id: u16, kind: u8, min: [f32; 3], max: [f32; 3]) -> Platform {
    Platform {
        id,
        kind,
        bounds: Aabb {
            min: v3(min[0], min[1], min[2]),
            max: v3(max[0], max[1], max[2]),
        },
    }
}

fn test_level(platforms: Vec<Platform>, seed: u32) -> LevelData {
    LevelData {
        seed,
        level: 1,
        kill_y: -20.0,
        platforms,
    }
}

fn empty_level(seed: u32) -> LevelData {
    test_level(Vec::new(), seed)
}

fn params(behavior: u8) -> EnemyParams {
    EnemyParams {
        behavior,
        speed_mul: 1.0,
        size: 0.5,
        grace_ticks: 0,
    }
}

fn enemy(id: u32, behavior: u8, x: f32, y: f32, z: f32) -> Enemy {
    spawn_enemy_with(id, 0, v3(x, y, z), params(behavior))
}

fn sense(id: u32, x: f32, y: f32, z: f32) -> PlayerSense {
    PlayerSense {
        id,
        feet: v3(x, y, z),
        vel: v3(0.0, 0.0, 0.0),
        warded: false,
    }
}

fn sense_vel(id: u32, feet: Vec3, vel: Vec3, warded: bool) -> PlayerSense {
    PlayerSense { id, feet, vel, warded }
}

fn tick(
    hz: &mut Hazards,
    enemies: &mut [Enemy],
    level: &LevelData,
    world: &World,
    players: &[PlayerSense],
    pickups: &[Vec3],
) -> Vec<HazardEffect> {
    hz.step(enemies, world, level, players, pickups, 1.0)
}

/// Burn the level-wide 4.0 s no-harm window with an empty roster.
fn warm(hz: &mut Hazards, level: &LevelData, world: &World) {
    for _ in 0..80 {
        let mut none = Vec::new();
        hz.step(&mut none, world, level, &[], &[], 1.0);
    }
}

fn kills(fx: &[HazardEffect]) -> Vec<u32> {
    fx.iter()
        .filter_map(|e| match e {
            HazardEffect::Kill { player, .. } => Some(*player),
            _ => None,
        })
        .collect()
}

fn kill_causes(fx: &[HazardEffect]) -> Vec<u8> {
    fx.iter()
        .filter_map(|e| match e {
            HazardEffect::Kill { cause, .. } => Some(*cause),
            _ => None,
        })
        .collect()
}

fn shove_count(fx: &[HazardEffect]) -> usize {
    fx.iter()
        .filter(|e| matches!(e, HazardEffect::Shove { .. }))
        .count()
}

fn view_of(hz: &Hazards, enemies: &[Enemy], id: u32) -> protocol::EnemyView {
    hz.views(enemies)
        .into_iter()
        .find(|v| v.id == id)
        .expect("view exists")
}

fn has_heavy(v: &protocol::EnemyView) -> bool {
    v.flags & ENEMY_FLAG_HEAVY != 0
}

    warm(&mut hz, &level, &world);
    let p = EnemyParams {
        grace_ticks: 5,
        ..params(BEHAVIOR_SNAPSHRIKE)
    };
    let mut enemies = vec![spawn_enemy_with(1, 0, v3(1.0, 0.0, 0.0), p)];
    let players = vec![sense(1, 0.0, 0.0, 0.0)];
    let start = enemies[0].pos.clone();

    assert_ne!(view_of(&hz, &enemies, 1).state, ENEMY_DORMANT);

#[test]
fn pallbearer_hunt_kneel_lunge_recover_sequence() {
    let level = empty_level(21);
    let world = World::from_level(&level);
    let mut hz = Hazards::new();
    warm(&mut hz, &level, &world);
    let mut enemies = vec![enemy(1, BEHAVIOR_PALLBEARER, 4.0, 0.0, 0.0)];
    let players = vec![sense(1, 0.0, 0.0, 0.0)];
    let mut kneel_at = None;
    let mut kill_at = None;
    let mut saw_attack = false;
    let mut killed_any = false;
    for t in 0..400 {
        let fx = tick(&mut hz, &mut enemies, &level, &world, &players, &[]);
        if !kills(&fx).is_empty() {
            killed_any = true;
        }
        let v = view_of(&hz, &enemies, 1);
        match v.state {
            ENEMY_TELEGRAPH => {
                if kneel_at.is_none() {
                    kneel_at = Some(t);
                }
            }
            ENEMY_ATTACK => {
                saw_attack = true;
                if kill_at.is_none() && kills(&fx).contains(&1) {
                    kill_at = Some(t);
                    break;
                }
            }
            _ => {}
        }
    }
    let kneel = kneel_at.expect("pallbearer rears back before lunging");
    let kill = kill_at.expect("lunge connects with a stationary target");
    assert!(saw_attack);
    assert!(
        kill - kneel >= 12,
        ">= 0.6 s telegraph before the lethal lunge"
    );
    assert!(killed_any);
    // After the lunge it drifts harmlessly, then returns to hunting.
    let mut saw_recover = false;
    let mut saw_hunt = false;
    for _ in 0..120 {
        tick(&mut hz, &mut enemies, &level, &world, &[], &[]);
        let v = view_of(&hz, &enemies, 1);
        if v.state == ENEMY_RECOVER {
            saw_recover = true;
        }
        if saw_recover && v.state == ENEMY_HUNTING {
            saw_hunt = true;
            break;
        }
    }
    assert!(saw_recover && saw_hunt);
}

#[test]
fn snapshrike_aim_is_frozen_for_final_035s_and_a_perpendicular_step_dodges() {
    let level = empty_level(32);
    let world = World::from_level(&level);
    let mut hz = Hazards::new();
    warm(&mut hz, &level, &world);
    let mut enemies = vec![enemy(1, BEHAVIOR_SNAPSHRIKE, 6.0, 0.0, 0.0)];
    let mut players = vec![sense_vel(1, v3(0.0, 0.0, 0.0), v3(0.0, 0.0, 6.0), false)];
    let mut targets = Vec::new();
    for t in 0..40 {
        if t == 3 {
            // Change velocity during the lock so the tracked aim provably moves.
            players[0].vel = v3(0.0, 0.0, -6.0);
        }
        players[0].feet.z += players[0].vel.z * 0.05;
        let _ = tick(&mut hz, &mut enemies, &level, &world, &players, &[]);
        let v = view_of(&hz, &enemies, 1);
        if v.state == ENEMY_TELEGRAPH {
            targets.push(v.target);
        } else if !targets.is_empty() {
            break;
        }
    }
    let n = targets.len();
    assert!(n >= 13, "lock telegraph ~0.75 s, got {n} samples");
    let last7 = &targets[n - 7..];
    for t in last7 {
        assert_eq!(t, &last7[0], "aim frozen for the final 0.35 s");
    }
    assert_ne!(targets[0], last7[0], "tracked aim moved before freezing");
    // A perpendicular step during the frozen window dodges the dash.
    players[0].feet.x += 2.6;
    let mut killed = false;
    for _ in 0..30 {
        let fx = tick(&mut hz, &mut enemies, &level, &world, &players, &[]);
        if !kills(&fx).is_empty() {
            killed = true;
            break;
        }
    }
    assert!(!killed, "a perpendicular step dodges the frozen aim");
}

#[test]
fn rift_warden_ignite_then_radial_shove() {
    let level = empty_level(51);
    let world = World::from_level(&level);
    let mut hz = Hazards::new();
    warm(&mut hz, &level, &world);
    let mut enemies = vec![enemy(1, BEHAVIOR_RIFT_WARDEN, 3.0, 0.0, 0.0)];
    let players = vec![sense(1, 0.0, 0.0, 0.0)];
    let mut tele_at = None;
    let mut shove = None;
    let mut killed_any = false;
    for t in 0..200 {
        let fx = tick(&mut hz, &mut enemies, &level, &world, &players, &[]);
        if !kills(&fx).is_empty() {
            killed_any = true;
        }
        let v = view_of(&hz, &enemies, 1);
        if v.state == ENEMY_TELEGRAPH && tele_at.is_none() {
            tele_at = Some(t);
        }
        for e in &fx {
            if let HazardEffect::Shove {
                player,
                vel,
                stagger_ticks,
            } = e
            {
                assert_eq!(*player, 1);
                shove = Some((t, vel.clone(), *stagger_ticks));
            }
        }
        if shove.is_some() {
            break;
        }
    }
    let (t, vel, stagger) = shove.expect("the ignition wave displaces the player");
    let tele = tele_at.expect("1.1 s rune telegraph");
    assert!(t - tele >= 12, "shove follows the >= 0.6 s telegraph");
    assert_eq!(stagger, 18, "0.9 s stagger");
    assert!((vel.y - 2.0).abs() < 1e-6, "shove launches upward");
    assert!((vel.x + 8.0).abs() < 1e-6, "8 u/s radially away");
    assert_eq!(vel.z, 0.0);
    assert!(!killed_any, "the rift warden never kills");
}

    /// Advance tiles, enemies, projectiles and crumble detection by one server tick.
    ///
    /// `players` are the living players; everything else (aim points, stillness
    /// checks, target centres) is derived from them. `pickups` are the positions
    /// of gifts collected this tick (lantern_leech). `speed_mul` is the phase-2
    /// enemy speed multiplier (1.0 or 1.1). Effects are ordered by enemy spawn
    /// order, then by player order. The first 4.0 s of a level and dormant
    /// enemies never produce effects.
    pub fn step(
        &mut self,
        enemies: &mut [Enemy],
        world: &World,
        level: &LevelData,
        players: &[PlayerSense],
        pickups: &[Vec3],
        speed_mul: f32,
    ) -> Vec<HazardEffect> {
        if !self.rng_seeded {
            let seed = u64::from(level.seed).wrapping_add(1).wrapping_mul(0x9E37_79B9_7F4A_7C15);
            self.rng = RNG_SEED ^ seed | 1;
            self.rng_seeded = true;
        }
        let damage_ok = self.ticks_elapsed >= LEVEL_GRACE_TICKS;
        self.ticks_elapsed = self.ticks_elapsed.saturating_add(1);

        let expired = self.age_tiles();
        self.sample_histories(players);

        // Snapshot positions so the rift warden can find its neighbours without
        // holding a second live borrow of the enemy slice.
        let enemy_positions: Vec<Vec3> = enemies.iter().map(|enemy| enemy.pos.clone()).collect();
        let ctx = Ctx {
            world,
            level,
            players,
            pickups,
            speed_mul,
            damage_ok,
            positions: &enemy_positions,
        };
        let mut fx: Vec<HazardEffect> = Vec::new();
        for i in 0..enemies.len() {
            if let Some(enemy) = enemies.get_mut(i) {
                step_one(self, i, enemy, &ctx, &mut fx);
            }
        }
        // Rift warden shoves ripple through other enemies (not the warden itself).
        for (index, vel) in self.take_pending_enemy_shoves() {
            if let Some(other) = enemies.get_mut(index) {
                other.knock_vel = vel;
                other.knock_ticks = RIFT_STAGGER_TICKS;
            }
        }
        self.step_projectiles(&ctx, &mut fx);
        self.mark_crumble(level, players, &expired);

        if !damage_ok {
            fx.clear();
        }
        fx
    }

fn moteling_step(
    h: &mut Hazards,
    id: u32,
    pos: &mut Vec3,
    b: &mut MotelingBrain,

    home: &Vec3,
    ctx: &Ctx,

                _ => {
                    let dir = h.rng_flat_direction();
                    let reach = 2.0 + (h.next_rng() % 7000) as f32 / 1000.0;
                    b.wander_goal = Some(Vec3::new(pos.x + dir.x * reach, home.y, pos.z + dir.z * reach));
                }
            }
            if let Some(target) = moteling_pick_target(h, ctx.players, pos) {
                b.marked = Some(target);
                b.phase = MotelingPhase::Orbit;
                b.orbit_ticks = 0;
                b.acquired = false;
                b.heavy_wait = 0;
            }
        }
        MotelingPhase::Orbit => {
            if b.marked.is_none() {
                b.marked = moteling_pick_target(h, ctx.players, pos);
                if b.marked.is_none() {
                    b.phase = MotelingPhase::Wander;
                    return;
                }
            }
            let Some(marked) = b.marked else {
                b.phase = MotelingPhase::Wander;
                return;
            };
            let found = ctx.players.iter().find(|p| p.id == marked);
            let Some(target) = found else {
                b.marked = None; // target died
                b.phase = MotelingPhase::Wander;
                return;
            };
            let centre = player_centre(&target.feet);
            b.anchor = centre.clone();
            b.orbit_ticks += 1;
            let dx = pos.x - centre.x;
            let dz = pos.z - centre.z;
            let len = (dx * dx + dz * dz).sqrt();
            let (rx, rz) = if len > 1e-4 { (dx / len, dz / len) } else { (1.0, 0.0) };
            let corr = (MOTELING_ORBIT_RADIUS - len).clamp(-2.5, 2.5);
            pos.x += (-rz * MOTELING_ORBIT_SPEED + rx * corr) * speed_mul * DT;
            pos.z += (rx * MOTELING_ORBIT_SPEED + rz * corr) * speed_mul * DT;
            if b.orbit_ticks >= MOTELING_ORBIT_PERIOD_TICKS {
                b.phase = MotelingPhase::Inhale;
                b.timer = MOTELING_INHALE_TICKS;
                b.acquired = false;
                b.heavy_wait = 0;
            }
        }
        MotelingPhase::Inhale => {
            let Some(marked) = b.marked else {
                b.phase = MotelingPhase::Cooldown;
                b.timer = MOTELING_COOLDOWN_TICKS;
                return;
            };
            let found = ctx.players.iter().find(|p| p.id == marked);
            let Some(target) = found else {
                b.marked = None;
                release_heavy(h, id, b);
                b.phase = MotelingPhase::Cooldown;
                b.timer = MOTELING_COOLDOWN_TICKS;
                return;
            };
            b.anchor = player_centre(&target.feet);
            if !b.acquired {
                if h.heavy_holder.is_none() {
                    b.acquired = true;
                    h.heavy_holder = Some(id);
                } else {
                    // Heavy slot is taken: wait frozen at the first frame.
                    b.heavy_wait += 1;
                    if b.heavy_wait > HEAVY_WAIT_MAX_TICKS {
                        b.marked = None;
                        b.orbit_ticks = 0;
                        b.heavy_wait = 0;
                        b.phase = MotelingPhase::Cooldown;
                        b.timer = MOTELING_COOLDOWN_TICKS;
                    }
                    return;
                }
            }
            b.timer = b.timer.saturating_sub(1);
            if b.timer == 0 {
                release_heavy(h, id, b);
                let hs = (target.vel.x * target.vel.x + target.vel.z * target.vel.z).sqrt();
                // The ring snaps shut: only a marked player who kept moving is hurt.
                // Routed through push_kill so a warded marked player survives.
                if hs > MOTELING_STILL_SPEED {
                    push_kill(fx, target, DEATH_ENEMY);
                }
                b.marked = None;
                b.orbit_ticks = 0;
                b.heavy_wait = 0;
                b.phase = MotelingPhase::Cooldown;
                b.timer = MOTELING_COOLDOWN_TICKS;
            }
        }
        MotelingPhase::Cooldown => {
            move_toward(pos, home, MOTELING_DRIFT_HOME * speed_mul * DT);
            b.timer = b.timer.saturating_sub(1);
            if b.timer == 0 {
                b.phase = MotelingPhase::Orbit;
                b.orbit_ticks = MOTELING_ORBIT_PERIOD_TICKS / 2; // re-approaches the ring before the next inhale
                b.marked = None;
            }
        }
    }
    // Virtue: never report the shriek radius anywhere except the Kill effect.
    let _ = MOTELING_SHRIEK_RADIUS;
}

=== FN: server/crates/sim/src/hazards.rs :: rift_warden_step ===
#[allow(clippy::too_many_arguments)]
fn rift_warden_step(
    h: &mut Hazards,
    self_index: usize,
    pos: &mut Vec3,
    b: &mut RiftWardenBrain,
    speed_mul: f32,
    contact_radius: f32,
    home: &Vec3,
    ctx: &Ctx,
    fx: &mut Vec<HazardEffect>,
) {
    match b.phase {
        RiftPhase::Haunt => {
            let target = nearest_index_3d(ctx.players, pos);
            match target.and_then(|i| ctx.players.get(i)) {
                None => move_toward(pos, home, RIFT_HAUNT_SPEED * speed_mul * DT),
                Some(target) => {
                    let centre = player_centre(&target.feet);
                    move_toward(pos, &centre, RIFT_HAUNT_SPEED * speed_mul * DT);
                    if sphere_hits_player_box(pos, contact_radius, &target.feet) {
                        let dir = flat_direction(pos, &centre);
                        push_shove(
                            fx,
                            target,
                            Vec3::new(dir.x * RIFT_CONTACT_SHOVE, 0.0, dir.z * RIFT_CONTACT_SHOVE),
                            0,
                        );
                    }
                    if distance(pos, &centre) <= RIFT_TRIGGER_RADIUS {
                        b.phase = RiftPhase::Ignite;
                        b.timer = RIFT_IGNITE_TICKS;
                    }
                }
            }
        }
        RiftPhase::Ignite => {
            b.timer = b.timer.saturating_sub(1);
            if b.timer == 0 {
                b.phase = RiftPhase::Shove;
                b.timer = RIFT_SHOVE_TICKS;
                // Radial displacement: players are shoved for real, enemies are knocked back.
                for player in ctx.players {
                    let centre = player_centre(&player.feet);
                    let d = distance(pos, &centre);
                    if d > RIFT_SHOVE_RADIUS {
                        continue;
                    }
                    let dir = flat_direction(pos, &centre);
                    push_shove(
                        fx,
                        player,
                        Vec3::new(dir.x * RIFT_SHOVE_SPEED, RIFT_SHOVE_UP, dir.z * RIFT_SHOVE_SPEED),
                        RIFT_STAGGER_TICKS,
                    );
                }
                for (index, vel) in enemies_from_ctx_players(ctx, pos, self_index, RIFT_SHOVE_RADIUS) {
                    h.pending_enemy_shoves.push((index, vel));
                }
            }
        }
        RiftPhase::Shove => {
            b.timer = b.timer.saturating_sub(1);
            if b.timer == 0 {
                b.phase = RiftPhase::Settle;
                b.timer = RIFT_SETTLE_TICKS;
            }
        }
        RiftPhase::Settle => {
            move_toward(pos, home, RIFT_CREEP_SPEED * speed_mul * DT);
            b.timer = b.timer.saturating_sub(1);
            if b.timer == 0 {
                b.phase = RiftPhase::Haunt;
            }
        }
    }
}

=== FN: server/crates/sim/src/hazards.rs :: enemies_from_ctx_players ===
/// Collect `(index, knockback velocity)` pairs for every enemy other than
/// `self_index` within `radius` of `centre`. Used by the rift warden's pulse; the
/// velocities are queued in [`Hazards::take_pending_enemy_shoves`] and applied after
/// the enemy loop, because stepping an enemy cannot hold a second live borrow of the
/// slice. Y carries the same upward kick as the player shove (`+2 u/s`).
fn enemies_from_ctx_players(ctx: &Ctx, centre: &Vec3, self_index: usize, radius: f32) -> Vec<(usize, Vec3)> {
    let mut out = Vec::new();
    for (index, pos) in ctx.positions.iter().enumerate() {
        if index == self_index {
            continue;
        }
        if distance(centre, pos) > radius {
            continue;
        }
        let dir = flat_direction(centre, pos);
        out.push((
            index,
            Vec3::new(dir.x * RIFT_SHOVE_SPEED, RIFT_SHOVE_UP, dir.z * RIFT_SHOVE_SPEED),
        ));
    }
    out
}

=== FN: server/crates/sim/src/hazards.rs :: step_one ===
fn step_one(h: &mut Hazards, index: usize, e: &mut Enemy, ctx: &Ctx, fx: &mut Vec<HazardEffect>) {
    if e.knock_ticks > 0 {
        e.pos.x += e.knock_vel.x * DT;
        e.pos.y += e.knock_vel.y * DT;
        e.pos.z += e.knock_vel.z * DT;
        e.knock_ticks -= 1;
        return;
    }
    if e.active_in > 0 {
        e.active_in -= 1;
        return;
    }
    let id = e.id;
    let speed_mul = e.params.speed_mul * ctx.speed_mul;
    let contact_radius = e.params.size * ENEMY_HIT_FORGIVENESS;
    let home = e.home.clone();
    match &mut e.brain {
        Brain::Pallbearer(b) => pallbearer_step(h, &mut e.pos, b, speed_mul, contact_radius, &home, ctx, fx),
        Brain::Snapshrike(b) => snapshrike_step(h, id, &mut e.pos, b, speed_mul, contact_radius, &home, ctx, fx),
        Brain::Moteling(b) => moteling_step(h, id, &mut e.pos, b, speed_mul, &home, ctx, fx),
        Brain::RiftWarden(b) => rift_warden_step(h, index, &mut e.pos, b, speed_mul, contact_radius, &home, ctx, fx),
        Brain::GaleWight(b) => gale_wight_step(h, id, &mut e.pos, b, speed_mul, &home, ctx),
        Brain::LanternLeech(b) => lantern_leech_step(&mut e.pos, b, speed_mul, &home, ctx, fx),
        Brain::Turret(b) => turret_step(h, &mut e.pos, b, speed_mul, ctx),
        Brain::Mimic(b) => mimic_step(&mut e.pos, b, speed_mul, ctx, fx),
        Brain::Corrupter(b) => corrupter_step(h, &mut e.pos, b, speed_mul, &home, ctx),
    }
    // The pallbearer never descends below y = 0.
    if e.pos.y < 0.0 {
        e.pos.y = 0.0;
    }
}
