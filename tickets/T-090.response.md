#[cfg(test)]
mod qa {
    //! T-090: scripted playthrough sweeps. These are the automated QA pass: they
    //! drive whole runs across every difficulty mode and assert the invariants a
    //! player would notice breaking — no softlock, no stuck phase, no runaway state.
    use super::*;

    /// Hard cap on the number of levels one sweep may drive.
    const MAX_LEVELS: usize = 40;
    /// Hard cap on the number of gifts one level may contain.
    const MAX_GIFTS: usize = 64;
    /// Hard cap on the number of ticks spent standing on a beacon.
    const MAX_BEACON_TICKS: usize = 8;
    /// Hard cap on the number of draft rounds cleared in one step.
    const MAX_DRAFTS: usize = 4;
    /// A wallet above this value means the `u32` balance wrapped around.
    const CURRENCY_CEILING: u32 = 1_000_000;
    /// Vertical gap between a ground marker and the player origin.
    const GROUND_OFFSET: f32 = 0.9;

    fn game() -> Game {
        Game::starting_at(2, 42).unwrap()
    }

    fn connect(game: &mut Game, id: ConnId) -> mpsc::Receiver<Vec<u8>> {
        let (tx, rx) = mpsc::channel(OUTBOX_CAPACITY);
        game.handle(GameEvent::Connected { id, out: tx });
        rx
    }

    fn hello(game: &mut Game, id: ConnId) {
        game.handle(GameEvent::Message {
            id,
            msg: ClientMsg::Hello {
                protocol_version: PROTOCOL_VERSION,
                name: "  tester\u{7}  ".into(),
            },
        });
    }

    fn drain(rx: &mut mpsc::Receiver<Vec<u8>>) -> Vec<ServerMsg> {
        let mut out = vec![];
        while let Ok(b) = rx.try_recv() {
            out.push(protocol::decode::<ServerMsg>(&b).unwrap());
        }
        out
    }

    fn player(g: &mut Game, id: ConnId) -> &mut Player {
        g.conns.get_mut(&id).unwrap().player.as_mut().unwrap()
    }

    fn vote(g: &mut Game, id: ConnId) {
        let picks = valid_picks(&g.run);
        g.handle(GameEvent::Message {
            id,
            msg: ClientMsg::DraftPick { picks },
        });
    }

    /// Keeps every enemy asleep so a sweep measures level flow, not combat.
    fn dormant_enemies(g: &mut Game) {
        for e in &mut g.run.level.enemies {
            e.active_in = 1_000;
        }
    }

    /// Places the player on a ground marker and lets one tick resolve the pickup.
    fn stand_on(g: &mut Game, id: ConnId, pos: Vec3, rx: &mut mpsc::Receiver<Vec<u8>>) {
        let p = player(g, id);
        p.state.pos = pos;
        p.state.vel = Vec3::ZERO;
        p.starved_ticks = 0;
        dormant_enemies(g);
        g.tick();
        drain(rx);
    }

    /// Player position that sits on top of the first gift still on the floor.
    fn next_gift_pos(g: &Game) -> Option<Vec3> {
        let gift = g.run.level.gifts.first()?;
        Some(Vec3::new(gift.pos.x, gift.pos.y - GROUND_OFFSET, gift.pos.z))
    }

    /// Teleports the player onto every remaining gift, one tick per gift.
    ///
    /// The loop always retries the *first* gift still present, so a pickup that
    /// does not register is retried a bounded number of times instead of hanging.
    fn collect_gifts(g: &mut Game, id: ConnId, rx: &mut mpsc::Receiver<Vec<u8>>) {
        for _ in 0..MAX_GIFTS {
            let Some(pos) = next_gift_pos(g) else {
                return;
            };
            stand_on(g, id, pos, rx);
        }
    }

    /// Stands the player on the level beacon until the level resolves.
    fn walk_to_beacon(g: &mut Game, id: ConnId, rx: &mut mpsc::Receiver<Vec<u8>>) {
        for _ in 0..MAX_BEACON_TICKS {
            let beacon = Vec3::new(
                g.run.level.beacon.x,
                g.run.level.beacon.y - GROUND_OFFSET,
                g.run.level.beacon.z,
            );
            stand_on(g, id, beacon, rx);
            if g.run.phase != PHASE_PLAYING {
                return;
            }
        }
    }

    /// Collects every gift and then walks to the beacon: one full level.
    fn clear_level(g: &mut Game, id: ConnId, rx: &mut mpsc::Receiver<Vec<u8>>) {
        collect_gifts(g, id, rx);
        walk_to_beacon(g, id, rx);
    }

    /// Votes through every draft the run is currently sitting in.
    fn clear_drafts(g: &mut Game, id: ConnId, rx: &mut mpsc::Receiver<Vec<u8>>) {
        for _ in 0..MAX_DRAFTS {
            if g.run.phase != PHASE_ENEMY_DRAFT && g.run.phase != PHASE_REWARD_DRAFT {
                return;
            }
            vote(g, id);
            drain(rx);
        }
    }

    /// Fails with the level number when the run sits in a phase it can never leave.
    fn assert_legal_phase(g: &Game, context: &str) {
        assert!(
            g.run.phase == PHASE_ENEMY_DRAFT
                || g.run.phase == PHASE_PLAYING
                || g.run.phase == PHASE_REWARD_DRAFT
                || g.run.phase == PHASE_OVER,
            "{context}: level {} is stuck in phase {:?}",
            g.run.level_num,
            g.run.phase
        );
    }

    /// True when the message carries a run-state snapshot.
    fn is_run_state(msg: &ServerMsg) -> bool {
        matches!(msg, ServerMsg::RunState { .. })
    }

    /// True when the message tells a client the run moved on.
    fn is_run_progress(msg: &ServerMsg) -> bool {
        matches!(msg, ServerMsg::RunOver { .. } | ServerMsg::RunState { .. })
    }

    #[test]
    fn standard_run_reaches_level_five_without_softlock() {
        let mut g = game();
        let mut rx = connect(&mut g, 1);
        hello(&mut g, 1);
        vote(&mut g, 1);
        drain(&mut rx);

        let mut cleared = 0usize;
        let mut highest = g.run.level_num;

        for _ in 0..MAX_LEVELS {
            if g.run.phase == PHASE_OVER {
                break;
            }
            assert_legal_phase(&g, "before clearing a level");
            clear_drafts(&mut g, 1, &mut rx);

            let before = g.run.level_num;
            clear_level(&mut g, 1, &mut rx);
            highest = highest.max(g.run.level_num);

            if g.run.phase == PHASE_OVER {
                break;
            }
            assert_eq!(
                g.run.phase,
                PHASE_ENEMY_DRAFT,
                "level {} did not return to the draft",
                g.run.level_num
            );
            assert!(
                g.run.level_num > before,
                "level {} did not advance past {before}",
                g.run.level_num
            );
            assert_legal_phase(&g, "after clearing a level");
            cleared += 1;
        }

        assert!(cleared >= 3, "only cleared {cleared} levels");
        assert!(highest >= 5, "run only reached level {highest}");
    }

    #[test]
    fn every_mode_completes_a_level() {
        let modes: Vec<sim::mode::DifficultyMode> =
            sim::mode::DifficultyMode::all().iter().copied().collect();
        assert!(!modes.is_empty(), "no difficulty modes to sweep");

        for mode in modes {
            let mut g = game();
            g.run.mode = mode;
            let mut rx = connect(&mut g, 1);
            hello(&mut g, 1);
            vote(&mut g, 1);
            drain(&mut rx);
            clear_drafts(&mut g, 1, &mut rx);

            let before = g.run.level_num;
            clear_level(&mut g, 1, &mut rx);

            assert_eq!(
                g.run.phase,
                PHASE_ENEMY_DRAFT,
                "mode {:?} did not finish level {before}",
                g.run.mode
            );
            assert_eq!(
                g.run.level_num,
                before + 1,
                "mode {:?} did not advance exactly one level",
                g.run.mode
            );
        }
    }

    #[test]
    fn eight_players_all_receive_run_state_and_can_vote() {
        let mut g = game();
        let mut clients: Vec<(ConnId, mpsc::Receiver<Vec<u8>>)> = Vec::new();
        for id in 1..=8 {
            let rx = connect(&mut g, id);
            clients.push((id, rx));
        }
        assert_eq!(g.conns.len(), 8, "not every client was admitted");

        for (id, _) in &clients {
            hello(&mut g, *id);
        }
        assert_eq!(g.conns.len(), 8, "a hello dropped a connection");

        for (id, _) in &clients {
            vote(&mut g, *id);
        }
        assert_eq!(g.conns.len(), 8, "a vote dropped a connection");

        for (id, rx) in &mut clients {
            let msgs = drain(rx);
            assert!(
                msgs.iter().any(is_run_state),
                "client {id} never received a RunState"
            );
        }
        assert_eq!(g.conns.len(), 8, "the draft dropped a connection");

        clear_drafts(&mut g, 1, &mut clients[0].1);
        clear_level(&mut g, 1, &mut clients[0].1);
        assert_eq!(g.conns.len(), 8, "the level transition dropped a connection");

        for (id, rx) in &mut clients {
            let msgs = drain(rx);
            assert!(
                msgs.iter().any(is_run_progress),
                "client {id} was dropped from the post-level broadcast"
            );
        }
        assert_eq!(g.conns.len(), 8, "the run dropped a connection");
    }

    #[test]
    fn shop_and_altars_do_not_break_the_run() {
        let mut g = game();
        let mut rx = connect(&mut g, 1);
        hello(&mut g, 1);
        vote(&mut g, 1);
        drain(&mut rx);

        let mut reached_shop = false;
        for _ in 0..MAX_LEVELS {
            if g.run.phase == PHASE_OVER {
                break;
            }
            if g.run.shop.is_some() && g.run.level_num >= 3 {
                reached_shop = true;
                break;
            }
            clear_drafts(&mut g, 1, &mut rx);
            clear_level(&mut g, 1, &mut rx);
            assert_legal_phase(&g, "while driving to the shop");
        }
        assert!(
            reached_shop,
            "never reached a level with a shop (stopped at level {})",
            g.run.level_num
        );

        clear_drafts(&mut g, 1, &mut rx);
        assert_legal_phase(&g, "before shopping");

        // The ticket does not pin the shop's internal shape, so we probe a
        // bounded slot range instead of indexing the offer list directly.
        for item in 0..8 {
            g.handle(GameEvent::Message {
                id: 1,
                msg: ClientMsg::Buy { item },
            });
            g.tick();
            drain(&mut rx);
            assert!(
                g.run.currency < CURRENCY_CEILING,
                "level {}: currency wrapped to {} after buying slot {item}",
                g.run.level_num,
                g.run.currency
            );
            assert_legal_phase(&g, "after buying a shop item");
        }

        clear_drafts(&mut g, 1, &mut rx);
        let before = g.run.level_num;
        clear_level(&mut g, 1, &mut rx);
        assert_eq!(
            g.run.phase,
            PHASE_ENEMY_DRAFT,
            "level {before} did not complete after shopping"
        );
        assert!(
            g.run.level_num > before,
            "level {before} did not advance after shopping"
        );
    }
}

=== NOTES ===
- Assumed `g.run.level.beacon` is a `Vec3` (ticket says "walk the player to `g.run.level.beacon`" with no field access). If it is a struct, only `walk_to_beacon` needs `.pos`.
- Assumed enemies live in `g.run.level.enemies` with an integer `active_in` (ticket only shows `e.active_in = 1_000`).
- Assumed `ClientMsg::Buy { item }` takes an integer slot index. The shop's internal shape is not specified, so the test probes slots `0..8` rather than reading the offer list; out-of-range/too-expensive buys must be rejected, which is exactly the invariant under test.
- Assumed `DifficultyMode::all()` returns a slice/Vec (used `.iter().copied()`); `ServerMsg::RunState`/`RunOver` matched with `{ .. }` so unit or struct variants both compile.
- Gift sweep makes one pass over `g.run.level.gifts`, always retrying the first remaining gift; this covers normal and golden gifts because the ticket exposes no golden discriminator. Bounded by `MAX_GIFTS`.
- Beacon walk re-teleports and ticks up to 8 times, stopping as soon as the phase leaves `PHASE_PLAYING`.
- A run that ends in `PHASE_OVER` is treated as a legitimate end, not a softlock; the `PHASE_ENEMY_DRAFT` assertion only applies while the run is still live.
- All loops are `for _ in 0..N` with named caps; the only `while` is the ticket-supplied `drain` body, which terminates on `try_recv` error.
- No `#[ignore]` used: nothing here was observed to be broken (tests were not executed in this environment).
- Acceptance criteria: four tests present, every loop capped, assertions name level/phase, no non-test code touched.
