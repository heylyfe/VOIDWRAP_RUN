// Deterministic pseudo-random helpers shared by procedural scene content.
// No dependencies, no global state: every generator is created from an explicit seed.

/**
 * Mulberry32 PRNG. Returns a function producing floats in [0, 1).
 * Same seed always yields the same sequence.
 */
export function mulberry32(seed: number): () => number {
  let state = seed >>> 0;
  return function next(): number {
    state = (state + 0x6d2b79f5) >>> 0;
    let t = state;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/** 32-bit FNV-1a hash of a string, as an unsigned integer. */
export function hashString(s: string): number {
  let hash = 0x811c9dc5;
  for (let i = 0; i < s.length; i += 1) {
    hash ^= s.charCodeAt(i);
    hash = Math.imul(hash, 0x01000193);
  }
  return hash >>> 0;
}

"use client";

import { useEffect, useMemo, useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import { BackSide, Color, ShaderMaterial, SphereGeometry, Vector3 } from "three";
import type { Mesh } from "three";
import type { Theme } from "./theme";

const SKY_VERTEX_SHADER = `
varying vec3 vWorldPosition;

void main() {
  vec4 worldPosition = modelMatrix * vec4(position, 1.0);
  vWorldPosition = worldPosition.xyz;
  gl_Position = projectionMatrix * viewMatrix * worldPosition;
}
`;

const SKY_FRAGMENT_SHADER = `
uniform vec3 uZenith;
uniform vec3 uHorizon;
uniform vec3 uCloud;
uniform vec3 uCloudDark;
uniform vec3 uCenter;
uniform float uTime;

varying vec3 vWorldPosition;

// Hash-based value noise (no textures, fully procedural).
float hash13(vec3 p) {
  p = fract(p * 0.1031);
  p += dot(p, p.zyx + 31.32);
  return fract((p.x + p.y) * p.z);
}

float valueNoise(vec3 p) {
  vec3 i = floor(p);
  vec3 f = fract(p);
  vec3 u = f * f * (3.0 - 2.0 * f);

  float n000 = hash13(i + vec3(0.0, 0.0, 0.0));
  float n100 = hash13(i + vec3(1.0, 0.0, 0.0));
  float n010 = hash13(i + vec3(0.0, 1.0, 0.0));
  float n110 = hash13(i + vec3(1.0, 1.0, 0.0));
  float n001 = hash13(i + vec3(0.0, 0.0, 1.0));
  float n101 = hash13(i + vec3(1.0, 0.0, 1.0));
  float n011 = hash13(i + vec3(0.0, 1.0, 1.0));
  float n111 = hash13(i + vec3(1.0, 1.0, 1.0));

  return mix(
    mix(mix(n000, n100, u.x), mix(n010, n110, u.x), u.y),
    mix(mix(n001, n101, u.x), mix(n011, n111, u.x), u.y),
    u.z
  );
}

float fbm(vec3 p) {
  float sum = 0.0;
  float amplitude = 0.5;
  for (int i = 0; i < 5; i++) {
    sum += amplitude * valueNoise(p);
    p = p * 2.02 + vec3(11.3, 7.7, 5.1);
    amplitude *= 0.5;
  }
  return sum;
}

void main() {
  vec3 dir = normalize(vWorldPosition - uCenter);

  // Vertical gradient: horizon colour low, zenith colour high. No seam because
  // the direction is continuous over the whole sphere.
  float height = smoothstep(-0.15, 0.85, dir.y);
  vec3 base = mix(uHorizon, uZenith, height);

  // Soft billowy clouds: slowly scrolling 3D domain sampled with fbm.
  vec3 cloudDomain = dir * 3.0 + vec3(uTime * 0.01, 0.0, uTime * 0.006);
  float cloudNoise = fbm(cloudDomain);
  float cloudMask = smoothstep(0.42, 0.82, cloudNoise);
  vec3 colour = mix(base, uCloud, cloudMask * 0.65);

  // Darker marbled streaks: lower frequency, strongest near the horizon.
  vec3 marbleDomain = dir * 1.35 + vec3(uTime * 0.004, uTime * 0.0025, 0.0);
  float marbleNoise = fbm(marbleDomain);
  float marbleMask = smoothstep(0.48, 0.9, marbleNoise);
  float horizonWeight = 1.0 - smoothstep(-0.1, 0.8, dir.y);
  colour = mix(colour, uCloudDark, marbleMask * horizonWeight * 0.55);

  gl_FragColor = vec4(colour, 1.0);

  #include <colorspace_fragment>
}
`;

type SkyUniforms = {
  uZenith: { value: Color };
  uHorizon: { value: Color };
  uCloud: { value: Color };
  uCloudDark: { value: Color };
  uCenter: { value: Vector3 };
  uTime: { value: number };
};

const DOME_RADIUS = 450;

export function Atmosphere(props: { theme: Theme }): JSX.Element {
  const { theme } = props;

  const domeRef = useRef<Mesh>(null);

  const uniforms = useMemo<SkyUniforms>(
    () => ({
      uZenith: { value: new Color() },
      uHorizon: { value: new Color() },
      uCloud: { value: new Color() },
      uCloudDark: { value: new Color() },
      uCenter: { value: new Vector3() },
      uTime: { value: 0 },
    }),
    [],
  );

  const geometry = useMemo(() => new SphereGeometry(DOME_RADIUS, 48, 24), []);

  const material = useMemo(
    () =>
      new ShaderMaterial({
        uniforms,
        vertexShader: SKY_VERTEX_SHADER,
        fragmentShader: SKY_FRAGMENT_SHADER,
        side: BackSide,
        depthWrite: false,
        fog: false,
      }),
    [uniforms],
  );

  // Theme changes only mutate uniform values: the material is never re-created.
  useEffect(() => {
    uniforms.uZenith.value.set(theme.sky.zenith);
    uniforms.uHorizon.value.set(theme.sky.horizon);
    uniforms.uCloud.value.set(theme.sky.cloud);
    uniforms.uCloudDark.value.set(theme.sky.cloudDark);
  }, [theme, uniforms]);

  useEffect(
    () => () => {
      geometry.dispose();
    },
    [geometry],
  );

  useEffect(
    () => () => {
      material.dispose();
    },
    [material],
  );

  useFrame((state) => {
    const dome = domeRef.current;
    if (!dome) return;
    const { x, y, z } = state.camera.position;
    dome.position.set(x, y, z);
    uniforms.uCenter.value.set(x, y, z);
    uniforms.uTime.value = state.clock.elapsedTime;
  });

  return (
    <>
      <color attach="background" args={[theme.sky.horizon]} />
      <fog attach="fog" args={[theme.fog.color, theme.fog.near, theme.fog.far]} />
      <mesh
        ref={domeRef}
        geometry={geometry}
        material={material}
        frustumCulled={false}
        renderOrder={-1}
      />
    </>
  );
}

"use client";

import { useEffect, useMemo, useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import { BoxGeometry, Color, MeshStandardMaterial, Object3D } from "three";
import type { InstancedMesh } from "three";
import type { Theme } from "./theme";
import { mulberry32 } from "./rng";

export interface DebrisItem {
  position: [number, number, number];
  scale: number;
  spin: [number, number, number];
  orbitSpeed: number;
  bobPhase: number;
}

const MIN_SCALE = 0.15;
const MAX_SCALE = 0.9;
const MIN_SPIN = -1.2;
const MAX_SPIN = 1.2;
const MIN_ORBIT = -0.03;
const MAX_ORBIT = 0.03;
const BOB_AMPLITUDE = 0.6;
const BOB_SPEED = 0.8;
const CLUSTER_FRACTION = 0.3;
const CLUSTER_SPREAD = 2.0;

function clamp(value: number, min: number, max: number): number {
  return Math.min(Math.max(value, min), max);
}

/**
 * Deterministic debris layout: a thick shell of cubes around `center`, with
 * roughly 30% of them grouped into small swarms.
 */
export function debrisLayout(
  seed: number,
  count: number,
  center: { x: number; y: number; z: number },
  radius: number,
): DebrisItem[] {
  const rand = mulberry32(seed);
  const total = Math.max(0, Math.floor(count));
  const items: DebrisItem[] = [];
  if (total === 0) return items;

  const innerRadius = radius * 0.55;
  const outerRadius = radius * 1.15;
  const minY = center.y - 12;
  const maxY = center.y + 18;

  const makeItem = (x: number, y: number, z: number): DebrisItem => {
    const scaleBias = rand();
    return {
      position: [x, y, z],
      scale: MIN_SCALE + scaleBias * scaleBias * (MAX_SCALE - MIN_SCALE),
      spin: [
        MIN_SPIN + rand() * (MAX_SPIN - MIN_SPIN),
        MIN_SPIN + rand() * (MAX_SPIN - MIN_SPIN),
        MIN_SPIN + rand() * (MAX_SPIN - MIN_SPIN),
      ],
      orbitSpeed: MIN_ORBIT + rand() * (MAX_ORBIT - MIN_ORBIT),
      bobPhase: rand() * Math.PI * 2,
    };
  };

  const shellPoint = (): [number, number, number] => {
    const angle = rand() * Math.PI * 2;
    const distance = innerRadius + rand() * (outerRadius - innerRadius);
    const y = minY + rand() * (maxY - minY);
    return [center.x + Math.cos(angle) * distance, y, center.z + Math.sin(angle) * distance];
  };

  const clusterPoint = (anchor: [number, number, number]): [number, number, number] => {
    const offsetX = anchor[0] - center.x;
    const offsetZ = anchor[2] - center.z;
    const baseAngle = Math.atan2(offsetZ, offsetX);
    const baseDistance = Math.hypot(offsetX, offsetZ);
    const tangential = (rand() - 0.5) * CLUSTER_SPREAD * 2;
    const angle = baseAngle + tangential / Math.max(baseDistance, 1);
    const distance = clamp(
      baseDistance + (rand() - 0.5) * CLUSTER_SPREAD * 2,
      innerRadius,
      outerRadius,
    );
    const y = clamp(anchor[1] + (rand() - 0.5) * CLUSTER_SPREAD * 2, minY, maxY);
    return [center.x + Math.cos(angle) * distance, y, center.z + Math.sin(angle) * distance];
  };

  const clusterTarget = Math.round(total * CLUSTER_FRACTION);
  while (items.length < clusterTarget) {
    const anchor = shellPoint();
    const clusterSize = 3 + Math.floor(rand() * 4);
    const remaining = Math.min(clusterSize, clusterTarget - items.length, total - items.length);
    for (let i = 0; i < remaining; i += 1) {
      const point = i === 0 ? anchor : clusterPoint(anchor);
      items.push(makeItem(point[0], point[1], point[2]));
    }
  }

  while (items.length < total) {
    const point = shellPoint();
    items.push(makeItem(point[0], point[1], point[2]));
  }

  return items;
}

export function Debris(props: {
  theme: Theme;
  center: { x: number; y: number; z: number };
  radius: number;
  count?: number;
  seed?: number;
}): JSX.Element {
  const { theme, center, radius, count = 140, seed = 1337 } = props;
  const { x: centerX, y: centerY, z: centerZ } = center;

  const layout = useMemo(
    () => debrisLayout(seed, count, { x: centerX, y: centerY, z: centerZ }, radius),
    [seed, count, centerX, centerY, centerZ, radius],
  );

  const geometry = useMemo(() => new BoxGeometry(1, 1, 1), []);

  const material = useMemo(
    () =>
      new MeshStandardMaterial({
        color: new Color(theme.debris),
        roughness: 0.6,
        metalness: 0.2,
        emissive: new Color(theme.accent),
        emissiveIntensity: 0.05,
      }),
    [theme.debris, theme.accent],
  );

  const dummy = useMemo(() => new Object3D(), []);
  const meshRef = useRef<InstancedMesh>(null);

  useEffect(
    () => () => {
      geometry.dispose();
    },
    [geometry],
  );

  useEffect(
    () => () => {
      material.dispose();
    },
    [material],
  );

  useFrame((state) => {
    const mesh = meshRef.current;
    if (!mesh) return;
    const time = state.clock.elapsedTime;

    for (let i = 0; i < layout.length; i += 1) {
      const item = layout[i];
      if (!item) continue;

      const angle = item.orbitSpeed * time;
      const cos = Math.cos(angle);
      const sin = Math.sin(angle);
      const offsetX = item.position[0] - centerX;
      const offsetZ = item.position[2] - centerZ;

      dummy.position.set(
        centerX + offsetX * cos - offsetZ * sin,
        item.position[1] + Math.sin(time * BOB_SPEED + item.bobPhase) * BOB_AMPLITUDE,
        centerZ + offsetX * sin + offsetZ * cos,
      );
      dummy.rotation.set(item.spin[0] * time, item.spin[1] * time, item.spin[2] * time);
      dummy.scale.setScalar(item.scale);
      dummy.updateMatrix();
      mesh.setMatrixAt(i, dummy.matrix);
    }

    mesh.instanceMatrix.needsUpdate = true;
  });

  return (
    <instancedMesh
      ref={meshRef}
      args={[undefined, undefined, layout.length]}
      geometry={geometry}
      material={material}
      frustumCulled={false}
    />
  );
}

import { describe, expect, it } from "vitest";
import { debrisLayout } from "./Debris";
import { mulberry32 } from "./rng";

const CENTER = { x: 10, y: 4, z: -6 };
const RADIUS = 40;
const EPSILON = 1e-6;

function distance(a: [number, number, number], b: [number, number, number]): number {
  return Math.hypot(a[0] - b[0], a[1] - b[1], a[2] - b[2]);
}

describe("mulberry32", () => {
  it("is deterministic for the same seed", () => {
    const first = mulberry32(42);
    const second = mulberry32(42);
    const firstSequence = Array.from({ length: 16 }, () => first());
    const secondSequence = Array.from({ length: 16 }, () => second());
    expect(firstSequence).toEqual(secondSequence);
  });

  it("produces different sequences for different seeds", () => {
    const first = mulberry32(1);
    const second = mulberry32(2);
    const firstSequence = Array.from({ length: 16 }, () => first());
    const secondSequence = Array.from({ length: 16 }, () => second());
    expect(firstSequence).not.toEqual(secondSequence);
  });

  it("returns floats in [0, 1)", () => {
    const rand = mulberry32(7);
    for (let i = 0; i < 1000; i += 1) {
      const value = rand();
      expect(value).toBeGreaterThanOrEqual(0);
      expect(value).toBeLessThan(1);
    }
  });
});

describe("debrisLayout", () => {
  it("is deterministic for the same seed", () => {
    const first = debrisLayout(1234, 60, CENTER, RADIUS);
    const second = debrisLayout(1234, 60, CENTER, RADIUS);
    expect(first).toEqual(second);
  });

  it("differs for a different seed", () => {
    const first = debrisLayout(1234, 60, CENTER, RADIUS);
    const second = debrisLayout(4321, 60, CENTER, RADIUS);
    expect(first).not.toEqual(second);
  });

  it("returns exactly the requested number of items", () => {
    expect(debrisLayout(9, 140, CENTER, RADIUS)).toHaveLength(140);
    expect(debrisLayout(9, 1, CENTER, RADIUS)).toHaveLength(1);
    expect(debrisLayout(9, 0, CENTER, RADIUS)).toHaveLength(0);
  });

  it("keeps every item inside the horizontal shell and the y band", () => {
    const items = debrisLayout(77, 200, CENTER, RADIUS);
    for (const item of items) {
      const dx = item.position[0] - CENTER.x;
      const dz = item.position[2] - CENTER.z;
      const horizontal = Math.hypot(dx, dz);
      expect(horizontal).toBeGreaterThanOrEqual(RADIUS * 0.55 - EPSILON);
      expect(horizontal).toBeLessThanOrEqual(RADIUS * 1.15 + EPSILON);
      expect(item.position[1]).toBeGreaterThanOrEqual(CENTER.y - 12 - EPSILON);
      expect(item.position[1]).toBeLessThanOrEqual(CENTER.y + 18 + EPSILON);
    }
  });

  it("keeps scale, spin, orbit speed and bob phase in range", () => {
    const items = debrisLayout(5, 200, CENTER, RADIUS);
    for (const item of items) {
      expect(item.scale).toBeGreaterThanOrEqual(0.15);
      expect(item.scale).toBeLessThanOrEqual(0.9);
      for (const axis of item.spin) {
        expect(axis).toBeGreaterThanOrEqual(-1.2);
        expect(axis).toBeLessThanOrEqual(1.2);
      }
      expect(item.orbitSpeed).toBeGreaterThanOrEqual(-0.03);
      expect(item.orbitSpeed).toBeLessThanOrEqual(0.03);
      expect(item.bobPhase).toBeGreaterThanOrEqual(0);
      expect(item.bobPhase).toBeLessThan(Math.PI * 2);
    }
  });

  it("groups roughly 30% of the items into clusters", () => {
    const items = debrisLayout(2024, 100, CENTER, RADIUS);
    let clustered = 0;
    for (const item of items) {
      const hasNeighbour = items.some(
        (other) => other !== item && distance(item.position, other.position) < 6,
      );
      if (hasNeighbour) clustered += 1;
    }
    expect(clustered).toBeGreaterThanOrEqual(20);
  });
});

=== NOTES ===
- Colour management: uniforms are THREE.Color built from the theme hex strings, so three converts them sRGB -> linear working space; the fragment shader therefore outputs linear colour and ends with `#include <colorspace_fragment>` (renderer-provided `linearToOutputTexel`) to convert to the output colour space. Tone mapping is intentionally not applied to the sky.
- `JSX.Element` is imported as `import type { JSX } from "react"` because React 19 types no longer declare a global JSX namespace.
- Sky dome: full sphere (no seam possible), BackSide, depthWrite false, fog false, frustumCulled false, renderOrder -1; position and uCenter are copied from the camera each frame so the gradient is always camera-relative. Theme changes only call `Color.set` on existing uniforms (material created once in useMemo, disposed on unmount).
- Debris: single InstancedMesh (one draw call), unit BoxGeometry + MeshStandardMaterial (roughness 0.6, metalness 0.2, emissive = theme.accent at 0.05). Material is re-created only when theme.debris/theme.accent change and the previous one is disposed by the effect cleanup; geometry disposed on unmount.
- Zero per-frame allocations: one preallocated Object3D reused for every instance; orbit/bob/tumble computed from clock time and the memoised layout.
- debrisLayout: mulberry32-driven; ~30% of items placed in clusters of 3-6 around a shell anchor, cluster members jittered in world units and clamped back into the [0.55r, 1.15r] shell and the y band, so the shell/bounds invariants hold for every item.
- `args={[undefined, undefined, layout.length]}` on instancedMesh keeps the mesh alive across theme changes (geometry/material supplied as props instead of constructor args).
- Test file imports Debris.tsx, which pulls in @react-three/fiber at module load; that import is DOM-free, so it is safe in the node vitest environment. No WebGL/document usage in the tested code.
