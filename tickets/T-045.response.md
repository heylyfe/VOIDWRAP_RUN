//! Draft content (enemies, upgrades, curses, classes, shop, events, altars) loaded from design/content.json —
//! the same file the client reads, so card ids, numbers and text cannot drift. Compiled into the binary;
//! validated at startup and in tests. The M2 sections are optional in the file so older content still loads.

use protocol::Tuning;
use serde::Deserialize;
use sim::rules::{BEHAVIOR_CHASER, BEHAVIOR_CORRUPTER, BEHAVIOR_MIMIC, BEHAVIOR_TURRET, EnemyParams};
use std::collections::BTreeMap;

const CONTENT_JSON: &str = include_str!("../../../../design/content.json");

/// Every mod key the run understands; the validator rejects anything else.
const KNOWN_MODS: [&str; 15] = [
    "run_speed_mul",
    "jump_mul",
    "gravity_mul",
    "dash_speed_mul",
    "dash_cooldown_mul",
    "gift_radius_mul",
    "enemy_speed_mul",
    "enemy_aggro_mul",
    "enemy_grace_mul",
    "enemy_size_mul",
    "fog_density_mul",
    "beacon_radius_mul",
    "air_jumps_add",
    "extra_lives_add",
    "extra_enemies_add",
];

fn chaser() -> String {
    "chaser".to_string()
}

fn one() -> f32 {
    1.0
}

#[derive(Debug, Clone, Deserialize)]
pub struct EnemyStats {
    pub speed: f32,
    pub aggro: f32,
    pub grace_s: f32,
    pub size: f32,
    /// "chaser" | "turret" | "mimic" | "corrupter"; unknown values degrade to chaser.
    #[serde(default = "chaser")]
    pub behavior: String,
    /// Seconds between turret shots (turrets only).
    #[serde(default)]
    pub fire_interval_s: Option<f32>,
    /// Projectile speed in units/s (turrets only).
    #[serde(default)]
    pub projectile_speed: Option<f32>,
    /// Distance at which a disguised mimic reveals itself.
    #[serde(default)]
    pub reveal_radius: Option<f32>,
    /// Seconds between a corrupter's platform rots.
    #[serde(default)]
    pub corrupt_interval_s: Option<f32>,
}

#[derive(Debug, Clone, Deserialize)]
pub struct EnemyDef {
    pub id: String,
    pub params: EnemyStats,
}

#[derive(Debug, Clone, Deserialize)]
pub struct ModCard {
    pub id: String,
    pub mods: BTreeMap<String, f32>,
    #[serde(default)]
    pub greater: bool,
}

/// A playable class; order in content must match the CLASS_* protocol constants.
#[derive(Debug, Clone, Deserialize)]
pub struct ClassDef {
    pub id: String,
}

/// An item sold in the lobby shop. `kind`: "life" | "reroll" | "mods".
#[derive(Debug, Clone, Deserialize)]
pub struct ShopItemDef {
    pub id: String,
    pub cost: u32,
    pub stock: u8,
    pub kind: String,
    #[serde(default)]
    pub mods: BTreeMap<String, f32>,
}

/// A level event (EVENT_*); `friction_mul` only means something for EVENT_ICE.
#[derive(Debug, Clone, Deserialize)]
pub struct EventDef {
    pub id: String,
    pub event: u8,
    #[serde(default = "one")]
    pub friction_mul: f32,
}

/// An interactable shrine; its id matches nothing on the wire (levelgen altar ids do).
#[derive(Debug, Clone, Deserialize)]
pub struct AltarDef {
    pub id: String,
}

#[derive(Debug, Clone, Deserialize)]
pub struct Content {
    pub enemies: Vec<EnemyDef>,
    pub upgrades: Vec<ModCard>,
    pub curses: Vec<ModCard>,
    #[serde(default)]
    pub classes: Vec<ClassDef>,
    #[serde(default)]
    pub shop: Vec<ShopItemDef>,
    #[serde(default)]
    pub events: Vec<EventDef>,
    #[serde(default)]
    pub altars: Vec<AltarDef>,
}

impl Content {
    /// Parse the embedded content. A malformed file is a build-time mistake caught by tests and `just verify`.
    pub fn load() -> Result<Content, String> {
        let c: Content = serde_json::from_str(CONTENT_JSON).map_err(|e| format!("design/content.json: {e}"))?;
        if c.enemies.len() < 4 || c.upgrades.len() < 3 || c.curses.iter().filter(|k| !k.greater).count() < 3 {
            return Err("design/content.json: not enough cards for a draft".into());
        }
        if c.curses.iter().filter(|k| k.greater).count() < 3 {
            return Err("design/content.json: need at least 3 greater curses".into());
        }
        // Shop: unique ids, known kinds, known mod keys.
        let mut shop_ids = std::collections::BTreeSet::new();
        for item in &c.shop {
            if !shop_ids.insert(item.id.as_str()) {
                return Err(format!("design/content.json: duplicate shop item {}", item.id));
            }
            match item.kind.as_str() {
                "life" | "reroll" => {}
                "mods" => {
                    if item.mods.is_empty() {
                        return Err(format!("design/content.json: shop item {} has no mods", item.id));
                    }
                    for key in item.mods.keys() {
                        if !KNOWN_MODS.contains(&key.as_str()) {
                            return Err(format!("design/content.json: shop item {}: unknown mod {key}", item.id));
                        }
                    }
                }
                other => {
                    return Err(format!(
                        "design/content.json: shop item {}: unknown kind {other}",
                        item.id
                    ));
                }
            }
        }
        // Events: unique event numbers and a sane friction multiplier.
        let mut seen_events = std::collections::BTreeSet::new();
        for event in &c.events {
            if !seen_events.insert(event.event) {
                return Err(format!("design/content.json: duplicate event {}", event.event));
            }
            if !event.friction_mul.is_finite() || event.friction_mul <= 0.0 {
                return Err(format!("design/content.json: event {} has invalid friction_mul", event.id));
            }
        }
        // Classes: the first four, in order, are the CLASS_* protocol constants.
        if !c.classes.is_empty() {
            let expected = ["striker", "grappler", "blinker", "glider"];
            let valid = c.classes.len() >= expected.len()
                && c.classes.iter().zip(expected).all(|(class, want)| class.id == want);
            if !valid {
                return Err("design/content.json: classes must start with striker, grappler, blinker, glider".into());
            }
        }
        // Altars: unique ids.
        let mut altar_ids = std::collections::BTreeSet::new();
        for altar in &c.altars {
            if !altar_ids.insert(altar.id.as_str()) {
                return Err(format!("design/content.json: duplicate altar {}", altar.id));
            }
        }
        Ok(c)
    }

    /// Index of an enemy in content order (sent to clients as `EnemyView.kind`) and its definition.
    pub fn enemy(&self, id: &str) -> Option<(usize, &EnemyDef)> {
        self.enemies.iter().enumerate().find(|(_, e)| e.id == id)
    }

    pub fn upgrade(&self, id: &str) -> Option<&ModCard> {
        self.upgrades.iter().find(|e| e.id == id)
    }

    pub fn curse(&self, id: &str) -> Option<&ModCard> {
        self.curses.iter().find(|e| e.id == id)
    }

    /// Shop item definition by id.
    pub fn shop_item(&self, id: &str) -> Option<&ShopItemDef> {
        self.shop.iter().find(|e| e.id == id)
    }

    /// Event definition (EVENT_*) by event number, e.g. `EVENT_ICE`.
    pub fn event(&self, event: u8) -> Option<&EventDef> {
        self.events.iter().find(|e| e.event == event)
    }
}

/// Accumulated effect of every upgrade and curse drafted this run (multipliers multiply, additions add).
#[derive(Debug, Clone, PartialEq)]
pub struct RunMods {
    pub run_speed_mul: f32,
    pub jump_mul: f32,
    pub gravity_mul: f32,
    pub dash_speed_mul: f32,
    pub dash_cooldown_mul: f32,
    pub air_jumps: u32,
    pub gift_radius_mul: f32,
    pub extra_lives: u32,
    pub enemy_speed_mul: f32,
    pub enemy_aggro_mul: f32,
    pub enemy_grace_mul: f32,
    pub enemy_size_mul: f32,
    pub extra_enemies: u32,
    pub fog_density_mul: f32,
    pub beacon_radius_mul: f32,
}

impl Default for RunMods {
    fn default() -> Self {
        RunMods {
            run_speed_mul: 1.0,
            jump_mul: 1.0,
            gravity_mul: 1.0,
            dash_speed_mul: 1.0,
            dash_cooldown_mul: 1.0,
            air_jumps: 0,
            gift_radius_mul: 1.0,
            extra_lives: 0,
            enemy_speed_mul: 1.0,
            enemy_aggro_mul: 1.0,
            enemy_grace_mul: 1.0,
            enemy_size_mul: 1.0,
            extra_enemies: 0,
            fog_density_mul: 1.0,
            beacon_radius_mul: 1.0,
        }
    }
}

/// Hard caps on stacked effects so no combination of cards can make a level impossible or absurd.
const MOVE_MUL_RANGE: (f32, f32) = (0.8, 1.8);
const JUMP_MUL_RANGE: (f32, f32) = (0.85, 1.6);
const GRAVITY_MUL_RANGE: (f32, f32) = (0.5, 1.25);
const GENERIC_MUL_RANGE: (f32, f32) = (0.4, 2.5);
const MAX_AIR_JUMPS: u32 = 3;
const MAX_EXTRA_ENEMIES: u32 = 6;
const MAX_LIVES: u32 = 9;

impl RunMods {
    /// Fold one card's mods in. Unknown keys are ignored (the content validator rejects them before shipping).
    pub fn apply(&mut self, mods: &BTreeMap<String, f32>) {
        for (key, &v) in mods {
            let v = if v.is_finite() { v } else { continue };
            match key.as_str() {
                "run_speed_mul" => self.run_speed_mul *= v,
                "jump_mul" => self.jump_mul *= v,
                "gravity_mul" => self.gravity_mul *= v,
                "dash_speed_mul" => self.dash_speed_mul *= v,
                "dash_cooldown_mul" => self.dash_cooldown_mul *= v,
                "gift_radius_mul" => self.gift_radius_mul *= v,
                "enemy_speed_mul" => self.enemy_speed_mul *= v,
                "enemy_aggro_mul" => self.enemy_aggro_mul *= v,
                "enemy_grace_mul" => self.enemy_grace_mul *= v,
                "enemy_size_mul" => self.enemy_size_mul *= v,
                "fog_density_mul" => self.fog_density_mul *= v,
                "beacon_radius_mul" => self.beacon_radius_mul *= v,
                "air_jumps_add" => self.air_jumps = self.air_jumps.saturating_add(v.max(0.0) as u32),
                "extra_lives_add" => self.extra_lives = self.extra_lives.saturating_add(v.max(0.0) as u32),
                "extra_enemies_add" => self.extra_enemies = self.extra_enemies.saturating_add(v.max(0.0) as u32),
                _ => {}
            }
        }
        self.clamp();
    }

    fn clamp(&mut self) {
        let c = |v: f32, (lo, hi): (f32, f32)| v.clamp(lo, hi);
        self.run_speed_mul = c(self.run_speed_mul, MOVE_MUL_RANGE);
        self.jump_mul = c(self.jump_mul, JUMP_MUL_RANGE);
        self.gravity_mul = c(self.gravity_mul, GRAVITY_MUL_RANGE);
        self.dash_speed_mul = c(self.dash_speed_mul, GENERIC_MUL_RANGE);
        self.dash_cooldown_mul = c(self.dash_cooldown_mul, GENERIC_MUL_RANGE);
        self.gift_radius_mul = c(self.gift_radius_mul, GENERIC_MUL_RANGE);
        self.enemy_speed_mul = c(self.enemy_speed_mul, GENERIC_MUL_RANGE);
        self.enemy_aggro_mul = c(self.enemy_aggro_mul, GENERIC_MUL_RANGE);
        self.enemy_grace_mul = c(self.enemy_grace_mul, GENERIC_MUL_RANGE);
        self.enemy_size_mul = c(self.enemy_size_mul, GENERIC_MUL_RANGE);
        self.fog_density_mul = c(self.fog_density_mul, GENERIC_MUL_RANGE);
        self.beacon_radius_mul = c(self.beacon_radius_mul, GENERIC_MUL_RANGE);
        self.air_jumps = self.air_jumps.min(MAX_AIR_JUMPS);
        self.extra_enemies = self.extra_enemies.min(MAX_EXTRA_ENEMIES);
        self.extra_lives = self.extra_lives.min(MAX_LIVES);
    }

    /// The movement tuning shared with the client predictor. `friction_mul` is 1.0 here;
    /// the level event (EVENT_ICE) overrides it, see `Run::tuning`.
    pub fn tuning(&self) -> Tuning {
        Tuning {
            run_speed_mul: self.run_speed_mul,
            jump_mul: self.jump_mul,
            gravity_mul: self.gravity_mul,
            dash_speed_mul: self.dash_speed_mul,
            dash_cooldown_mul: self.dash_cooldown_mul,
            air_jumps: self.air_jumps as u8,
            friction_mul: 1.0,
        }
    }

    /// Enemy parameters after curses. Seconds become whole ticks at TICK_HZ (rounded, at least 1).
    pub fn enemy_params(&self, stats: &EnemyStats) -> EnemyParams {
        let mut params = EnemyParams {
            speed: stats.speed * self.enemy_speed_mul,
            aggro: stats.aggro * self.enemy_aggro_mul,
            size: stats.size * self.enemy_size_mul,
            grace_ticks: secs_to_ticks(stats.grace_s * self.enemy_grace_mul),
            behavior: behavior_from_str(&stats.behavior),
            ..EnemyParams::default()
        };
        if let Some(s) = stats.fire_interval_s.filter(|s| s.is_finite()) {
            params.fire_interval_ticks = secs_to_ticks(s);
        }
        if let Some(v) = stats.projectile_speed.filter(|v| v.is_finite()) {
            params.projectile_speed = v;
        }
        if let Some(v) = stats.reveal_radius.filter(|v| v.is_finite()) {
            params.reveal_radius = v;
        }
        if let Some(s) = stats.corrupt_interval_s.filter(|s| s.is_finite()) {
            params.corrupt_interval_ticks = secs_to_ticks(s);
        }
        params
    }
}

/// Map a behavior string to a BEHAVIOR_* constant; unknown strings degrade to chaser.
fn behavior_from_str(s: &str) -> u8 {
    match s {
        "turret" => BEHAVIOR_TURRET,
        "mimic" => BEHAVIOR_MIMIC,
        "corrupter" => BEHAVIOR_CORRUPTER,
        _ => BEHAVIOR_CHASER,
    }
}

/// Seconds to whole ticks at TICK_HZ, rounded, at least 1 tick (and NaN-safe).
fn secs_to_ticks(s: f32) -> u32 {
    (s * protocol::TICK_HZ as f32).round().max(1.0) as u32
}

    #[test]
    fn mods_stack_and_clamp_to_playable_limits() {
        let mut m = RunMods::default();
        let slow: BTreeMap<String, f32> = [("run_speed_mul".to_string(), 0.5)].into();
        for _ in 0..5 {
            m.apply(&slow);
        }
        assert_eq!(m.run_speed_mul, MOVE_MUL_RANGE.0);
        let jumps: BTreeMap<String, f32> = [("air_jumps_add".to_string(), 2.0)].into();
        m.apply(&jumps);
        m.apply(&jumps);
        assert_eq!(m.air_jumps, MAX_AIR_JUMPS);
        assert_eq!(m.tuning().air_jumps, 3);
        let nan: BTreeMap<String, f32> = [("jump_mul".to_string(), f32::NAN)].into();
        m.apply(&nan);
        assert_eq!(m.jump_mul, 1.0);
    }

//! Run state and rules: lobby drafts, generated levels, shared gift progress, golden-gift currency, lives,
//! the enemies the team drafted, the lobby shop, level events and altars. Pure game logic on top of `sim` —
//! no I/O, fully unit-testable.
//!
//! Flow: Lobby (spawn hub + draft offer and, from level 2 on, the shop) → all players vote → generate a level → Playing (collect every normal
//! gift → golden gifts appear and the beacon opens) → beacon → Lobby for the next level. Death with no lives left
//! ends that player's run; when everyone is down the run is Over and Restart begins a new one.

use crate::content::{Content, RunMods};
use protocol::{
    CARD_CURSE, CARD_ENEMY, CARD_UPGRADE, DraftCard, EVENT_ICE, EnemyView, GIFT_GOLDEN, LevelData,
    PHASE_ENEMY_DRAFT, PHASE_OVER, PHASE_PLAYING, Platform, RunInfo, Tuning, Vec3,
};
use sim::hazards::Hazards;
use sim::levelgen;
use sim::rules::{self, BEACON_RADIUS, BEHAVIOR_MIMIC, Enemy, GIFT_RADIUS, enemy_state};
use sim::world::World;
use std::collections::BTreeMap;

/// Currency granted per golden gift.
pub const GOLDEN_VALUE: u32 = 5;
/// Enemy cards offered at the start of a run, and how many the team keeps.
const ENEMY_OFFER: usize = 4;
const ENEMY_PICKS: u8 = 2;
/// Reward cards offered before each level (one is kept).
const REWARD_PICKS: u8 = 1;
/// Every Nth level offers greater curses only.
const GREATER_EVERY: u32 = 10;
/// Shop items offered per lobby (first, plain lobby gets none).
const SHOP_OFFER: usize = 4;

/// Small deterministic PRNG (SplitMix32) for draft offers, shop offers, altar rolls and level seeds. Server-only.
#[derive(Debug, Clone)]
pub struct Rng(u32);

impl Rng {
    pub fn new(seed: u32) -> Self {
        Rng(seed)
    }
    pub fn next_u32(&mut self) -> u32 {
        self.0 = self.0.wrapping_add(0x9e37_79b9);
        let mut z = self.0;
        z = (z ^ (z >> 16)).wrapping_mul(0x85eb_ca6b);
        z = (z ^ (z >> 13)).wrapping_mul(0xc2b2_ae35);
        z ^ (z >> 16)
    }
    /// Uniform index in 0..n (n > 0).
    pub fn below(&mut self, n: usize) -> usize {
        (self.next_u32() as usize) % n.max(1)
    }
    /// `k` distinct indices from 0..n, in random order.
    pub fn sample(&mut self, n: usize, k: usize) -> Vec<usize> {
        let mut pool: Vec<usize> = (0..n).collect();
        for i in (1..pool.len()).rev() {
            let j = self.below(i + 1);
            pool.swap(i, j);
        }
        pool.truncate(k.min(n));
        pool
    }
}

#[derive(Debug, Clone, PartialEq)]
pub struct Offer {
    pub enemy_picks: u8,
    pub reward_picks: u8,
    pub cards: Vec<DraftCard>,
}

/// One item on offer in the lobby shop, with the stock left for this lobby.
#[derive(Debug, Clone, PartialEq)]
pub struct ShopOffer {
    /// Id of the item in design/content.json.
    pub id: String,
    /// Cost in currency.
    pub cost: u32,
    /// Items left to sell in this lobby.
    pub stock: u8,
}

/// What a successful shop purchase did.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ShopPurchase {
    /// Bought a life: +1 life.
    Life,
    /// Rerolled the reward cards of the current draft offer.
    Reroll,
    /// Applied a mods item like an upgrade (not recorded in RunInfo lists).
    Mods,
}

/// Why a purchase was refused.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ShopError {
    /// The shop is not open (not in a lobby, or this lobby has no shop).
    NotInLobby,
    /// No such item in the current offer.
    UnknownItem,
    /// The item is sold out.
    SoldOut,
    /// Not enough (team-wide) currency.
    NotEnoughCurrency,
}

pub struct Run {
    pub phase: u8,
    /// The level being played (Playing) or about to be played (Lobby).
    pub level_num: u32,
    pub level: LevelData,
    pub world: World,
    pub gifts_taken: Vec<u16>,
    /// Normal gifts collected over the whole run.
    pub gifts_collected: u32,
    pub currency: u32,
    pub lives: u32,
    pub enemies: Vec<Enemy>,
    pub drafted_enemies: Vec<String>,
    pub upgrades: Vec<String>,
    pub curses: Vec<String>,
    pub mods: RunMods,
    pub offer: Option<Offer>,
    /// Projectiles, crumbling tiles and turret/mimic/corrupter behaviour for the current level.
    pub hazards: Hazards,
    /// Levelgen altar ids already used this level.
    pub altars_used: Vec<u16>,
    /// Friction multiplier from the level's event (EVENT_ICE pulls it from content; else 1.0).
    pub event_friction_mul: f32,
    /// The lobby shop offer (from level 2 on); `None` outside the shop lobby and while playing.
    pub shop: Option<Vec<ShopOffer>>,
    rng: Rng,
}

impl Run {
    /// A new run, waiting in the lobby for the start-of-run draft.
    pub fn new(content: &Content, seed: u32, start_level: u32) -> Self {
        let level = levelgen::hub();
        let world = World::from_level(&level);
        let mut run = Run {
            phase: PHASE_ENEMY_DRAFT,
            level_num: start_level.max(1),
            level,
            world,
            gifts_taken: Vec::new(),
            gifts_collected: 0,
            currency: 0,
            lives: 0,
            enemies: Vec::new(),
            drafted_enemies: Vec::new(),
            upgrades: Vec::new(),
            curses: Vec::new(),
            mods: RunMods::default(),
            offer: None,
            hazards: Hazards::new(),
            altars_used: Vec::new(),
            event_friction_mul: 1.0,
            shop: None,
            rng: Rng::new(seed),
        };
        run.offer = Some(run.make_offer(content));
        run.shop = run.make_shop(content);
        run
    }

    /// The reward cards for the current lobby (greater-curse level or mixed upgrade/curse), drawn from the run RNG.
    fn reward_cards(&mut self, content: &Content) -> Vec<DraftCard> {
        if self.level_num.is_multiple_of(GREATER_EVERY) {
            let greater: Vec<_> = content.curses.iter().filter(|c| c.greater).collect();
            let mut cards = Vec::with_capacity(3);
            for i in self.rng.sample(greater.len(), 3) {
                cards.push(DraftCard {
                    kind: CARD_CURSE,
                    id: greater[i].id.clone(),
                });
            }
            return cards;
        }
        // At least one upgrade and one curse; the third card comes from either pool.
        let normal_curses: Vec<_> = content.curses.iter().filter(|c| !c.greater).collect();
        let ups = self.rng.sample(content.upgrades.len(), 2);
        let curses = self.rng.sample(normal_curses.len(), 2);
        let upgrade = |i: usize| DraftCard {
            kind: CARD_UPGRADE,
            id: content.upgrades[i].id.clone(),
        };
        let curse = |i: usize| DraftCard {
            kind: CARD_CURSE,
            id: normal_curses[i].id.clone(),
        };
        let third = if self.rng.below(2) == 0 {
            upgrade(ups[1])
        } else {
            curse(curses[1])
        };
        let mut rewards = vec![upgrade(ups[0]), curse(curses[0]), third];
        let r = self.rng.below(rewards.len());
        rewards.swap(0, r);
        rewards
    }

    fn make_offer(&mut self, content: &Content) -> Offer {
        // The first round of a run is plain: no hunters, no upgrades, no curses. The lobby
        // still opens so everyone can gather and press Continue together.
        if self.level_num == 1 {
            return Offer {
                enemy_picks: 0,
                reward_picks: 0,
                cards: Vec::new(),
            };
        }
        let mut cards = Vec::new();
        let enemy_picks = if self.drafted_enemies.is_empty() {
            for i in self.rng.sample(content.enemies.len(), ENEMY_OFFER) {
                cards.push(DraftCard {
                    kind: CARD_ENEMY,
                    id: content.enemies[i].id.clone(),
                });
            }
            ENEMY_PICKS
        } else {
            0
        };
        cards.extend(self.reward_cards(content));
        Offer {
            enemy_picks,
            reward_picks: REWARD_PICKS,
            cards,
        }
    }

    /// Pick the lobby shop: up to SHOP_OFFER distinct items from content, stock from content, run RNG.
    fn make_shop(&mut self, content: &Content) -> Option<Vec<ShopOffer>> {
        if self.level_num == 1 || content.shop.is_empty() {
            return None;
        }
        let picks = self.rng.sample(content.shop.len(), SHOP_OFFER.min(content.shop.len()));
        Some(
            picks
                .into_iter()
                .map(|i| {
                    let def = &content.shop[i];
                    ShopOffer {
                        id: def.id.clone(),
                        cost: def.cost,
                        stock: def.stock,
                    }
                })
                .collect(),
        )
    }

    /// Replace the reward cards of the current offer, keeping enemy cards (and enemy picks) at the same indices.
    /// Returns false when no draft offer is open.
    pub fn reroll_rewards(&mut self, content: &Content) -> bool {
        if self.offer.is_none() {
            return false;
        }
        let rewards = self.reward_cards(content);
        if let Some(offer) = self.offer.as_mut() {
            offer.cards.retain(|c| c.kind == CARD_ENEMY);
            offer.cards.extend(rewards);
        }
        true
    }

    /// Buy an item from the current lobby shop (team-wide currency). Nothing is charged when the purchase fails.
    pub fn buy_shop_item(&mut self, content: &Content, item: &str) -> Result<ShopPurchase, ShopError> {
        if self.phase != PHASE_ENEMY_DRAFT {
            return Err(ShopError::NotInLobby);
        }
        // Validate against the current offer first (stock, then currency), without holding any borrows.
        let (cost, stock) = match self.shop.as_ref() {
            None => return Err(ShopError::UnknownItem),
            Some(offer) => match offer.iter().find(|entry| entry.id == item) {
                None => return Err(ShopError::UnknownItem),
                Some(entry) => (entry.cost, entry.stock),
            },
        };
        if stock == 0 {
            return Err(ShopError::SoldOut);
        }
        if self.currency < cost {
            return Err(ShopError::NotEnoughCurrency);
        }
        let Some(def) = content.shop_item(item) else {
            return Err(ShopError::UnknownItem);
        };
        // Apply the effect.
        let purchase = match def.kind.as_str() {
            "life" => {
                self.lives = self.lives.saturating_add(1);
                ShopPurchase::Life
            }
            "reroll" => {
                if !self.reroll_rewards(content) {
                    return Err(ShopError::UnknownItem);
                }
                ShopPurchase::Reroll
            }
            "mods" => {
                let before = self.mods.extra_lives;
                self.mods.apply(&def.mods);
                self.lives = self.lives.saturating_add(self.mods.extra_lives.saturating_sub(before));
                ShopPurchase::Mods
            }
            _ => return Err(ShopError::UnknownItem),
        };
        // Charge only after the effect went through.
        if let Some(offer) = self.shop.as_mut() {
            if let Some(entry) = offer.iter_mut().find(|entry| entry.id == item) {
                entry.stock = entry.stock.saturating_sub(1);
            }
        }
        self.currency = self.currency.saturating_sub(cost);
        Ok(purchase)
    }

    /// Check one player's picks against the current offer.
    pub fn validate_picks(&self, picks: &[u8]) -> Result<(), &'static str> {
        let Some(offer) = &self.offer else {
            return Err("no draft is open");
        };
        if self.phase != PHASE_ENEMY_DRAFT {
            return Err("not in the lobby");
        }
        let mut seen = [false; 256];
        let (mut enemies, mut rewards) = (0u8, 0u8);
        for &p in picks {
            let Some(card) = offer.cards.get(p as usize) else {
                return Err("pick out of range");
            };
            if std::mem::replace(&mut seen[p as usize], true) {
                return Err("duplicate pick");
            }
            if card.kind == CARD_ENEMY {
                enemies = enemies.saturating_add(1);
            } else {
                rewards = rewards.saturating_add(1);
            }
        }
        if enemies != offer.enemy_picks || rewards != offer.reward_picks {
            return Err("wrong number of picks");
        }
        Ok(())
    }

    /// Tally everyone's (validated) votes, apply the winning cards and start the generated level.
    /// Ties go to the card offered first.
    pub fn resolve(&mut self, content: &Content, votes: &[Vec<u8>]) {
        let Some(offer) = self.offer.take() else { return };
        let mut tally: BTreeMap<usize, u32> = BTreeMap::new();
        for v in votes {
            for &p in v {
                *tally.entry(p as usize).or_default() += 1;
            }
        }
        let mut ranked: Vec<usize> = (0..offer.cards.len()).collect();
        ranked.sort_by_key(|i| (std::cmp::Reverse(tally.get(i).copied().unwrap_or(0)), *i));
        let winners = |enemy: bool, n: u8| -> Vec<usize> {
            ranked
                .iter()
                .copied()
                .filter(|&i| (offer.cards[i].kind == CARD_ENEMY) == enemy)
                .take(n as usize)
                .collect()
        };
        for i in winners(true, offer.enemy_picks) {
            self.drafted_enemies.push(offer.cards[i].id.clone());
        }
        for i in winners(false, offer.reward_picks) {
            let card = &offer.cards[i];
            let def = if card.kind == CARD_UPGRADE {
                content.upgrade(&card.id)
            } else {
                content.curse(&card.id)
            };
            if let Some(def) = def {
                let lives_before = self.mods.extra_lives;
                self.mods.apply(&def.mods);
                self.lives = self
                    .lives
                    .saturating_add(self.mods.extra_lives.saturating_sub(lives_before));
                if card.kind == CARD_UPGRADE {
                    self.upgrades.push(card.id.clone());
                } else {
                    self.curses.push(card.id.clone());
                }
            }
        }
        self.start_level(content);
    }

    fn start_level(&mut self, content: &Content) {
        let seed = self.rng.next_u32();
        let generated = levelgen::generate(seed, self.level_num);
        self.level = generated.data;
        self.world = World::from_level(&self.level);
        self.gifts_taken.clear();
        self.enemies.clear();
        self.hazards.reset();
        self.altars_used.clear();
        // The shop lives in the lobby only; starting a level closes it.
        self.shop = None;
        self.event_friction_mul = event_friction_mul(content, self.level.event);
        let mut roster: Vec<&str> = self.drafted_enemies.iter().map(String::as_str).collect();
        let drafted = self.drafted_enemies.len();
        if drafted > 0 {
            for k in 0..self.mods.extra_enemies as usize {
                roster.push(self.drafted_enemies[k % drafted].as_str());
            }
        }
        for (slot, (id, pos)) in roster.iter().zip(generated.enemy_spawns.iter()).enumerate() {
            if let Some((index, def)) = content.enemy(id) {
                let params = self.mods.enemy_params(&def.params);
                let mut pos = pos.clone();
                // Mimics pose as gifts: spawn them on the platform below the generated spawn
                // (pos.y = platform top + 0.5). If the spawn is not above any platform, keep the position.
                if params.behavior == BEHAVIOR_MIMIC
                    && let Some(top) = platform_top_below(&self.level.platforms, pos.x, pos.z, pos.y)
                {
                    pos.y = top + 0.5;
                }
                let id = u32::try_from(slot).unwrap_or(u32::MAX - 1) + 1;
                // kind = index into content.json enemies, so the client draws the right model.
                self.enemies.push(rules::spawn_enemy_with(
                    id,
                    u8::try_from(index).unwrap_or(u8::MAX),
                    pos,
                    params,
                ));
            }
        }
        self.phase = PHASE_PLAYING;
    }

    /// The beacon was reached: back to the lobby to draft for the next level.
    pub fn finish_level(&mut self, content: &Content) {
        self.level_num = self.level_num.saturating_add(1);
        self.level = levelgen::hub();
        self.world = World::from_level(&self.level);
        self.gifts_taken.clear();
        self.enemies.clear();
        self.hazards.reset();
        self.altars_used.clear();
        self.event_friction_mul = 1.0;
        self.phase = PHASE_ENEMY_DRAFT;
        self.offer = Some(self.make_offer(content));
        self.shop = self.make_shop(content);
    }

    pub fn end(&mut self) {
        self.phase = PHASE_OVER;
        self.offer = None;
        self.shop = None;
    }

    /// Use the first unused altar within 2.0 units horizontally and 1.5 vertically of `feet`:
    /// apply one random upgrade and one random non-greater curse (run RNG) and record both in the
    /// RunInfo lists. Returns the applied (upgrade id, curse id), or None when nothing was used.
    pub fn try_use_altar(&mut self, content: &Content, feet: &Vec3) -> Option<(String, String)> {
        if self.phase != PHASE_PLAYING {
            return None;
        }
        let mut hit = None;
        for altar in &self.level.altars {
            if self.altars_used.contains(&altar.id) {
                continue;
            }
            let dx = feet.x - altar.pos.x;
            let dz = feet.z - altar.pos.z;
            let dy = (feet.y - altar.pos.y).abs();
            if dx * dx + dz * dz <= 4.0 && dy <= 1.5 {
                hit = Some(altar.id);
                break;
            }
        }
        let altar_id = hit?;
        if content.upgrades.is_empty() {
            return None;
        }
        // One random upgrade...
        let up_i = self.rng.below(content.upgrades.len());
        let upgrade_id = content.upgrades[up_i].id.clone();
        let upgrade_mods = content.upgrades[up_i].mods.clone();
        // ...and one random non-greater curse.
        let normal_curses: Vec<&crate::content::ModCard> =
            content.curses.iter().filter(|c| !c.greater).collect();
        if normal_curses.is_empty() {
            return None;
        }
        let cur_i = self.rng.below(normal_curses.len());
        let curse_id = normal_curses[cur_i].id.clone();
        let curse_mods = normal_curses[cur_i].mods.clone();
        let lives_before = self.mods.extra_lives;
        self.mods.apply(&upgrade_mods);
        self.lives = self
            .lives
            .saturating_add(self.mods.extra_lives.saturating_sub(lives_before));
        let lives_before = self.mods.extra_lives;
        self.mods.apply(&curse_mods);
        self.lives = self
            .lives
            .saturating_add(self.mods.extra_lives.saturating_sub(lives_before));
        self.upgrades.push(upgrade_id.clone());
        self.curses.push(curse_id.clone());
        self.altars_used.push(altar_id);
        Some((upgrade_id, curse_id))
    }

    fn normal_left(&self) -> usize {
        self.level
            .gifts
            .iter()
            .filter(|g| g.kind != GIFT_GOLDEN && !self.gifts_taken.contains(&g.id))
            .count()
    }

    pub fn beacon_open(&self) -> bool {
        self.phase == PHASE_PLAYING && self.normal_left() == 0
    }

    /// Collect gifts the player at `feet` touches. Golden gifts only count once every normal gift is taken.
    pub fn collect_gifts(&mut self, feet: &Vec3) {
        if self.phase != PHASE_PLAYING {
            return;
        }
        let radius = GIFT_RADIUS * self.mods.gift_radius_mul;
        let golden_open = self.normal_left() == 0;
        for gift in &self.level.gifts {
            let golden = gift.kind == GIFT_GOLDEN;
            if self.gifts_taken.contains(&gift.id) || (golden && !golden_open) {
                continue;
            }
            if rules::gift_touched_within(feet, gift, radius) {
                self.gifts_taken.push(gift.id);
                if golden {
                    self.currency = self.currency.saturating_add(GOLDEN_VALUE);
                } else {
                    self.gifts_collected = self.gifts_collected.saturating_add(1);
                }
            }
        }
    }

    /// The shared tuning: movement mods plus the level event's friction override.
    pub fn tuning(&self) -> Tuning {
        let mut tuning = self.mods.tuning();
        tuning.friction_mul = self.event_friction_mul;
        tuning
    }

    pub fn beacon_reached(&self, feet: &Vec3) -> bool {
        self.beacon_open()
            && rules::beacon_reached_within(feet, &self.level.beacon, BEACON_RADIUS * self.mods.beacon_radius_mul)
    }

    pub fn enemy_views(&self) -> Vec<EnemyView> {
        self.enemies
            .iter()
            .map(|e| EnemyView {
                id: e.id,
                kind: e.kind,
                pos: e.pos.clone(),
                radius: e.params.size,
                // DORMANT / HUNTING / DISGUISED (mimic still posing as a gift).
                state: enemy_state(e),
            })
            .collect()
    }

    pub fn info(&self) -> RunInfo {
        RunInfo {
            phase: self.phase,
            level: self.level_num,
            lives: self.lives,
            currency: self.currency,
            tuning: self.tuning(),
            fog_density_mul: self.mods.fog_density_mul,
            enemies: self.drafted_enemies.clone(),
            upgrades: self.upgrades.clone(),
            curses: self.curses.clone(),
        }
    }
}

/// Friction multiplier for `event`: EVENT_ICE pulls it from content, anything else is 1.0.
fn event_friction_mul(content: &Content, event: u8) -> f32 {
    if event == EVENT_ICE {
        content
            .event(EVENT_ICE)
            .map(|e| e.friction_mul)
            .filter(|f| f.is_finite() && *f > 0.0)
            .unwrap_or(1.0)
    } else {
        1.0
    }
}

/// Highest platform top at or below `y` whose xz bounds contain (x, z). Tops may match `y` within a small tolerance
/// (spawns sitting exactly on a platform stay put).
fn platform_top_below(platforms: &[Platform], x: f32, z: f32, y: f32) -> Option<f32> {
    let mut best: Option<f32> = None;
    for platform in platforms {
        let top = platform.bounds.max.y;
        if x >= platform.bounds.min.x
            && x <= platform.bounds.max.x
            && z >= platform.bounds.min.z
            && z <= platform.bounds.max.z
            && top <= y + 0.01
        {
            best = Some(match best {
                Some(b) => b.max(top),
                None => top,
            });
        }
    }
    best
}

#[cfg(test)]
pub(crate) fn valid_picks(run: &Run) -> Vec<u8> {
    let Some(offer) = run.offer.as_ref() else { return vec![] };
    let mut picks = vec![];
    let (mut e, mut r) = (0, 0);
    for (i, c) in offer.cards.iter().enumerate() {
        if c.kind == CARD_ENEMY && e < offer.enemy_picks {
            picks.push(i as u8);
            e += 1;
        } else if c.kind != CARD_ENEMY && r < offer.reward_picks {
            picks.push(i as u8);
            r += 1;
        }
    }
    picks
}

#[cfg(test)]
mod tests {
    use super::*;
    use protocol::{
        EVENT_CRUMBLE, EVENT_HIGH_RISE, EVENT_NONE,
    };
    use sim::rules::ENEMY_KIND_CHASER;

    fn content() -> Content {
        Content::load().unwrap()
    }

    #[test]
    fn first_round_is_plain_then_hunters_are_drafted() {
        let c = content();
        let mut run = Run::new(&c, 7, 1);
        let offer = run.offer.clone().unwrap();
        assert!(offer.cards.is_empty());
        assert_eq!((offer.enemy_picks, offer.reward_picks), (0, 0));
        assert_eq!(run.shop.is_none(), true, "the first, plain lobby has no shop");
        assert_eq!(run.validate_picks(&[]), Ok(()));
        run.resolve(&c, &[vec![]]);
        assert_eq!(run.phase, PHASE_PLAYING);
        assert!(run.enemies.is_empty() && run.upgrades.is_empty() && run.curses.is_empty());
        run.finish_level(&c);
        let next = run.offer.clone().unwrap();
        assert_eq!(next.enemy_picks, 2);
    }

    #[test]
    fn second_level_offer_has_enemies_and_mixed_rewards() {
        let c = content();
        let run = Run::new(&c, 7, 2);
        let offer = run.offer.clone().unwrap();
        assert_eq!(offer.enemy_picks, 2);

//! The game task: owns all simulation state and runs the authoritative tick loop.
//!
//! Input model: clients send one [`InputFrame`] per fixed physics step (1/PHYSICS_HZ s),
//! tagged with an increasing `seq`. Each 20 Hz tick, every player gets
//! `STEPS_PER_TICK` step tokens (capped, so short network jitter can catch up but a
//! client can never run faster than real time), and the server consumes that many
//! queued frames. Snapshots carry the last applied `seq` so the client can reconcile.
//!
//! Run flow (see [`crate::run`]): lobby on the spawn hub with a draft offer (and, from level 2 on, the shop)
//! -> every player votes -> a generated level -> collect all normal gifts -> golden gifts + beacon -> lobby.
//! Tick order: movement (altar uses on rising-edge BTN_INTERACT) -> (playing) gifts -> hazards -> deaths/lives -> beacon -> end check -> snapshots.
//!
//! Multiplayer: a player who enters the open beacon is FINISHED and one who runs out of lives is DEAD; both
//! become invisible spectators. The level ends when nobody is ACTIVE: back to the lobby if anyone finished,
//! otherwise the run is over. Solo is the same rule with one player.

use crate::content::Content;
use crate::run::{Run, ShopError, ShopPurchase};
use protocol::{
    BTN_ABILITY, BTN_INTERACT, BTN_JUMP, CARD_ENEMY, CLASS_GLIDER, CLASS_STRIKER, ClientMsg, ControllerState,
    DEATH_VOID, EVENT_NONE, InputFrame, MAX_DRAFT_PICKS, MAX_INPUTS_PER_MSG, MAX_NAME_LEN, NOTICE_ALTAR,
    NOTICE_BOUGHT, NOTICE_EVENT, PHASE_ENEMY_DRAFT, PHASE_OVER, PHASE_PLAYING, PHYSICS_HZ, PLAYER_ACTIVE,
    PLAYER_DEAD, PLAYER_FINISHED, PLAYER_WAITING, PROTOCOL_VERSION, PlayerView, RunSummary, ServerMsg, ShopItem,
    TICK_HZ, Vec3,
};
use sim::controller;
use std::collections::{BTreeMap, VecDeque};
use std::time::{Duration, Instant};
use tokio::sync::mpsc;

pub type ConnId = u64;

/// Outbound frames buffered per connection before we consider the client too slow.
pub const OUTBOX_CAPACITY: usize = 64;
pub const MAX_PLAYERS: usize = 8;
/// One second of buffered input at most.
const MAX_QUEUED_INPUTS: usize = PHYSICS_HZ as usize;
const STEPS_PER_TICK: u32 = PHYSICS_HZ / TICK_HZ;
/// Token cap: up to 3 ticks' worth of catch-up after jitter.
const MAX_STEP_TOKENS: u32 = STEPS_PER_TICK * 3;
/// After this many ticks with no input, simulate neutral input so gravity keeps acting.
const STARVE_TICKS: u32 = 4;
const BUTTON_MASK: u8 = BTN_JUMP | BTN_ABILITY | BTN_INTERACT;
/// Players spawn spread around the shared spawn point so they never stack.
const SPAWN_RING: [(f32, f32); MAX_PLAYERS] = [
    (0.0, 0.0),
    (1.6, 0.0),
    (-1.6, 0.0),
    (0.0, 1.6),
    (0.0, -1.6),
    (1.6, 1.6),
    (-1.6, -1.6),
    (1.6, -1.6),
];

pub enum GameEvent {
    Connected { id: ConnId, out: mpsc::Sender<Vec<u8>> },
    Message { id: ConnId, msg: ClientMsg },
    Disconnected { id: ConnId },
}

struct Player {
    id: u32,
    name: String,
    state: ControllerState,
    inputs: VecDeque<InputFrame>,
    last_recv_seq: Option<u32>,
    last_applied_seq: u32,
    tokens: u32,
    starved_ticks: u32,
    /// PLAYER_ACTIVE / PLAYER_FINISHED / PLAYER_DEAD / PLAYER_WAITING.
    mode: u8,
    /// How this player last died (for the run summary).
    cause: u8,
    /// The class the player chose (CLASS_*); copied into the controller state at every spawn/respawn.
    class: u8,
}

impl Player {
    fn active(&self) -> bool {
        self.mode == PLAYER_ACTIVE
    }

    fn centre(&self) -> Vec3 {
        Vec3::new(
            self.state.pos.x,
            self.state.pos.y + sim::tuning::PLAYER_HEIGHT / 2.0,
            self.state.pos.z,
        )
    }
}

struct Conn {
    out: mpsc::Sender<Vec<u8>>,
    player: Option<Player>,
}

/// Spawn state at `at` with the player's class baked into the controller state.
fn player_spawn_state(at: Vec3, class: u8) -> ControllerState {
    let mut state = controller::spawn_state(at);
    state.class = class;
    state
}

pub struct Game {
    tick: u32,
    conns: BTreeMap<ConnId, Conn>,
    content: Content,
    run: Run,
    /// Draft votes of the current lobby, by connection.
    votes: BTreeMap<ConnId, Vec<u8>>,
    next_player_id: u32,
    /// Level every new run starts at (1 in production; `VOIDWRAP_START_LEVEL` for visual QA).
    start_level: u32,
    /// Seeds successive runs (draft offers and level layouts).
    seed: u32,
}

impl Game {
    pub fn starting_at(start_level: u32, seed: u32) -> Result<Self, String> {
        let content = Content::load()?;
        let start_level = start_level.max(1);
        let run = Run::new(&content, seed, start_level);
        Ok(Game {
            tick: 0,
            conns: BTreeMap::new(),
            content,
            run,
            votes: BTreeMap::new(),
            next_player_id: 1,
            start_level,
            seed,
        })
    }

    pub fn tick_count(&self) -> u32 {
        self.tick
    }

    pub fn player_count(&self) -> usize {
        self.conns.values().filter(|c| c.player.is_some()).count()
    }

    fn joined_ids(&self) -> Vec<ConnId> {
        self.conns
            .iter()
            .filter(|(_, c)| c.player.is_some())
            .map(|(id, _)| *id)
            .collect()
    }

    /// Spawn position for the n-th joined player.
    fn spawn_point(&self, slot: usize) -> Vec3 {
        let (dx, dz) = SPAWN_RING[slot % MAX_PLAYERS];
        let s = &self.run.level.spawn;
        Vec3::new(s.x + dx, s.y, s.z + dz)
    }

    pub fn handle(&mut self, ev: GameEvent) {
        match ev {
            GameEvent::Connected { id, out } => {
                self.conns.insert(id, Conn { out, player: None });
            }
            GameEvent::Disconnected { id } => {
                if let Some(Conn { player: Some(p), .. }) = self.conns.remove(&id) {
                    tracing::info!(conn = id, player = p.id, name = %p.name, "player left");
                }
                self.votes.remove(&id);
                self.maybe_resolve_draft();
            }
            GameEvent::Message { id, msg } => self.on_message(id, msg),
        }
    }

    fn send(&mut self, id: ConnId, msg: &ServerMsg) {
        let Some(conn) = self.conns.get(&id) else { return };
        let bytes = match protocol::encode(msg) {
            Ok(b) => b,
            Err(e) => {
                tracing::error!("encode failed: {e}");
                return;
            }
        };
        if conn.out.try_send(bytes).is_err() {
            // Outbox full (client not reading) or writer gone: drop the connection.
            tracing::warn!(conn = id, "dropping slow or closed connection");
            self.conns.remove(&id);
            self.votes.remove(&id);
        }
    }

    fn broadcast(&mut self, msg: &ServerMsg) {
        for id in self.joined_ids() {
            self.send(id, msg);
        }
    }

    /// Send an error and drop the connection.
    fn reject(&mut self, id: ConnId, code: u16, message: &str) {
        tracing::warn!(conn = id, code, "rejecting: {message}");
        self.send(
            id,
            &ServerMsg::Error {
                code,
                message: message.to_owned(),
            },
        );
        self.conns.remove(&id);
        self.votes.remove(&id);
    }

    fn draft_msg(&self) -> Option<ServerMsg> {
        let offer = self.run.offer.as_ref()?;
        (self.run.phase == PHASE_ENEMY_DRAFT).then(|| ServerMsg::Draft {
            enemy_picks: offer.enemy_picks,
            reward_picks: offer.reward_picks,
            cards: offer.cards.clone(),
        })
    }

    /// The lobby shop offer as a message, when a shop is open.
    fn shop_msg(&self) -> Option<ServerMsg> {
        let shop = self.run.shop.as_ref()?;
        Some(ServerMsg::Shop {
            items: shop
                .iter()
                .map(|item| ShopItem {
                    id: item.id.clone(),
                    cost: item.cost,
                    stock: item.stock,
                })
                .collect(),
        })
    }

    /// Everyone back on the spawn platform of the current level (hub or generated), alive, then tell them.
    fn respawn_all_and_broadcast(&mut self) {
        let ids = self.joined_ids();
        for (slot, id) in ids.iter().enumerate() {
            let at = self.spawn_point(slot);
            if let Some(p) = self.conns.get_mut(id).and_then(|c| c.player.as_mut()) {
                p.state = player_spawn_state(at, p.class);
                p.mode = PLAYER_ACTIVE;
                p.starved_ticks = 0;
            }
        }
        let level = ServerMsg::Level {
            data: self.run.level.clone(),
        };
        self.broadcast(&level);
        // Level start (generated levels only; the hub has no event): announce it.
        if self.run.level.event != EVENT_NONE
            && let Some(def) = self.content.event(self.run.level.event)
        {
            self.broadcast(&ServerMsg::Notice {
                kind: NOTICE_EVENT,
                ids: vec![def.id.clone()],
            });
        }
        let info = ServerMsg::RunState { info: self.run.info() };
        self.broadcast(&info);
        if let Some(draft) = self.draft_msg() {
            self.broadcast(&draft);
        }
        if let Some(shop) = self.shop_msg() {
            self.broadcast(&shop);
        }
    }

    fn broadcast_run_state(&mut self) {
        let info = ServerMsg::RunState { info: self.run.info() };
        self.broadcast(&info);
    }

    /// Resolve the lobby draft once every joined player has a currently-valid vote.
    /// (After a shop reroll, stored enemy-only votes are invalid until the player re-votes.)
    fn maybe_resolve_draft(&mut self) {
        if self.run.phase != PHASE_ENEMY_DRAFT || self.run.offer.is_none() {
            return;
        }
        let ids = self.joined_ids();
        let all_valid = !ids.is_empty()
            && ids
                .iter()
                .all(|id| self.votes.get(id).is_some_and(|v| self.run.validate_picks(v).is_ok()));
        if !all_valid {
            return;
        }
        let votes: Vec<Vec<u8>> = ids.iter().filter_map(|id| self.votes.get(id).cloned()).collect();
        self.votes.clear();
        self.run.resolve(&self.content, &votes);
        tracing::info!(
            level = self.run.level_num,
            enemies = ?self.run.drafted_enemies,
            upgrades = ?self.run.upgrades,
            curses = ?self.run.curses,
            "draft resolved; level generated"
        );
        self.respawn_all_and_broadcast();
    }

    fn new_run(&mut self) {
        self.seed = self.seed.wrapping_mul(1_664_525).wrapping_add(1_013_904_223);
        self.run = Run::new(&self.content, self.seed, self.start_level);
        self.votes.clear();
        // Players keep their last class; they can change it in the next first lobby.
        self.respawn_all_and_broadcast();
    }

    fn on_message(&mut self, id: ConnId, msg: ClientMsg) {
        let Some(conn) = self.conns.get_mut(&id) else { return };
        let joined = conn.player.is_some();
        match msg {
            ClientMsg::Hello { protocol_version, name } => {
                if joined {
                    return self.reject(id, 400, "duplicate hello");
                }
                if protocol_version != PROTOCOL_VERSION {
                    return self.reject(id, 426, "protocol version mismatch; reload the page");
                }
                if self.player_count() >= MAX_PLAYERS {
                    return self.reject(id, 503, "server full");
                }
                let name = sanitize_name(&name);
                let player_id = self.next_player_id;
                self.next_player_id = self.next_player_id.wrapping_add(1).max(1);
                let spawn = self.spawn_point(self.player_count());
                tracing::info!(conn = id, player = player_id, name = %name, "player joined");
                // Joining mid-level: you spectate until the next lobby; in the lobby you play.
                let mode = if self.run.phase == PHASE_ENEMY_DRAFT {
                    PLAYER_ACTIVE
                } else {
                    PLAYER_WAITING
                };
                let class = CLASS_STRIKER;
                let state = player_spawn_state(spawn, class);
                if let Some(conn) = self.conns.get_mut(&id) {
                    conn.player = Some(Player {
                        id: player_id,
                        name,
                        state,
                        inputs: VecDeque::new(),
                        last_recv_seq: None,
                        last_applied_seq: 0,
                        tokens: 0,
                        starved_ticks: 0,
                        mode,
                        cause: DEATH_VOID,
                        class,
                    });
                }
                self.send(
                    id,
                    &ServerMsg::Welcome {
                        protocol_version: PROTOCOL_VERSION,
                        player_id,
                        tick: self.tick,
                    },
                );
                let level = ServerMsg::Level {
                    data: self.run.level.clone(),
                };
                self.send(id, &level);
                let info = ServerMsg::RunState { info: self.run.info() };
                self.send(id, &info);
                if let Some(draft) = self.draft_msg() {
                    self.send(id, &draft);
                }
                if let Some(shop) = self.shop_msg() {
                    self.send(id, &shop);
                }
            }
            _ if !joined => self.reject(id, 400, "hello required first"),
            ClientMsg::Inputs { frames } => {
                if frames.len() > MAX_INPUTS_PER_MSG as usize {
                    return self.reject(id, 400, "too many inputs in one message");
                }
                if let Some(p) = conn.player.as_mut() {
                    for mut f in frames {
                        if p.last_recv_seq.is_some_and(|last| f.seq <= last) {
                            continue; // duplicate or reordered
                        }
                        if p.inputs.len() >= MAX_QUEUED_INPUTS {
                            break; // flooding: drop the excess
                        }
                        f.buttons &= BUTTON_MASK;
                        p.last_recv_seq = Some(f.seq);
                        p.inputs.push_back(f);
                    }
                }
            }
            ClientMsg::Ping { t } => self.send(id, &ServerMsg::Pong { t }),
            ClientMsg::Buy { item } => {
                match self.run.buy_shop_item(&self.content, &item) {
                    Ok(purchase) => {
                        // A reroll replaces the reward cards: enemy cards stay, reward votes are cleared,
                        // everyone gets the fresh Draft again.
                        if purchase == ShopPurchase::Reroll {
                            let cards = self.run.offer.as_ref().map(|o| o.cards.clone()).unwrap_or_default();
                            for vote in self.votes.values_mut() {
                                vote.retain(|&p| cards.get(p as usize).is_some_and(|c| c.kind == CARD_ENEMY));
                            }
                            if let Some(draft) = self.draft_msg() {
                                self.broadcast(&draft);
                            }
                        }
                        self.broadcast(&ServerMsg::Notice {
                            kind: NOTICE_BOUGHT,
                            ids: vec![item],
                        });
                        self.broadcast_run_state();
                        // Re-send the shop so everyone sees the new stock and currency.
                        if let Some(shop) = self.shop_msg() {
                            self.broadcast(&shop);
                        }
                    }
                    Err(why) => {
                        let (code, message) = match why {
                            ShopError::NotInLobby => (409, "the shop is not open"),
                            ShopError::UnknownItem => (404, "unknown shop item"),
                            ShopError::SoldOut => (409, "sold out"),
                            ShopError::NotEnoughCurrency => (402, "not enough currency"),
                        };
                        self.send(
                            id,
                            &ServerMsg::Error {
                                code,
                                message: message.to_owned(),
                            },
                        );
                    }
                }
            }
            ClientMsg::ChooseClass { class } => {
                // Only in the lobby, and only for the first lobby of a run (level 1).
                if self.run.phase != PHASE_ENEMY_DRAFT || self.run.level_num != 1 {
                    return self.send(
                        id,
                        &ServerMsg::Error {
                            code: 409,
                            message: "class can only be changed in the first lobby".to_owned(),
                        },
                    );
                }
                if class > CLASS_GLIDER {
                    return self.send(
                        id,
                        &ServerMsg::Error {
                            code: 400,
                            message: "unknown class".to_owned(),
                        },
                    );
                }
                if let Some(p) = self.conns.get_mut(&id).and_then(|c| c.player.as_mut()) {
                    p.class = class;
                    p.state.class = class;
                }
            }
            ClientMsg::DraftPick { picks } => {
                if picks.len() > MAX_DRAFT_PICKS as usize {
                    return self.reject(id, 400, "too many draft picks");
                }
                match self.run.validate_picks(&picks) {
                    Ok(()) => {
                        self.votes.insert(id, picks);
                        self.maybe_resolve_draft();
                    }
                    Err(why) => self.send(
                        id,
                        &ServerMsg::Error {
                            code: 422,
                            message: why.to_owned(),
                        },
                    ),
                }
            }
            ClientMsg::Restart => {
                // A new run starts once the run is over (the whole team is down); otherwise ignore.
                if self.run.phase == PHASE_OVER {
                    tracing::info!(conn = id, "new run");
                    self.new_run();
                }
            }
        }
    }

    /// Advance the simulation one 20 Hz tick and broadcast snapshots.
    pub fn tick(&mut self) {
        self.tick = self.tick.wrapping_add(1);
        let tuning = self.run.tuning();
        // Altar uses are discovered while stepping input frames; notices go out after the movement loop.
        let mut altar_notices: Vec<(String, String)> = Vec::new();

        // 1. Movement (lobby and playing).
        for p in self.conns.values_mut().filter_map(|c| c.player.as_mut()) {
            p.tokens = (p.tokens + STEPS_PER_TICK).min(MAX_STEP_TOKENS);
            if !p.active() {
                // Spectators' inputs are acknowledged and dropped (their camera is client-side).
                if let Some(last) = p.inputs.back() {
                    p.last_applied_seq = last.seq;
                }
                p.inputs.clear();
                continue;
            }
            let mut stepped = 0;
            while p.tokens > 0 {
                let Some(f) = p.inputs.pop_front() else { break };
                // Altars: rising-edge BTN_INTERACT from the input frames, ACTIVE players, playing levels only.
                if self.run.phase == PHASE_PLAYING
                    && f.buttons & BTN_INTERACT != 0
                    && p.state.prev_buttons & BTN_INTERACT == 0
                    && let Some(pair) = self.run.try_use_altar(&self.content, &p.state.pos)
                {
                    altar_notices.push(pair);
                }
                controller::step_with(&mut p.state, &f, &self.run.world, &tuning);
                p.last_applied_seq = f.seq;
                p.tokens -= 1;
                stepped += 1;
            }
            if stepped == 0 {
                p.starved_ticks += 1;
                if p.starved_ticks >= STARVE_TICKS {
                    let neutral = InputFrame {
                        seq: p.last_applied_seq,
                        move_x: 0,
                        move_z: 0,
                        buttons: 0,
                    };
                    for _ in 0..STEPS_PER_TICK.min(p.tokens) {
                        controller::step_with(&mut p.state, &neutral, &self.run.world, &tuning);
                        p.tokens -= 1;
                    }
                }
            } else {
                p.starved_ticks = 0;
            }
        }
        for (upgrade, curse) in altar_notices {
            self.broadcast(&ServerMsg::Notice {
                kind: NOTICE_ALTAR,
                ids: vec![upgrade, curse],
            });
            self.broadcast_run_state();
        }

        if self.run.phase == PHASE_PLAYING {
            self.tick_playing();
        } else {
            // Lobby / over: the hub is a safe space - falling off just puts you back on the spawn.
            let kill_y = self.run.world.kill_y;
            let ids = self.joined_ids();
            for (slot, id) in ids.iter().enumerate() {
                let at = self.spawn_point(slot);
                if let Some(p) = self.conns.get_mut(id).and_then(|c| c.player.as_mut())
                    && p.state.pos.y < kill_y
                {
                    p.state = player_spawn_state(at, p.class);
                }
            }
        }

        // Snapshots.
        let players: Vec<PlayerView> = self
            .conns
            .values()
            .filter_map(|c| c.player.as_ref())
            .map(|p| PlayerView {
                id: p.id,
                pos: p.state.pos.clone(),
                vel: p.state.vel.clone(),
                mode: p.mode,
                class: p.state.class,
            })
            .collect();
        let enemies = self.run.enemy_views();
        let projectiles = self.run.hazards.projectiles();
        let tiles = self.run.hazards.tiles();
        let altars_used = self.run.altars_used.clone();
        for id in self.joined_ids() {
            let Some(p) = self.conns.get(&id).and_then(|c| c.player.as_ref()) else {
                continue;
            };
            let msg = ServerMsg::Snapshot {
                tick: self.tick,
                level: self.run.level.level,
                last_input_seq: p.last_applied_seq,
                you: p.state.clone(),
                players: players.clone(),
                enemies: enemies.clone(),
                gifts_taken: self.run.gifts_taken.clone(),
                projectiles: projectiles.clone(),
                tiles: tiles.clone(),
                altars_used: altars_used.clone(),
            };
            self.send(id, &msg);
        }
    }

    fn tick_playing(&mut self) {
        // Gifts (shared by the team; golden ones unlock after the normal ones).
        let currency_before = self.run.currency;
        for conn in self.conns.values() {
            if let Some(p) = conn.player.as_ref().filter(|p| p.active()) {
                self.run.collect_gifts(&p.state.pos);
            }
        }
        let mut info_changed = self.run.currency != currency_before;

        // Hazards: enemy behaviour (chaser/turret/mimic/corrupter), projectiles, crumbling tiles.
        // `players` are the feet positions of the living; Hazards derives centres from them.
        let feet: Vec<Vec3> = self
            .conns
            .values()
            .filter_map(|c| c.player.as_ref())
            .filter(|p| p.active())
            .map(|p| p.state.pos.clone())
            .collect();
        self.run
            .hazards
            .step(&mut self.run.enemies, &self.run.world, &self.run.level, &feet);
        // Crumbled platforms stop being solid for every player's controller from the next step.
        self.run.world.set_tiles(&self.run.hazards.tiles());

        // Deaths: a spare life (shared by the team) respawns you at the level start; otherwise you spectate.
        let ids = self.joined_ids();
        for (slot, id) in ids.iter().enumerate() {
            let at = self.spawn_point(slot);
            let Some(p) = self
                .conns
                .get_mut(id)
                .and_then(|c| c.player.as_mut())
                .filter(|p| p.active())
            else {
                continue;
            };
            let cause = if p.state.pos.y < self.run.world.kill_y {
                Some(DEATH_VOID)
            } else {
                // Dormant and disguised enemies are harmless; projectiles kill.
                self.run.hazards.hit(&self.run.enemies, &p.state.pos)
            };
            let Some(cause) = cause else {
                continue;
            };
            if self.run.lives > 0 {
                self.run.lives -= 1;
                p.state = player_spawn_state(at, p.class);
                tracing::info!(conn = id, cause, lives = self.run.lives, "life lost");
            } else {
                p.mode = PLAYER_DEAD;
                p.cause = cause;
                p.state.vel = Vec3::ZERO;
                tracing::info!(conn = id, cause, level = self.run.level_num, "player out; spectating");
            }
            info_changed = true;
        }

        // Beacon: each active player who walks into the open beacon finishes the level and spectates.
        for p in self
            .conns
            .values_mut()
            .filter_map(|c| c.player.as_mut())
            .filter(|p| p.active())
        {
            if self.run.beacon_reached(&p.state.pos) {
                p.mode = PLAYER_FINISHED;
                p.state.vel = Vec3::ZERO;
                info_changed = true;
            }
        }

        // The level ends when nobody is still playing it.
        let players: Vec<&Player> = self.conns.values().filter_map(|c| c.player.as_ref()).collect();
        if !players.is_empty() && !players.iter().any(|p| p.active()) {
            if players.iter().any(|p| p.mode == PLAYER_FINISHED) {
                self.run.finish_level(&self.content);
                tracing::info!(
                    next_level = self.run.level_num,
                    currency = self.run.currency,
                    "level cleared"
                );
                self.votes.clear();
                self.respawn_all_and_broadcast();
            } else {
                self.end_run();
            }
            return;
        }
        if info_changed {
            self.broadcast_run_state();
        }
    }

    /// Everyone is out: the run is over. Each player gets a summary with their own cause of death
    /// (DEATH_VOID, DEATH_ENEMY or DEATH_PROJECTILE).
    fn end_run(&mut self) {
        self.run.end();
        tracing::info!(level = self.run.level_num, "run over");
        let summaries: Vec<(ConnId, u8)> = self
            .conns
            .iter()
            .filter_map(|(id, c)| c.player.as_ref().map(|p| (*id, p.cause)))
            .collect();
        for (id, cause) in summaries {
            let summary = RunSummary {
                level_reached: self.run.level_num,
                gifts_collected: self.run.gifts_collected,
                cause,
            };
            self.send(id, &ServerMsg::RunOver { summary });
        }
        self.broadcast_run_state();
    }
}

/// Trim, drop control characters, cap length; never empty.
fn sanitize_name(raw: &str) -> String {
    let name: String = raw
        .chars()
        .filter(|c| !c.is_control())
        .take(MAX_NAME_LEN as usize)
        .collect();
    let name = name.trim();
    if name.is_empty() {
        "blocky".to_owned()
    } else {
        name.to_owned()
    }
}

/// Runs the game until every sender is gone.
pub async fn run(mut rx: mpsc::Receiver<GameEvent>, mut game: Game) {
    let period = Duration::from_millis(1000 / u64::from(TICK_HZ));
    let mut interval = tokio::time::interval(period);
    interval.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Burst);
    let mut window_start = Instant::now();
    let mut window_ticks = 0u32;
    loop {
        tokio::select! {
            biased;
            _ = interval.tick() => {
                game.tick();
                window_ticks += 1;
                tracing::trace!(tick = game.tick_count(), "tick");
                let elapsed = window_start.elapsed();
                if elapsed >= Duration::from_secs(1) {
                    let hz = f64::from(window_ticks) / elapsed.as_secs_f64();
                    tracing::info!("tick {} | {hz:.1} Hz | players {}", game.tick_count(), game.player_count());
                    window_start = Instant::now();
                    window_ticks = 0;
                }
            }
            ev = rx.recv() => match ev {
                Some(ev) => game.handle(ev),
                None => break,
            },
        }
    }
}

        let mut out = vec![];
        while let Ok(b) = rx.try_recv() {
            out.push(protocol::decode::<ServerMsg>(&b).unwrap());
        }
        out
    }

        }
    }
