import { describe, expect, it } from 'vitest';

import { selectionComplete, toggleSelection } from './draftSelection';

// 4 enemies (indices 0-3) + 2 upgrades (4-5) + 1 curse (6).
const CARDS = [
  { kind: 0 },
  { kind: 0 },
  { kind: 0 },
  { kind: 0 },
  { kind: 1 },
  { kind: 1 },
  { kind: 2 },
];

describe('toggleSelection', () => {
  it('selects cards up to the per-group limit', () => {
    expect(toggleSelection([], 0, CARDS, 2, 1)).toEqual([0]);
    expect(toggleSelection([0], 1, CARDS, 2, 1)).toEqual([0, 1]);
    expect(toggleSelection([0, 1], 4, CARDS, 2, 1)).toEqual([0, 1, 4]);
  });

  it('replaces the oldest selection within the same group', () => {
    // Enemies: limit 2, selecting a third enemy drops the oldest (index 0).
    expect(toggleSelection([0, 1], 2, CARDS, 2, 1)).toEqual([1, 2]);
    // Mixed selection: oldest ENEMY is replaced, reward stays.
    expect(toggleSelection([1, 2, 4], 0, CARDS, 2, 1)).toEqual([2, 4, 0]);
    // Rewards: limit 1, selecting another reward drops the previous one.
    expect(toggleSelection([2, 4], 5, CARDS, 2, 1)).toEqual([2, 5]);
  });

  it('keeps enemy and reward limits independent', () => {
    // Enemies full (limit 2) but a reward pick still lands.
    expect(toggleSelection([0, 1], 6, CARDS, 2, 1)).toEqual([0, 1, 6]);
    // Rewards full (limit 1) but an enemy pick still lands.
    expect(toggleSelection([2, 6], 0, CARDS, 2, 1)).toEqual([2, 6, 0]);
    // A full reward group replaces its own pick, leaving enemies untouched.
    expect(toggleSelection([1, 2, 4], 6, CARDS, 2, 1)).toEqual([1, 2, 6]);
  });

  it('deselects a selected card', () => {
    expect(toggleSelection([1, 2, 4], 1, CARDS, 2, 1)).toEqual([2, 4]);
    expect(toggleSelection([1, 2, 4], 4, CARDS, 2, 1)).toEqual([1, 2]);
  });

  it('ignores out-of-range indices', () => {
    expect(toggleSelection([], 9, CARDS, 2, 1)).toEqual([]);
    expect(toggleSelection([], CARDS.length, CARDS, 2, 1)).toEqual([]);
    expect(toggleSelection([0], -1, CARDS, 2, 1)).toEqual([0]);
  });

  it('never selects into a group with a zero limit', () => {
    expect(toggleSelection([], 0, CARDS, 0, 1)).toEqual([]);
    expect(toggleSelection([], 4, CARDS, 2, 0)).toEqual([]);
  });
});

describe('selectionComplete', () => {
  it('is complete only with exact counts in both groups', () => {
    expect(selectionComplete([0, 1, 4], CARDS, 2, 1)).toBe(true);
    expect(selectionComplete([2, 3, 6], CARDS, 2, 1)).toBe(true);
  });

  it('is incomplete when either group is under the limit', () => {
    expect(selectionComplete([0, 4], CARDS, 2, 1)).toBe(false);
    expect(selectionComplete([0, 1], CARDS, 2, 1)).toBe(false);
    expect(selectionComplete([], CARDS, 2, 1)).toBe(false);
  });

  it('is incomplete with too many picks in a group', () => {
    expect(selectionComplete([0, 1, 2, 4], CARDS, 2, 1)).toBe(false);
    expect(selectionComplete([0, 1, 4, 5], CARDS, 2, 1)).toBe(false);
  });

  it('handles groups with zero picks required', () => {
    expect(selectionComplete([4], CARDS, 0, 1)).toBe(true);
    expect(selectionComplete([0, 4], CARDS, 0, 1)).toBe(false);
  });
});
