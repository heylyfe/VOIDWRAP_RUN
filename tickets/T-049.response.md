#!/usr/bin/env node
// Renders the icon SVGs into transparent PNGs and writes design/icon/icon.ico (all seven sizes,
// PNG-compressed entries, ICONDIR/ICONDIRENTRY written by hand), design/icon/icon-256.png,
// web/public/icon.png and design/icon/icon-64.rgba (raw RGBA, 64*64*4 = 16384 bytes).
// Binary outputs are committed; re-run this after changing the SVGs.
//   node tools/make-icon.mjs
import { createRequire } from "node:module";
import { mkdirSync, readFileSync, writeFileSync } from "node:fs";
import { dirname, join } from "node:path";
import { fileURLToPath } from "node:url";

const root = join(dirname(fileURLToPath(import.meta.url)), "..");
const require = createRequire(join(root, "web/package.json"));
const { chromium } = require("@playwright/test");

const iconDir = join(root, "design/icon");
const bigUrl = `data:image/svg+xml;base64,${readFileSync(join(iconDir, "icon.svg")).toString("base64")}`;
const smallUrl = `data:image/svg+xml;base64,${readFileSync(join(iconDir, "icon-small.svg")).toString("base64")}`;

// 16/24/32 come from the small SVG (it reads better at tiny sizes), the rest from the big one.
const jobs = [
  [16, smallUrl],
  [24, smallUrl],
  [32, smallUrl],
  [48, bigUrl],
  [64, bigUrl],
  [128, bigUrl],
  [256, bigUrl],
];

const browser = await chromium.launch();

/** Render one SVG data URL at a pixel size; returns the PNG buffer (transparent background). */
async function renderPng(size, url) {
  const page = await browser.newPage({ viewport: { width: size, height: size } });
  await page.setContent(`<!doctype html><html><body style="margin:0;background:transparent"><img src="${url}" width="${size}" height="${size}" style="display:block"></body></html>`);
  await page.evaluate(
    () =>
      new Promise((resolve, reject) => {
        const img = document.querySelector("img");
        if (img.complete) resolve(undefined);
        else {
          img.onload = () => resolve(undefined);
          img.onerror = () => reject(new Error("image failed to decode"));
        }
      }),
  );
  const png = await page.screenshot({ omitBackground: true, clip: { x: 0, y: 0, width: size, height: size } });
  await page.close();
  return png;
}

const pngs = new Map();
for (const [size, url] of jobs) {
  pngs.set(size, await renderPng(size, url));
}

// Raw 64x64 RGBA bytes via a canvas inside the page (premultiplied roundtrip; see NOTES).
const rgba64 = await (async () => {
  const page = await browser.newPage({ viewport: { width: 64, height: 64 } });
  await page.setContent(`<!doctype html><canvas id="c" width="64" height="64"></canvas>`);
  const base64 = await page.evaluate(async (pngDataUrl) => {
    const img = new Image();
    img.src = pngDataUrl;
    await img.decode();
    const ctx = document.getElementById("c").getContext("2d");
    ctx.clearRect(0, 0, 64, 64);
    ctx.drawImage(img, 0, 0);
    const d = ctx.getImageData(0, 0, 64, 64).data;
    let s = "";
    const CHUNK = 8192;
    for (let i = 0; i < d.length; i += CHUNK) s += String.fromCharCode(...d.subarray(i, i + CHUNK));
    return btoa(s);
  }, `data:image/png;base64,${pngs.get(64).toString("base64")}`);
  const buf = Buffer.from(base64, "base64");
  await page.close();
  return buf;
})();
if (rgba64.length !== 64 * 64 * 4) {
  throw new Error(`expected 16384 raw RGBA bytes, got ${rgba64.length}`);
}
await browser.close();

// --- ICOn (ICONDIR + ICONDIRENTRY + PNG blobs), written by hand. -------------------------
const SIZES = jobs.map(([size]) => size);
const offsets = [];
let offset = 6 + 16 * SIZES.length;
for (const size of SIZES) {
  offsets.push(offset);
  offset += pngs.get(size).length;
}
const icoHeader = Buffer.alloc(6 + 16 * SIZES.length);
icoHeader.writeUInt16LE(0, 0); // reserved
icoHeader.writeUInt16LE(1, 2); // type 1 = icon
icoHeader.writeUInt16LE(SIZES.length, 4);
let entryPos = 6;
for (let i = 0; i < SIZES.length; i += 1) {
  const size = SIZES[i];
  const png = pngs.get(size);
  const dim = size >= 256 ? 0 : size; // 0 means 256 in ICONDIRENTRY
  icoHeader.writeUInt8(dim, entryPos); // width
  icoHeader.writeUInt8(dim, entryPos + 1); // height
  icoHeader.writeUInt8(0, entryPos + 2); // palette colors (0 = true color)
  icoHeader.writeUInt8(0, entryPos + 3); // reserved
  icoHeader.writeUInt16LE(1, entryPos + 4); // planes
  icoHeader.writeUInt16LE(32, entryPos + 6); // bits per pixel
  icoHeader.writeUInt32LE(png.length, entryPos + 8); // bytes in resource
  icoHeader.writeUInt32LE(offsets[i], entryPos + 12); // image data offset
  entryPos += 16;
}
const ico = Buffer.concat([icoHeader, ...SIZES.map((size) => pngs.get(size))]);

// --- Write outputs. ------------------------------------------------------------------------
mkdirSync(iconDir, { recursive: true });
mkdirSync(join(root, "web/public"), { recursive: true });
writeFileSync(join(iconDir, "icon.ico"), ico);
writeFileSync(join(iconDir, "icon-256.png"), pngs.get(256));
writeFileSync(join(root, "web/public/icon.png"), pngs.get(256));
writeFileSync(join(iconDir, "icon-64.rgba"), rgba64);

for (const [size, png] of pngs) console.log(`make-icon: rendered ${size}x${size} PNG (${png.length} bytes)`);
console.log(`make-icon: wrote design/icon/icon.ico (${ico.length} bytes, ${SIZES.length} sizes: ${SIZES.join(", ")})`);
console.log(`make-icon: wrote design/icon/icon-256.png (${pngs.get(256).length} bytes)`);
console.log(`make-icon: wrote web/public/icon.png (${pngs.get(256).length} bytes)`);
console.log(`make-icon: wrote design/icon/icon-64.rgba (${rgba64.length} bytes)`);
console.log("make-icon: done");

//! Build script for the `server` crate.
//!
//! On Windows this embeds the application icon and version metadata into the finished
//! executable. Everywhere else (and on a non-Windows host cross-compiling for Windows)
//! it only watches the icon file for changes, so Linux builds keep working.

/// Entry point for every build.
fn main() {
    println!("cargo:rerun-if-changed=../../../../design/icon/icon.ico");
    embed_windows_resources();
}

/// Compile and link the Windows resources (icon + version strings).
///
/// Only meaningful when both the host and the target are Windows; on other hosts this
/// function does not exist, so `winresource` is never referenced where it is unavailable.
#[cfg(windows)]
fn embed_windows_resources() {
    if std::env::var("CARGO_CFG_TARGET_OS").as_deref() != Ok("windows") {
        return;
    }
    let version = std::env::var("CARGO_PKG_VERSION").unwrap_or_else(|_| String::from("0.0.0"));
    let mut resources = winresource::WindowsResource::new();
    resources.set_icon("../../../../design/icon/icon.ico");
    resources.set("ProductName", "VOIDWRAP RUN");
    resources.set("FileDescription", "VOIDWRAP RUN");
    resources.set("CompanyName", "VOIDWRAP RUN");
    resources.set("LegalCopyright", "VOIDWRAP RUN contributors");
    resources.set("OriginalFilename", "VOIDWRAP RUN.exe");
    resources.set("FileVersion", &version);
    resources.set("ProductVersion", &version);
    if let Err(error) = resources.compile() {
        // Never fail the build: version metadata is a nicety, and the lead merges what
        // compiles cleanly everywhere, not only on a perfectly set-up Windows box.
        println!("cargo:warning=could not embed the Windows resources: {error}");
    }
}

/// No-op on non-Windows hosts: the winresource crate only exists for Windows.
#[cfg(not(windows))]
fn embed_windows_resources() {}

//! Native desktop window for the game (enabled by the `desktop` cargo feature).
//!
//! The server runs on a tokio runtime in a background thread; the main thread owns the
//! tao event loop and a wry WebView loading `http://localhost:<port>/`. Closing the
//! window asks the server to shut down (best effort) and then exits.
//!
//! The window icon is embedded from `design/icon/icon-64.rgba` (generated by
//! `tools/make-icon.mjs`, 64x64 raw RGBA); if it cannot be decoded the window still
//! opens, without an icon, after logging a warning.

#![cfg(feature = "desktop")]

use std::time::{Duration, Instant};

/// Title of the native window.
pub const WINDOW_TITLE: &str = "VOIDWRAP RUN";
/// Window width in logical pixels.
const WINDOW_WIDTH: f64 = 1280.0;
/// Window height in logical pixels.
const WINDOW_HEIGHT: f64 = 800.0;
/// Raw 64x64 RGBA pixels for the window icon (path is relative to this source file).
const WINDOW_ICON_RGBA: &[u8] = include_bytes!("../../../../design/icon/icon-64.rgba");

/// Decide whether to run in a native window. The window is disabled by `--browser` and
/// `--headless`, and when `PORT` is set (hosting platforms, background agents).
pub fn should_open_window(args: &[String], port_env_set: bool) -> bool {
    if port_env_set {
        return false;
    }
    !args.iter().any(|a| a == "--browser" || a == "--headless")
}

/// Wait until the server accepts TCP connections on `port`, or `timeout` passes
/// (in which case the window is opened anyway).
pub fn wait_for_server(port: u16, timeout: Duration) {
    let deadline = Instant::now() + timeout;
    loop {
        if std::net::TcpStream::connect((std::net::Ipv4Addr::LOCALHOST, port)).is_ok() {
            return;
        }
        if Instant::now() >= deadline {
            tracing::warn!(
                "server did not accept connections within {} s; opening the window anyway",
                timeout.as_secs()
            );
            return;
        }
        std::thread::sleep(Duration::from_millis(50));
    }
}

/// Build the tao window icon from the raw RGBA bytes.
///
/// Returns `None` (after logging a warning) when the bytes cannot be decoded, so the
/// window can still open without an icon.
fn window_icon() -> Option<tao::window::Icon> {
    match tao::window::Icon::from_rgba(WINDOW_ICON_RGBA.to_vec(), 64, 64) {
        Ok(icon) => Some(icon),
        Err(error) => {
            tracing::warn!("could not build the window icon: {error}");
            None
        }
    }
}

/// Open the native window and block until the user closes it.
/// Sends `shutdown` to the server when the window closes (best effort).
pub fn open_window(port: u16, shutdown: tokio::sync::oneshot::Sender<()>, server_done: std::sync::mpsc::Receiver<()>) {
    use tao::dpi::LogicalSize;
    use tao::event::{Event, WindowEvent};
    use tao::event_loop::{ControlFlow, EventLoop};
    use tao::window::WindowBuilder;

    let event_loop = EventLoop::<()>::new();
    let mut window_builder = WindowBuilder::new()
        .with_title(WINDOW_TITLE)
        .with_inner_size(LogicalSize::new(WINDOW_WIDTH, WINDOW_HEIGHT))
        .with_resizable(true);
    if let Some(icon) = window_icon() {
        window_builder = window_builder.with_window_icon(Some(icon));
    }
    let window = match window_builder.build(&event_loop) {
        Ok(window) => window,
        Err(e) => {
            tracing::error!("could not create the window: {e}");
            let _ = shutdown.send(());
            return;
        }
    };
    let url = format!("http://localhost:{port}/");
    // `window` and `_webview` must stay alive for the whole event loop.
    let _webview = match wry::WebViewBuilder::new().with_url(&url).build(&window) {
        Ok(webview) => webview,
        Err(e) => {
            tracing::error!("could not create the web view ({e}); open {url} in a browser instead");
            let _ = shutdown.send(());
            return;
        }
    };
    let mut shutdown = Some(shutdown);
    event_loop.run(move |event, _, control_flow| {
        *control_flow = ControlFlow::Wait;
        if let Event::WindowEvent {
            event: WindowEvent::CloseRequested,
            ..
        } = event
        {
            if let Some(tx) = shutdown.take() {
                let _ = tx.send(());
                // Give the server time to close the hosting listener and remove the UPnP mapping.
                let _ = server_done.recv_timeout(Duration::from_secs(8));
            }
            *control_flow = ControlFlow::Exit;
        }
    });
}

#!/usr/bin/env node
// Builds a ready-to-play folder: dist/VOIDWRAP_RUN/{voidwrap-server(.exe), web/, PLAY launchers, HOW-TO-PLAY}.
// On Windows (desktop build) it also builds dist/VOIDWRAP_RUN-<version>-windows-setup.exe when NSIS is available.
// Used by the release workflow and by `npm start`.
//   node tools/package.mjs [--skip-build]
import { spawnSync } from "node:child_process";
import { chmodSync, cpSync, existsSync, mkdirSync, rmSync } from "node:fs";
import { dirname, join } from "node:path";
import { fileURLToPath } from "node:url";

const root = join(dirname(fileURLToPath(import.meta.url)), "..");
const win = process.platform === "win32";
const run = (cmd, args, cwd = root, env = {}) => {
  console.log(`package: ${cmd} ${args.join(" ")}`);
  const r = spawnSync(cmd, args, { cwd, stdio: "inherit", shell: win, env: { ...process.env, ...env } });
  if (r.status !== 0) process.exit(r.status ?? 1);
};

// Windows gets the desktop window (feature "desktop"); Linux packages stay browser-based for now
// (the webview there needs extra system libraries). VOIDWRAP_DESKTOP=0/1 overrides.
const desktop = (process.env.VOIDWRAP_DESKTOP ?? (win ? "1" : "0")) === "1";

if (!process.argv.includes("--skip-build")) {
  run("node", ["tools/build-wasm.mjs"]);
  run("pnpm", ["--dir", "web", "build"], root, { NODE_ENV: "production" });
  run("cargo", ["build", "--release", "-p", "server", ...(desktop ? ["--features", "desktop"] : [])], join(root, "server"));
}

const out = join(root, "dist/VOIDWRAP_RUN");
rmSync(out, { recursive: true, force: true });
mkdirSync(out, { recursive: true });
const targetDir = process.env.CARGO_TARGET_DIR ?? join(root, "server/target");
const exe = win ? "voidwrap-server.exe" : "voidwrap-server";
// The desktop build opens its own game window, so on Windows the program itself is what players double-click.
const shipped = win && desktop ? "VOIDWRAP RUN.exe" : exe;
cpSync(join(targetDir, "release", exe), join(out, shipped));
cpSync(join(root, "web/out"), join(out, "web"), { recursive: true });
if (win && desktop) {
  cpSync(join(root, "release/PLAY-IN-BROWSER.cmd"), join(out, "PLAY-IN-BROWSER.cmd"));
} else if (win) {
  cpSync(join(root, "release/PLAY.cmd"), join(out, "PLAY.cmd"));
  cpSync(join(root, "release/PLAY-WITH-FRIENDS.cmd"), join(out, "PLAY-WITH-FRIENDS.cmd"));
} else {
  cpSync(join(root, "release/play.sh"), join(out, "play.sh"));
  chmodSync(join(out, "play.sh"), 0o755);
  chmodSync(join(out, exe), 0o755);
}
for (const doc of ["HOW-TO-PLAY.md", "LICENSE", "ENEMIES.md"]) {
  const src = join(root, doc);
  if (existsSync(src)) cpSync(src, join(out, doc));
}

// On Windows with the desktop build, also produce a per-user installer when NSIS is available.
if (win && desktop) {
  const envVersion = process.env.VOIDWRAP_VERSION;
  const version = envVersion && envVersion.length > 0 ? envVersion : "dev";
  const nsisBundled = "C:\\Program Files (x86)\\NSIS\\makensis.exe";
  const where = spawnSync("where", ["makensis"], { encoding: "utf8" });
  let makensis = null;
  if (where.status === 0) {
    makensis = (where.stdout ?? "").split(/\r?\n/).map((line) => line.trim()).find((line) => line.length > 0) ?? null;
  } else if (existsSync(nsisBundled)) {
    makensis = nsisBundled;
  }
  if (makensis) {
    const setup = join(root, "dist", `VOIDWRAP_RUN-${version}-windows-setup.exe`);
    console.log(`package: building the Windows installer -> ${setup}`);
    // cwd = release\ so the icon path (..\design\icon\icon.ico) inside installer.nsi resolves.
    const r = spawnSync(makensis, [`/DSRC=${out}`, `/DVERSION=${version}`, `/DOUTFILE=${setup}`, "installer.nsi"], {
      cwd: join(root, "release"),
      stdio: "inherit",
    });
    if (r.status !== 0) process.exit(r.status ?? 1);
  } else {
    console.log("package: makensis not found; skipping the Windows installer (install NSIS 3 to enable it)");
  }
}

console.log(`package: ready -> ${out}`);
