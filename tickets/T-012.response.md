//! Pure, deterministic M1 game rules: gift pickups, the beacon, and chaser enemies.
//!
//! Everything here is a plain function over protocol data so the authoritative server
//! and the client-side predictor can run the exact same code. Only `+ - * /` and
//! `sqrt` are used, so results are bit-identical on every target.

use protocol::{Aabb, Gift, Vec3, TICK_HZ};

use crate::controller::player_box;
use crate::world::overlaps;

/// A gift is collected when the player centre is within this distance of it.
pub const GIFT_RADIUS: f32 = 1.1;

/// Horizontal radius in which the beacon counts as reached.
pub const BEACON_RADIUS: f32 = 1.5;

/// Vertical tolerance for standing on the beacon's platform.
pub const BEACON_HEIGHT: f32 = 1.0;

/// Enemy kind tag for the flying chaser.
pub const ENEMY_KIND_CHASER: u8 = 1;

/// Chaser flight speed in units per second (the player runs at 9.0).
pub const CHASER_SPEED: f32 = 5.5;

/// A chaser only pursues targets within this 3D distance.
pub const CHASER_AGGRO_RADIUS: f32 = 22.0;

/// Half extent of the chaser's cubic hit box, centred on its position.
pub const CHASER_HALF_SIZE: f32 = 0.5;

/// Ticks a freshly spawned chaser stays dormant (3 s at 20 Hz).
pub const CHASER_GRACE_TICKS: u32 = 60;

/// Height of the player's centre above their feet.
const PLAYER_CENTRE_HEIGHT: f32 = 0.9;

/// A single enemy. `home` is where it drifts back to when nothing is in range.
#[derive(Debug, Clone, PartialEq)]
pub struct Enemy {
    pub id: u32,
    pub kind: u8,
    pub pos: Vec3,
    pub home: Vec3,
    pub active_in: u32,
}

/// Euclidean distance between two points.
fn distance(a: &Vec3, b: &Vec3) -> f32 {
    let dx = a.x - b.x;
    let dy = a.y - b.y;
    let dz = a.z - b.z;
    (dx * dx + dy * dy + dz * dz).sqrt()
}

/// The player's centre point, derived from their feet position.
fn player_centre(player_feet: &Vec3) -> Vec3 {
    Vec3::new(
        player_feet.x,
        player_feet.y + PLAYER_CENTRE_HEIGHT,
        player_feet.z,
    )
}

/// Move `pos` toward `target` by at most `max_step`, never overshooting.
fn move_toward(pos: &mut Vec3, target: &Vec3, max_step: f32) {
    let dx = target.x - pos.x;
    let dy = target.y - pos.y;
    let dz = target.z - pos.z;
    let d = (dx * dx + dy * dy + dz * dz).sqrt();
    if d <= max_step {
        *pos = target.clone();
        return;
    }
    let scale = max_step / d;
    pos.x += dx * scale;
    pos.y += dy * scale;
    pos.z += dz * scale;
}

/// True when the player's centre is within [`GIFT_RADIUS`] of the gift centre.
pub fn gift_touched(player_feet: &Vec3, gift: &Gift) -> bool {
    distance(&player_centre(player_feet), &gift.pos) <= GIFT_RADIUS
}

/// True when the player stands on the beacon: close horizontally and roughly level.
pub fn beacon_reached(player_feet: &Vec3, beacon: &Vec3) -> bool {
    let dx = player_feet.x - beacon.x;
    let dz = player_feet.z - beacon.z;
    let horizontal = (dx * dx + dz * dz).sqrt();
    horizontal <= BEACON_RADIUS && (player_feet.y - beacon.y).abs() <= BEACON_HEIGHT
}

/// Create an enemy sitting at `pos`, with `pos` as its home and a full grace period.
pub fn spawn_enemy(id: u32, kind: u8, pos: Vec3) -> Enemy {
    Enemy {
        id,
        kind,
        home: pos.clone(),
        pos,
        active_in: CHASER_GRACE_TICKS,
    }
}

/// Advance one enemy by one server tick (`dt = 1 / TICK_HZ`).
///
/// `targets` are the centres of living players. While `active_in > 0` the enemy only
/// counts down. Afterwards it flies straight at the nearest target inside
/// [`CHASER_AGGRO_RADIUS`] (ties go to the lowest index), or drifts home at half speed.
/// Unknown kinds do nothing at all.
pub fn step_enemy(enemy: &mut Enemy, targets: &[Vec3]) {
    if enemy.kind != ENEMY_KIND_CHASER {
        return;
    }
    if enemy.active_in > 0 {
        enemy.active_in -= 1;
        return;
    }

    let step = CHASER_SPEED / TICK_HZ as f32;

    let mut nearest: Option<(usize, f32)> = None;
    for (index, target) in targets.iter().enumerate() {
        let d = distance(&enemy.pos, target);
        if d > CHASER_AGGRO_RADIUS {
            continue;
        }
        match nearest {
            Some((_, best)) if best <= d => {}
            _ => nearest = Some((index, d)),
        }
    }

    match nearest {
        Some((index, _)) => {
            let target = targets[index].clone();
            move_toward(&mut enemy.pos, &target, step);
        }
        None => {
            let home = enemy.home.clone();
            move_toward(&mut enemy.pos, &home, step * 0.5);
        }
    }
}

/// True when the enemy's cube overlaps the player's collision box.
pub fn enemy_touches(enemy: &Enemy, player_feet: &Vec3) -> bool {
    let half = CHASER_HALF_SIZE;
    let cube = Aabb {
        min: Vec3::new(enemy.pos.x - half, enemy.pos.y - half, enemy.pos.z - half),
        max: Vec3::new(enemy.pos.x + half, enemy.pos.y + half, enemy.pos.z + half),
    };
    overlaps(&cube, &player_box(player_feet))
}

//! Hand-authored levels.
//!
//! Three M1 designs live here. Each is a set of floating slabs in the void that read as
//! a branching route rather than a corridor: the player can wander, backtrack, and take
//! optional detours for gifts. All geometry respects the movement envelope (a running
//! jump clears 4.0 units of horizontal gap at equal height, or rises 1.5 units from 3.0
//! units away; a jump plus dash clears 7.0 units).

use protocol::{Aabb, Gift, LevelData, Platform, Vec3};

use crate::rules::ENEMY_KIND_CHASER;

/// Number of hand-authored designs. Levels above this reuse them cyclically.
pub const LEVEL_COUNT: u32 = 3;

/// A rectangular slab whose top face sits at `top` and which is one unit thick.
fn slab(id: u16, cx: f32, top: f32, cz: f32, half_x: f32, half_z: f32) -> Platform {
    Platform {
        id,
        kind: 0,
        bounds: Aabb {
            min: Vec3::new(cx - half_x, top - 1.0, cz - half_z),
            max: Vec3::new(cx + half_x, top, cz + half_z),
        },
    }
}

/// A gift floating one unit above a platform top, centred on `(x, z)`.
fn gift(id: u16, x: f32, top: f32, z: f32) -> Gift {
    Gift {
        id,
        pos: Vec3::new(x, top + 1.0, z),
    }
}

/// M0 bootstrap level: one grey 40×40 plane floating in the void.
pub fn bootstrap() -> LevelData {
    LevelData {
        level: 0,
        seed: 0,
        spawn: Vec3::new(0.0, 0.0, 0.0),
        beacon: Vec3::new(0.0, 0.0, -15.0),
        kill_y: -30.0,
        platforms: vec![slab(0, 0.0, 0.0, 0.0, 20.0, 20.0)],
        gifts: vec![],
    }
}

/// Which hand-authored design backs level `n`. Level 0 behaves like level 1.
fn design_of(n: u32) -> u32 {
    if n == 0 {
        1
    } else {
        ((n - 1) % LEVEL_COUNT) + 1
    }
}

/// Level `n`. Designs 1..=3 are hand-authored; higher levels cycle through them.
pub fn level(n: u32) -> LevelData {
    let mut data = match design_of(n) {
        1 => design_one(),
        2 => design_two(),
        _ => design_three(),
    };
    data.level = n;
    data.seed = n;
    data
}

/// Enemy spawn points for level `n`, as `(kind, position)` pairs.
pub fn enemy_spawns(n: u32) -> Vec<(u8, Vec3)> {
    match design_of(n) {
        1 => vec![(ENEMY_KIND_CHASER, Vec3::new(12.0, 2.0, 10.0))],
        2 => vec![(ENEMY_KIND_CHASER, Vec3::new(12.0, 5.0, 10.0))],
        _ => vec![
            (ENEMY_KIND_CHASER, Vec3::new(0.0, 3.5, -27.0)),
            (ENEMY_KIND_CHASER, Vec3::new(0.0, 5.0, -33.0)),
        ],
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::world::overlaps;

    const EPS: f32 = 1e-4;

    /// Horizontal (x/z) distance between two platform rectangles; 0 when they overlap.
    fn gap_xz(a: &Platform, b: &Platform) -> f32 {
        let gx = (b.bounds.min.x - a.bounds.max.x)
            .max(a.bounds.min.x - b.bounds.max.x)
            .max(0.0);
        let gz = (b.bounds.min.z - a.bounds.max.z)
            .max(a.bounds.min.z - b.bounds.max.z)
            .max(0.0);
        (gx * gx + gz * gz).sqrt()
    }

    /// Index of the platform whose top is `top` and whose x/z extent contains `(x, z)`.
    fn platform_at_top(level: &LevelData, x: f32, z: f32, top: f32) -> Option<usize> {
        level.platforms.iter().position(|p| {
            x >= p.bounds.min.x - EPS
                && x <= p.bounds.max.x + EPS
                && z >= p.bounds.min.z - EPS
                && z <= p.bounds.max.z + EPS
                && (p.bounds.max.y - top).abs() < EPS
        })
    }

    /// Directed reachability: A -> B when the gap is <= 4.0 and B is at most 1.5 higher.
    fn reachable_from(level: &LevelData, from: usize) -> Vec<bool> {
        let mut seen = vec![false; level.platforms.len()];
        let mut stack = vec![from];
        seen[from] = true;
        while let Some(i) = stack.pop() {
            let a = &level.platforms[i];
            for (j, b) in level.platforms.iter().enumerate() {
                if seen[j] {
                    continue;
                }
                if gap_xz(a, b) <= 4.0 && b.bounds.max.y <= a.bounds.max.y + 1.5 {
                    seen[j] = true;
                    stack.push(j);
                }
            }
        }
        seen
    }

    /// Every structural invariant shared by all hand-authored levels.
    fn check_level(level: &LevelData) {
        assert!(!level.platforms.is_empty(), "level has no platforms");

        let mut ids: Vec<u16> = level.platforms.iter().map(|p| p.id).collect();
        ids.sort_unstable();
        ids.dedup();
        assert_eq!(
            ids.len(),
            level.platforms.len(),
            "platform ids must be unique"
        );

        for (i, a) in level.platforms.iter().enumerate() {
            for b in level.platforms.iter().skip(i + 1) {
                assert!(
                    !overlaps(&a.bounds, &b.bounds),
                    "platforms {} and {} overlap",
                    a.id,
                    b.id
                );
            }
        }

        let first = &level.platforms[0];
        assert!(
            (level.spawn.y - first.bounds.max.y).abs() < EPS,
            "spawn is not on platform 0's top"
        );
        assert!(
            level.spawn.x >= first.bounds.min.x - EPS
                && level.spawn.x <= first.bounds.max.x + EPS
                && level.spawn.z >= first.bounds.min.z - EPS
                && level.spawn.z <= first.bounds.max.z + EPS,
            "spawn is outside platform 0's x/z extent"
        );

        assert_eq!(level.gifts.len(), 5, "every level has exactly 5 gifts");
        for (i, g) in level.gifts.iter().enumerate() {
            assert_eq!(g.id, i as u16, "gift ids must be 0..5 in order");
            assert!(
                platform_at_top(level, g.pos.x, g.pos.z, g.pos.y - 1.0).is_some(),
                "gift {} is not 1.0 above a platform top",
                g.id
            );
        }

        assert!(
            platform_at_top(level, level.beacon.x, level.beacon.z, level.beacon.y).is_some(),
            "beacon is not on a platform top"
        );

        let lowest = level
            .platforms
            .iter()
            .map(|p| p.bounds.min.y)
            .fold(f32::INFINITY, f32::min);
        assert!(
            level.kill_y <= lowest - 10.0,
            "kill_y must be at least 10 below the lowest platform bottom"
        );

        for p in &level.platforms {
            assert!(
                p.bounds.max.y >= 0.0 && p.bounds.max.y <= 8.0,
                "platform {} top is outside [0, 8]",
                p.id
            );
        }
    }

    /// Every gift platform and the beacon platform must be reachable from platform 0.
    fn check_reachability(level: &LevelData) {
        let seen = reachable_from(level, 0);
        for g in &level.gifts {
            let idx = platform_at_top(level, g.pos.x, g.pos.z, g.pos.y - 1.0)
                .expect("gift platform exists");
            assert!(seen[idx], "gift platform {} is unreachable", idx);
        }
        let beacon = platform_at_top(level, level.beacon.x, level.beacon.z, level.beacon.y)
            .expect("beacon platform exists");
        assert!(seen[beacon], "beacon platform {} is unreachable", beacon);
    }

    #[test]
    fn bootstrap_is_unchanged() {
        let level = bootstrap();
        assert_eq!(level.level, 0);
        assert_eq!(level.seed, 0);
        assert_eq!(level.spawn, Vec3::new(0.0, 0.0, 0.0));
        assert_eq!(level.beacon, Vec3::new(0.0, 0.0, -15.0));
        assert_eq!(level.kill_y, -30.0);
        assert_eq!(level.platforms.len(), 1);
        assert_eq!(level.platforms[0].id, 0);
