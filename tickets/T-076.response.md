const FALLBACK_LOOK: EnemyLook = { behavior: "pallbearer", accent: "#ff3355", glow: "#ffdf9e" };

export function isEnemyKind(kind: number): boolean {
  return Number.isInteger(kind) && kind >= 0 && kind < ENEMY_LOOKS.length;
}

/** Unknown/missing kinds fall back to the pallbearer look (never crash a render). */
export function enemyLookFor(kind: number | undefined): EnemyLook {
  if (kind === undefined) return FALLBACK_LOOK;
  const look = ENEMY_LOOKS[kind];
  return look ?? FALLBACK_LOOK;
}

export function isTelegraphState(state: number | undefined): boolean {
  return state === ENEMY_TELEGRAPH || state === ENEMY_ATTACK;
}

export function clamp01(v: number): number {
  if (!Number.isFinite(v)) return 0;
  return v < 0 ? 0 : v > 1 ? 1 : v;
}

/** Wire `prog` (0–255) → 0..1. */
export function prog01(prog: number | undefined): number {
  if (prog === undefined || !Number.isFinite(prog)) return 0;
  return clamp01(prog / 255);
}

export function lerp(a: number, b: number, t: number): number {
  const c = clamp01(t);
  return a + (b - a) * c;
}

/* ---------- telegraph curves ---------- */

/** Alpha ramp shared by every telegraph: dim at prog 0, solid at 255. */
export function telegraphAlpha(t01: number): number {
  return 0.3 + 0.7 * clamp01(t01);
}

/** Moteling (Stillness Shrieker) INHALE: warning ring starts at orbit radius and collapses onto the victim. */
export const MOTELING_RING_START = 5.0;
export const MOTELING_RING_END = 0.45;
export function motelingRingRadius(t01: number): number {
  return lerp(MOTELING_RING_START, MOTELING_RING_END, t01);
}

/** Rings that grow outward as an attack resolves (shriek snap / warden shove / fret spread). */
export function expandingRingRadius(t01: number, maxRadius: number): number {
  return Math.max(0.25, clamp01(t01) * maxRadius);
}

export const SNAPSHRIKE_LOCK_PROG = 0.4 / 0.75; // ≈ 0.533
/** Alpha for the LOCK line: ramps while tracking, snaps solid once the line freezes. */
export function snapshrikeAlpha(t01: number): number {
  const t = clamp01(t01);
  return t >= SNAPSHRIKE_LOCK_PROG ? 1 : telegraphAlpha(t) * 0.85;
}

/** MIMIC CACHED tell: a 0.5 s magenta shimmer every 3.0 s. Purely client-clock driven. */
export const MIMIC_SHIMMER_PERIOD = 3.0;
export const MIMIC_SHIMMER_WINDOW = 0.5;
function shimmerPhase(timeS: number): number {
  return ((timeS % MIMIC_SHIMMER_PERIOD) + MIMIC_SHIMMER_PERIOD) % MIMIC_SHIMMER_PERIOD;
}
export function mimicShimmerVisible(timeS: number): boolean {
  if (!Number.isFinite(timeS)) return false;
  return shimmerPhase(timeS) < MIMIC_SHIMMER_WINDOW;
}
/** 0 → peak → 0 across the shimmer window; 0 outside it. */
export function mimicShimmerOpacity(timeS: number): number {
  if (!mimicShimmerVisible(timeS)) return 0;
  return Math.sin((shimmerPhase(timeS) / MIMIC_SHIMMER_WINDOW) * Math.PI) * 0.9;
}

/** Yaw that faces a model (+Z forward) from (fx,fz) toward (tx,tz). */
export function yawToward(fx: number, fz: number, tx: number, tz: number): number {
  return Math.atan2(tx - fx, tz - fz);
}

import { clamp01, prog01, yawToward } from "../enemyLook";

const ECHOS = 4;

/**
 * Rewind — a thin replay ghost: a translucent body trailing a short ribbon of its
 * own past positions (the echoes). Accent #46e0ff (temporal cyan). Unit-radius model;
 * the parent scales by body radius. Idle: slow bob and echo drift.
 * Telegraph (SNAP back): the body trembles and the echoes flare. Attack: snap-back lurch.
 */
export function Rewind({ anim }: EnemyModelProps): JSX.Element {
  const root = useRef<Group>(null);
  const inner = useRef<Group>(null);
  const bodyMat = useRef<MeshStandardMaterial>(null);
  const headMat = useRef<MeshStandardMaterial>(null);
  const echoRefs = useRef<(Group | null)[]>([]);
  const echoMats = useRef<(MeshStandardMaterial | null)[]>([]);

  useFrame((state) => {
    const r = root.current;
    if (!r || (r.parent !== null && !r.parent.visible)) return;
    const a = anim.current;
    const t = state.clock.elapsedTime;
    const dormant = a.state === ENEMY_DORMANT;
    const telegraph = a.state === ENEMY_TELEGRAPH;
    const attack = a.state === ENEMY_ATTACK;
    const p = prog01(a.prog);
    const dim = dormant ? 0.14 : 1;
    const winding = telegraph || attack;

    if (winding) {
      r.rotation.y = yawToward(a.ex, a.ez, a.tx, a.tz);
    }

    if (inner.current) {
      inner.current.position.x = telegraph ? Math.sin(t * 36) * 0.05 * (0.35 + p) : 0;
      inner.current.position.y = Math.sin(t * 1.9) * 0.07;
      inner.current.position.z = attack ? -Math.sin(clamp01(p) * Math.PI) * 0.45 : 0;
      inner.current.rotation.z = telegraph
        ? Math.sin(t * 41 + 0.7) * 0.06 * (0.3 + p)
        : Math.sin(t * 1.1) * 0.04;
    }
    const heat = 0.55 + Math.sin(t * 3.1) * 0.25 + (winding ? p * 2.3 : 0);
    if (bodyMat.current) bodyMat.current.emissiveIntensity = heat * dim;
    if (headMat.current) headMat.current.emissiveIntensity = (heat + 0.5) * dim;
    for (let i = 0; i < ECHOS; i += 1) {
      const g = echoRefs.current[i];
      const m = echoMats.current[i];
      const lag = i + 1;
      if (g) {
        const drift = Math.sin(t * 2.3 - lag * 0.9);
        g.position.z = -0.3 - lag * 0.21 - drift * 0.05;
        g.position.y = 0.15 + Math.sin(t * 2.3 - lag * 0.9 + 1.1) * 0.05;
        const s = 1 - lag * 0.16;
        g.scale.set(s * 0.75, s * 1.45, s * 0.75);
      }
      if (m) {
        m.opacity = clamp01((0.42 - lag * 0.09) * (winding ? 1 + p * 1.4 : 1)) * dim;
        m.emissiveIntensity = (0.9 + (winding ? p * 2 : 0)) * dim;
      }
    }
  });

  return (
    <group ref={root}>
      <group ref={inner}>
        {/* trunk — a tall tapered shade */}
        <mesh position={[0, 0.35, 0]}>
          <cylinderGeometry args={[0.15, 0.34, 1.5, 10]} />
          <meshStandardMaterial
            ref={bodyMat}
            color="#0e2a33"
            emissive="#46e0ff"
            emissiveIntensity={0.6}
            transparent
            opacity={0.62}
            roughness={0.3}
            metalness={0.05}
            toneMapped={false}
            depthWrite={false}
          />
        </mesh>
        {/* head */}
        <mesh position={[0, 1.18, 0]}>
          <sphereGeometry args={[0.17, 12, 10]} />
          <meshStandardMaterial
            ref={headMat}
            color="#12333d"
            emissive="#cdf6ff"
            emissiveIntensity={1.1}
            transparent
            opacity={0.7}
            roughness={0.25}
            toneMapped={false}
            depthWrite={false}
          />
        </mesh>
        {/* tattered tail */}
        <mesh position={[0, -0.62, 0]} rotation={[Math.PI, 0, 0]}>
          <coneGeometry args={[0.3, 0.75, 10]} />
          <meshStandardMaterial
            color="#0e2a33"
            emissive="#46e0ff"
            emissiveIntensity={0.45}

        ))}

  );

"use client";

import { useRef } from "react";
import type { JSX } from "react";
import { useFrame } from "@react-three/fiber";
import type { Group, Mesh, MeshStandardMaterial } from "three";
import { DoubleSide } from "three";
import { ENEMY_ATTACK, ENEMY_DORMANT, ENEMY_TELEGRAPH } from "@/protocol/generated";
import type { EnemyModelProps } from "../enemyLook";
import { clamp01, prog01, yawToward } from "../enemyLook";

const SHELL_H = 1.15;

/**
 * Peal — a bronze bell whose two shell halves swing open at the lip as the ring
 * charges. Accent #e63946 (bell-mouth red). Unit-radius model; the parent scales by
 * body radius. Idle: slow sway. Telegraph (PEAL): lip opens and the mouth glows.
 * Attack (the ring): the shell clangs through a full open-shut arc.
 */
export function Peal({ anim }: EnemyModelProps): JSX.Element {
  const root = useRef<Group>(null);
  const inner = useRef<Group>(null);
  const leftHalf = useRef<Group>(null);
  const rightHalf = useRef<Group>(null);
  const clapper = useRef<Group>(null);
  const lipMesh = useRef<Mesh>(null);
  const lipMat = useRef<MeshStandardMaterial>(null);
  const glowMat = useRef<MeshStandardMaterial>(null);
  const shellMats = useRef<(MeshStandardMaterial | null)[]>([]);

  useFrame((state) => {
    const r = root.current;
    if (!r || (r.parent !== null && !r.parent.visible)) return;
    const a = anim.current;
    const t = state.clock.elapsedTime;
    const dormant = a.state === ENEMY_DORMANT;
    const telegraph = a.state === ENEMY_TELEGRAPH;
    const attack = a.state === ENEMY_ATTACK;
    const p = prog01(a.prog);
    const dim = dormant ? 0.16 : 1;
    const winding = telegraph || attack;

    if (winding) {
      r.rotation.y = yawToward(a.ex, a.ez, a.tx, a.tz);
    }

    // How open the mouth is: telegraph winds it open; attack clangs open→shut.
    const open = telegraph
      ? p
      : attack
        ? Math.sin(clamp01(p) * Math.PI)
        : 0;

    if (leftHalf.current) leftHalf.current.rotation.z = -open * 0.42;
    if (rightHalf.current) rightHalf.current.rotation.z = open * 0.42;
    if (inner.current) {
      inner.current.position.y = Math.sin(t * 1.7) * 0.06;
      inner.current.rotation.x = winding ? -p * 0.22 : 0;
      inner.current.rotation.z = Math.sin(t * 0.8) * 0.03;
    }
    if (clapper.current) {
      clapper.current.rotation.z = Math.sin(t * (winding ? 11 : 1.6)) *
        (winding ? 0.35 + p * 0.75 : 0.12);
    }
    if (lipMesh.current) {
      const s = 1 + open * 0.18;
      lipMesh.current.scale.set(s, s, 1);
    }
    if (lipMat.current) {
      lipMat.current.emissiveIntensity =
        (0.9 + Math.sin(t * 2.6) * 0.25 + open * 3.2) * dim;
    }
    if (glowMat.current) {
      glowMat.current.opacity = (0.18 + open * 0.72) * dim;
      glowMat.current.emissiveIntensity = (1 + open * 2.6) * dim;
    }
    for (const m of shellMats.current) {
      if (m) m.emissiveIntensity = (0.18 + open * 1.1) * dim;
    }
  });

  const shellMaterial = (slot: number) => (
    <meshStandardMaterial
      ref={(m) => {
        shellMats.current[slot] = m;
      }}
      color="#2a2118"
      emissive="#e63946"
      emissiveIntensity={0.18}
      roughness={0.35}
      metalness={0.75}
      side={DoubleSide}
    />
  );

  return (
    <group ref={root}>
      <group ref={inner}>
        {/* crown */}
        <mesh position={[0, 0.66, 0]}>
          <cylinderGeometry args={[0.09, 0.14, 0.22, 10]} />
          <meshStandardMaterial color="#3a2c1c" roughness={0.4} metalness={0.8} />
        </mesh>
        <mesh position={[0, 0.82, 0]}>
          <sphereGeometry args={[0.085, 10, 8]} />
          <meshStandardMaterial color="#4a3823" roughness={0.35} metalness={0.85} />
        </mesh>
        {/* two shell halves, pivoting at the crown so the lip splays open */}
        <group ref={leftHalf} position={[0, 0.56, 0]}>
          <mesh position={[0, -SHELL_H / 2, 0]}>
            <cylinderGeometry args={[0.24, 0.62, SHELL_H, 12, 1, true, 0, Math.PI]} />
            {shellMaterial(0)}
          </mesh>

export const MAX_TELEGRAPHS = 24;

interface TelegraphSlot {
  outer: Group | null;
  ring: Mesh | null;
  disc: Mesh | null;
  line: Mesh | null;
  ringMat: MeshBasicMaterial;
  discMat: MeshBasicMaterial;
  lineMat: MeshBasicMaterial;
}

function createSlot(): TelegraphSlot {
  const base = {
    transparent: true,
    opacity: 0,
    blending: AdditiveBlending,
    depthWrite: false,
    toneMapped: false,
    side: DoubleSide,
  } as const;
  return {
    outer: null,
    ring: null,
    disc: null,
    line: null,
    ringMat: new MeshBasicMaterial({ ...base, color: "#ffffff" }),
    discMat: new MeshBasicMaterial({ ...base, color: "#ffffff" }),
    lineMat: new MeshBasicMaterial({ ...base, color: "#ffffff" }),
  };
}

/**
 * Pooled telegraph drawings, one slot per enemy (max 24). Everything is additive,
 * bright, and tinted with the enemy's accent colour; alpha and radius are driven by
 * the wire `prog` (0..255). Slots exist only while the enemy is TELEGRAPH/ATTACK —
 * otherwise hidden. `target` is the telegraph anchor from the server payload.
 * The switch covers all twenty roster ids.
 */
export function Telegraphs({ session }: { session: GameSession }): JSX.Element {
  const slots = useRef<TelegraphSlot[]>([]);
  if (slots.current.length === 0) {
    slots.current = Array.from({ length: MAX_TELEGRAPHS }, createSlot);
  }

  const ringGeo = useMemo(() => new RingGeometry(0.86, 1, 48), []);
  const discGeo = useMemo(() => new CircleGeometry(1, 40), []);
  const lineGeo = useMemo(() => new PlaneGeometry(1, 1), []);
  const palette = useMemo(() => ENEMY_LOOKS.map((l) => new Color(l.accent)), []);
  const fallbackColor = useMemo(() => new Color("#ffffff"), []);

  useEffect(
    () => () => {
      for (const slot of slots.current) {
        slot.ringMat.dispose();
        slot.discMat.dispose();
        slot.lineMat.dispose();
      }
      ringGeo.dispose();
      discGeo.dispose();
      lineGeo.dispose();
    },
    [ringGeo, discGeo, lineGeo],
  );

  // Ribbon lines orient with Euler order Y→X→Z (yaw then pitch). Set once.
  useEffect(() => {
    for (const slot of slots.current) {
      if (slot.line) slot.line.rotation.order = "YXZ";
    }
  }, []);

  useFrame((state) => {
    const now = performance.now();
    const t = state.clock.elapsedTime;
    let i = 0;
    for (const [id, o] of session.enemyPositions(now)) {
      if (i >= MAX_TELEGRAPHS) break;
      const slot = slots.current[i];
      if (!slot || !slot.outer) {
        i += 1;
        continue;

      const info = session.enemyInfo.get(id);
      if (!info || !isTelegraphState(info.state)) {
        slot.outer.visible = false;
        i += 1;
        continue;

      const kind = info.kind;
      const look = enemyLookFor(kind);
      const col =
        kind !== undefined && kind >= 0 && kind < palette.length
          ? palette[kind] ?? fallbackColor
          : fallbackColor;

      const attack = info.state === ENEMY_ATTACK;
      const p = prog01(info.prog);
      const anchorX = info.target.x;
      const anchorY = info.target.y;
      const anchorZ = info.target.z;

      let ringVisible = false;
      let discVisible = false;
      let lineVisible = false;
      let ringX = anchorX;
      let ringY = anchorY + 0.06;
      let ringZ = anchorZ;
      let ringRadius = 1;
      let ringAlpha = telegraphAlpha(p);
      let discRadius = 1;
      let discAlpha = 0;
      let lineWidth = 0.2;
      let lineAlpha = telegraphAlpha(p);

      switch (look.behavior) {
        case "pallbearer":

          lineWidth = 0.2;

          ringAlpha = telegraphAlpha(p);

        case "snapshrike":

          lineVisible = true;

            lineAlpha = 0.6 * (1 - p);
          } else {

            lineAlpha = snapshrikeAlpha(p);
          }
          break;
        case "moteling":
          // INHALE: collapsing ring around the marked player; SHRIEK: ring snap
          ringVisible = true;
          if (attack) {
            ringRadius = expandingRingRadius(p, 4.0);
            ringAlpha = 0.9 * (1 - p);
