  name: string;
  flavor: string;
  behavior: string;

  name: string;
  flavor: string;
  effect: string;
  rarity: "common" | "rare";
  mods: Mods;
  threat?: number;
  greater?: boolean;

export interface ScreenGrade {
  tint: string;
  tint_strength: number;
  saturation: number;
  contrast: number;
  brightness: number;
  vignette_color: string;
  vignette_darkness: number;
  aberration: number;
  grain: number;
  scanlines: number;
}

export const ENEMIES: readonly EnemyCard[] = content.enemies;
export const UPGRADES: readonly ModCard[] = content.upgrades as ModCard[];
export const CURSES: readonly ModCard[] = content.curses as ModCard[];
const GRADES: Record<string, ScreenGrade> = content.grades;

/** Card kinds as used on the wire (protocol DraftCard.kind). */
export const CARD_ENEMY = 0;
export const CARD_UPGRADE = 1;
export const CARD_CURSE = 2;

export type AnyCard =
  | { kind: typeof CARD_ENEMY; card: EnemyCard }
  | { kind: typeof CARD_UPGRADE; card: ModCard }
  | { kind: typeof CARD_CURSE; card: ModCard };

/** Resolve a wire card (kind + id) to its content; null for unknown ids (never trust the wire blindly). */
export function lookupCard(kind: number, id: string): AnyCard | null {
  if (kind === CARD_ENEMY) {
    const card = ENEMIES.find((e) => e.id === id);
    return card ? { kind: CARD_ENEMY, card } : null;
  }
  if (kind === CARD_UPGRADE) {
    const card = UPGRADES.find((e) => e.id === id);
    return card ? { kind: CARD_UPGRADE, card } : null;
  }
  if (kind === CARD_CURSE) {
    const card = CURSES.find((e) => e.id === id);
    return card ? { kind: CARD_CURSE, card } : null;
  }
  return null;
}

const NEUTRAL_GRADE: ScreenGrade = {
  tint: "#ffffff",
  tint_strength: 0,
  saturation: 0,
  contrast: 0,
  brightness: 0,
  vignette_color: "#000000",
  vignette_darkness: 0.4,
  aberration: 0.001,
  grain: 0.02,
  scanlines: 0,
};

export function gradeForTheme(themeId: string): ScreenGrade {
  return GRADES[themeId] ?? NEUTRAL_GRADE;
}

// Pure draft-selection logic: enemy cards (kind 0) and reward cards (kind 1/2)
// have separate pick limits; selecting beyond a limit replaces the oldest
// selection in that same group. No DOM, no React, fully deterministic.

/** Protocol DraftCard.kind for enemy cards (see content.ts CARD_ENEMY). */
const ENEMY_KIND = 0;

/** Safe kind lookup: undefined for out-of-range indices (avoids unchecked indexing). */
function kindAt(cards: readonly { kind: number }[], index: number): number | undefined {
  if (index < 0 || index >= cards.length) {
    return undefined;
  }
  return cards[index]?.kind;
}

function isEnemyKind(cards: readonly { kind: number }[], index: number): boolean {
  return kindAt(cards, index) === ENEMY_KIND;
}

export function toggleSelection(
  selected: readonly number[],
  index: number,
  cards: readonly { kind: number }[],
  enemyPicks: number,
  rewardPicks: number,
): number[] {
  // Out-of-range index -> unchanged.
  if (index < 0 || index >= cards.length) {
    return [...selected];
  }
  // Clicking a selected card deselects it.
  if (selected.includes(index)) {
    return selected.filter((i) => i !== index);
  }

  const enemy = isEnemyKind(cards, index);
  const limit = enemy ? enemyPicks : rewardPicks;
  // A group with no allowance can never be selected into.
  if (limit <= 0) {
    return [...selected];
  }

  const next = [...selected];
  const sameGroup = next.filter((i) => isEnemyKind(cards, i) === enemy);
  // Selecting beyond the limit for this group replaces the OLDEST selection
  // in the same group (oldest = first one still in the ordered selection).
  if (sameGroup.length >= limit) {
    for (const i of next) {
      if (isEnemyKind(cards, i) === enemy) {
        next.splice(next.indexOf(i), 1);
        break;
      }
    }
  }
  next.push(index);
  return next;
}

export function selectionComplete(
  selected: readonly number[],
  cards: readonly { kind: number }[],
  enemyPicks: number,
  rewardPicks: number,
): boolean {
  const inRange = selected.filter((i) => i >= 0 && i < cards.length);
  const enemies = inRange.filter((i) => isEnemyKind(cards, i));
  const rewards = inRange.filter((i) => !isEnemyKind(cards, i));
  return enemies.length === enemyPicks && rewards.length === rewardPicks;
}

  const clamped = Math.max(0, Math.min(5, Math.round(props.threat)));

              {resolved.card.rarity.toUpperCase()}

      : null;

  return (
    <button
      type="button"
      aria-pressed={selected}
      onClick={onToggle}
      className={[

          color: selected ? accent : MUTED,

'use client';

// Full-screen draft overlay: enemy picks + reward picks, hotkeys 1..n, Enter
// to submit. The selection state lives in an inner component that is keyed by
// the card manifest, so a new draft restarts the choice cleanly (no setState
// in effects).
import { useCallback, useEffect, useState } from 'react';
import type { JSX } from 'react';
import '../content';
import { CARD_ENEMY } from '../content';
import { DraftCard } from './DraftCard';
import { selectionComplete, toggleSelection } from './draftSelection';

const DANGER = '#ff4d4d';
const MAGENTA = '#ff3db8';
const GOLD = '#ffd23d';
const MUTED = 'rgba(255,255,255,0.55)';

export interface DraftScreenProps {
  cards: readonly { kind: number; id: string }[];
  enemyPicks: number;
  rewardPicks: number;
  nextLevel: number;
  currency: number;
  lives: number;
  waiting: boolean;
  onSubmit: (picks: number[]) => void;
}

function DraftScreenInner(props: DraftScreenProps): JSX.Element {
  const { cards, enemyPicks, rewardPicks, nextLevel, currency, lives, waiting, onSubmit } = props;

  const enemyCount = cards.filter((c) => c.kind === CARD_ENEMY).length;
  const rewardCount = cards.length - enemyCount;
  // Never allow more picks than there are cards in a group.
  const enemyLimit = Math.min(enemyPicks, enemyCount);
  const rewardLimit = Math.min(rewardPicks, rewardCount);

  const [selected, setSelected] = useState<readonly number[]>([]);

  const toggle = useCallback(
    (index: number) => {
      if (waiting) {
        return;
      }
      setSelected((prev) => toggleSelection(prev, index, cards, enemyLimit, rewardLimit));
    },
    [cards, enemyLimit, rewardLimit, waiting],
  );

  const complete = selectionComplete(selected, cards, enemyLimit, rewardLimit);

  const submit = useCallback(() => {
    if (waiting || !complete) {
      return;
    }
    onSubmit([...selected].sort((a, b) => a - b));
  }, [complete, onSubmit, selected, waiting]);

  useEffect(() => {
    if (waiting) {
      return;
    }
    const onKeyDown = (event: KeyboardEvent): void => {
      if (event.repeat) {
        return;
      }
      if (event.ctrlKey || event.metaKey || event.altKey) {
        return;
      }
      if (event.key === 'Enter') {
        // A focused Continue button handles its own click via Enter.
        if (event.target instanceof HTMLElement && event.target.tagName === 'BUTTON') {
          return;
        }
        submit();
        return;
      }
      const digit = Number.parseInt(event.key, 10);
      if (!Number.isNaN(digit) && digit >= 1 && digit <= 9) {
        toggle(digit - 1);
      }
    };
    window.addEventListener('keydown', onKeyDown);
    return () => {
      window.removeEventListener('keydown', onKeyDown);
    };
  }, [submit, toggle, waiting]);

  const enemySelected = selected.filter((i) => i >= 0 && i < cards.length && cards[i]?.kind === CARD_ENEMY).length;
  const rewardSelected = selected.length - enemySelected;

  return (
    <div
      data-testid="draft-screen"
      className="pointer-events-auto fixed inset-0 z-40 overflow-y-auto bg-black/55 font-mono backdrop-blur-sm"
    >
      <div className="mx-auto flex min-h-full w-full max-w-4xl flex-col gap-5 px-4 py-6 text-white">
        <header className="flex flex-wrap items-baseline justify-between gap-2 border-b border-white/10 pb-3">
          <h1 className="text-2xl font-bold tracking-[0.3em]">LEVEL {nextLevel}</h1>
          <div className="flex items-center gap-4 text-sm">
            <span className="flex items-center gap-1" title="currency">
              <span aria-hidden="true" style={{ color: GOLD }}>●</span>
              <span className="text-white/90">{currency}</span>
            </span>
            <span style={{ color: DANGER }}>❤ {lives}</span>
          </div>
        </header>

        <p className="text-xs tracking-widest" style={{ color: MUTED }}>
          {enemySelected}/{enemyLimit} hunters · {rewardSelected}/{rewardLimit} rewards · keys 1-9 toggle · enter continues
        </p>

        {enemyCount > 0 && (
          <section className="flex flex-col gap-3">
            <h2 className="text-sm font-bold tracking-[0.25em] text-white/85">
              WHO HUNTS YOU <span style={{ color: DANGER }}>— choose {enemyPicks}</span>
            </h2>
            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-4">
              {cards.map((card, index) =>
                card.kind === CARD_ENEMY ? (
                  <span
                    key={`${index}:${card.id}`}
                    className={waiting ? 'pointer-events-none opacity-60' : undefined}
                  >
                    <DraftCard
                      kind={card.kind}
                      id={card.id}
                      selected={selected.includes(index)}
                      hotkey={index + 1}
                      onToggle={() => toggle(index)}
                    />
                  </span>
                ) : null,
              )}
            </div>
          </section>
        )}

        {rewardCount > 0 && (
          <section className="flex flex-col gap-3">
            <h2 className="text-sm font-bold tracking-[0.25em] text-white/85">
              TAKE ONE FROM THE VOID <span style={{ color: MAGENTA }}>— choose {rewardPicks}</span>
            </h2>
            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-4">
              {cards.map((card, index) =>
                card.kind === CARD_ENEMY ? null : (
                  <span
                    key={`${index}:${card.id}`}
                    className={waiting ? 'pointer-events-none opacity-60' : undefined}
                  >
                    <DraftCard
                      kind={card.kind}
                      id={card.id}
                      selected={selected.includes(index)}
                      hotkey={index + 1}
                      onToggle={() => toggle(index)}
                    />
                  </span>
                ),
              )}
            </div>
          </section>
        )}

        <footer className="mt-auto flex flex-col items-center gap-2 pt-4">
          {waiting ? (
            <div
              data-testid="draft-waiting"
              className="rounded-lg border border-white/15 bg-[#120f1f] px-6 py-3 text-sm tracking-[0.2em] text-white/70"
            >
              Waiting for other players...
            </div>
          ) : (
            <button
              type="button"
              data-testid="draft-continue"
              disabled={!complete}
              onClick={submit}
              className={[
                'rounded-lg border px-8 py-3 text-sm font-bold tracking-[0.3em] transition',
                complete
                  ? 'border-[#ff3db8] text-[#ff3db8] hover:bg-[#ff3db8]/15 focus-visible:ring-2 focus-visible:ring-[#ff3db8]/60 focus:outline-none'
                  : 'cursor-not-allowed border-white/15 text-white/30',
              ].join(' ')}
            >
              CONTINUE
            </button>
          )}
        </footer>
      </div>
    </div>
  );
}

export function DraftScreen(props: DraftScreenProps): JSX.Element {
  // Key the inner component by the card manifest so a fresh draft (new cards
  // array with new content) resets the selection without setState in effects.
  const cardsKey = props.cards.map((c) => `${c.kind}:${c.id}`).join('|');
  return <DraftScreenInner key={`draft:${cardsKey}`} {...props} />;
}

interface ChipData {
  key: string;
  label: string;
  detail: string;
  color: string;
  background: string;
  borderColor: string;
}

function buildChip(kind: number, id: string, color: string, background: string, borderColor: string): ChipData {
  const resolved = lookupCard(kind, id);
  if (resolved === null) {
    return {
      key: `${kind}:${id}`,
      label: id,
      detail: 'Unknown card',
      color,
      background,
      borderColor,
    };
  }
  return {
    key: `${kind}:${id}`,
    label: resolved.card.name,
    detail: resolved.kind === CARD_ENEMY ? resolved.card.behavior : resolved.card.effect,
    color,
    background,
    borderColor,
  };
}

import { describe, expect, it } from 'vitest';

import { selectionComplete, toggleSelection } from './draftSelection';

// 4 enemies (indices 0-3) + 2 upgrades (4-5) + 1 curse (6).
const CARDS = [
  { kind: 0 },
  { kind: 0 },
  { kind: 0 },
  { kind: 0 },
  { kind: 1 },
  { kind: 1 },
  { kind: 2 },
];

describe('toggleSelection', () => {
  it('selects cards up to the per-group limit', () => {
    expect(toggleSelection([], 0, CARDS, 2, 1)).toEqual([0]);
    expect(toggleSelection([0], 1, CARDS, 2, 1)).toEqual([0, 1]);
    expect(toggleSelection([0, 1], 4, CARDS, 2, 1)).toEqual([0, 1, 4]);
  });

  it('replaces the oldest selection within the same group', () => {
    // Enemies: limit 2, selecting a third enemy drops the oldest (index 0).
    expect(toggleSelection([0, 1], 2, CARDS, 2, 1)).toEqual([1, 2]);
    // Mixed selection: oldest ENEMY is replaced, reward stays.
    expect(toggleSelection([1, 2, 4], 0, CARDS, 2, 1)).toEqual([2, 4, 0]);
    // Rewards: limit 1, selecting another reward drops the previous one.
    expect(toggleSelection([2, 4], 5, CARDS, 2, 1)).toEqual([2, 5]);
  });

  it('keeps enemy and reward limits independent', () => {
    // Full on enemies but a reward pick still lands.
    expect(toggleSelection([1, 2, 4], 6, CARDS, 2, 1)).toEqual([1, 2, 4, 6]);
    // Full on rewards but an enemy pick still lands.
    expect(toggleSelection([2, 6], 0, CARDS, 2, 1)).toEqual([2, 6, 0]);
  });

  it('deselects a selected card', () => {
    expect(toggleSelection([1, 2, 4], 1, CARDS, 2, 1)).toEqual([2, 4]);
    expect(toggleSelection([1, 2, 4], 4, CARDS, 2, 1)).toEqual([1, 2]);
