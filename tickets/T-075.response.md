/// Seconds per simulation tick (60 Hz, matching `crate::tuning::DT`).
pub const SIM_DT: f32 = 1.0 / 60.0;

/// OVERTUNED: the debuff Coil inflicts that makes Peal's final ring lethal
/// (registry synergy hook `peal_lethal_when_overtuned`).
///
/// 10 seconds at 60 Hz. Batch A is adding `overtuned_ticks: u32` to the player
/// state; once it exists, the merge is one field on the player plus calls to
/// the three helpers below. If batch A has not landed yet, callers can track the
/// timer caller-side with the same helpers.
pub const OVERTUNED_TICKS: u32 = 600;

/// Apply or refresh OVERTUNED on a bare timer.
pub fn apply_overtuned(ticks: &mut u32) {
    *ticks = OVERTUNED_TICKS;
}

/// Advance an OVERTUNED timer by one tick; returns true while it is still live.
pub fn tick_overtuned(ticks: &mut u32) -> bool {
    if *ticks > 0 {
        *ticks -= 1;
    }
    *ticks > 0
}

/// True while an OVERTUNED timer is live. Peal's final ring reads this.
pub fn overtuned_active(ticks: u32) -> bool {
    ticks > 0
}

/// Per-player snapshot the brains read for one tick.
#[derive(Debug, Clone)]
pub struct PlayerSample<'a> {
    /// Stable player id.
    pub id: u32,
    /// Feet position.
    pub feet: Vec3,
    /// Class id from `design/content.json` (`striker`, `grappler`, `bulwark`, `glider`).
    pub class: &'a str,
    /// True on the tick this player launched off a jump pad.
    pub jump_launched: bool,
    /// True on the tick this player used their class ability.
    pub used_ability: bool,
    /// True on the tick this player used a support-class ability (cleanses debuffs).
    pub used_support_ability: bool,
    /// True while the glider class ability is held and the player is falling.
    pub gliding: bool,
}

/// World inputs shared by one enemy tick.
#[derive(Debug, Clone)]
pub struct TickContext<'a> {
    /// All players in the level.
    pub players: &'a [PlayerSample<'a>],
    /// True during the collapse phase (players rushing the exit).
    pub collapse_phase: bool,
    /// Minimum corner of the playable map (used for leap / re-dorm placement).
    pub map_min: Vec3,
    /// Maximum corner of the playable map.
    pub map_max: Vec3,
}

/// Nearest player to `from` (horizontal distance) that satisfies `pred`.
pub(crate) fn nearest_player_where<'a, F: Fn(&PlayerSample<'a>) -> bool>(
    players: &'a [PlayerSample<'a>],
    from: &Vec3,
    pred: F,
) -> Option<&'a PlayerSample<'a>> {
    let mut best: Option<&PlayerSample<'a>> = None;
    let mut best_d = f32::INFINITY;
    for p in players {
        if !pred(p) {
            continue;
        }
        let d = horizontal_distance(from, &p.feet);
        if d < best_d {
            best_d = d;
            best = Some(p);
        }
    }
    best
}

/// Deterministic xorshift32; `state` must not be zero (zero is remapped).
pub(crate) fn xorshift32(state: &mut u32) -> u32 {
    let mut x = *state;
    if x == 0 {
        x = 0x9E37_79B9;
    }
    x ^= x << 13;
    x ^= x >> 17;
    x ^= x << 5;
    *state = x;
    x
}

/// Next deterministic value in `[0, 1)`.
pub(crate) fn unit_f32(state: &mut u32) -> f32 {
    (xorshift32(state) % 1024) as f32 / 1024.0
}

/// Stable seed derived from a spawn position (the step API has no enemy id).
pub(crate) fn seed_from_pos(pos: &Vec3) -> u32 {
    let s = pos.x.to_bits() ^ pos.z.to_bits().rotate_left(13) ^ 0x9E37_79B9;
    if s == 0 {
        0xA53A_9D1F
    } else {
        s
    }
}

//! Blight — the squelching infection debuff (cue tier 3, min level 8, cap 1).
//!
//! Ambient enemy: no per-attack wind-up. The telegraph is the infected ground
//! itself — a discoloured aura on Blight that spreads one tile per beat and
//! never disappears, so the level is readable without watching Blight.
//!
//! Punish: crossing infected ground (or being hit by Blight) INFECTS: the
//! victim is slowed to [`INFECTION_SPEED_MUL`] and the infection pauses Rewind's
//! and Kite's signature timers while it lasts (registry synergy hook
//! `infection_pauses_timer`).
//!
//! Cleanse: a jump-pad launch or a support-class ability. Glider is immune
//! (airborne, per registry); a grappler using their ability is clean while
//! grappling above it (registry relation `grappler: counters`).

use protocol::Vec3;

use super::{nearest_player_where, TickContext, SIM_DT};
use crate::rules::{horizontal_distance, move_toward};

/// Approach speed while stalking a player (u/s).
pub const BLIGHT_APPROACH_SPEED: f32 = 2.6;
/// Radius of the infected aura at spawn (units; one tile = 1.0 unit).
pub const BLIGHT_INFECT_RADIUS_START: f32 = 2.0;
/// Maximum radius the infected aura spreads to (units).
pub const BLIGHT_INFECT_RADIUS_MAX: f32 = 6.0;
/// Ticks per tile of aura spread (one beat, 1.0 s at 60 Hz).
pub const BLIGHT_INFECT_SPREAD_TICKS: u32 = 60;
/// Infection duration once applied (6.0 s at 60 Hz).
pub const BLIGHT_INFECTION_TICKS: u32 = 360;
/// Movement multiplier applied to infected players.
pub const INFECTION_SPEED_MUL: f32 = 0.6;

/// Blight phases. Tier 3: ambient, no wind-up.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum BlightPhase {
    /// Squelching approach; the aura does the rest.
    Stalk,
}

/// Blight brain state.
#[derive(Debug, Clone, PartialEq)]
pub struct BlightBrain {
    /// Current phase.
    pub phase: BlightPhase,
    /// Radius of the infected ground (units). The persistent telegraph.
    pub aura_radius: f32,
    /// Ticks until the aura spreads one tile.
    pub spread_in: u32,
    /// Infected players: `(player_id, ticks remaining)`.
    pub infected: Vec<(u32, u32)>,
}

impl Default for BlightBrain {
    fn default() -> Self {
        Self {
            phase: BlightPhase::Stalk,
            aura_radius: BLIGHT_INFECT_RADIUS_START,
            spread_in: BLIGHT_INFECT_SPREAD_TICKS,
            infected: Vec::new(),
        }
    }
}

/// What happened this tick.
#[derive(Debug, Clone, PartialEq, Default)]
pub struct BlightStep {
    /// Players newly infected this tick.
    pub newly_infected: Vec<u32>,
    /// Players cleansed this tick (jump-pad launch or support ability).
    pub cleansed: Vec<u32>,
}

/// True while Blight's infection is on the player.
///
/// Rewind's rewind clock and Kite's tether clock must not decrement while this
/// is true for the afflicted player (registry synergy hook
/// `infection_pauses_timer`). The predicate is exported so batch A can gate
/// their timer decrement without duplicating the rule.
pub fn infection_pauses_timer(infected_ticks: u32) -> bool {
    infected_ticks > 0
}

impl BlightBrain {
    /// True while the player is infected.
    pub fn is_infected(&self, player_id: u32) -> bool {
        self.infected.iter().any(|&(id, t)| id == player_id && t > 0)
    }

    /// Infection ticks remaining for the player (0 = clean).
    pub fn infected_ticks(&self, player_id: u32) -> u32 {
        self.infected
            .iter()
            .find(|&&(id, _)| id == player_id)
            .map(|&(_, t)| t)
            .unwrap_or(0)
    }

    /// Movement multiplier for the player ([`INFECTION_SPEED_MUL`] while infected).
    pub fn speed_mul_for(&self, player_id: u32) -> f32 {
        if self.is_infected(player_id) {
            INFECTION_SPEED_MUL
        } else {
            1.0
        }
    }
}

/// Apply or refresh the infection; returns true if freshly applied.
fn apply_infection(brain: &mut BlightBrain, id: u32) -> bool {
    if let Some(entry) = brain.infected.iter_mut().find(|e| e.0 == id) {
        entry.1 = BLIGHT_INFECTION_TICKS;
        false
    } else {
        brain.infected.push((id, BLIGHT_INFECTION_TICKS));
        true
    }
}

/// Remove the infection; returns true if the player was infected.
fn remove_infection(brain: &mut BlightBrain, id: u32) -> bool {
    let before = brain.infected.len();
    brain.infected.retain(|&(eid, _)| eid != id);
    before != brain.infected.len()
}

/// Advance Blight by one tick.
pub fn step_blight(
    enemy_pos: &mut Vec3,
    brain: &mut BlightBrain,
    ctx: &TickContext,
) -> BlightStep {
    let mut out = BlightStep::default();

    // Infected ground spreads one tile per beat, up to the cap.
    if brain.spread_in > 0 {
        brain.spread_in -= 1;
    } else if brain.aura_radius < BLIGHT_INFECT_RADIUS_MAX {
        brain.aura_radius = (brain.aura_radius + 1.0).min(BLIGHT_INFECT_RADIUS_MAX);
        brain.spread_in = BLIGHT_INFECT_SPREAD_TICKS;
    }

    // Tick existing infections down BEFORE applying new ones.
    for entry in &mut brain.infected {
        entry.1 = entry.1.saturating_sub(1);
    }
    brain.infected.retain(|&(_, t)| t > 0);

    // Cleanse hooks: jump-pad launch or a support-class ability.
    for p in ctx.players {
        if p.jump_launched || p.used_support_ability {
            if remove_infection(brain, p.id) {
                out.cleansed.push(p.id);
            }
        }
    }

    // Crossing the infected ground applies it.
    // Glider is immune (airborne). A grappler mid-grapple is over the infection.
    for p in ctx.players {
        if p.class == "glider" {
            continue;
        }
        if p.class == "grappler" && p.used_ability {
            continue;
        }
        if horizontal_distance(enemy_pos, &p.feet) <= brain.aura_radius {
            if apply_infection(brain, p.id) {
                out.newly_infected.push(p.id);
            }
        }
    }

    // Squelching approach toward the nearest player.
    if let Some(target) = nearest_player_where(ctx.players, enemy_pos, |_| true) {
        let goal = Vec3::new(target.feet.x, enemy_pos.y, target.feet.z);
        move_toward(enemy_pos, &goal, BLIGHT_APPROACH_SPEED * SIM_DT);
    }

    out
}

//! Metronome — the counting beacon (cue tier 1, min level 10, cap 1).
//!
//! A beacon shows a number that ticks up. The marked player must use their
//! class ability exactly on their number or be struck. Numbers are assigned
//! deterministically per cycle (`1 + cycle % 4`), there is no RNG.
//!
//! Telegraph: the arm flashes for 0.5 s ([`METRONOME_ARM_FLASH_TICKS`]) before
//! each swing resolves — squarely inside tier 1's 0.35–0.95 s band. The window
//! during which the ability must be used IS the flash.
//!
//! Relations (registry): `bulwark: immune` (the ward ignores the beat — never
//! struck); `glider: counters` (gliding crosses the swing — a glider that is
//! gliding at any point during the window is not struck even without ability
//! use). No class has an `ignores` relation against Metronome in the frozen
//! registry, so none is implemented here.

use protocol::Vec3;

use super::{nearest_player_where, TickContext};

/// Ticks per number on the beacon (0.5 s at 60 Hz).
pub const METRONOME_BEAT_TICKS: u32 = 30;
/// Arm-flash telegraph / ability window, in ticks (0.5 s at 60 Hz).
pub const METRONOME_ARM_FLASH_TICKS: u32 = 30;
/// Rest between cycles, in ticks (1.0 s at 60 Hz).
pub const METRONOME_REST_TICKS: u32 = 60;
/// Highest number a cycle can target; cycles through `1..=METRONOME_NUMBER_MAX`.
pub const METRONOME_NUMBER_MAX: u32 = 4;

/// Metronome phases.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum MetronomePhase {
    /// Resting between cycles; marks the nearest player when the rest ends.
    Rest,
    /// The beacon is ticking up toward the target number.
    Count,
    /// The arm flashes; the ability window is open; the strike resolves at the end.
    Flash,
}

/// Metronome brain state. The beacon is the enemy's own position.
#[derive(Debug, Clone, PartialEq)]
pub struct MetronomeBrain {
    /// Current phase.
    pub phase: MetronomePhase,
    /// Phase countdown in ticks (Flash duration when in `Flash`).
    pub timer: u32,
    /// The player marked this cycle.
    pub marked: Option<u32>,
    /// Number currently shown on the beacon.
    pub display: u32,
    /// The number on which the marked player must use their class ability.
    pub target_number: u32,
    /// Ticks until the number advances.
    pub count_in: u32,
    /// True once ability use was seen during the Flash window.
    pub ability_seen: bool,
    /// True once gliding was seen during the Flash window (glider counter).
    pub gliding_seen: bool,
    /// Cycle counter; drives the deterministic `target_number` sequence.
    pub cycle: u32,
}

impl Default for MetronomeBrain {
    fn default() -> Self {
        Self {
            phase: MetronomePhase::Rest,
            timer: 0,
            marked: None,
            display: 1,
            target_number: 1,
            count_in: METRONOME_BEAT_TICKS,
            ability_seen: false,
            gliding_seen: false,
            cycle: 0,
        }
    }
}

/// What happened this tick.
#[derive(Debug, Clone, PartialEq, Default)]
pub struct MetronomeStep {
    /// True while the arm-flash telegraph is up (VFX hook; also the test hook).
    pub telegraphing: bool,
    /// The number currently on the beacon (HUD hook).
    pub display: u32,
    /// The player to strike this tick; the caller applies damage/debuff.
    /// `None` when the ability was used on the number, the class is immune
    /// (bulwark), or the glider crossed it.
    pub struck: Option<u32>,
}

/// Advance Metronome by one tick. The beacon does not move.
pub fn step_metronome(
    enemy_pos: &Vec3,
    brain: &mut MetronomeBrain,
    ctx: &TickContext,
) -> MetronomeStep {
    let mut out = MetronomeStep::default();
    out.display = brain.display;

    match brain.phase {
        MetronomePhase::Rest => {
            if brain.timer > 0 {
                brain.timer -= 1;
                return out;
            }
            if let Some(p) = nearest_player_where(ctx.players, enemy_pos, |_| true) {
                brain.marked = Some(p.id);
                brain.target_number = 1 + (brain.cycle % METRONOME_NUMBER_MAX);
                brain.display = 1;
                brain.ability_seen = false;
                brain.gliding_seen = false;
                out.display = brain.display;
                if brain.target_number == 1 {
                    brain.phase = MetronomePhase::Flash;
                    brain.timer = METRONOME_ARM_FLASH_TICKS;
                } else {
                    brain.phase = MetronomePhase::Count;
                    brain.count_in = METRONOME_BEAT_TICKS;
                }
            } else {
                brain.timer = METRONOME_BEAT_TICKS;
            }
        }
        MetronomePhase::Count => {
            if brain.count_in > 0 {
                brain.count_in -= 1;
            }
            if brain.count_in == 0 {
                brain.display += 1;
                if brain.display == brain.target_number {
                    brain.phase = MetronomePhase::Flash;
                    brain.timer = METRONOME_ARM_FLASH_TICKS;
                } else {
                    brain.count_in = METRONOME_BEAT_TICKS;
                }
                out.display = brain.display;
            }
        }
        MetronomePhase::Flash => {
            out.telegraphing = true;
            if let Some(id) = brain.marked {
                if let Some(p) = ctx.players.iter().find(|p| p.id == id) {
                    if p.used_ability {
                        brain.ability_seen = true;
                    }
                    if p.class == "glider" && p.gliding {
                        brain.gliding_seen = true;
                    }
                }
            }
            if brain.timer > 0 {
                brain.timer -= 1;
            }
            if brain.timer == 0 {
                let mut struck = None;
                if let Some(id) = brain.marked {
                    if let Some(p) = ctx.players.iter().find(|p| p.id == id) {
                        let immune = p.class == "bulwark";
                        let answered =
                            brain.ability_seen || (p.class == "glider" && brain.gliding_seen);
                        if !immune && !answered {
                            struck = Some(id);
                        }
                    }
                }
                out.struck = struck;
                brain.phase = MetronomePhase::Rest;
                brain.timer = METRONOME_REST_TICKS;
                brain.cycle += 1;
                brain.marked = None;
            }
        }
    }
    out
}

//! Overlook — the teleporting marker and one-way laser (cue tier 2,
//! min level 12, cap 1).
//!
//! Flanks around a marked player, teleporting to random offsets and leaving a
//! glowing mark at each spot, then locks a 1.6 s sight line onto the marked
//! player (tier 2: decal-class telegraph, exact hit line, ≥ 1.0 s), then fires a
//! slow tracking laser that ONLY the marked player can see and be harmed by.
//!
//! Teammates can draw the laser's path during the collapse phase: the brain
//! records the beam tip every tick (up to [`OVERLOOK_PATH_MAX`] entries) and
//! [`OverlookBrain::drawn_path`] returns it only when asked with
//! `collapse_phase == true`.
//!
//! Glider is never attracted (registry relation `glider: ignores`): gliders are
//! never marked, so the laser never targets them. Striker counters by closing
//! and killing it (kill plumbing is out of scope for the brain; noted).

use protocol::Vec3;

use super::{nearest_player_where, unit_f32, TickContext};
use crate::rules::flat_direction;

/// Ticks between teleports while flanking (0.75 s at 60 Hz).
pub const OVERLOOK_FLANK_INTERVAL_TICKS: u32 = 45;
/// Teleports before the sight line locks.
pub const OVERLOOK_FLANKS: u32 = 4;
/// Teleport offset radius around the marked player (units).
pub const OVERLOOK_TELEPORT_RADIUS: f32 = 6.0;
/// Sight-line telegraph duration, in ticks (1.6 s at 60 Hz; tier 2: ≥ 1.0 s).
pub const OVERLOOK_SIGHT_LINE_TICKS: u32 = 96;
/// Laser duration once fired, in ticks (3.0 s at 60 Hz).
pub const OVERLOOK_LASER_TICKS: u32 = 180;
/// Laser length (units).
pub const OVERLOOK_LASER_RANGE: f32 = 30.0;
/// Laser beam radius — the exact hit volume shown during the telegraph.
pub const OVERLOOK_LASER_RADIUS: f32 = 0.6;
/// Fraction of the way the beam turns toward the marked player each tick.
pub const OVERLOOK_LASER_STEER: f32 = 0.10;
/// Cooldown after the laser ends, in ticks (2.0 s at 60 Hz).
pub const OVERLOOK_COOLDOWN_TICKS: u32 = 120;
/// Cap on stored glowing marks (oldest dropped).
pub const OVERLOOK_MARKS_MAX: usize = 8;
/// Cap on recorded laser path points (3.0 s at 60 Hz).
pub const OVERLOOK_PATH_MAX: usize = 180;

/// Overlook phases.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum OverlookPhase {
    /// Teleports around the marked player, leaving glowing marks.
    Flank,
    /// 1.6 s sight line to the marked player; the glowing mark flashes.
    Mark,
    /// Slow tracking laser; only the marked player is harmed.
    Fire,
    /// Cooldown before the next hunt.
    Cooldown,
}

/// Overlook brain state.
#[derive(Debug, Clone, PartialEq)]
pub struct OverlookBrain {
    /// Current phase.
    pub phase: OverlookPhase,
    /// Phase countdown in ticks.
    pub timer: u32,
    /// The player the laser is for (and only for).
    pub marked: Option<u32>,
    /// Glowing marks left at each teleport spot (newest last).
    pub marks: Vec<Vec3>,
    /// Unit horizontal direction of the current sight line / laser.
    pub sight_dir: Vec3,
    /// Locked laser direction (unit, horizontal); steered during `Fire`.
    pub laser_dir: Vec3,
    /// Recorded beam tips; exposed to teammates only during collapse.
    pub laser_path: Vec<Vec3>,
    /// Random seed for teleport placement.
    pub seed: u32,
    /// Teleports left in this hunt.
    pub flanks_left: u32,
    /// Ticks until the next teleport.
    pub flank_in: u32,
}

impl Default for OverlookBrain {
    fn default() -> Self {
        Self {
            phase: OverlookPhase::Flank,
            timer: 0,
            marked: None,
            marks: Vec::new(),
            sight_dir: Vec3::new(0.0, 0.0, 1.0),
            laser_dir: Vec3::new(1.0, 0.0, 0.0),
            laser_path: Vec::new(),
            seed: 0,
            flanks_left: OVERLOOK_FLANKS,
            flank_in: 0,
        }
    }
}

impl OverlookBrain {
    /// The laser path for teammates to draw. Empty outside the collapse phase
    /// or before the laser has fired.
    pub fn drawn_path(&self, collapse_phase: bool) -> &[Vec3] {
        if collapse_phase {
            &self.laser_path
        } else {
            &[]
        }
    }
}

/// What happened this tick.
#[derive(Debug, Clone, PartialEq, Default)]
pub struct OverlookStep {
    /// True while the sight line telegraph is up (VFX/test hook).
    pub telegraphing: bool,
    /// True when the laser crossed the MARKED player this tick.
    pub marked_hit: bool,
    /// True when the laser path may be drawn by teammates (collapse phase).
    pub path_public: bool,
}

/// Random point in a circle around the player, rejection-sampled (no trig).
fn teleport_offset(brain: &mut OverlookBrain) -> Vec3 {
    for _ in 0..8 {
        let x = (unit_f32(&mut brain.seed) * 2.0 - 1.0) * OVERLOOK_TELEPORT_RADIUS;
        let z = (unit_f32(&mut brain.seed) * 2.0 - 1.0) * OVERLOOK_TELEPORT_RADIUS;
        if x * x + z * z <= OVERLOOK_TELEPORT_RADIUS * OVERLOOK_TELEPORT_RADIUS {
            return Vec3::new(x, 0.0, z);
        }
    }
    Vec3::new(OVERLOOK_TELEPORT_RADIUS, 0.0, 0.0)
}

/// Normalised horizontal direction; falls back to +X.
fn normalize_flat(d: Vec3) -> Vec3 {
    let d2 = d.x * d.x + d.z * d.z;
    if d2 <= 1e-9 {
        return Vec3::new(1.0, 0.0, 0.0);
    }
    let inv = 1.0 / d2.sqrt();
    Vec3::new(d.x * inv, 0.0, d.z * inv)
}

/// Distance from `p` to segment `a`..`b` (sqrt only).
fn point_segment_distance(p: &Vec3, a: &Vec3, b: &Vec3) -> f32 {
    let abx = b.x - a.x;
    let aby = b.y - a.y;
    let abz = b.z - a.z;
    let apx = p.x - a.x;
    let apy = p.y - a.y;
    let apz = p.z - a.z;
    let ab_len2 = abx * abx + aby * aby + abz * abz;
    let t = if ab_len2 > 1e-9 {
        ((apx * abx + apy * aby + apz * abz) / ab_len2).clamp(0.0, 1.0)
    } else {
        0.0
    };
    let cx = a.x + abx * t;
    let cy = a.y + aby * t;
    let cz = a.z + abz * t;
    let dx = p.x - cx;
    let dy = p.y - cy;
    let dz = p.z - cz;
    (dx * dx + dy * dy + dz * dz).sqrt()
}

/// Seed the RNG from the spawn position once (the step API has no enemy id).
fn seed_if_needed(enemy_pos: &Vec3, brain: &mut OverlookBrain) {
    if brain.seed == 0 {
        brain.seed = super::seed_from_pos(enemy_pos);
    }
}

/// Advance Overlook by one tick.
pub fn step_overlook(
    enemy_pos: &mut Vec3,
    brain: &mut OverlookBrain,
    ctx: &TickContext,
) -> OverlookStep {
    seed_if_needed(enemy_pos, brain);
    let mut out = OverlookStep::default();

    match brain.phase {
        OverlookPhase::Flank => {
            if brain.marked.is_none() {
                // Gliders are never attracted and never marked.
                brain.marked =
                    nearest_player_where(ctx.players, enemy_pos, |p| p.class != "glider")
                        .map(|p| p.id);
                if brain.marked.is_none() {
                    brain.phase = OverlookPhase::Cooldown;
                    brain.timer = 30;
                    return out;
                }
                brain.flanks_left = OVERLOOK_FLANKS;
                brain.flank_in = 0;
            }
            if brain.flank_in > 0 {
                brain.flank_in -= 1;
                return out;
            }
            let prey = ctx
                .players
                .iter()
                .find(|p| Some(p.id) == brain.marked)
                .map(|p| p.feet);
            let Some(prey) = prey else {
                brain.marked = None;
                brain.phase = OverlookPhase::Cooldown;
                brain.timer = 30;
                return out;
            };
            let off = teleport_offset(brain);
            enemy_pos.x = prey.x + off.x;
            enemy_pos.z = prey.z + off.z;
            brain.marks.push(*enemy_pos);
            if brain.marks.len() > OVERLOOK_MARKS_MAX {
                brain.marks.remove(0);
            }
            brain.flanks_left = brain.flanks_left.saturating_sub(1);
            if brain.flanks_left == 0 {
                brain.phase = OverlookPhase::Mark;
                brain.timer = OVERLOOK_SIGHT_LINE_TICKS;
            } else {
                brain.flank_in = OVERLOOK_FLANK_INTERVAL_TICKS;
            }
        }
        OverlookPhase::Mark => {
            out.telegraphing = true;
            if let Some(prey) = ctx
                .players
                .iter()
                .find(|p| Some(p.id) == brain.marked)
                .map(|p| p.feet)
            {
                brain.sight_dir = flat_direction(enemy_pos, &prey);
            }
            if brain.timer > 0 {
                brain.timer -= 1;
            }
            if brain.timer == 0 {
                brain.phase = OverlookPhase::Fire;
                brain.timer = OVERLOOK_LASER_TICKS;
                brain.laser_dir = brain.sight_dir;
                brain.laser_path.clear();
            }
        }
        OverlookPhase::Fire => {
            let prey_feet = ctx
                .players
                .iter()
                .find(|p| Some(p.id) == brain.marked)
                .map(|p| p.feet);
            if let Some(prey) = prey_feet {
                let desired = flat_direction(enemy_pos, &prey);
                brain.laser_dir = normalize_flat(Vec3::new(
                    brain.laser_dir.x + (desired.x - brain.laser_dir.x) * OVERLOOK_LASER_STEER,
                    0.0,
                    brain.laser_dir.z + (desired.z - brain.laser_dir.z) * OVERLOOK_LASER_STEER,
                ));
                // ONLY the marked player can see and be harmed by the laser.
                let beam_end = Vec3::new(
                    enemy_pos.x + brain.laser_dir.x * OVERLOOK_LASER_RANGE,
                    enemy_pos.y,
                    enemy_pos.z + brain.laser_dir.z * OVERLOOK_LASER_RANGE,
                );
                let centre = Vec3::new(prey.x, prey.y + 0.9, prey.z);
                if point_segment_distance(&centre, enemy_pos, &beam_end) <= OVERLOOK_LASER_RADIUS {
                    out.marked_hit = true;
                }
            }
            let tip = Vec3::new(
                enemy_pos.x + brain.laser_dir.x * OVERLOOK_LASER_RANGE,
                enemy_pos.y,
                enemy_pos.z + brain.laser_dir.z * OVERLOOK_LASER_RANGE,
            );
            brain.laser_path.push(tip);
            if brain.laser_path.len() > OVERLOOK_PATH_MAX {
                brain.laser_path.remove(0);
            }
            out.path_public = ctx.collapse_phase;
            if brain.timer > 0 {
                brain.timer -= 1;
            }
            if brain.timer == 0 {
                brain.phase = OverlookPhase::Cooldown;
                brain.timer = OVERLOOK_COOLDOWN_TICKS;
                brain.marked = None;
                brain.flanks_left = OVERLOOK_FLANKS;
            }
        }
        OverlookPhase::Cooldown => {
            if brain.timer > 0 {
                brain.timer -= 1;
            } else {
                brain.phase = OverlookPhase::Flank;
                brain.flank_in = 0;
            }
        }
    }
    out
}

//! Chorus — the gated choir (cue tier 3, min level 10, cap 1).
//!
//! Only a candidate once the run holds at least two `small` permanents at
//! level ≥ 10 (registry synergy hook `chorus_needs_small_pair`); the pool filter
//! is data-driven in the server, and [`chorus_candidate_allowed`] is the same
//! rule for the sim side.
//!
//! Ambient, tier 3: the singers are audible long before anything can hurt you
//! ([`ChorusBrain::murmur_ticks`]); there is no per-attack telegraph and no
//! unavoidable instant hit. Instruments spawn as special gifts on a cadence;
//! each one collected freezes the choir and extends its timer. When the timer
//! runs out, Chorus moves and hunts — one-way.
//!
//! Counter: Striker answers the choir (singer cleanup is out of scope for the
//! brain — no singer actors are modelled here; noted). Bulwark is immune
//! (registry); likewise the singer shove is out of scope.

use protocol::Vec3;

use super::{nearest_player_where, TickContext, SIM_DT};
use crate::rules::move_toward;

/// Base ticks until the choir hunts (45 s at 60 Hz).
pub const CHORUS_TIMER_TICKS: u32 = 2700;
/// Time granted per collected instrument (20 s at 60 Hz).
pub const CHORUS_INSTRUMENT_TIME_TICKS: u32 = 1200;
/// Ticks the choir is frozen by a collected instrument (2.0 s at 60 Hz).
pub const CHORUS_INSTRUMENT_FREEZE_TICKS: u32 = 120;
/// Ticks between instrument gift requests (8.0 s at 60 Hz).
pub const CHORUS_INSTRUMENT_INTERVAL_TICKS: u32 = 480;
/// Maximum uncollected instrument requests outstanding.
pub const CHORUS_INSTRUMENTS_MAX_OUTSTANDING: u32 = 2;
/// Speed while hunting once the timer expires (u/s).
pub const CHORUS_HUNT_SPEED: f32 = 4.5;
/// Small permanents required before Chorus may be offered.
pub const CHORUS_SMALL_PAIR_REQUIRED: u32 = 2;
/// Minimum level before Chorus may be offered (mirrors the registry entry).
pub const CHORUS_MIN_LEVEL: u32 = 10;

/// Pool gate for the `chorus_needs_small_pair` hook plus the level gate.
pub fn chorus_candidate_allowed(small_permanents: u32, level: u32) -> bool {
    small_permanents >= CHORUS_SMALL_PAIR_REQUIRED && level >= CHORUS_MIN_LEVEL
}

/// Chorus phases.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ChorusPhase {
    /// Singers audible; the timer ticks down; instruments extend it.
    Hush,
    /// The timer expired; the choir moves and hunts. One-way.
    Hunt,
}

/// Chorus brain state.
#[derive(Debug, Clone, PartialEq)]
pub struct ChorusBrain {
    /// Current phase.
    pub phase: ChorusPhase,
    /// Ticks until the choir hunts; extended by collected instruments.
    pub timer: u32,
    /// Ticks the choir is frozen by a collected instrument. While frozen the
    /// timer does not advance.
    pub frozen_ticks: u32,
    /// Instrument gift requests not yet collected.
    pub instruments_outstanding: u32,
    /// Instruments collected in total.
    pub instruments_collected: u32,
    /// Ticks until the next instrument request.
    pub instrument_in: u32,
    /// Ambient murmur pulse counter; the persistent tier-3 cue. VFX only.
    pub murmur_ticks: u32,
}

impl Default for ChorusBrain {
    fn default() -> Self {
        Self {
            phase: ChorusPhase::Hush,
            timer: CHORUS_TIMER_TICKS,
            frozen_ticks: 0,
            instruments_outstanding: 0,
            instruments_collected: 0,
            instrument_in: CHORUS_INSTRUMENT_INTERVAL_TICKS,
            murmur_ticks: 0,
        }
    }
}

/// What happened this tick.
#[derive(Debug, Clone, PartialEq, Default)]
pub struct ChorusStep {
    /// Instrument gifts the server should spawn this tick.
    pub instruments_to_spawn: u32,
    /// True while a collected instrument freezes the choir.
    pub frozen: bool,
    /// True on the tick the choir starts hunting.
    pub began_hunting: bool,
    /// True while the choir is hunting.
    pub hunting: bool,
}

/// Advance Chorus by one tick. `instruments_collected` is how many instrument
/// gifts were collected by any player this tick.
pub fn step_chorus(
    enemy_pos: &mut Vec3,
    brain: &mut ChorusBrain,
    ctx: &TickContext,
    instruments_collected: u32,
) -> ChorusStep {
    let mut out = ChorusStep::default();
    brain.murmur_ticks = brain.murmur_ticks.saturating_add(1);

    for _ in 0..instruments_collected {
        brain.instruments_outstanding = brain.instruments_outstanding.saturating_sub(1);
        brain.timer = brain.timer.saturating_add(CHORUS_INSTRUMENT_TIME_TICKS);
        brain.frozen_ticks = CHORUS_INSTRUMENT_FREEZE_TICKS;
        brain.instruments_collected = brain.instruments_collected.saturating_add(1);
    }

    match brain.phase {
        ChorusPhase::Hush => {
            if brain.frozen_ticks > 0 {
                brain.frozen_ticks -= 1;
                out.frozen = true;
            } else if brain.timer > 0 {
                brain.timer -= 1;
            }
            if brain.instrument_in > 0 {
                brain.instrument_in -= 1;
            } else if brain.instruments_outstanding < CHORUS_INSTRUMENTS_MAX_OUTSTANDING {
                out.instruments_to_spawn = 1;
                brain.instruments_outstanding += 1;
                brain.instrument_in = CHORUS_INSTRUMENT_INTERVAL_TICKS;
            }
            if brain.timer == 0 {
                brain.phase = ChorusPhase::Hunt;
                out.began_hunting = true;
                out.hunting = true;
            }
        }
        ChorusPhase::Hunt => {
            out.hunting = true;
            if let Some(p) = nearest_player_where(ctx.players, enemy_pos, |_| true) {
                let goal = Vec3::new(p.feet.x, enemy_pos.y, p.feet.z);
                move_toward(enemy_pos, &goal, CHORUS_HUNT_SPEED * SIM_DT);
            }
        }
    }
    out
}

//! Coil — the winding shockwave leap (cue tier 1, min level 4, cap 2).
//! Deliberate pair with Peal via the OVERTUNED debuff.
//!
//! Coils in place, winds up for 0.8 s ([`COIL_WIND_TICKS`]) while a player
//! stays near, then releases a radial shockwave that inflicts OVERTUNED (the
//! debuff that makes Peal's final ring lethal instead of a debuff — registry
//! synergy hook `peal_lethal_when_overtuned`). Afterwards it recovers and
//! leaps to a random map spot (deterministic RNG — see NOTES).
//!
//! The longer a player stays near while it winds, the tighter it coils: every
//! [`COIL_TIGHTEN_INTERVAL_TICKS`] of continuous proximity shaves
//! [`COIL_TIGHTEN_BONUS_TICKS`] off the remaining wind. The wind never drops
//! below [`COIL_MIN_WIND_TICKS`], so the tier-1 wind-up promise holds even
//! under tightening.
//!
//! Glider is never attracted (registry relation `glider: ignores`): gliders do
//! not trigger the wind-up and are not chased. Striker dashes out (escape by
//! leaving the radius before the release).

use protocol::Vec3;

use super::{nearest_player_where, seed_from_pos, unit_f32, TickContext, SIM_DT};
use crate::rules::{horizontal_distance, move_toward};

/// Aggro radius for the slow drift toward a player (units).
pub const COIL_AGGRO_RADIUS: f32 = 18.0;
/// Drift speed while coiled (u/s).
pub const COIL_DRIFT_SPEED: f32 = 1.8;
/// Radius inside which staying near starts the wind-up (units).
pub const COIL_TAUNT_RADIUS: f32 = 4.0;
/// Wind-up duration, in ticks (0.8 s at 60 Hz; tier 1).
pub const COIL_WIND_TICKS: u32 = 48;
/// Hard floor for the wind-up under tightening pressure (0.35 s; tier-1 minimum).
pub const COIL_MIN_WIND_TICKS: u32 = 21;
/// Continuous-proximity cadence for tightening (20 ticks ≈ 0.33 s).
///
/// A full "every second" cadence cannot fit inside the frozen 0.8 s tier-1
/// wind; the cadence is tightened so the mechanic exists at all. The tier-1
/// floor is preserved regardless.
pub const COIL_TIGHTEN_INTERVAL_TICKS: u32 = 20;
/// Ticks removed from the remaining wind per tighten step.
pub const COIL_TIGHTEN_BONUS_TICKS: u32 = 8;
/// Shockwave radius (units).
pub const COIL_SHOCKWAVE_RADIUS: f32 = 5.0;
/// Recovery pause before the leap (1.0 s at 60 Hz).
pub const COIL_RECOVER_TICKS: u32 = 60;
/// Airborne leap duration (0.8 s at 60 Hz).
pub const COIL_LEAP_TICKS: u32 = 48;
/// Leap arc apex height above the lerp line (units).
pub const COIL_LEAP_HEIGHT: f32 = 4.0;
/// Cooldown between wind-ups after a release (2.5 s at 60 Hz).
pub const COIL_RELEASE_COOLDOWN_TICKS: u32 = 150;

/// Coil phases.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CoilPhase {
    /// Coiled; slow drift toward the nearest attracted player.
    Coiled,
    /// Winding up; telegraph for the release.
    Wind,
    /// One tick: the shockwave resolves and OVERTUNED is inflicted.
    Release,
    /// Recovering before the leap.
    Recover,
    /// Airborne arc to the chosen map spot.
    Leap,
}

/// Coil brain state.
#[derive(Debug, Clone, PartialEq)]
pub struct CoilBrain {
    /// Current phase.
    pub phase: CoilPhase,
    /// Phase countdown in ticks.
    pub timer: u32,
    /// Continuous ticks a player has stayed in the taunt radius.
    pub proximity_ticks: u32,
    /// Ticks until the next tighten step.
    pub tighten_in: u32,
    /// Cooldown between wind-ups.
    pub cooldown_in: u32,
    /// Leap arc start (current position when the leap began).
    pub leap_from: Vec3,
    /// Leap arc end (random map spot).
    pub leap_to: Vec3,
    /// Random seed for leap placement.
    pub seed: u32,
}

impl Default for CoilBrain {
    fn default() -> Self {
        Self {
            phase: CoilPhase::Coiled,
            timer: 0,
            proximity_ticks: 0,
            tighten_in: COIL_TIGHTEN_INTERVAL_TICKS,
            cooldown_in: 0,
            leap_from: Vec3::new(0.0, 0.0, 0.0),
            leap_to: Vec3::new(0.0, 0.0, 0.0),
            seed: 0,

    }

/// What happened this tick.
#[derive(Debug, Clone, PartialEq, Default)]
pub struct CoilStep {
    /// True while winding (VFX/test hook).
    pub telegraphing: bool,
    /// Players caught in the shockwave this tick; the caller applies OVERTUNED
    /// to each via `enemies::apply_overtuned`.
    pub overtuned_ids: Vec<u32>,
    /// True while airborne on the leap arc.
    pub leaping: bool,
}

/// Advance Coil by one tick.
pub fn step_coil(
    enemy_pos: &mut Vec3,
    brain: &mut CoilBrain,
    ctx: &TickContext,
) -> CoilStep {
    if brain.seed == 0 {
        brain.seed = seed_from_pos(enemy_pos);
    }
    let mut out = CoilStep::default();

    match brain.phase {
        CoilPhase::Coiled => {
            if brain.cooldown_in > 0 {
                brain.cooldown_in -= 1;
            }
            // Gliders are never attracted: no wind-up, no chase.
            let target =
                nearest_player_where(ctx.players, enemy_pos, |p| p.class != "glider");
            match target {
                Some(p) => {
                    let d = horizontal_distance(enemy_pos, &p.feet);
                    if d <= COIL_TAUNT_RADIUS && brain.cooldown_in == 0 {
                        brain.phase = CoilPhase::Wind;
                        brain.timer = COIL_WIND_TICKS;
                        brain.proximity_ticks = 0;
                        brain.tighten_in = COIL_TIGHTEN_INTERVAL_TICKS;
                    } else if d <= COIL_AGGRO_RADIUS {
                        let goal = Vec3::new(p.feet.x, enemy_pos.y, p.feet.z);
                        move_toward(enemy_pos, &goal, COIL_DRIFT_SPEED * SIM_DT);
                    }
                }
                None => {}
            }
        }
        CoilPhase::Wind => {
            out.telegraphing = true;
            let someone_near = ctx.players.iter().any(|p| {
                p.class != "glider" && horizontal_distance(enemy_pos, &p.feet) <= COIL_TAUNT_RADIUS
            });
            if someone_near {
                brain.proximity_ticks += 1;
                if brain.tighten_in > 0 {
                    brain.tighten_in -= 1;
                }
                if brain.tighten_in == 0 {
                    brain.timer = brain
                        .timer
                        .saturating_sub(COIL_TIGHTEN_BONUS_TICKS)
                        .max(COIL_MIN_WIND_TICKS);
                    brain.tighten_in = COIL_TIGHTEN_INTERVAL_TICKS;
                }
            } else {
                brain.proximity_ticks = 0;
            }
            if brain.timer > 0 {
                brain.timer -= 1;
            }
            if brain.timer == 0 {
                brain.phase = CoilPhase::Release;
            }
        }
        CoilPhase::Release => {
            out.overtuned_ids = ctx
                .players
                .iter()
                .filter(|p| horizontal_distance(enemy_pos, &p.feet) <= COIL_SHOCKWAVE_RADIUS)
                .map(|p| p.id)
                .collect();
            brain.phase = CoilPhase::Recover;
            brain.timer = COIL_RECOVER_TICKS;
            brain.cooldown_in = COIL_RELEASE_COOLDOWN_TICKS;
        }
        CoilPhase::Recover => {
            if brain.timer > 0 {
                brain.timer -= 1;
            }
            if brain.timer == 0 {
                let rx = unit_f32(&mut brain.seed);
                let rz = unit_f32(&mut brain.seed);
                brain.leap_from = *enemy_pos;
                brain.leap_to = Vec3::new(
                    ctx.map_min.x + rx * (ctx.map_max.x - ctx.map_min.x),
                    enemy_pos.y,
                    ctx.map_min.z + rz * (ctx.map_max.z - ctx.map_min.z),
                );
                brain.phase = CoilPhase::Leap;
                brain.timer = COIL_LEAP_TICKS;
            }
        }
        CoilPhase::Leap => {
            out.leaping = true;
            let p = 1.0 - brain.timer as f32 / COIL_LEAP_TICKS as f32;
            let h = COIL_LEAP_HEIGHT * 4.0 * p * (1.0 - p);
            enemy_pos.x = brain.leap_from.x + (brain.leap_to.x - brain.leap_from.x) * p;
            enemy_pos.y = brain.leap_from.y + (brain.leap_to.y - brain.leap_from.y) * p + h;
            enemy_pos.z = brain.leap_from.z + (brain.leap_to.z - brain.leap_from.z) * p;
            if brain.timer > 0 {
                brain.timer -= 1;
            }
            if brain.timer == 0 {
                *enemy_pos = brain.leap_to;
                brain.phase = CoilPhase::Coiled;
            }
        }
    }
    out
}

//! Lure — the redesigned fog-gift ambush (cue tier 1, min level 3, cap 2;
//! tags fog/ambush/collapse).
//!
//! It kept its star silhouette and purple-fog identity and lost “does nothing”.
//!
//! - DORMANT: sits in fog emitting a soft gift-like glimmer; the fog pulses
//!   slowly ([`glimmer_intensity`]) — that pulse IS the telegraph. No movement,
//!   no chase. At gameplay distance it reads as a collectible.
//! - TRIGGER: a player inside the short radius gets a brief flare-up
//!   ([`LURE_FLARE_TICKS`], 0.6 s — tier 1, under 1 s), then a small shockwave
//!   (horizontal knockback, no damage), then the Lure pursues the nearest
//!   player for a few seconds before re-dorming elsewhere.
//! - The ambush layer: distinct from Static (hunts) and Coil (random leaps).
//!   Its whole threat is being mistaken for a gift.
//! - One class sees the true silhouette through the glimmer
//!   ([`lure_sees_true_silhouette`]); a jump-pad launch escapes the pursuit
//!   (a grappler using their ability also pulls past it).
//! - During the collapse phase it re-dorms aggressively when players rush the
//!   exit: pursuit lasts [`LURE_COLLAPSE_PURSUE_TICKS`] instead of
//!   [`LURE_PURSUE_TICKS`].
//!
//! Pursuit is deliberately slower than `RUN_SPEED` (8.5 u/s vs 9.0 u/s), so a
//! walking player outruns it; the flare shockwave is the dangerous part.

pub const LURE_FLARE_TICKS: u32 = 36;

pub const LURE_SHOCKWAVE_RADIUS: f32 = 3.0;

pub const LURE_SHOCKWAVE_IMPULSE: f32 = 7.0;
/// Normal pursuit duration, in ticks (3.0 s at 60 Hz).
pub const LURE_PURSUE_TICKS: u32 = 180;
/// Collapse-phase pursuit duration — re-dorms aggressively (1.0 s at 60 Hz).
pub const LURE_COLLAPSE_PURSUE_TICKS: u32 = 60;
/// Pursued speed (u/s); slower than a walking player's run speed.
pub const LURE_PURSUE_SPEED: f32 = 8.5;

pub const LURE_REDORM_TICKS: u32 = 30;

pub const LURE_REDORM_HEIGHT: f32 = 1.5;

/// What happened this tick.
#[derive(Debug, Clone, PartialEq, Default)]
pub struct LureStep {
    /// True while flaring (tier-1 telegraph; VFX/test hook).
    pub telegraphing: bool,
    /// Shockwave impulse per affected player, `(id, impulse)`, emitted once at
    /// the end of the flare. The caller applies the knockback.
    pub shockwave: Option<Vec<(u32, Vec3)>>,
    /// True while pursuing.
    pub pursuing: bool,
    /// True on the tick the pursued player escaped (jump-pad launch, or a
    /// grappler using their ability); the Lure begins re-dorming.
    pub escaped: bool,

/// True for the class that sees the true star silhouette through the glimmer.
///
/// The ticket does not name the class; this is mapped to `glider`, the Lure's
/// `immune` relation in the frozen registry (“crosses above”). Flagged in
/// NOTES for design confirmation.
pub fn lure_sees_true_silhouette(class: &str) -> bool {
    class == "glider"
}

/// Fog pulse intensity in `[0.35, 0.85]`; a triangle wave over
/// [`LURE_GLIMMER_PERIOD`] ticks. This pulse IS the DORMANT telegraph.
pub fn glimmer_intensity(brain: &LureBrain) -> f32 {
    let t = brain.glimmer_ticks as f32 / LURE_GLIMMER_PERIOD as f32;
    let tri = if t < 0.5 { t * 2.0 } else { (1.0 - t) * 2.0 };
    0.35 + 0.5 * tri
}

/// Deterministic re-dorm spot in map bounds; rerolls up to 8 times to land
/// at least [`LURE_REDORM_MIN_DIST`] away.
fn pick_redorm(brain: &mut LureBrain, ctx: &TickContext, from: &Vec3) -> Vec3 {
    let mut candidate = *from;
    for _ in 0..8 {
        let x = ctx.map_min.x + unit_f32(&mut brain.seed) * (ctx.map_max.x - ctx.map_min.x);
        let z = ctx.map_min.z + unit_f32(&mut brain.seed) * (ctx.map_max.z - ctx.map_min.z);
        candidate = Vec3::new(x, from.y, z);
        if horizontal_distance(&candidate, from) >= LURE_REDORM_MIN_DIST {
            break;
        }
    }
    candidate
}

/// Advance Lure by one tick.
pub fn step_lure(
    enemy_pos: &mut Vec3,
    brain: &mut LureBrain,
    ctx: &TickContext,
) -> LureStep {
    if brain.seed == 0 {
        brain.seed = seed_from_pos(enemy_pos);
    }
    let mut out = LureStep::default();
    brain.glimmer_ticks = (brain.glimmer_ticks + 1) % LURE_GLIMMER_PERIOD;

    match brain.phase {
        LurePhase::Dormant => {
            if let Some(p) = nearest_player_where(ctx.players, enemy_pos, |_| true) {
                if horizontal_distance(enemy_pos, &p.feet) <= LURE_TRIGGER_RADIUS {
                    brain.prey = Some(p.id);
                    brain.phase = LurePhase::Flare;
                    brain.timer = LURE_FLARE_TICKS;
                }
            }
        }
        LurePhase::Flare => {
            out.telegraphing = true;
            if brain.timer > 0 {
                brain.timer -= 1;
            }
            if brain.timer == 0 {
                let mut hits = Vec::new();
                for p in ctx.players {
                    if horizontal_distance(enemy_pos, &p.feet) <= LURE_SHOCKWAVE_RADIUS {
                        let dir = flat_direction(
                            enemy_pos,
                            &Vec3::new(p.feet.x, enemy_pos.y, p.feet.z),
                        );
                        hits.push((
                            p.id,
                            Vec3::new(
                                dir.x * LURE_SHOCKWAVE_IMPULSE,
                                0.0,
                                dir.z * LURE_SHOCKWAVE_IMPULSE,
                            ),
                        ));
                    }
                }
                out.shockwave = Some(hits);
                brain.phase = LurePhase::Pursue;
                brain.timer = if ctx.collapse_phase {
                    LURE_COLLAPSE_PURSUE_TICKS
                } else {
                    LURE_PURSUE_TICKS
                };
            }
        }
        LurePhase::Pursue => {
            out.pursuing = true;
            let mut escaped = false;
            if let Some(id) = brain.prey {
                if let Some(p) = ctx.players.iter().find(|p| p.id == id) {
                    if p.jump_launched || (p.class == "grappler" && p.used_ability) {
                        escaped = true;
                    }
                }
            }
            if escaped {
                out.escaped = true;
                brain.phase = LurePhase::ReDorm;
                brain.timer = LURE_REDORM_TICKS;
                brain.redorm_from = *enemy_pos;
                brain.redorm_to = pick_redorm(brain, ctx, enemy_pos);
                return out;
            }
            if let Some(p) = nearest_player_where(ctx.players, enemy_pos, |_| true) {
                let goal = Vec3::new(p.feet.x, enemy_pos.y, p.feet.z);
                move_toward(enemy_pos, &goal, LURE_PURSUE_SPEED * SIM_DT);
            }
            if brain.timer > 0 {
                brain.timer -= 1;
            }
            if brain.timer == 0 {
                brain.phase = LurePhase::ReDorm;
                brain.timer = LURE_REDORM_TICKS;
                brain.redorm_from = *enemy_pos;
                brain.redorm_to = pick_redorm(brain, ctx, enemy_pos);
            }
        }
        LurePhase::ReDorm => {
            let p = 1.0 - brain.timer as f32 / LURE_REDORM_TICKS as f32;
            let h = LURE_REDORM_HEIGHT * 4.0 * p * (1.0 - p);
            enemy_pos.x = brain.redorm_from.x + (brain.redorm_to.x - brain.redorm_from.x) * p;
            enemy_pos.y = brain.redorm_from.y + (brain.redorm_to.y - brain.redorm_from.y) * p + h;
            enemy_pos.z = brain.redorm_from.z + (brain.redorm_to.z - brain.redorm_from.z) * p;
            if brain.timer > 0 {
                brain.timer -= 1;
            }
            if brain.timer == 0 {
                *enemy_pos = brain.redorm_to;
                brain.phase = LurePhase::Dormant;
                brain.prey = None;
            }
        }
    }
    out
}

/// A single enemy. `home` is where it drifts back to when nothing is in range.
#[derive(Debug, Clone, PartialEq)]
pub struct Enemy {

    pub id: u32,
    /// Content kind id (`design/content.json`); reported verbatim in `EnemyView.kind`.
    pub kind: u8,
    /// World position of the enemy's centre.
    pub pos: Vec3,
    /// Home position it drifts back to when idle.
    pub home: Vec3,
    /// Ticks until the enemy becomes active (spawn grace); 0 = active.

/// Euclidean distance between two points.
pub(crate) fn distance(a: &Vec3, b: &Vec3) -> f32 {
    let dx = a.x - b.x;
    let dy = a.y - b.y;
    let dz = a.z - b.z;
    (dx * dx + dy * dy + dz * dz).sqrt()
}

/// Horizontal (XZ) distance between two points.
pub(crate) fn horizontal_distance(a: &Vec3, b: &Vec3) -> f32 {
    let dx = a.x - b.x;
    let dz = a.z - b.z;
    (dx * dx + dz * dz).sqrt()
}

/// The player's centre point, derived from their feet position.
pub(crate) fn player_centre(player_feet: &Vec3) -> Vec3 {
    Vec3::new(player_feet.x, player_feet.y + 0.9, player_feet.z)
}

/// Move `pos` toward `target` by at most `max_step` (3D), never overshooting.
pub(crate) fn move_toward(pos: &mut Vec3, target: &Vec3, max_step: f32) {
    let dx = target.x - pos.x;
    let dy = target.y - pos.y;
    let dz = target.z - pos.z;
    let d = (dx * dx + dy * dy + dz * dz).sqrt();
    if d <= max_step {
        *pos = *target;
        return;
    }
    if d <= 1e-9 {
        return;
    }
    let scale = max_step / d;
    pos.x += dx * scale;
    pos.y += dy * scale;
    pos.z += dz * scale;
}

/// Normalised 3D direction from `from` to `to`; falls back to +X when they coincide.
pub(crate) fn dir_normalized(from: &Vec3, to: &Vec3) -> Vec3 {
    let dx = to.x - from.x;
    let dy = to.y - from.y;
    let dz = to.z - from.z;
    let d = (dx * dx + dy * dy + dz * dz).sqrt();
    if d <= 1e-9 {
        return Vec3::new(1.0, 0.0, 0.0);
    }
    Vec3::new(dx / d, dy / d, dz / d)
}

/// Normalised horizontal (XZ) direction from `from` to `to`; falls back to +X when coincident.
pub(crate) fn flat_direction(from: &Vec3, to: &Vec3) -> Vec3 {
    let dx = to.x - from.x;
    let dz = to.z - from.z;
    let d = (dx * dx + dz * dz).sqrt();
    if d <= 1e-9 {
        return Vec3::new(1.0, 0.0, 0.0);
    }
    Vec3::new(dx / d, 0.0, dz / d)
}

/// Squared distance from a point to an AABB (0 when the point is inside).
pub(crate) fn aabb_distance(p: &Vec3, b: &Aabb) -> f32 {
    let dx = (b.min.x - p.x).max(0.0).max(p.x - b.max.x);
    let dy = (b.min.y - p.y).max(0.0).max(p.y - b.max.y);
    let dz = (b.min.z - p.z).max(0.0).max(p.z - b.max.z);
    (dx * dx + dy * dy + dz * dz).sqrt()
}

/// True when the segment `a`..`b` intersects (or touches) the AABB `bounds`.
pub(crate) fn segment_intersects_aabb(a: &Vec3, b: &Vec3, bounds: &Aabb) -> bool {
    let mut t_min = 0.0_f32;
    let mut t_max = 1.0_f32;
    if !clip_axis(a.x, b.x - a.x, bounds.min.x, bounds.max.x, &mut t_min, &mut t_max) {
        return false;
    }
    if !clip_axis(a.y, b.y - a.y, bounds.min.y, bounds.max.y, &mut t_min, &mut t_max) {
        return false;
    }
    if !clip_axis(a.z, b.z - a.z, bounds.min.z, bounds.max.z, &mut t_min, &mut t_max) {
        return false;
    }
    true
}

/// Clip the slab `[min, max]` on one axis into the `[t_min, t_max]` window; false = no hit.
fn clip_axis(origin: f32, dir: f32, min: f32, max: f32, t_min: &mut f32, t_max: &mut f32) -> bool {
    if dir.abs() <= 1e-9 {
        return origin >= min && origin <= max;
    }
    let inv = 1.0 / dir;
    let mut t0 = (min - origin) * inv;
    let mut t1 = (max - origin) * inv;
    if t0 > t1 {
        std::mem::swap(&mut t0, &mut t1);
    }
    if t0 > *t_min {
        *t_min = t0;
    }
    if t1 < *t_max {
        *t_max = t1;
    }
    *t_min <= *t_max
}

/// True when a sphere at `centre` with `radius` overlaps the player's collision box.
pub(crate) fn sphere_hits_player_box(centre: &Vec3, radius: f32, feet: &Vec3) -> bool {
    let b = player_box(feet);
    let dx = centre.x - centre.x.clamp(b.min.x, b.max.x);
    let dy = centre.y - centre.y.clamp(b.min.y, b.max.y);
    let dz = centre.z - centre.z.clamp(b.min.z, b.max.z);
    dx * dx + dy * dy + dz * dz <= radius * radius
}

/// True when the player's centre is within [`GIFT_RADIUS`] of the gift centre.
pub fn gift_touched(player_feet: &Vec3, gift: &Gift) -> bool {
    gift_touched_within(player_feet, gift, GIFT_RADIUS)
}

/// True when the player's centre is within `radius` of the gift centre.
pub fn gift_touched_within(player_feet: &Vec3, gift: &Gift, radius: f32) -> bool {
    distance(&player_centre(player_feet), &gift.pos) <= radius
}

/// True when the player stands on the beacon: close horizontally and roughly level.
pub fn beacon_reached(player_feet: &Vec3, beacon: &Vec3) -> bool {
    beacon_reached_within(player_feet, beacon, BEACON_RADIUS)
}

/// True when the player is within `radius` horizontally of the beacon and roughly
/// level with it. The vertical tolerance stays [`BEACON_HEIGHT`].
pub fn beacon_reached_within(player_feet: &Vec3, beacon: &Vec3, radius: f32) -> bool {
    let dx = player_feet.x - beacon.x;
    let dz = player_feet.z - beacon.z;
    let horizontal = (dx * dx + dz * dz).sqrt();
    horizontal <= radius && (player_feet.y - beacon.y).abs() <= BEACON_HEIGHT
}

/// Create an enemy sitting at `pos`, with `pos` as its home, default params and a
/// full default spawn grace (pallbearers wait 4.5 s).
pub fn spawn_enemy(id: u32, kind: u8, pos: Vec3) -> Enemy {
    spawn_enemy_with(id, kind, pos, EnemyParams::default())
}

/// Create an enemy with explicit [`EnemyParams`]; `active_in` starts at
/// `params.grace_ticks`, with a 4.5 s floor for pallbearers.
pub fn spawn_enemy_with(id: u32, kind: u8, pos: Vec3, params: EnemyParams) -> Enemy {
    let active_in = if params.behavior == BEHAVIOR_PALLBEARER {
        params.grace_ticks.max(PALLBEARER_WAKE_TICKS)
    } else {
        params.grace_ticks
    };
    Enemy {
        id,
        kind,
        home: pos,
        pos,
        active_in,
        brain: Brain::for_behavior(params.behavior),
        knock_vel: Vec3::new(0.0, 0.0, 0.0),
        knock_ticks: 0,
        params,
    }
}

/// Fraction of the drawn body radius that actually kills for `enemy_touches`.
pub fn enemy_hit_radius(enemy: &Enemy) -> f32 {
    enemy.params.size * ENEMY_HIT_FORGIVENESS
}

/// True when the enemy's hit sphere overlaps the player's collision box.
pub fn enemy_touches(enemy: &Enemy, player_feet: &Vec3) -> bool {
    sphere_hits_player_box(&enemy.pos, enemy_hit_radius(enemy), player_feet)
}
