//! Run state and rules: lobby drafts, generated levels, shared gift progress, golden-gift currency, lives,
//! the enemies the team drafted, the lobby shop, level events and altars. Pure game logic on top of `sim` —
//! no I/O, fully unit-testable.
//!
//! Flow: Lobby (spawn hub + draft offer and, from level 2 on, the shop) → all players vote → generate a level →
//! Playing (two phases: LEVEL_COLLECT until every gift is gathered, then LEVEL_COLLAPSE_PENDING — the siren — and
//! LEVEL_COLLAPSING; golden gifts exist at the collected positions, the beacon opens, the non-hub platforms fall
//! in six farthest-first batches) → beacon → Lobby for the next level. The spawn hub is exempt from collapse.
//! During the collapse/exit window every connected player votes on one permanent addition for the rest of the run
//! (T-078); the automatic one-enemy-per-level scaling is gone.
//! Death with no lives left ends that player's run; when everyone is down the run is Over and Restart begins a new one.

/// A vote session: the options on offer, the ballots keyed by player id, the host and the countdown.
///
/// Ballots are keyed by player id, never by connection, so a disconnect does not destroy a cast ballot.
/// A reconnected player is a fresh player id; the old ballot still counts. All draws and tie-breaks are
/// seeded from the run seed mixed with the level — never from the clock.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct VoteSession {
    /// The level the vote belongs to (the level being exited).
    pub level: u32,
    /// The options on offer (never more than `VOTE_POOL_OFFER`).
    options: Vec<SessionOption>,
    /// Ballots keyed by player id; re-voting replaces the player's earlier ballot.
    ballots: BTreeMap<u32, u8>,
    /// Ticks remaining before the session closes on its own.
    ticks_left: u32,
    /// The host's player id (the lowest connected player id when the session opened).
    host: Option<u32>,
    /// The run seed mixed with the level; drives every deterministic coin flip in this session.
    draw_seed: u32,
    /// Last whole second that was broadcast in `VoteState` (`None` before the first broadcast).
    pub last_sent_second: Option<u16>,
}

impl VoteSession {
    /// Open a session for `level` with the offered `options`.
    pub fn open(level: u32, seed: u32, options: Vec<SessionOption>, host: Option<u32>) -> Self {
        VoteSession {
            level,
            options,
            ballots: BTreeMap::new(),
            ticks_left: VOTE_COUNTDOWN_TICKS,
            host,
            draw_seed: mix_seed(seed, level),
            last_sent_second: None,
        }
    }

    /// The options on offer.
    pub fn options(&self) -> &[SessionOption] {
        &self.options
    }

    /// Whole seconds left in the countdown (rounds up; 20 right after opening).
    pub fn seconds_left(&self) -> u16 {
        u16::try_from(self.ticks_left.div_ceil(TICK_HZ)).unwrap_or(0)
    }

    /// Ticks left in the countdown.
    pub fn ticks_left(&self) -> u32 {
        self.ticks_left
    }

    /// Cast (or re-cast) a ballot. Fails when the session is closed or the option index is out of range.
    pub fn cast(&mut self, player_id: u32, option: u8) -> Result<(), &'static str> {
        if self.ticks_left == 0 {
            return Err("the vote is closed");
        }
        if usize::from(option) >= self.options.len() {
            return Err("option out of range");
        }
        self.ballots.insert(player_id, option);
        Ok(())
    }

    /// Player ids holding a ballot, ascending.
    pub fn voted_ids(&self) -> Vec<u32> {
        self.ballots.keys().copied().collect()
    }

    /// True when `player_id` has cast a ballot in this session.
    pub fn has_voted(&self, player_id: u32) -> bool {
        self.ballots.contains_key(&player_id)
    }

    /// Per-option ballot counts, in option order.
    pub fn counts(&self) -> Vec<u32> {
        let mut counts = vec![0u32; self.options.len()];
        for &option in self.ballots.values() {
            if let Some(slot) = counts.get_mut(usize::from(option)) {
                *slot = slot.saturating_add(1);
            }
        }
        counts
    }

    /// Advance the countdown one tick. Returns true on the tick the session closes.
    pub fn tick(&mut self) -> bool {
        if self.ticks_left > 0 {
            self.ticks_left -= 1;
        }
        self.ticks_left == 0
    }

    /// Resolve the ballots: majority wins; a tie goes to the host's ballot when the host voted for a tied
    /// option, otherwise to a deterministic coin flip from the run seed and the level. With no ballots at all
    /// the coin is flipped across every option. Returns `None` only when no option was offered.
    pub fn resolve(&self) -> Option<VoteOutcome> {
        if self.options.is_empty() {
            return None;
        }
        let counts = self.counts();
        let max = counts.iter().copied().max().unwrap_or(0);
        let tied: Vec<usize> = counts
            .iter()
            .enumerate()
            .filter(|&(_, &c)| c == max)
            .map(|(i, _)| i)
            .collect();
        let winner = if tied.len() == 1 {
            tied[0]
        } else {
            match self
                .host
                .and_then(|h| self.ballots.get(&h).copied())
                .map(usize::from)
            {
                Some(ballot) if tied.contains(&ballot) => ballot,
                _ => {
                    let mut rng = Rng::new(self.draw_seed.wrapping_add(VOTE_COIN_SALT));
                    tied[rng.below(tied.len())]
                }
            }
        };
        let option = &self.options[winner];
        Some(VoteOutcome {
            winner: option.index,
            id: option.id.clone(),
            name: option.name.clone(),
            kind: option.kind,
            permanent: option.kind == CARD_ENEMY && !option.temporary,
            temporary: option.temporary,
            counts,
        })
    }
}

/// One registry entry reduced to the fields the pool rules use (registry §9).
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PoolEntry {
    /// Stable enemy id.
    pub id: String,
    /// Plain name for the vote UI.
    pub name: String,
    /// First level the enemy may be offered.
    pub min_level: u32,
    /// Permanent copies allowed in one run.
    pub selection_cap: u32,
    /// Tags (used by `requires` hooks).
    pub tags: Vec<String>,
    /// Class id → relation; only ever compared against `"ignores"` here.
    pub class_interactions: BTreeMap<String, String>,
    /// `requires` hooks as (tag, count) pairs.
    pub requires: Vec<(String, u32)>,
    /// Position in design/content.json (network kind index); ties in the sort are broken by this.
    pub content_order: usize,
}

/// Tag → count over the run's permanent copies (temporary summons never count; each copy counts once).
fn permanent_tag_counts<'a>(entries: &'a [PoolEntry], permanents: &[String]) -> BTreeMap<&'a str, u32> {
    let mut tag_counts: BTreeMap<&'a str, u32> = BTreeMap::new();
    for id in permanents {
        if let Some(entry) = entries.iter().find(|e| &e.id == id) {
            for tag in &entry.tags {
                *tag_counts.entry(tag.as_str()).or_default() += 1;
            }
        }
    }
    tag_counts
}

/// True when every `requires` hook of `entry` is satisfied by the run's permanents.
fn requires_satisfied(entry: &PoolEntry, tag_counts: &BTreeMap<&str, u32>) -> bool {
    entry
        .requires
        .iter()
        .all(|(tag, count)| tag_counts.get(tag.as_str()).copied().unwrap_or(0) >= *count)
}

/// True when `entry` refuses to spawn against any of the class ids present in the lobby.
fn class_excluded(entry: &PoolEntry, classes: &[&str]) -> bool {
    classes.iter().any(|class| {
        entry
            .class_interactions
            .get(*class)
            .is_some_and(|relation| relation == "ignores")
    })
}

/// The filtered, deterministically ordered enemy pool (registry §9): `min_level` gate, `selection_cap`
/// against the run's permanents, `requires` hooks, class composition. Ordered by `min_level` ascending,
/// ties broken by content order. No RNG.
pub fn vote_pool(
    entries: &[PoolEntry],
    level: u32,
    classes: &[&str],
    permanents: &[String],
) -> Vec<PoolEntry> {
    let tag_counts = permanent_tag_counts(entries, permanents);
    let mut pool: Vec<PoolEntry> = entries
        .iter()
        .filter(|e| e.min_level <= level)
        .filter(|e| {
            let held = permanents.iter().filter(|id| id.as_str() == e.id).count();
            held < e.selection_cap as usize
        })
        .filter(|e| requires_satisfied(e, &tag_counts))
        .filter(|e| !class_excluded(e, classes))
        .cloned()
        .collect();
    pool.sort_by(|a, b| a.min_level.cmp(&b.min_level).then(a.content_order.cmp(&b.content_order)));
    pool
}

/// Registry §9 overflow: every entry is at its cap, so the vote offers TEMPORARY summons for the next
/// level. Candidates ignore `selection_cap` (temporaries never count) but still respect `min_level`, the
/// `requires` hooks and class composition; ordered by `min_level` descending, ties broken by content order.
pub fn overflow_pool(
    entries: &[PoolEntry],
    level: u32,
    classes: &[&str],
    permanents: &[String],
) -> Vec<PoolEntry> {
    let tag_counts = permanent_tag_counts(entries, permanents);
    let mut pool: Vec<PoolEntry> = entries
        .iter()
        .filter(|e| e.min_level <= level)
        .filter(|e| requires_satisfied(e, &tag_counts))
        .filter(|e| !class_excluded(e, classes))
        .cloned()
        .collect();
    pool.sort_by(|a, b| b.min_level.cmp(&a.min_level).then(a.content_order.cmp(&b.content_order)));
    pool.truncate(VOTE_POOL_OFFER);
    pool
}

/// Deterministic seed mix (SplitMix32-style) for vote draws and tie-breaks.
fn mix_seed(seed: u32, level: u32) -> u32 {
    let mut z = seed ^ level.wrapping_mul(0x9E37_79B9);
    z = (z ^ (z >> 16)).wrapping_mul(0x85EB_CA6B);
    z = (z ^ (z >> 13)).wrapping_mul(0xC2B2_AE35);
    z ^ (z >> 16)
}

/// True when `level` has a milestone row in the escalation table (design §5: level 4+; 25+ is the endless row).
pub fn escalation_milestone_level(level: u32) -> bool {
    const ROWS: [u32; 13] = [4, 5, 6, 7, 8, 9, 10, 12, 14, 15, 17, 18, 20];
    ROWS.contains(&level) || level >= 25
}

/// Per-tile collapse warning time for `level` (design §5 level 9+: 2.0 s, else 3.0 s).
pub(crate) fn tile_warning_seconds(level: u32) -> f32 {
    if level >= 9 { 2.0 } else { 3.0 }
}

/// Collapse duration D in seconds for `level` (design §1.4; level 17+ loses 3 s with the tier-30 floor, level 25+ loses 1 s per level beyond and the floor drops to 25).
pub(crate) fn collapse_duration_seconds(level: u32) -> f32 {
    let base = (52.0 - 2.0 * level.saturating_sub(1) as f32).max(30.0);
    let hungering = if level >= 17 { 3.0 } else { 0.0 };
    let endless = if level > 25 { (level - 25) as f32 } else { 0.0 };
    let floor = if level > 25 { 25.0 } else { 30.0 };
    (base - hungering - endless).max(floor)
}

/// Fog multiplier from the escalation table (level 12: Grey Tide +25%, level 18: Deep Night +15%; multiplies with curses).
pub(crate) fn escalation_fog_mul(level: u32) -> f32 {
    let mut fog = 1.0;
    if level >= 12 {
        fog *= 1.25;
    }
    if level >= 18 {
        fog *= 1.15;
    }
    fog
}

/// Extra copies of drafted enemies granted by the escalation table (design §5, cumulative rows, from level 4 on).
fn escalation_extra_copies(level: u32, drafted: &[String], most_voted: Option<&str>) -> Vec<String> {
    let mut extra: Vec<String> = Vec::new();
    if level >= 4
        && let Some(mv) = most_voted
    {
        extra.push(mv.to_string());
    }
    if level >= 6 {
        if drafted.iter().any(|d| d == "pallbearer") {
            extra.push("pallbearer".to_string());
        }
        // Crowded Wake: +1 copy of every drafted enemy every 3 levels after level 6 (cap +6).
        let per3 = ((level - 6) / 3 + 1).min(6);
        for _ in 0..per3 {
            extra.extend(drafted.iter().cloned());
        }
    }
    if level >= 10 {
        // Second Front: +1 copy per 2 drafted enemy types, floor, max +4 (round-robin over the drafted list).
        let types = drafted.len();
        let n = (types / 2).min(4);
        for k in 0..n {
            extra.push(drafted[k % types].clone());
        }
    }
    if level >= 20 {
        // Full Chorus: one extra copy of every drafted enemy.
        extra.extend(drafted.iter().cloned());
    }
    extra
}

/// Distance from a point to an AABB in 3D (0 when the point is inside).
fn point_aabb_distance(p: &Vec3, b: &Aabb) -> f32 {
    let dx = (b.min.x - p.x).max(0.0).max(p.x - b.max.x);
    let dy = (b.min.y - p.y).max(0.0).max(p.y - b.max.y);
    let dz = (b.min.z - p.z).max(0.0).max(p.z - b.max.z);
    (dx * dx + dy * dy + dz * dz).sqrt()
}

/// The spawn hub is exempt from collapse: the platform layer whose XZ footprint contains the beacon or the spawn
/// and whose top is within 2 u of it (the dais/plaza deck) never falls.
fn is_hub_platform(platform: &Platform, beacon: &Vec3, spawn: &Vec3) -> bool {
    let covers = |p: &Vec3| {
        p.x >= platform.bounds.min.x
            && p.x <= platform.bounds.max.x
            && p.z >= platform.bounds.min.z
            && p.z <= platform.bounds.max.z
    };
    let top = platform.bounds.max.y;
    (covers(beacon) && (top - beacon.y).abs() <= 2.0) || (covers(spawn) && (top - spawn.y).abs() <= 2.0)
}

/// Plan the collapse of a generated level: every non-hub platform, farthest from the beacon first
/// (ties by platform id ascending), split into 6 contiguous batches; batch i warns at `i*D/6` into phase 2.
fn build_collapse_plan(platforms: &[Platform], beacon: &Vec3, spawn: &Vec3, level: u32) -> Vec<CollapseEntry> {
    let duration = collapse_duration_seconds(level);
    let warn_ticks = (tile_warning_seconds(level) * TICK_HZ as f32).round() as u32;
    let mut planned: Vec<(f32, u16)> = platforms
        .iter()
        .filter(|p| !is_hub_platform(p, beacon, spawn))
        .map(|p| (point_aabb_distance(beacon, &p.bounds), p.id))
        .collect();
    planned.sort_by(|a, b| b.0.total_cmp(&a.0).then_with(|| a.1.cmp(&b.1)));
    let n = planned.len();
    let mut entries = Vec::with_capacity(n);
    for (idx, (_, id)) in planned.iter().enumerate() {
        let batch = (idx as u32 * 6) / (n as u32).max(1);
        let warn_start_tick = ((batch as f32 * duration / 6.0) * TICK_HZ as f32).round() as u32;
        entries.push(CollapseEntry {
            id: *id,
            warn_start_tick,
            gone_tick: warn_start_tick + warn_ticks,
            state: CollapseTile::Normal,
        });
    }
    entries
}

/// Map every gift's id to the platform it rests on (highest top at or below the gift, within its XZ footprint).
fn gift_platform_map(platforms: &[Platform], gifts: &[protocol::Gift]) -> BTreeMap<u16, u16> {
    let mut map = BTreeMap::new();
    for gift in gifts {
        let target_y = gift.pos.y - 0.3 + 0.01;
        let mut best: Option<&Platform> = None;
        for platform in platforms {
            let top = platform.bounds.max.y;
            let covers = gift.pos.x >= platform.bounds.min.x
                && gift.pos.x <= platform.bounds.max.x
                && gift.pos.z >= platform.bounds.min.z
                && gift.pos.z <= platform.bounds.max.z;
            if !covers || top > target_y {
                continue;
            }
            let better = match best {
                Some(b) => top > b.bounds.max.y,
                None => true,
            };
            if better {
                best = Some(platform);
            }
        }
        if let Some(platform) = best {
            map.insert(gift.id, platform.id);
        }
    }
    map
}

/// Small deterministic PRNG (SplitMix32) for draft offers, shop offers, altar rolls, vote coins and level seeds. Server-only.
#[derive(Debug, Clone)]
pub struct Rng(u32);

impl Rng {
    pub fn new(seed: u32) -> Self {
        Rng(seed)
    }
    pub fn next_u32(&mut self) -> u32 {
        self.0 = self.0.wrapping_add(0x9e37_79b9);
        let mut z = self.0;
        z = (z ^ (z >> 16)).wrapping_mul(0x85eb_ca6b);
        z = (z ^ (z >> 13)).wrapping_mul(0xc2b2_ae35);
        z ^ (z >> 16)
    }
    /// Uniform index in 0..n (n > 0).
    pub fn below(&mut self, n: usize) -> usize {
        (self.next_u32() as usize) % n.max(1)
    }
    /// `k` distinct indices from 0..n, in random order.
    pub fn sample(&mut self, n: usize, k: usize) -> Vec<usize> {
        let mut pool: Vec<usize> = (0..n).collect();
        for i in (1..pool.len()).rev() {
            let j = self.below(i + 1);
            pool.swap(i, j);
        }
        pool.truncate(k.min(n));
        pool
    }
}

    /// Phase of the current level: LEVEL_COLLECT, LEVEL_COLLAPSE_PENDING or LEVEL_COLLAPSING.
    pub level_phase: u8,
    /// Ids of the golden gifts collected this level (same ids as the collected normal gifts).
    pub golden_taken: Vec<u16>,
    /// Ticks since phase 2 (LEVEL_COLLAPSE_PENDING) began; 0 in LEVEL_COLLECT.
    collapse_elapsed_ticks: u32,
    /// Collapse plan: one entry per non-hub platform, farthest from the beacon first.
    collapse: Vec<CollapseEntry>,
    /// Gift id -> platform id it rests on (for golden gifts vanishing with a fallen tile).
    gift_platform: BTreeMap<u16, u16>,
    /// The single most-voted drafted enemy of the last draft (escalation "Second Step", level 4+).
    most_voted_enemy: Option<String>,
    rng: Rng,
}

impl Run {
    /// A new run, waiting in the lobby for the start-of-run draft.
    pub fn new(content: &Content, seed: u32, start_level: u32) -> Self {
        let level = levelgen::hub();
        let world = World::from_level(&level);
        let mut run = Run {
            phase: PHASE_ENEMY_DRAFT,
            level_num: start_level.max(1),
            level,
            world,
            gifts_taken: Vec::new(),
            gifts_collected: 0,
            currency: 0,
            lives: 0,
            enemies: Vec::new(),
            drafted_enemies: Vec::new(),
            permanents: Vec::new(),
            pending_temporary: Vec::new(),
            vote_session: None,
            seed,
            upgrades: Vec::new(),
            curses: Vec::new(),
            mods: RunMods::default(),
            offer: None,
            hazards: Hazards::new(),
            altars_used: Vec::new(),
            event_friction_mul: 1.0,
            shop: None,
            level_phase: LEVEL_COLLECT,
            golden_taken: Vec::new(),
            collapse_elapsed_ticks: 0,
            collapse: Vec::new(),
            gift_platform: BTreeMap::new(),
            most_voted_enemy: None,
            rng: Rng::new(seed),
        };
        run.offer = Some(run.make_offer(content));
        run.shop = run.make_shop(content);
        run
    }

    /// Everything permanent in the run, drafted enemies first, voted permanents after; used for the
    /// pool's `selection_cap` count and the `requires` hooks. Temporary summons never appear here.
    pub fn all_permanents(&self) -> Vec<String> {
        let mut all = self.drafted_enemies.clone();
        all.extend(self.permanents.iter().cloned());
        all
    }

    /// Build the pool entries from the registry content, in content order.
    fn pool_entries(content: &Content) -> Vec<PoolEntry> {
        content
            .enemies
            .iter()
            .enumerate()
            .map(|(content_order, e)| PoolEntry {
                id: e.id.clone(),
                name: e.name.clone(),
                min_level: e.min_level.into(),
                selection_cap: e.selection_cap.into(),
                tags: e.tags.clone(),
                class_interactions: e.class_interactions.clone(),
                requires: e
                    .synergy_hooks
                    .iter()
                    .filter_map(|h| h.requires.as_ref())
                    .map(|r| (r.tag.clone(), r.count.into()))
                    .collect(),
                content_order,
            })
            .collect()
    }

    /// Build the vote options for the level being exited. Deterministic: enemy candidates come from the
    /// registry order (registry §9), a curse may replace the last enemy option on a seeded coin (eligible
    /// curses are non-greater, or greater-only on `GREATER_EVERY` levels), and an empty enemy pool overflows
    /// into temporary summons. Never pads with entries the filters exclude.
    pub fn build_vote_options(&self, content: &Content, classes: &[&str]) -> Vec<SessionOption> {
        let entries = Self::pool_entries(content);
        let permanents = self.all_permanents();
        let level = self.level_num;
        let pool = vote_pool(&entries, level, classes, &permanents);
        let mut options: Vec<SessionOption> = pool
            .iter()
            .take(VOTE_POOL_OFFER)
            .map(|e| SessionOption {
                index: 0,
                id: e.id.clone(),
                name: e.name.clone(),
                kind: CARD_ENEMY,
                temporary: false,
            })
            .collect();
        if options.is_empty() {
            options = overflow_pool(&entries, level, classes, &permanents)
                .into_iter()
                .map(|e| SessionOption {
                    index: 0,
                    id: e.id,
                    name: e.name,
                    kind: CARD_ENEMY,
                    temporary: true,
                })
                .collect();
        }
        // One of the three options may be a curse instead of an enemy: seeded coin, 1 in 2.
        let mut rng = Rng::new(mix_seed(self.seed, level) ^ VOTE_CURSE_SALT);
        let greater = level.is_multiple_of(GREATER_EVERY);
        let curse_pool: Vec<&crate::content::ModCard> =
            content.curses.iter().filter(|c| c.greater == greater).collect();
        if !curse_pool.is_empty() && rng.below(2) == 0 {
            let pick = curse_pool[rng.below(curse_pool.len())];
            if !options.is_empty() {
                options.pop();
            }
            options.push(SessionOption {
                index: 0,
                id: pick.id.clone(),
                // No curse name is assumed to exist in content; the wire carries the id.
                name: pick.id.clone(),
                kind: CARD_CURSE,
                temporary: false,
            });
        }
        for (i, option) in options.iter_mut().enumerate() {
            option.index = i as u8;
        }
        options
    }

    /// Open the vote session for the level being exited; `classes` are the class ids present in the lobby.
    /// Returns false when a session is already open or no option can be offered at all.
    pub fn open_vote(&mut self, content: &Content, classes: &[&str], host: Option<u32>) -> bool {
        if self.vote_session.is_some() {
            return false;
        }
        let options = self.build_vote_options(content, classes);
        if options.is_empty() {
            return false;
        }
        self.vote_session = Some(VoteSession::open(self.level_num, self.seed, options, host));
        true
    }

    /// Close the vote session (countdown elapsed, everyone voted, or the level ended) and apply the outcome.
    /// Returns `None` when no session was open.
    pub fn close_vote(&mut self, content: &Content) -> Option<VoteOutcome> {
        let session = self.vote_session.take()?;
        let outcome = session.resolve()?;
        self.apply_vote_outcome(content, &outcome);
        Some(outcome)
    }

    /// Apply a closed vote: enemies join the permanents (or the one-level temporary list on overflow),
    /// curses act like drafted curses (mods + the run's curse list).
    fn apply_vote_outcome(&mut self, content: &Content, outcome: &VoteOutcome) {
        match outcome.kind {
            CARD_ENEMY => {
                if outcome.temporary {
                    self.pending_temporary.push(outcome.id.clone());
                } else {
                    self.permanents.push(outcome.id.clone());
                }
            }
            _ => {
                if let Some(def) = content.curse(&outcome.id) {
                    let lives_before = self.mods.extra_lives;
                    self.mods.apply(&def.mods);
                    self.lives = self
                        .lives
                        .saturating_add(self.mods.extra_lives.saturating_sub(lives_before));
                    self.curses.push(outcome.id.clone());
                }
            }
        }
    }

    /// The reward cards for the current lobby (greater-curse level or mixed upgrade/curse), drawn from the run RNG.
    fn reward_cards(&mut self, content: &Content) -> Vec<DraftCard> {
        if self.level_num.is_multiple_of(GREATER_EVERY) {
            let greater: Vec<_> = content.curses.iter().filter(|c| c.greater).collect();
            let mut cards = Vec::with_capacity(3);
            for i in self.rng.sample(greater.len(), 3) {
                cards.push(DraftCard {
                    kind: CARD_CURSE,
                    id: greater[i].id.clone(),
                });
            }
            return cards;
        }
        // At least one upgrade and one curse; the third card comes from either pool.
        let normal_curses: Vec<_> = content.curses.iter().filter(|c| !c.greater).collect();
        let ups = self.rng.sample(content.upgrades.len(), 2);
        let curses = self.rng.sample(normal_curses.len(), 2);
        let upgrade = |i: usize| DraftCard {
            kind: CARD_UPGRADE,
            id: content.upgrades[i].id.clone(),
        };
        let curse = |i: usize| DraftCard {
            kind: CARD_CURSE,
            id: normal_curses[i].id.clone(),
        };
        let third = if self.rng.below(2) == 0 {
            upgrade(ups[1])
        } else {
            curse(curses[1])
        };
        let mut rewards = vec![upgrade(ups[0]), curse(curses[0]), third];
        let r = self.rng.below(rewards.len());
        rewards.swap(0, r);
        rewards
    }

    fn make_offer(&mut self, content: &Content) -> Offer {
        // The first round of a run is plain: no hunters, no upgrades, no curses. The lobby
        // still opens so everyone can gather and press Continue together.
        if self.level_num == 1 {
            return Offer {
                enemy_picks: 0,
                reward_picks: 0,
                cards: Vec::new(),
            };
        }
        let mut cards = Vec::new();
        let enemy_picks = if self.drafted_enemies.is_empty() {
            for i in self.rng.sample(content.enemies.len(), ENEMY_OFFER) {
                cards.push(DraftCard {
                    kind: CARD_ENEMY,
                    id: content.enemies[i].id.clone(),
                });
            }
            ENEMY_PICKS
        } else {
            0
        };
        cards.extend(self.reward_cards(content));
        Offer {
            enemy_picks,
            reward_picks: REWARD_PICKS,
            cards,
        }
    }

    /// Pick the lobby shop: up to SHOP_OFFER distinct items from content, stock from content, run RNG.
    fn make_shop(&mut self, content: &Content) -> Option<Vec<ShopOffer>> {
        if self.level_num == 1 || content.shop.is_empty() {
            return None;
        }
        let picks = self.rng.sample(content.shop.len(), SHOP_OFFER.min(content.shop.len()));
        Some(
            picks
                .into_iter()
                .map(|i| {
                    let def = &content.shop[i];
                    ShopOffer {
                        id: def.id.clone(),
                        cost: def.cost,
                        stock: def.stock,
                    }
                })
                .collect(),
        )
    }

    /// Replace the reward cards of the current offer, keeping enemy cards (and enemy picks) at the same indices.
    /// Returns false when no draft offer is open.
    pub fn reroll_rewards(&mut self, content: &Content) -> bool {
        if self.offer.is_none() {
            return false;
        }
        let rewards = self.reward_cards(content);
        if let Some(offer) = self.offer.as_mut() {
            offer.cards.retain(|c| c.kind == CARD_ENEMY);
            offer.cards.extend(rewards);
        }
        true
    }

    /// Buy an item from the current lobby shop (team-wide currency). Nothing is charged when the purchase fails.
    pub fn buy_shop_item(&mut self, content: &Content, item: &str) -> Result<ShopPurchase, ShopError> {
        if self.phase != PHASE_ENEMY_DRAFT {
            return Err(ShopError::NotInLobby);
        }
        // Validate against the current offer first (stock, then currency), without holding any borrows.
        let (cost, stock) = match self.shop.as_ref() {
            None => return Err(ShopError::UnknownItem),
            Some(offer) => match offer.iter().find(|entry| entry.id == item) {
                None => return Err(ShopError::UnknownItem),
                Some(entry) => (entry.cost, entry.stock),
            },
        };
        if stock == 0 {
            return Err(ShopError::SoldOut);
        }
        if self.currency < cost {
            return Err(ShopError::NotEnoughCurrency);
        }
        let Some(def) = content.shop_item(item) else {
            return Err(ShopError::UnknownItem);
        };
        // Apply the effect.
        let purchase = match def.kind.as_str() {
            "life" => {
                self.lives = self.lives.saturating_add(1);
                ShopPurchase::Life
            }
            "reroll" => {
                if !self.reroll_rewards(content) {
                    return Err(ShopError::UnknownItem);
                }
                ShopPurchase::Reroll
            }
            "mods" => {
                let before = self.mods.extra_lives;
                self.mods.apply(&def.mods);
                self.lives = self.lives.saturating_add(self.mods.extra_lives.saturating_sub(before));
                ShopPurchase::Mods
            }
            _ => return Err(ShopError::UnknownItem),
        };
        // Charge only after the effect went through.
        if let Some(entry) = self
            .shop
            .as_mut()
            .and_then(|offer| offer.iter_mut().find(|entry| entry.id == item))
        {
            entry.stock = entry.stock.saturating_sub(1);
        }
        self.currency = self.currency.saturating_sub(cost);
        Ok(purchase)
    }

    /// Check one player's picks against the current offer.
    pub fn validate_picks(&self, picks: &[u8]) -> Result<(), &'static str> {
        let Some(offer) = &self.offer else {
            return Err("no draft is open");
        };
        if self.phase != PHASE_ENEMY_DRAFT {
            return Err("not in the lobby");
        }
        let mut seen = [false; 256];
        let (mut enemies, mut rewards) = (0u8, 0u8);
        for &p in picks {
            let Some(card) = offer.cards.get(p as usize) else {
                return Err("pick out of range");
            };
            if std::mem::replace(&mut seen[p as usize], true) {
                return Err("duplicate pick");
            }
            if card.kind == CARD_ENEMY {
                enemies = enemies.saturating_add(1);
            } else {
                rewards = rewards.saturating_add(1);
            }
        }
        if enemies != offer.enemy_picks || rewards != offer.reward_picks {
            return Err("wrong number of picks");
        }
        Ok(())
    }

    /// Tally everyone's (validated) votes, apply the winning cards and start the generated level.
    /// Ties go to the card offered first. The number of voters is the player count used for gift generation.
    pub fn resolve(&mut self, content: &Content, votes: &[Vec<u8>]) {
        let Some(offer) = self.offer.take() else { return };
        let mut tally: BTreeMap<usize, u32> = BTreeMap::new();
        for v in votes {
            for &p in v {
                *tally.entry(p as usize).or_default() += 1;
            }
        }
        let mut ranked: Vec<usize> = (0..offer.cards.len()).collect();
        ranked.sort_by_key(|i| (std::cmp::Reverse(tally.get(i).copied().unwrap_or(0)), *i));
        let winners = |enemy: bool, n: u8| -> Vec<usize> {
            ranked
                .iter()
                .copied()
                .filter(|&i| (offer.cards[i].kind == CARD_ENEMY) == enemy)
                .take(n as usize)
                .collect()
        };
        let enemy_winners = winners(true, offer.enemy_picks);
        self.most_voted_enemy = enemy_winners.first().map(|&i| offer.cards[i].id.clone());
        for i in enemy_winners {
            self.drafted_enemies.push(offer.cards[i].id.clone());
        }
        for i in winners(false, offer.reward_picks) {
            let card = &offer.cards[i];
            let def = if card.kind == CARD_UPGRADE {
                content.upgrade(&card.id)
            } else {
                content.curse(&card.id)
            };
            if let Some(def) = def {
                let lives_before = self.mods.extra_lives;
                self.mods.apply(&def.mods);
                self.lives = self
                    .lives
                    .saturating_add(self.mods.extra_lives.saturating_sub(lives_before));
                if card.kind == CARD_UPGRADE {
                    self.upgrades.push(card.id.clone());
                } else {
                    self.curses.push(card.id.clone());
                }
            }
        }
        let players = u32::try_from(votes.len()).unwrap_or(1).max(1);
        self.start_level(content, players);
    }

    fn start_level(&mut self, content: &Content, players: u32) {
        let seed = self.rng.next_u32();
        // T-062: the Clustered Gifts curse scales the minimum gift spacing for this level.
        let generated = levelgen::generate_for_with_spacing(seed, self.level_num, players, self.mods.gift_spacing_mul);
        let spawn_slots = generated.enemy_spawns.len();
        self.level = generated.data;
        self.world = World::from_level(&self.level);
        self.gifts_taken.clear();
        self.golden_taken.clear();
        self.enemies.clear();
        self.hazards.reset();
        self.altars_used.clear();
        self.level_phase = LEVEL_COLLECT;
        self.collapse_elapsed_ticks = 0;
        self.collapse = build_collapse_plan(
            &self.level.platforms,
            &self.level.beacon,
            &self.level.spawn,
            self.level_num,
        );
        self.gift_platform = gift_platform_map(&self.level.platforms, &self.level.gifts);
        // The shop lives in the lobby only; starting a level closes it.
        self.shop = None;
        self.event_friction_mul = event_friction_mul(content, self.level.event);
        // Roster: drafted enemies + voted permanents + curse copies + milestone escalation copies
        // (design §5, from level 4 on). T-078: no per-level automatic scaling remains.
        let base = self.all_permanents();
        let base_len = base.len();
        let mut roster: Vec<String> = base.clone();
        if base_len > 0 {
            for k in 0..self.mods.extra_enemies as usize {
                roster.push(base[k % base_len].clone());
            }
        }
        roster.extend(escalation_extra_copies(
            self.level_num,
            &base,
            self.most_voted_enemy.as_deref(),
        ));
        // Level 1 stays plain (design A4): no enemies at all.
        if self.level_num == 1 {
            roster.clear();
        } else {
            // One-level temporary summons from the last vote's overflow (registry §9).
            roster.extend(self.pending_temporary.drain(..));
        }
        // The hub has 8 fixed spawn slots; extra copies cycle onto the slots.
        for (index, id) in roster.iter().enumerate() {
            let Some((content_index, def)) = content.enemy(id) else {
                continue;
            };
            let mut params = self.mods.enemy_params(&def.params);
            params.speed_mul *= escalation_speed_mul(self.level_num);
            let mut pos = if spawn_slots == 0 {
                self.level.spawn
            } else {
                generated.enemy_spawns[index % spawn_slots]
            };
            // Mimics pose as gifts: spawn them on the platform below the generated spawn
            // (pos.y = platform top + 0.5). If the spawn is not above any platform, keep the position.
            if params.behavior == BEHAVIOR_MIMIC
                && let Some(top) = platform_top_below(&self.level.platforms, pos.x, pos.z, pos.y)
            {
                pos.y = top + 0.5;
            } else if let Some(top) = platform_top_below(&self.level.platforms, pos.x, pos.z, pos.y + 0.01) {
                // Everything else starts with its body clear of the floor and its centre at least at a
                // player's chest height: hub enemy slots lie ON the floor (y = top), and a turret whose
                // centre sits on the floor surface had its line of sight and its flat bolts start inside
                // the floor, so it never fired.
                pos.y = pos.y.max(top + params.size.max(0.9) + 0.1);
            }
            let enemy_id = u32::try_from(index).unwrap_or(u32::MAX - 1) + 1;
            // kind = index into content.json enemies, so the client draws the right model.
            self.enemies.push(rules::spawn_enemy_with(
                enemy_id,
                u8::try_from(content_index).unwrap_or(u8::MAX),
                pos,
                params,
            ));
        }
        self.phase = PHASE_PLAYING;
    }

    /// The beacon was reached: back to the lobby to draft for the next level.
    pub fn finish_level(&mut self, content: &Content) {
        self.level_num = self.level_num.saturating_add(1);
        self.level = levelgen::hub();
        self.world = World::from_level(&self.level);
        self.gifts_taken.clear();
        self.golden_taken.clear();
        self.enemies.clear();
        self.hazards.reset();
        self.altars_used.clear();
        self.level_phase = LEVEL_COLLECT;
        self.collapse_elapsed_ticks = 0;
        self.collapse.clear();
        self.gift_platform.clear();
        self.event_friction_mul = 1.0;
        self.phase = PHASE_ENEMY_DRAFT;
        self.offer = Some(self.make_offer(content));
        self.shop = self.make_shop(content);
    }

    pub fn end(&mut self) {
        self.phase = PHASE_OVER;
        self.offer = None;
        self.shop = None;
    }

    /// Use the first unused altar within 2.0 units horizontally and 1.5 vertically of `feet`:
    /// apply one random upgrade and one random non-greater curse (run RNG) and record both in the
    /// RunInfo lists. Returns the applied (upgrade id, curse id), or None when nothing was used.
    pub fn try_use_altar(&mut self, content: &Content, feet: &Vec3) -> Option<(String, String)> {
        if self.phase != PHASE_PLAYING {
            return None;
        }
        let mut hit = None;
        for altar in &self.level.altars {
            if self.altars_used.contains(&altar.id) {
                continue;
            }
            let dx = feet.x - altar.pos.x;
            let dz = feet.z - altar.pos.z;
            let dy = (feet.y - altar.pos.y).abs();
            if dx * dx + dz * dz <= 4.0 && dy <= 1.5 {
                hit = Some(altar.id);
                break;
            }
        }
        let altar_id = hit?;
        if content.upgrades.is_empty() {
            return None;
        }
        // One random upgrade...
        let up_i = self.rng.below(content.upgrades.len());
        let upgrade_id = content.upgrades[up_i].id.clone();
        let upgrade_mods = content.upgrades[up_i].mods.clone();
        // ...and one random non-greater curse.
        let normal_curses: Vec<&crate::content::ModCard> = content.curses.iter().filter(|c| !c.greater).collect();
        if normal_curses.is_empty() {
            return None;
        }
        let cur_i = self.rng.below(normal_curses.len());
        let curse_id = normal_curses[cur_i].id.clone();
        let curse_mods = normal_curses[cur_i].mods.clone();
        let lives_before = self.mods.extra_lives;
        self.mods.apply(&upgrade_mods);
        self.lives = self
            .lives
            .saturating_add(self.mods.extra_lives.saturating_sub(lives_before));
        let lives_before = self.mods.extra_lives;
        self.mods.apply(&curse_mods);
        self.lives = self
            .lives
            .saturating_add(self.mods.extra_lives.saturating_sub(lives_before));
        self.upgrades.push(upgrade_id.clone());
        self.curses.push(curse_id.clone());
        self.altars_used.push(altar_id);
        Some((upgrade_id, curse_id))
    }

    fn normal_left(&self) -> usize {
        self.level
            .gifts
            .iter()
            .filter(|g| !self.gifts_taken.contains(&g.id))
            .count()
    }

    pub fn beacon_open(&self) -> bool {
        self.phase == PHASE_PLAYING && self.level_phase != LEVEL_COLLECT
    }

    /// Collect gifts the player at `feet` touches. In LEVEL_COLLECT the remaining normal gifts are gathered
    /// (the last one starts phase 2). From LEVEL_COLLAPSE_PENDING on, golden gifts sitting at the positions of
    /// the collected normal gifts are collected instead (1 gold each, same gift ids, `golden_taken` lists them).
    /// Returns the gift positions collected this call (the sim uses them for the pickup events).
    pub fn collect_gifts(&mut self, feet: &Vec3) -> Vec<Vec3> {
        let mut picked = Vec::new();
        if self.phase != PHASE_PLAYING {
            return picked;
        }
        let radius = GIFT_RADIUS * self.mods.gift_radius_mul;
        if self.level_phase == LEVEL_COLLECT {
            for gift in &self.level.gifts {
                if self.gifts_taken.contains(&gift.id) {
                    continue;
                }
                if rules::gift_touched_within(feet, gift, radius) {
                    self.gifts_taken.push(gift.id);
                    self.gifts_collected = self.gifts_collected.saturating_add(1);
                    picked.push(gift.pos);
                }
            }
            // The last gift (including the vacuous zero-gift case) starts phase 2.
            if self.normal_left() == 0 {
                self.enter_collapse_pending();
            }
            return picked;
        }
        // Phase 2: one golden gift per collected normal gift, at its exact position.
        for gift in &self.level.gifts {
            if !self.gifts_taken.contains(&gift.id) {
                continue;
            }
            if self.golden_taken.contains(&gift.id) {
                continue;
            }
            if self.platform_gone_for_gift(gift.id) {
                // The tile fell: the uncollected golden gift vanished forever.
                continue;
            }
            if rules::gift_touched_within(feet, gift, radius) {
                self.golden_taken.push(gift.id);
                self.currency = self.currency.saturating_add(GOLDEN_VALUE);
                picked.push(gift.pos);
            }
        }
        picked
    }

    /// Enter LEVEL_COLLAPSE_PENDING (phase 2 starts): the siren sounds, the beacon opens.
    fn enter_collapse_pending(&mut self) {
        if self.level_phase != LEVEL_COLLECT {
            return;
        }
        self.level_phase = LEVEL_COLLAPSE_PENDING;
        self.collapse_elapsed_ticks = 0;
    }

    /// True when the gift rests on a platform that has already fallen.
    fn platform_gone_for_gift(&self, gift_id: u16) -> bool {
        let Some(platform_id) = self.gift_platform.get(&gift_id) else {
            return false;
        };
        self.collapse
            .iter()
            .find(|e| e.id == *platform_id)
            .is_some_and(|e| e.state == CollapseTile::Gone)
    }

    /// Advance the collapse timers one tick: PENDING -> COLLAPSING after 3.0 s, batch warnings, TILE_GONE flips.
    /// Batches start their warning `i*D/6` after phase 2 begins (the clock starts at PENDING).
    pub fn tick_collapse(&mut self) {
        if self.level_phase == LEVEL_COLLECT {
            return;
        }
        self.collapse_elapsed_ticks = self.collapse_elapsed_ticks.saturating_add(1);
        if self.level_phase == LEVEL_COLLAPSE_PENDING && self.collapse_elapsed_ticks >= COLLAPSE_PENDING_TICKS {
            self.level_phase = LEVEL_COLLAPSING;
        }
        let clock = self.collapse_elapsed_ticks;
        for entry in &mut self.collapse {
            if entry.state == CollapseTile::Gone {
                continue;
            }
            if clock >= entry.gone_tick {
                entry.state = CollapseTile::Gone;
            } else if clock >= entry.warn_start_tick {
                entry.state = CollapseTile::Warn;
            }
        }
    }

    /// Milliseconds until the last collapse batch has fallen (0 before phase 2 and once the collapse is done).
    pub fn collapse_ms(&self) -> u32 {
        if self.level_phase == LEVEL_COLLECT {
            return 0;
        }
        let last_gone = self.collapse.iter().map(|e| e.gone_tick).max().unwrap_or(0);
        let remaining = last_gone.saturating_sub(self.collapse_elapsed_ticks);
        remaining.saturating_mul(1000 / TICK_HZ)
    }

    /// Tiles to simulate and report: the hazards tiles overlaid by the collapse (collapse wins on the same id).
    pub fn world_tiles(&self) -> Vec<TileState> {
        let mut merged: BTreeMap<u16, TileState> = BTreeMap::new();
        for tile in self.hazards.tiles() {
            merged.insert(tile.id, tile);
        }
        for entry in &self.collapse {
            let state = match entry.state {
                CollapseTile::Normal => None,
                CollapseTile::Warn => Some(TILE_WARN),
                CollapseTile::Gone => Some(TILE_GONE),
            };
            if let Some(state) = state {
                merged.insert(entry.id, TileState { id: entry.id, state });
            }
        }
        merged.into_values().collect()
    }

    /// The shared tuning: movement mods plus the level event's friction override.
    pub fn tuning(&self) -> Tuning {
        let mut tuning = self.mods.tuning();
        tuning.friction_mul = self.event_friction_mul;
        tuning
    }

    pub fn beacon_reached(&self, feet: &Vec3) -> bool {
        self.beacon_open()
            && rules::beacon_reached_within(feet, &self.level.beacon, BEACON_RADIUS * self.mods.beacon_radius_mul)
    }

    pub fn enemy_views(&self) -> Vec<EnemyView> {
        self.hazards.views(&self.enemies)
    }

    pub fn info(&self) -> RunInfo {
        RunInfo {
            phase: self.phase,
            level: self.level_num,
            lives: self.lives,
            currency: self.currency,
            tuning: self.tuning(),
            fog_density_mul: self.mods.fog_density_mul * escalation_fog_mul(self.level_num),
            enemies: self.all_permanents(),
            upgrades: self.upgrades.clone(),
            curses: self.curses.clone(),
        }
    }
}

/// Friction multiplier for `event`: EVENT_ICE pulls it from content, anything else is 1.0.
fn event_friction_mul(content: &Content, event: u8) -> f32 {
    if event == EVENT_ICE {
        content
            .event(EVENT_ICE)
            .map(|e| e.friction_mul)
            .filter(|f| f.is_finite() && *f > 0.0)
            .unwrap_or(1.0)
    } else {
        1.0
    }
}

/// Highest platform top at or below `y` whose xz bounds contain (x, z). Tops may match `y` within a small tolerance
/// (spawns sitting exactly on a platform stay put).
fn platform_top_below(platforms: &[Platform], x: f32, z: f32, y: f32) -> Option<f32> {
    let mut best: Option<f32> = None;
    for platform in platforms {
        let top = platform.bounds.max.y;
        if x >= platform.bounds.min.x
            && x <= platform.bounds.max.x
            && z >= platform.bounds.min.z
            && z <= platform.bounds.max.z
            && top <= y + 0.01
        {
            best = Some(match best {
                Some(b) => b.max(top),
                None => top,
            });
        }
    }
    best
}

#[cfg(test)]
pub(crate) fn valid_picks(run: &Run) -> Vec<u8> {
    let Some(offer) = run.offer.as_ref() else { return vec![] };
    let mut picks = vec![];
    let (mut e, mut r) = (0, 0);
    for (i, c) in offer.cards.iter().enumerate() {
        if c.kind == CARD_ENEMY && e < offer.enemy_picks {
            picks.push(i as u8);
            e += 1;
        } else if c.kind != CARD_ENEMY && r < offer.reward_picks {
            picks.push(i as u8);
            r += 1;
        }
    }
    picks
}

//! The game task: owns all simulation state and runs the authoritative tick loop.
//!
//! Input model: clients send one [`InputFrame`] per fixed physics step (1/PHYSICS_HZ s),
//! tagged with an increasing `seq`. Each 20 Hz tick, every player gets
//! `STEPS_PER_TICK` step tokens (capped, so short network jitter can catch up but a
//! client can never run faster than real time), and the server consumes that many
//! queued frames. Snapshots carry the last applied `seq` so the client can reconcile.
//!
//! Run flow (see [`crate::run`]): lobby on the spawn hub with a draft offer (and, from level 2 on, the shop)
//! -> every player votes -> a generated level -> collect all normal gifts -> golden gifts + beacon -> lobby.
//! During the collapse/exit window every connected player votes on one permanent addition (T-078); the
//! vote closes on its own 20 s countdown, early once every joined player has balloted, or when the level ends.
//! Tick order: movement (altar uses on rising-edge BTN_INTERACT) -> (playing) vote countdown -> gifts -> hazards -> deaths/lives -> beacon -> end check -> snapshots.
//!
//! Multiplayer: a player who enters the open beacon is FINISHED and one who runs out of lives is DEAD; both
//! become invisible spectators. The level ends when nobody is ACTIVE: back to the lobby if anyone finished,
//! otherwise the run is over. Solo is the same rule with one player.

use crate::content::Content;
use crate::matchmaker_client::GameStateHandle;
use crate::run::{Run, ShopError, ShopPurchase};
use protocol::{
    BTN_ABILITY, BTN_INTERACT, BTN_JUMP, CARD_ENEMY, CLASS_GLIDER, CLASS_STRIKER, ClientMsg, ControllerState,
    DEATH_VOID, EVENT_NONE, InputFrame, MAX_DRAFT_PICKS, MAX_INPUTS_PER_MSG, MAX_NAME_LEN, NOTICE_ALTAR, NOTICE_BOUGHT,
    NOTICE_EVENT, PHASE_ENEMY_DRAFT, PHASE_OVER, PHASE_PLAYING, PHYSICS_HZ, PLAYER_ACTIVE, PLAYER_DEAD,
    PLAYER_FINISHED, PLAYER_WAITING, PROTOCOL_VERSION, PlayerView, RunSummary, ServerMsg, ShopItem, TICK_HZ, Vec3,
};
use sim::controller;
use std::collections::{BTreeMap, VecDeque};
use std::sync::Arc;
use std::time::{Duration, Instant};
use tokio::sync::mpsc;

pub type ConnId = u64;

/// Outbound frames buffered per connection before we consider the client too slow.
pub const OUTBOX_CAPACITY: usize = 64;
pub const MAX_PLAYERS: usize = 8;
/// One second of buffered input at most.
const MAX_QUEUED_INPUTS: usize = PHYSICS_HZ as usize;
const STEPS_PER_TICK: u32 = PHYSICS_HZ / TICK_HZ;
/// Token cap: up to 3 ticks' worth of catch-up after jitter.
const MAX_STEP_TOKENS: u32 = STEPS_PER_TICK * 3;
/// After this many ticks with no input, simulate neutral input so gravity keeps acting.
const STARVE_TICKS: u32 = 4;
const BUTTON_MASK: u8 = BTN_JUMP | BTN_ABILITY | BTN_INTERACT;
/// Players spawn spread around the shared spawn point so they never stack.
const SPAWN_RING: [(f32, f32); MAX_PLAYERS] = [
    (0.0, 0.0),
    (1.6, 0.0),
    (-1.6, 0.0),
    (0.0, 1.6),
    (0.0, -1.6),
    (1.6, 1.6),
    (-1.6, -1.6),
    (1.6, -1.6),
];

pub enum GameEvent {
    Connected { id: ConnId, out: mpsc::Sender<Vec<u8>> },
    Message { id: ConnId, msg: ClientMsg },
    Disconnected { id: ConnId },
}

struct Player {
    id: u32,
    name: String,
    state: ControllerState,
    inputs: VecDeque<InputFrame>,
    last_recv_seq: Option<u32>,
    last_applied_seq: u32,
    tokens: u32,
    starved_ticks: u32,
    /// PLAYER_ACTIVE / PLAYER_FINISHED / PLAYER_DEAD / PLAYER_WAITING.
    mode: u8,
    /// How this player last died (for the run summary).
    cause: u8,
    /// The class the player chose (CLASS_*); copied into the controller state at every spawn/respawn.
    class: u8,
    /// Ticks of OVERTUNED left (Coil's shockwave, Peal's ring). Peal's ring is lethal only while set.
    overtuned_ticks: u16,
    /// Ticks of Blight's infection left. Slows the victim and pauses Rewind's and Kite's timers.
    infected_ticks: u16,
    /// True on the tick a jump pad launched this player (cleanses Blight, escapes Lure).
    pad_launched: bool,
}

impl Player {
    fn active(&self) -> bool {
        self.mode == PLAYER_ACTIVE
    }
}

struct Conn {
    out: mpsc::Sender<Vec<u8>>,
    player: Option<Player>,
}

/// Spawn state at `at` with the player's class baked into the controller state.
fn player_spawn_state(at: Vec3, class: u8) -> ControllerState {
    let mut state = controller::spawn_state(at);
    state.class = class;
    state
}

/// Map a CLASS_* protocol constant to the class id the enemy registry filters on.
fn class_id_str(class: u8) -> Option<&'static str> {
    match class {
        CLASS_STRIKER => Some("striker"),
        protocol::CLASS_GRAPPLER => Some("grappler"),
        protocol::CLASS_BULWARK => Some("bulwark"),
        CLASS_GLIDER => Some("glider"),
        _ => None,
    }
}

pub struct Game {
    tick: u32,
    conns: BTreeMap<ConnId, Conn>,
    content: Content,
    run: Run,
    /// Draft votes of the current lobby, by connection.
    votes: BTreeMap<ConnId, Vec<u8>>,
    next_player_id: u32,
    /// Level every new run starts at (1 in production; `VOIDWRAP_START_LEVEL` for visual QA).
    start_level: u32,
    /// Seeds successive runs (draft offers and level layouts).
    seed: u32,
}

impl Game {
    pub fn starting_at(start_level: u32, seed: u32) -> Result<Self, String> {
        let content = Content::load()?;
        let start_level = start_level.max(1);
        let run = Run::new(&content, seed, start_level);
        Ok(Game {
            tick: 0,
            conns: BTreeMap::new(),
            content,
            run,
            votes: BTreeMap::new(),
            next_player_id: 1,
            start_level,
            seed,
        })
    }

    pub fn tick_count(&self) -> u32 {
        self.tick
    }

    pub fn player_count(&self) -> usize {
        self.conns.values().filter(|c| c.player.is_some()).count()
    }

    fn joined_ids(&self) -> Vec<ConnId> {
        self.conns
            .iter()
            .filter(|(_, c)| c.player.is_some())
            .map(|(id, _)| *id)
            .collect()
    }

    /// Player ids of every joined player, ascending by connection order.
    fn joined_player_ids(&self) -> Vec<u32> {
        self.conns
            .values()
            .filter_map(|c| c.player.as_ref().map(|p| p.id))
            .collect()
    }

    /// Distinct class ids present in the lobby (the vote pool refuses enemies that ignore any of them).
    fn lobby_class_ids(&self) -> Vec<&'static str> {
        let mut classes: Vec<&'static str> = self
            .conns
            .values()
            .filter_map(|c| c.player.as_ref())
            .filter_map(|p| class_id_str(p.class))
            .collect();
        classes.sort_unstable();
        classes.dedup();
        classes
    }

    /// Spawn position for the n-th joined player.
    fn spawn_point(&self, slot: usize) -> Vec3 {
        let (dx, dz) = SPAWN_RING[slot % MAX_PLAYERS];
        let s = &self.run.level.spawn;
        Vec3::new(s.x + dx, s.y, s.z + dz)
    }

    pub fn handle(&mut self, ev: GameEvent) {
        match ev {
            GameEvent::Connected { id, out } => {
                self.conns.insert(id, Conn { out, player: None });
            }
            GameEvent::Disconnected { id } => {
                if let Some(Conn { player: Some(p), .. }) = self.conns.remove(&id) {
                    tracing::info!(conn = id, player = p.id, name = %p.name, "player left");
                }
                // Draft votes are connection-scoped and drop with the connection; vote-session
                // ballots are keyed by player id and survive (T-078).
                self.votes.remove(&id);
                self.maybe_resolve_draft();
            }
            GameEvent::Message { id, msg } => self.on_message(id, msg),
        }
    }

    fn send(&mut self, id: ConnId, msg: &ServerMsg) {
        let Some(conn) = self.conns.get(&id) else { return };
        let bytes = match protocol::encode(msg) {
            Ok(b) => b,
            Err(e) => {
                tracing::error!("encode failed: {e}");
                return;
            }
        };
        if conn.out.try_send(bytes).is_err() {
            // Outbox full (client not reading) or writer gone: drop the connection.
            tracing::warn!(conn = id, "dropping slow or closed connection");
            self.conns.remove(&id);
            self.votes.remove(&id);
            // A leaver may have been the only missing vote. Recursion is bounded: `conns`
            // shrinks with every removal, and resolving the draft leaves the draft phase.
            self.maybe_resolve_draft();
        }
    }

    fn broadcast(&mut self, msg: &ServerMsg) {
        for id in self.joined_ids() {
            self.send(id, msg);
        }
    }

    /// Send an error and drop the connection.
    fn reject(&mut self, id: ConnId, code: u16, message: &str) {
        tracing::warn!(conn = id, code, "rejecting: {message}");
        self.send(
            id,
            &ServerMsg::Error {
                code,
                message: message.to_owned(),
            },
        );
        self.conns.remove(&id);
        self.votes.remove(&id);
        // Same as in `send`: recursion is bounded because `conns` shrinks.
        self.maybe_resolve_draft();
    }

    fn draft_msg(&self) -> Option<ServerMsg> {
        let offer = self.run.offer.as_ref()?;
        (self.run.phase == PHASE_ENEMY_DRAFT).then(|| ServerMsg::Draft {
            enemy_picks: offer.enemy_picks,
            reward_picks: offer.reward_picks,
            cards: offer.cards.clone(),
        })
    }

    /// The lobby shop offer as a message, when a shop is open.
    fn shop_msg(&self) -> Option<ServerMsg> {
        let shop = self.run.shop.as_ref()?;
        Some(ServerMsg::Shop {
            items: shop
                .iter()
                .map(|item| ShopItem {
                    id: item.id.clone(),
                    cost: item.cost,
                    stock: item.stock,
                })
                .collect(),
        })
    }

    /// The VoteOpen message for the open session, if any.
    fn vote_open_msg(&self) -> Option<ServerMsg> {
        let session = self.run.vote_session.as_ref()?;
        Some(ServerMsg::VoteOpen {
            level: session.level,
            options: session
                .options()
                .iter()
                .map(|o| protocol::VoteOption {
                    index: o.index,
                    id: o.id.clone(),
                    name: o.name.clone(),
                    kind: o.kind,
                })
                .collect(),
            countdown_s: u16::try_from(crate::run::VOTE_COUNTDOWN_TICKS / TICK_HZ).unwrap_or(20),
        })
    }

    /// The VoteState message for the open session, if any.
    fn vote_state_msg(&self) -> Option<ServerMsg> {
        let session = self.run.vote_session.as_ref()?;
        Some(ServerMsg::VoteState {
            counts: session.counts(),
            voted: session.voted_ids(),
            seconds_left: session.seconds_left(),
        })
    }

    /// Broadcast the current VoteState (counts, who has voted, seconds left).
    fn broadcast_vote_state(&mut self) {
        if let Some(msg) = self.vote_state_msg() {
            self.broadcast(&msg);
        }
    }

    /// Open and broadcast a fresh vote session for the level being exited (no-op when one is open).
    fn open_vote_session(&mut self) {
        let classes = self.lobby_class_ids();
        let host = self.joined_player_ids().into_iter().min();
        if !self.run.open_vote(&self.content, &classes, host) {
            return;
        }
        if let Some(session) = self.run.vote_session.as_mut() {
            session.last_sent_second = Some(session.seconds_left());
        }
        if let Some(msg) = self.vote_open_msg() {
            self.broadcast(&msg);
        }
        self.broadcast_vote_state();
        tracing::info!(level = self.run.level_num, "vote session opened");
    }

    /// Close the vote session and broadcast the outcome (winner, new permanent).
    fn finish_vote(&mut self) {
        let Some(outcome) = self.run.close_vote(&self.content) else { return };
        self.broadcast(&ServerMsg::VoteResult {
            winner: outcome.winner,
            id: outcome.id.clone(),
            name: outcome.name.clone(),
            kind: outcome.kind,
            permanent: outcome.permanent,
        });
        tracing::info!(winner = %outcome.id, permanent = outcome.permanent, "vote resolved");
    }

    /// True when every joined player holds a ballot in the open session.
    fn all_joined_voted(&self) -> bool {
        let Some(session) = self.run.vote_session.as_ref() else { return false };
        let players = self.joined_player_ids();
        !players.is_empty() && players.iter().all(|id| session.has_voted(*id))
    }

    /// Everyone back on the spawn platform of the current level (hub or generated), alive, then tell them.
    fn respawn_all_and_broadcast(&mut self) {
        let ids = self.joined_ids();
        for (slot, id) in ids.iter().enumerate() {
            let at = self.spawn_point(slot);
            if let Some(p) = self.conns.get_mut(id).and_then(|c| c.player.as_mut()) {
                p.state = player_spawn_state(at, p.class);
                p.inputs.clear();
                p.mode = PLAYER_ACTIVE;
                p.starved_ticks = 0;
            }
        }
        let level = ServerMsg::Level {
            data: self.run.level.clone(),
        };
        self.broadcast(&level);
        // Level start (generated levels only; the hub has no event): announce it.
        if self.run.level.event != EVENT_NONE
            && let Some(def) = self.content.event(self.run.level.event)
        {
            self.broadcast(&ServerMsg::Notice {
                kind: NOTICE_EVENT,
                ids: vec![def.id.clone()],
            });
        }
        // Escalation milestone banner (design §5, from level 4 on): only on actual level starts, not in the lobby.
        if self.run.phase == PHASE_PLAYING && crate::run::escalation_milestone_level(self.run.level_num) {
            self.broadcast(&ServerMsg::Notice {
                kind: NOTICE_EVENT,
                ids: vec![format!("milestone_{}", self.run.level_num)],
            });
        }
        let info = ServerMsg::RunState { info: self.run.info() };
        self.broadcast(&info);
        if let Some(draft) = self.draft_msg() {
            self.broadcast(&draft);
        }
        if let Some(shop) = self.shop_msg() {
            self.broadcast(&shop);
        }
    }

    fn broadcast_run_state(&mut self) {
        let info = ServerMsg::RunState { info: self.run.info() };
        self.broadcast(&info);
    }

    /// Resolve the lobby draft once every joined player has a currently-valid vote.
    /// (After a shop reroll, stored enemy-only votes are invalid until the player re-votes.)
    fn maybe_resolve_draft(&mut self) {
        if self.run.phase != PHASE_ENEMY_DRAFT || self.run.offer.is_none() {
            return;
        }
        let ids = self.joined_ids();
        let all_valid = !ids.is_empty()
            && ids
                .iter()
                .all(|id| self.votes.get(id).is_some_and(|v| self.run.validate_picks(v).is_ok()));
        if !all_valid {
            return;
        }
        let votes: Vec<Vec<u8>> = ids.iter().filter_map(|id| self.votes.get(id).cloned()).collect();
        self.votes.clear();
        self.run.resolve(&self.content, &votes);
        tracing::info!(
            level = self.run.level_num,
            enemies = ?self.run.drafted_enemies,
            upgrades = ?self.run.upgrades,
            curses = ?self.run.curses,
            "draft resolved; level generated"
        );
        self.respawn_all_and_broadcast();
    }

    fn new_run(&mut self) {
        self.seed = self.seed.wrapping_mul(1_664_525).wrapping_add(1_013_904_223);
        self.run = Run::new(&self.content, self.seed, self.start_level);
        self.votes.clear();
        // Players keep their last class; they can change it in the next first lobby.
        self.respawn_all_and_broadcast();
    }

    fn on_message(&mut self, id: ConnId, msg: ClientMsg) {
        let Some(conn) = self.conns.get_mut(&id) else { return };
        let joined = conn.player.is_some();
        match msg {
            ClientMsg::Hello { protocol_version, name } => {
                if joined {
                    return self.reject(id, 400, "duplicate hello");
                }
                if protocol_version != PROTOCOL_VERSION {
                    return self.reject(id, 426, "protocol version mismatch; reload the page");
                }
                if self.player_count() >= MAX_PLAYERS {
                    return self.reject(id, 503, "server full");
                }
                let name = sanitize_name(&name);
                let player_id = self.next_player_id;
                self.next_player_id = self.next_player_id.wrapping_add(1).max(1);
                let spawn = self.spawn_point(self.player_count());
                tracing::info!(conn = id, player = player_id, name = %name, "player joined");
                // Joining mid-level: you spectate until the next lobby; in the lobby you play.
                let mode = if self.run.phase == PHASE_ENEMY_DRAFT {
                    PLAYER_ACTIVE
                } else {
                    PLAYER_WAITING
                };
                let class = CLASS_STRIKER;
                let state = player_spawn_state(spawn, class);
                if let Some(conn) = self.conns.get_mut(&id) {
                    conn.player = Some(Player {
                        id: player_id,
                        name,
                        state,
                        inputs: VecDeque::new(),
                        last_recv_seq: None,
                        last_applied_seq: 0,
                        tokens: 0,
                        starved_ticks: 0,
                        mode,
                        cause: DEATH_VOID,
                        class,
                        overtuned_ticks: 0,
                        infected_ticks: 0,
                        pad_launched: false,
                    });
                }
                self.send(
                    id,
                    &ServerMsg::Welcome {
                        protocol_version: PROTOCOL_VERSION,
                        player_id,
                        tick: self.tick,
                    },
                );
                let level = ServerMsg::Level {
                    data: self.run.level.clone(),
                };
                self.send(id, &level);
                let info = ServerMsg::RunState { info: self.run.info() };
                self.send(id, &info);
                if let Some(draft) = self.draft_msg() {
                    self.send(id, &draft);
                }
                if let Some(shop) = self.shop_msg() {
                    self.send(id, &shop);
                }
                // Rejoining during an open vote: re-offer the session so the new player can ballot.
                if let Some(open) = self.vote_open_msg() {
                    self.send(id, &open);
                }
                if let Some(state) = self.vote_state_msg() {
                    self.send(id, &state);
                }
            }
            _ if !joined => self.reject(id, 400, "hello required first"),
            ClientMsg::Inputs { frames } => {
                if frames.len() > MAX_INPUTS_PER_MSG as usize {
                    return self.reject(id, 400, "too many inputs in one message");
                }
                if let Some(p) = conn.player.as_mut() {
                    for mut f in frames {
                        if p.last_recv_seq.is_some_and(|last| f.seq <= last) {
                            continue; // duplicate or reordered
                        }
                        if p.inputs.len() >= MAX_QUEUED_INPUTS {
                            break; // flooding: drop the excess
                        }
                        f.buttons &= BUTTON_MASK;
                        p.last_recv_seq = Some(f.seq);
                        p.inputs.push_back(f);
                    }
                }
            }
            ClientMsg::Ping { t } => self.send(id, &ServerMsg::Pong { t }),
            ClientMsg::Buy { item } => {
                match self.run.buy_shop_item(&self.content, &item) {
                    Ok(purchase) => {
                        // A reroll replaces the reward cards: enemy cards stay, reward votes are cleared,
                        // everyone gets the fresh Draft again.
                        if purchase == ShopPurchase::Reroll {
                            let cards = self.run.offer.as_ref().map(|o| o.cards.clone()).unwrap_or_default();
                            for vote in self.votes.values_mut() {
                                vote.retain(|&p| cards.get(p as usize).is_some_and(|c| c.kind == CARD_ENEMY));
                            }
                            if let Some(draft) = self.draft_msg() {
                                self.broadcast(&draft);
                            }
                        }
                        self.broadcast(&ServerMsg::Notice {
                            kind: NOTICE_BOUGHT,
                            ids: vec![item],
                        });
                        self.broadcast_run_state();
                        // Re-send the shop so everyone sees the new stock and currency.
                        if let Some(shop) = self.shop_msg() {
                            self.broadcast(&shop);
                        }
                    }
                    Err(why) => {
                        let (code, message) = match why {
                            ShopError::NotInLobby => (409, "the shop is not open"),
                            ShopError::UnknownItem => (404, "unknown shop item"),
                            ShopError::SoldOut => (409, "sold out"),
                            ShopError::NotEnoughCurrency => (402, "not enough currency"),
                        };
                        self.send(
                            id,
                            &ServerMsg::Error {
                                code,
                                message: message.to_owned(),
                            },
                        );
                    }
                }
            }
            ClientMsg::ChooseClass { class } => {
                // Only in the lobby, and only for the first lobby of a run (level 1).
                if self.run.phase != PHASE_ENEMY_DRAFT || self.run.level_num != 1 {
                    return self.send(
                        id,
                        &ServerMsg::Error {
                            code: 409,
                            message: "class can only be changed in the first lobby".to_owned(),
                        },
                    );
                }
                if class > CLASS_GLIDER {
                    return self.send(
                        id,
                        &ServerMsg::Error {
                            code: 400,
                            message: "unknown class".to_owned(),
                        },
                    );
                }
                if let Some(p) = self.conns.get_mut(&id).and_then(|c| c.player.as_mut()) {
                    p.class = class;
                    p.state.class = class;
                }
            }
            ClientMsg::DraftPick { picks } => {
                if picks.len() > MAX_DRAFT_PICKS as usize {
                    return self.reject(id, 400, "too many draft picks");
                }
                match self.run.validate_picks(&picks) {
                    Ok(()) => {
                        self.votes.insert(id, picks);
                        self.maybe_resolve_draft();
                    }
                    Err(why) => self.send(
                        id,
                        &ServerMsg::Error {
                            code: 422,
                            message: why.to_owned(),
                        },
                    ),
                }
            }
            ClientMsg::Vote { option } => {
                let player_id = match conn.player.as_ref() {
                    Some(p) => p.id,
                    None => return self.reject(id, 400, "hello required first"),
                };
                let Some(session) = self.run.vote_session.as_mut() else {
                    return self.send(
                        id,
                        &ServerMsg::Error {
                            code: 409,
                            message: "no vote is open".to_owned(),
                        },
                    );
                };
                match session.cast(player_id, option) {
                    Ok(()) => {
                        self.broadcast_vote_state();
                        // The session closes early once every joined player holds a ballot.
                        if self.all_joined_voted() {
                            self.finish_vote();
                        }
                    }
                    Err(why) => self.send(
                        id,
                        &ServerMsg::Error {
                            code: 422,
                            message: why.to_owned(),
                        },
                    ),
                }
            }
            ClientMsg::Restart => {
                // A new run starts once the run is over (the whole team is down); otherwise ignore.
                if self.run.phase == PHASE_OVER {
                    tracing::info!(conn = id, "new run");
                    self.new_run();
                }
            }
        }
    }

    /// Advance the simulation one 20 Hz tick and broadcast snapshots.
    pub fn tick(&mut self) {
        self.tick = self.tick.wrapping_add(1);
        let tuning = self.run.tuning();
        // Altar uses are discovered while stepping input frames; notices go out after the movement loop.
        let mut altar_notices: Vec<(String, String)> = Vec::new();

        // 1. Movement (lobby and playing).
        for p in self.conns.values_mut().filter_map(|c| c.player.as_mut()) {
            p.pad_launched = false;
            p.tokens = (p.tokens + STEPS_PER_TICK).min(MAX_STEP_TOKENS);
            if !p.active() {
                // Spectators' inputs are acknowledged and dropped (their camera is client-side).
                if let Some(last) = p.inputs.back() {
                    p.last_applied_seq = last.seq;
                }
                p.inputs.clear();
                continue;
            }
            let mut stepped = 0;
            while p.tokens > 0 {
                let Some(f) = p.inputs.pop_front() else { break };
                // Altars: rising-edge BTN_INTERACT from the input frames, ACTIVE players, playing levels only.
                if self.run.phase == PHASE_PLAYING
                    && f.buttons & BTN_INTERACT != 0
                    && p.state.prev_buttons & BTN_INTERACT == 0
                    && let Some(pair) = self.run.try_use_altar(&self.content, &p.state.pos)
                {
                    altar_notices.push(pair);
                }
                controller::step_with(&mut p.state, &f, &self.run.world, &tuning);
                // A pad launch is the only way to leave the ground faster than a jump
                // (PAD_LAUNCH_VEL 16 vs JUMP_VELOCITY 12); the controller reports no flag.
                if p.state.vel.y >= sim::tuning::PAD_LAUNCH_VEL - 0.5 {
                    p.pad_launched = true;
                }
                p.last_applied_seq = f.seq;
                p.tokens -= 1;
                stepped += 1;
            }
            if stepped == 0 {
                p.starved_ticks += 1;
                if p.starved_ticks >= STARVE_TICKS {
                    let neutral = InputFrame {
                        seq: p.last_applied_seq,
                        move_x: 0,
                        move_z: 0,
                        buttons: 0,
                    };
                    for _ in 0..STEPS_PER_TICK.min(p.tokens) {
                        controller::step_with(&mut p.state, &neutral, &self.run.world, &tuning);
                        p.tokens -= 1;
                    }
                }
            } else {
                p.starved_ticks = 0;
            }
        }
        for (upgrade, curse) in altar_notices {
            self.broadcast(&ServerMsg::Notice {
                kind: NOTICE_ALTAR,
                ids: vec![upgrade, curse],
            });
            self.broadcast_run_state();
        }

        if self.run.phase == PHASE_PLAYING {
            self.tick_playing();
        } else {
            // Lobby / over: the hub is a safe space - falling off just puts you back on the spawn.
            let kill_y = self.run.world.kill_y;
            let ids = self.joined_ids();
            for (slot, id) in ids.iter().enumerate() {
                let at = self.spawn_point(slot);
                if let Some(p) = self.conns.get_mut(id).and_then(|c| c.player.as_mut())
                    && p.state.pos.y < kill_y
                {
                    p.state = player_spawn_state(at, p.class);
                    p.inputs.clear();
                }
            }
        }

        // Snapshots.
        let players: Vec<PlayerView> = self
            .conns
            .values()
            .filter_map(|c| c.player.as_ref())
            .map(|p| PlayerView {
                id: p.id,
                pos: p.state.pos,
                vel: p.state.vel,
                mode: p.mode,
                class: p.state.class,
            })
            .collect();
        let enemies = self.run.enemy_views();
        let projectiles = self.run.hazards.projectiles();
        let tiles = self.run.world_tiles();
        let altars_used = self.run.altars_used.clone();
        for id in self.joined_ids() {
            let Some(p) = self.conns.get(&id).and_then(|c| c.player.as_ref()) else {
                continue;
            };
            let msg = ServerMsg::Snapshot {
                tick: self.tick,
                level: self.run.level.level,
                last_input_seq: p.last_applied_seq,
                you: p.state.clone(),
                players: players.clone(),
                enemies: enemies.clone(),
                gifts_taken: self.run.gifts_taken.clone(),
                projectiles: projectiles.clone(),
                tiles: tiles.clone(),
                altars_used: altars_used.clone(),
                // M3 (T-056): two-phase levels.
                level_phase: self.run.level_phase,
                golden_taken: self.run.golden_taken.clone(),
                collapse_ms: self.run.collapse_ms(),
            };
            self.send(id, &msg);
        }
    }

    fn tick_playing(&mut self) {
        // Phase timers advance with the tick (PENDING -> COLLAPSING after 3.0 s; batch warnings and drops).
        self.run.tick_collapse();

        // The vote session countdown advances with the playing phase; it closes on zero, on
        // everyone having voted (at cast time), or when the level ends (below).
        let vote_tick = match self.run.vote_session.as_mut() {
            Some(session) => {
                let closed = session.tick();
                let secs = session.seconds_left();
                let changed = session.last_sent_second != Some(secs);
                session.last_sent_second = Some(secs);
                Some((closed, changed))
            }
            None => None,
        };
        match vote_tick {
            Some((true, _)) => self.finish_vote(),
            Some((false, true)) => self.broadcast_vote_state(),
            _ => {}
        }

        // Gifts (shared by the team). In phase 2 these are the golden gifts at the collected positions.
        let currency_before = self.run.currency;
        let mut pickups: Vec<Vec3> = Vec::new();
        let phase_before = self.run.level_phase;
        for conn in self.conns.values() {
            if let Some(p) = conn.player.as_ref().filter(|p| p.active()) {
                pickups.extend(self.run.collect_gifts(&p.state.pos));
            }
        }
        let mut info_changed = self.run.currency != currency_before;
        if phase_before != self.run.level_phase && self.run.level_phase == protocol::LEVEL_COLLAPSE_PENDING {
            self.broadcast(&ServerMsg::Notice {
                kind: NOTICE_EVENT,
                ids: vec!["collapse".to_string()],
            });
            // T-078: the vote session opens with the collapse/exit window.
            self.open_vote_session();
        }

        // Hazards: the senses of the ACTIVE players. `warded` = Bulwark Aegis Bloom is up (knockback-immune).
        // `speed_mul` = the phase-2 pressure multiplier (1.1 in LEVEL_COLLAPSING, design §1.4).
        let speed_mul = if self.run.level_phase == protocol::LEVEL_COLLAPSING {
            1.10
        } else {
            1.0
        };
        let senses: Vec<sim::hazards::PlayerSense> = self
            .conns
            .values()
            .filter_map(|c| c.player.as_ref())
            .filter(|p| p.active())
            .map(|p| sim::hazards::PlayerSense {
                id: p.id,
                feet: p.state.pos,
                vel: p.state.vel,
                warded: p.state.ward_frames > 0,
                class: p.class,
                jump_launched: p.pad_launched,
                used_ability: p.state.ability_active_frames > 0,
                used_support_ability: p.class == protocol::CLASS_BULWARK && p.state.ability_active_frames > 0,
                gliding: p.class == CLASS_GLIDER && p.state.ability_active_frames > 0 && !p.state.grounded,
                overtuned_ticks: p.overtuned_ticks,
                infected_ticks: p.infected_ticks,
            })
            .collect();
        // Registry brains read the phase and the run's permanents.
        let coil_permanent = self.run.all_permanents().iter().any(|id| id == "coil");
        self.run
            .hazards
            .set_run_context(self.run.level_phase == protocol::LEVEL_COLLAPSING, coil_permanent);
        let effects = self.run.hazards.step(
            &mut self.run.enemies,
            &self.run.world,
            &self.run.level,
            &senses,
            &pickups,
            speed_mul,
        );
        let mut shoves: Vec<(u32, Vec3, u8)> = Vec::new();
        let mut kills: BTreeMap<u32, u8> = BTreeMap::new();
        let mut overtuned: Vec<u32> = Vec::new();
        let mut infected: Vec<u32> = Vec::new();
        for effect in effects {
            match effect {
                sim::hazards::HazardEffect::Kill { player, cause } => {
                    kills.entry(player).or_insert(cause);
                }
                sim::hazards::HazardEffect::Shove {
                    player,
                    vel,
                    stagger_ticks,
                } => shoves.push((player, vel, stagger_ticks)),
                sim::hazards::HazardEffect::Overtuned { player } => overtuned.push(player),
                sim::hazards::HazardEffect::Infect { player } => infected.push(player),
            }
        }
        // Debuffs the registry brains applied this tick. OVERTUNED is what makes Peal's
        // ring lethal; the infection slows the victim and pauses Rewind's and Kite's timers.
        for p in self.conns.values_mut().filter_map(|c| c.player.as_mut()) {
            if overtuned.contains(&p.id) {
                p.overtuned_ticks = sim::rules::enemies::OVERTUNED_TICKS as u16;
            }
            if infected.contains(&p.id) {
                p.infected_ticks = sim::rules::enemies::blight::BLIGHT_INFECTION_TICKS as u16;
            }
            p.overtuned_ticks = p.overtuned_ticks.saturating_sub(1);
            p.infected_ticks = p.infected_ticks.saturating_sub(1);
        }

        // Bulwark Aegis Bloom: on the pulse frame the sim knocks enemies back; other ACTIVE, unwarded
        // players within 3.0 u (chest to chest) are shoved directly away at 7 u/s with 48 frames of stagger.
        let pulses: Vec<(u32, Vec3)> = self
            .conns
            .values()
            .filter_map(|c| c.player.as_ref())
            .filter(|p| p.active() && controller::bulwark_pulse_frame(&p.state))
            .map(|p| (p.id, Vec3::new(p.state.pos.x, p.state.pos.y + 0.9, p.state.pos.z)))
            .collect();
        for (caster_id, centre) in pulses {
            self.run
                .hazards
                .pulse(&mut self.run.enemies, &centre, BULWARK_PULSE_RADIUS);
            let others: Vec<PulsePlayer> = self
                .conns
                .values()
                .filter_map(|c| c.player.as_ref())
                .filter(|p| p.active())
                .map(|p| PulsePlayer {
                    id: p.id,
                    feet: p.state.pos,
                    warded: p.state.ward_frames > 0,
                })
                .collect();
            for (player_id, vel, stagger_frames) in pulse_stagger_targets(caster_id, &centre, &others) {
                if let Some(p) = self
                    .conns
                    .values_mut()
                    .filter_map(|c| c.player.as_mut())
                    .find(|p| p.id == player_id)
                {
                    p.state.vel = vel;
                    p.state.stagger_frames = stagger_frames;
                }
            }
        }

        // Shove effects (e.g. the Rift Warden): velocity is replaced; stagger upgraded to 60 Hz frames.
        for (player_id, vel, stagger_ticks) in shoves {
            if let Some(p) = self
                .conns
                .values_mut()
                .filter_map(|c| c.player.as_mut())
                .find(|p| p.id == player_id && p.active())
            {
                p.state.vel = vel;
                p.state.stagger_frames = stagger_ticks.saturating_mul(3);
            }
        }

        // World tiles: hazards (Crumble/Fretted) merged with the collapse (collapse wins on the same id).
        let tiles = self.run.world_tiles();
        self.run.world.set_tiles(&tiles);

        // Deaths: a spare life (shared by the team) respawns you at the level start; otherwise you spectate.
        // `kills` come from the hazards (enemy contact, projectiles); the void check stays the cheap early-out.
        let ids = self.joined_ids();
        for (slot, id) in ids.iter().enumerate() {
            let at = self.spawn_point(slot);
            let Some(p) = self
                .conns
                .get_mut(id)
                .and_then(|c| c.player.as_mut())
                .filter(|p| p.active())
            else {
                continue;
            };
            let cause = if p.state.pos.y < self.run.world.kill_y {
                Some(DEATH_VOID)
            } else {
                kills.get(&p.id).copied()
            };
            let Some(cause) = cause else {
                continue;
            };
            if self.run.lives > 0 {
                self.run.lives -= 1;
                p.state = player_spawn_state(at, p.class);
                p.inputs.clear();
                tracing::info!(conn = id, cause, lives = self.run.lives, "life lost");
            } else {
                p.mode = PLAYER_DEAD;
                p.cause = cause;
                p.state.vel = Vec3::ZERO;
                tracing::info!(conn = id, cause, level = self.run.level_num, "player out; spectating");
            }
            info_changed = true;
        }

        // Beacon: each active player who walks into the open beacon finishes the level and spectates.
        for p in self
            .conns
            .values_mut()
            .filter_map(|c| c.player.as_mut())
            .filter(|p| p.active())
        {
            if self.run.beacon_reached(&p.state.pos) {
                p.mode = PLAYER_FINISHED;
                p.state.vel = Vec3::ZERO;
                info_changed = true;
            }
        }

        // The level ends when nobody is still playing it.
        let players: Vec<&Player> = self.conns.values().filter_map(|c| c.player.as_ref()).collect();
        if !players.is_empty() && !players.iter().any(|p| p.active()) {
            if players.iter().any(|p| p.mode == PLAYER_FINISHED) {
                // Any open vote closes with the level: the winner still applies to the next level.
                self.finish_vote();
                self.run.finish_level(&self.content);
                tracing::info!(
                    next_level = self.run.level_num,
                    currency = self.run.currency,
                    "level cleared"
                );
                self.votes.clear();
                self.respawn_all_and_broadcast();
            } else {
                // Team wipe: the run is over; an open vote is discarded with it.
                self.run.vote_session = None;
                self.end_run();
            }
            return;
        }
        if info_changed {
            self.broadcast_run_state();
        }
    }

    /// Everyone is out: the run is over. Each player gets a summary with their own cause of death
    /// (DEATH_VOID, DEATH_ENEMY or DEATH_PROJECTILE).
    fn end_run(&mut self) {
        self.run.end();
        tracing::info!(level = self.run.level_num, "run over");
        let summaries: Vec<(ConnId, u8)> = self
            .conns
            .iter()
            .filter_map(|(id, c)| c.player.as_ref().map(|p| (*id, p.cause)))
            .collect();
        for (id, cause) in summaries {
            let summary = RunSummary {
                level_reached: self.run.level_num,
                gifts_collected: self.run.gifts_collected,
                cause,
            };
            self.send(id, &ServerMsg::RunOver { summary });
        }
        self.broadcast_run_state();
    }
}

/// Trim, drop control characters, cap length; never empty.
fn sanitize_name(raw: &str) -> String {
    let name: String = raw
        .chars()
        .filter(|c| !c.is_control())
        .take(MAX_NAME_LEN as usize)
        .collect();
    let name = name.trim();
    if name.is_empty() {
        "blocky".to_owned()
    } else {
        name.to_owned()
    }
}

/// Runs the game until every sender is gone.
pub async fn run(mut rx: mpsc::Receiver<GameEvent>, mut game: Game, game_state: Arc<GameStateHandle>) {
    let period = Duration::from_millis(1000 / u64::from(TICK_HZ));
    let mut interval = tokio::time::interval(period);
    interval.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Burst);
    let mut window_start = Instant::now();
    let mut window_ticks = 0u32;
    loop {
        tokio::select! {
            biased;
            _ = interval.tick() => {
                game.tick();
                game_state.publish(
                    game.player_count() as u32,
                    game.run.level_num,
                    game.run.phase,
                );
                window_ticks += 1;
                tracing::trace!(tick = game.tick_count(), "tick");
                let elapsed = window_start.elapsed();
                if elapsed >= Duration::from_secs(1) {
                    let hz = f64::from(window_ticks) / elapsed.as_secs_f64();
                    tracing::info!("tick {} | {hz:.1} Hz | players {}", game.tick_count(), game.player_count());
                    window_start = Instant::now();
                    window_ticks = 0;
                }
            }
            ev = rx.recv() => match ev {
                Some(ev) => game.handle(ev),
                None => break,
            },
        }
    }
}

/// Radius of a Bulwark Aegis Bloom pulse (design §3: 3.0 u).
const BULWARK_PULSE_RADIUS: f32 = 3.0;
/// Speed of the radial knockback applied by the pulse (7 u/s, away from the caster).
const BULWARK_PULSE_SPEED: f32 = 7.0;
/// Stagger applied by the pulse to other players, in 60 Hz frames (0.8 s).
const BULWARK_PULSE_STAGGER_FRAMES: u8 = 48;

/// One ACTIVE player as seen by a Bulwark pulse.
struct PulsePlayer {
    /// Player id (the caster is excluded by id).
    id: u32,
    /// Feet position (the chest is derived as `feet + 0.9`).
    feet: Vec3,
    /// True while the player is rooted in their own ward (knockback-immune).
    warded: bool,
}

/// Compute the player shoves for one Bulwark pulse: every ACTIVE, unwarded player other than the caster
/// within [`BULWARK_PULSE_RADIUS`] (measured chest to chest). Each is pushed horizontally directly away at
/// [`BULWARK_PULSE_SPEED`] u/s with [`BULWARK_PULSE_STAGGER_FRAMES`] frames of stagger.
fn pulse_stagger_targets(caster_id: u32, centre: &Vec3, others: &[PulsePlayer]) -> Vec<(u32, Vec3, u8)> {
    let mut out = Vec::new();
    for player in others {
        if player.id == caster_id || player.warded {
            continue;
        }
        let dx = player.feet.x - centre.x;
        let dy = (player.feet.y + 0.9) - centre.y;
        let dz = player.feet.z - centre.z;
        if dx * dx + dy * dy + dz * dz > BULWARK_PULSE_RADIUS * BULWARK_PULSE_RADIUS {
            continue;
        }
        let horizontal = (dx * dx + dz * dz).sqrt();
        let (ux, uz) = if horizontal <= 1e-6 {
            (1.0, 0.0)
        } else {
            (dx / horizontal, dz / horizontal)
        };
        out.push((
            player.id,
            Vec3::new(ux * BULWARK_PULSE_SPEED, 0.0, uz * BULWARK_PULSE_SPEED),
            BULWARK_PULSE_STAGGER_FRAMES,
        ));
    }
    out
}

    fn connect(game: &mut Game, id: ConnId) -> mpsc::Receiver<Vec<u8>> {
        let (tx, rx) = mpsc::channel(OUTBOX_CAPACITY);
        game.handle(GameEvent::Connected { id, out: tx });
        rx
    }

            },

    }

    fn drain(rx: &mut mpsc::Receiver<Vec<u8>>) -> Vec<ServerMsg> {
        let mut out = vec![];
        while let Ok(b) = rx.try_recv() {
            out.push(protocol::decode::<ServerMsg>(&b).unwrap());
        }
        out
    }

    /// Collect every remaining normal gift as `id`; the last one starts the collapse window
    /// (and with it the vote session). The final tick's messages are left in the outboxes.
    fn enter_collapse_and_open_vote(g: &mut Game, id: ConnId, rxs: &mut [&mut mpsc::Receiver<Vec<u8>>]) {
        let gifts: Vec<_> = g
            .run
            .level
            .gifts
            .iter()
            .filter(|g| g.kind != protocol::GIFT_GOLDEN)
            .cloned()
            .collect();
        for (i, gift) in gifts.iter().enumerate() {
            {
                let p = player(g, id);
                p.state.pos = Vec3::new(gift.pos.x, gift.pos.y - 0.9, gift.pos.z);
                p.state.vel = Vec3::ZERO;
                p.starved_ticks = 0;
            }
            for e in &mut g.run.enemies {
                e.active_in = 1_000;
            }
            g.tick();
            // Keep the outboxes under capacity; leave the last tick's burst for the caller.
            if i + 1 < gifts.len() {
                for rx in &mut *rxs {
                    let _ = drain(rx);
                }
            }
        }
        assert_eq!(g.run.level_phase, protocol::LEVEL_COLLAPSE_PENDING);
        assert!(g.run.vote_session.is_some(), "the vote session opens with the collapse window");
    }
