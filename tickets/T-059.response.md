=== FILE: web/src/content.ts ===
// Typed access to design/content.json: draft cards (enemies, upgrades, curses), run classes, lobby shop stock,
// level events, milestone notices and altars, plus per-theme screen grades.
// The Rust server reads the same file, so card ids, numbers and text can never drift between client and server.
import content from "../../design/content.json";

export type Mods = Partial<Record<string, number>>;

/** How an enemy moves once it is awake. Absent on a card means "chaser". */
export type EnemyBehavior = "chaser" | "turret" | "mimic" | "corrupter";

export interface EnemyParams {
  speed: number;
  aggro: number;
  grace_s: number;
  size: number;
  /** Movement archetype; "chaser" when omitted. */
  behavior?: EnemyBehavior;
  /** turret: seconds between bolts. */
  fire_interval_s?: number;
  /** turret: bolt travel speed in units per second. */
  projectile_speed?: number;
  /** mimic: distance at which the gift disguise drops. */
  reveal_radius?: number;
  /** corrupter: seconds between platform corruptions. */
  corrupt_interval_s?: number;
}

export interface EnemyCard {
  id: string;
  name: string;
  flavor: string;
  behavior: string;
  threat: number;
  params: EnemyParams;
}

export interface ModCard {
  id: string;
  name: string;
  flavor: string;
  effect: string;
  rarity: "common" | "rare";
  mods: Mods;
  threat?: number;
  greater?: boolean;
}

/** A playable class: one ability, one control line. */
export interface ClassCard {
  id: string;
  name: string;
  flavor: string;
  ability: string;
  controls: string;
}

export type ShopKind = "life" | "reroll" | "mods";

/** A lobby purchase paid for with golden gifts. */
export interface ShopItem {
  id: string;
  name: string;
  flavor: string;
  effect: string;
  cost: number;
  stock: number;
  kind: ShopKind;
  /** Only on kind "mods": permanent for the run. */
  mods?: Mods;
}

/** A level-wide twist, keyed by the wire event number. */
export interface LevelEvent {
  id: string;
  event: number;
  name: string;
  banner: string;
  /** slick_ice only: ground friction multiplier. */
  friction_mul?: number;
}

/**
 * A milestone notice: the phase-2 collapse, or one escalation step from design §5.
 * The server broadcasts these as NOTICE_EVENT ids ("collapse", "milestone_9", ...).
 */
export interface Milestone {
  id: string;
  /** Escalation level that fires it; 0 for the collapse. */
  level: number;
  name: string;
  banner: string;
}

/** Title + body of one notice line. */
export interface NoticeLine {
  title: string;
  body: string;
}

export interface Altar {
  id: string;
  name: string;
  flavor: string;
  effect: string;
}

export interface ScreenGrade {
  tint: string;
  tint_strength: number;
  saturation: number;
  contrast: number;
  brightness: number;
  vignette_color: string;
  vignette_darkness: number;
  aberration: number;
  grain: number;
  scanlines: number;
}

export const ENEMIES: readonly EnemyCard[] = content.enemies as EnemyCard[];
export const UPGRADES: readonly ModCard[] = content.upgrades as ModCard[];
export const CURSES: readonly ModCard[] = content.curses as ModCard[];
export const CLASSES: readonly ClassCard[] = content.classes as ClassCard[];
export const SHOP: readonly ShopItem[] = content.shop as ShopItem[];
export const EVENTS: readonly LevelEvent[] = content.events as LevelEvent[];
export const MILESTONES: readonly Milestone[] = content.milestones as Milestone[];
export const ALTARS: readonly Altar[] = content.altars as Altar[];
const GRADES: Record<string, ScreenGrade> = content.grades;

/** Card kinds as used on the wire (protocol DraftCard.kind). */
export const CARD_ENEMY = 0;
export const CARD_UPGRADE = 1;
export const CARD_CURSE = 2;

export type AnyCard =
  | { kind: typeof CARD_ENEMY; card: EnemyCard }
  | { kind: typeof CARD_UPGRADE; card: ModCard }
  | { kind: typeof CARD_CURSE; card: ModCard };

/** Resolve a wire card (kind + id) to its content; null for unknown ids (never trust the wire blindly). */
export function lookupCard(kind: number, id: string): AnyCard | null {
  if (kind === CARD_ENEMY) {
    const card = ENEMIES.find((e) => e.id === id);
    return card ? { kind: CARD_ENEMY, card } : null;
  }
  if (kind === CARD_UPGRADE) {

// Pure notice helpers: what a notice says, and how visible it is right now.
// Unit-tested; never throws.
import type { Notice } from '../game/store';
import { NOTICE_ALTAR, NOTICE_BOUGHT, NOTICE_EVENT } from '../protocol/generated';
import { CARD_CURSE, CARD_UPGRADE, SHOP, lookupCard, noticeEventText } from '../content';

export const NOTICE_LIFETIME_MS = 4000;
export const NOTICE_FADE_MS = 500;

export interface NoticeText {
  title: string;
  body: string;
}

/**
 * Resolve a notice to plain text. Returns null for unknown kinds, unknown
 * content ids, or malformed notices; never throws.
 */
export function noticeText(notice: Notice): NoticeText | null {
  try {
    if (notice.kind === NOTICE_ALTAR) {
      const [upgradeId, curseId] = notice.ids;
      if (!upgradeId || !curseId) {
        return null;
      }
      const upgrade = lookupCard(CARD_UPGRADE, upgradeId);
      const curse = lookupCard(CARD_CURSE, curseId);
      if (upgrade === null || upgrade.kind !== CARD_UPGRADE) {
        return null;
      }
      if (curse === null || curse.kind !== CARD_CURSE) {
        return null;
      }
      return {
        title: 'THE ALTAR',
        body: `The altar gave you ${upgrade.card.name} and ${curse.card.name}.`,
      };
    }
    if (notice.kind === NOTICE_BOUGHT) {
      const [itemId] = notice.ids;
      if (!itemId) {
        return null;
      }
      const item = SHOP.find((entry) => entry.id === itemId);
      if (!item) {
        return null;
      }
      return { title: 'SHOP', body: `Bought ${item.name}.` };
    }
    if (notice.kind === NOTICE_EVENT) {
      const [eventId] = notice.ids;
      if (!eventId) {
        return null;
      }
      // Level events first, then milestones (collapse + escalation steps); null when neither knows the id.
      return noticeEventText(eventId);
    }
    return null;
  } catch {
    return null;
  }
}

/**
 * Opacity for a notice of the given age. Opaque for the first
 * NOTICE_LIFETIME_MS - NOTICE_FADE_MS, then a linear fade to 0. Future or
 * invalid ages are clamped; non-finite input hides the notice.
 */
export function noticeOpacity(ageMs: number): number {
  if (!Number.isFinite(ageMs)) {
    return 0;
  }
  if (ageMs <= NOTICE_LIFETIME_MS - NOTICE_FADE_MS) {
    return 1;
  }
  if (ageMs >= NOTICE_LIFETIME_MS) {
    return 0;
  }
  return 1 - (ageMs - (NOTICE_LIFETIME_MS - NOTICE_FADE_MS)) / NOTICE_FADE_MS;
}

      const p = PLAYABLE[k];
      if (p && (v < p[0] || v > p[1])) err(`${where}.${k}=${v} breaks playability [${p[0]}, ${p[1]}]`);
    } else if (ADD.includes(k)) {
